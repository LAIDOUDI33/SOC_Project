   191→
   192→  incident            Incident?        @relation(fields: [incidentId], references: [id])

   194→
   195→  // Optimized indexes for production workloads
   196→  @@index([status, severity])                          // Dashboard: alerts by status/severity
   197→  @@index([firstSeen])                                 // Time-range queries

   199→  @@index([severity]) WHERE status = 'NEW'             // Triage queue
