   380→function ComplianceScoreCard({
   381→  score,
   382→  trend,
   383→  label,
   384→  subtitle
   385→}: {
   386→  score: number
   387→  trend?: number
   388→  label: string
   389→  subtitle?: string
   390→}) {
   391→  const getScoreColor = (s: number) => {
   392→    if (s >= 90) return 'text-green-400'
   393→    if (s >= 70) return 'text-yellow-400'
   394→    if (s >= 50) return 'text-orange-400'
   395→    return 'text-red-400'
   396→  }
   397→
   398→  const getProgressColor = (s: number) => {
   399→    if (s >= 90) return '[&>div]:bg-green-500'
   400→    if (s >= 70) return '[&>div]:bg-yellow-500'
   401→    if (s >= 50) return '[&>div]:bg-orange-500'
   402→    return '[&>div]:bg-red-500'
   403→  }
   404→
   405→  return (
   406→    <Card className="bg-slate-900 border-slate-700">
   407→      <CardContent className="pt-6">
   408→        <p className="text-sm text-slate-400 mb-1">{label}</p>
   409→        <div className={`text-3xl font-bold ${getScoreColor(score)} mb-2`}>
   410→          {score}%
   411→        </div>
   412→        <Progress value={score} className={`h-2 ${getProgressColor(score)}`} />
   413→        <div className="flex items-center justify-between mt-2">
   414→          {trend !== undefined && (
   415→            <MetricTrend value={trend} showArrow size="sm" />
   416→          )}
   417→          {subtitle && (
   418→            <span className="text-xs text-slate-500">{subtitle}</span>
   419→          )}
   420→        </div>
   421→      </CardContent>
   422→    </Card>
   423→  )
   424→}

   512→function AuditFindingsLifecycle({ findings }: { findings: AuditFinding[] }) {
   513→  const getSeverityIcon = (severity: AuditFinding['severity']) => {
   514→    switch (severity) {
   515→      case 'critical': return <AlertCircle className="h-4 w-4 text-red-400" />
   516→      case 'major': return <AlertTriangle className="h-4 w-4 text-orange-400" />
   517→      case 'minor': return <AlertCircle className="h-4 w-4 text-yellow-400" />
   518→      case 'observation': return <Eye className="h-4 w-4 text-blue-400" />
   519→    }
   520→  }
   521→
   522→  const getStatusColor = (status: AuditFinding['status']) => {
   523→    switch (status) {
   524→      case 'open': return 'border-red-500/30 bg-red-500/5'
   525→      case 'in-progress': return 'border-blue-500/30 bg-blue-500/5'
   526→      case 'remediated': return 'border-green-500/30 bg-green-500/5'
   527→      case 'closed': return 'border-slate-500/30 bg-slate-500/5'
   528→      case 'accepted-risk': return 'border-purple-500/30 bg-purple-500/5'
   529→    }
   530→  }
   531→
   532→  const openFindings = findings.filter(f => f.status === 'open' || f.status === 'in-progress')
   533→  const closedFindings = findings.filter(f => f.status === 'closed' || f.status === 'remediated')

   608→function DeadlineCountdown({ deadlines }: { deadlines: RegulatoryDeadline[] }) {
   609→  const getDeadlineStatus = (status: RegulatoryDeadline['status']) => {
   610→    switch (status) {
   611→      case 'on-track': return { color: 'bg-green-500/20 border-green-500/30', badge: 'bg-green-500/20 text-green-400' }
   612→      case 'at-risk': return { color: 'bg-yellow-500/20 border-yellow-500/30', badge: 'bg-yellow-500/20 text-yellow-400' }
   613→      case 'overdue': return { color: 'bg-red-500/20 border-red-500/30', badge: 'bg-red-500/20 text-red-400' }
   614→      case 'completed': return { color: 'bg-slate-500/20 border-slate-500/30', badge: 'bg-slate-500/20 text-slate-400' }
   615→    }
   616→  }
   617→
   618→  const daysRemaining = (date: Date) => Math.ceil((date.getTime() - Date.now()) / 86400000)
   619→
   620→  return (
   621→    <Card className="bg-slate-900 border-slate-700">
   622→      <CardHeader className="pb-3">
   623→        <CardTitle className="text-white text-base flex items-center gap-2">
   624→          <Timer className="h-4 w-4 text-red-400" />
   625→          Regulatory Deadline Countdown
   626→        </CardTitle>
   627→      </CardHeader>
   628→      <CardContent>
   629→        <div className="space-y-3">
   630→          {deadlines.map(deadline => {
   631→            const style = getDeadlineStatus(deadline.status)
   632→            const remaining = daysRemaining(deadline.deadline)
   633→            
   634→            return (
   635→              <div key={deadline.id} className={`p-3 rounded-lg border ${style.color}`}>
   636→                <div className="flex items-start justify-between mb-2">
   637→                  <div>
   638→                    <p className="font-medium text-white text-sm">{deadline.requirement}</p>
   639→                    <p className="text-xs text-slate-400">{deadline.regulation}</p>
   640→                  </div>
   641→                  <Badge variant="secondary" className={style.badge}>
   642→                    {deadline.status.replace('-', ' ').toUpperCase()}
   643→                  </Badge>
   644→                </div>
   645→                
   646→                <div className="flex items-center gap-4 mt-3">
   647→                  <div className="flex-1">
   648→                    <div className="flex items-center justify-between text-xs mb-1">
   649→                      <span className="text-slate-400">Progress</span>
   650→                      <span className="text-white font-medium">{deadline.progress}%</span>
   651→                    </div>
   652→                    <Progress value={deadline.progress} className="h-1.5" />
   653→                  </div>
   654→                  
   655→                  <div className="text-right min-w-[80px]">
   656→                    <p className={`text-lg font-bold ${
   657→                      remaining < 0 ? 'text-red-400' :
   658→                      remaining <= 7 ? 'text-orange-400' :
   659→                      remaining <= 30 ? 'text-yellow-400' :
   660→                      'text-green-400'
   661→                    }`}>
   662→                      {remaining < 0 ? `${Math.abs(remaining)}d overdue` : `${remaining}d left`}
   663→                    </p>
   664→                    <p className="text-[10px] text-slate-500">{deadline.owner}</p>
   665→                  </div>
   666→                </div>
   667→              </div>
   668→            )
   669→          })}
   670→        </div>
   671→      </CardContent>
   672→    </Card>
   673→  )
   674→}
