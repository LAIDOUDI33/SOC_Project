   362→   362→
   363→   363→/**
   364→   364→ * Security Event Audit Logger
   365→   365→ * 
   366→   366→ * Provides comprehensive audit logging with:
   367→   367→ * - Structured event capture
   368→   368→ * - Integrity verification via hash chains
   369→   369→ * - PII masking for privacy
   370→   370→ * - Multiple output backends
   371→   371→ * - SIEM forwarding
   372→   372→ * 
   373→   373→ * @example
   374→   374→ * ```typescript
   375→   375→ * const auditLogger = new AuditLogger(DEFAULT_AUDIT_CONFIG);
   376→   376→ * 
   377→   377→ * // Log authentication failure
   378→   378→ * await auditLogger.log({
   379→   379→ *   eventType: 'AUTH_FAILURE',
   380→   380→ *   sourceIp: clientIp,
   381→   381→ *   resource: '/api/auth/login',
   382→   382→ *   action: 'LOGIN_ATTEMPT',
   383→   383→ *   outcome: 'FAILURE',
   384→   384→ *   message: 'Invalid credentials for user john.doe',
   385→   385→ *   userId: 'user-123',
   386→   386→ *   username: 'john.doe',
   387→   387→ *   metadata: { reason: 'invalid_password' },
   388→   388→ * });
   389→   389→ * ```
   390→   390→ */
   391→   391→export class AuditLogger {
   392→   392→  private config: AuditLoggerConfig;
   393→   393→  private previousEventHash: string = '';
   394→   394→  private eventBuffer: AuditEvent[] = [];
   395→   395→  private flushTimer: ReturnType<typeof setInterval> | null = null;
   396→   396→
   397→   397→  constructor(config: Partial<AuditLoggerConfig> = {}) {
   398→   398→    this.config = { ...DEFAULT_AUDIT_CONFIG, ...config };
   399→   399→    
   400→   400→    // Start auto-flush timer if SIEM forwarding enabled
   401→   401→    if (this.config.siemForwarding?.enabled) {
   402→   402→      this.flushTimer = setInterval(
   403→   403→        () => this.flush(),
   404→   404→        this.config.siemForwarding.flushIntervalMs
   405→   405→      );
   406→   406→    }
   407→   407→  }
   408→   408→
   409→   409→  /**
   410→   410→   * Main logging method - creates and persists an audit event
   411→   411→   */
   412→   412→  async log(eventInput: Omit<AuditEvent, keyof AuditEventBase>): Promise<AuditEvent> {
   413→   413→    const event = await this.createEvent(eventInput);
   414→   414→    
   415→   415→    // Write to configured outputs
   416→   416→    await this.writeEvent(event);
   417→   417→    
   418→   418→    // Buffer for batch forwarding
   419→   419→    if (this.config.siemForwarding?.enabled) {
   420→   420→      this.eventBuffer.push(event);
   421→   421→      if (this.eventBuffer.length >= this.config.siemForwarding.batchSize) {
   422→   422→        await this.flush();
   423→   423→      }
   424→   424→    }
   425→   425→    
   426→   426→    return event;
   427→   427→  }
   428→   428→
   429→   429→  /**
   430→   430→   * Convenience methods for common event types
   431→   431→   */
   432→   432→
   433→   433→  /** Log successful authentication */
   434→   434→  async logAuthSuccess(params: {
   435→   435→    userId: string;
   436→   436→    username: string;

   438→   438→    sourceIp: string;

   440→   440→    sessionId: string;
   441→   441→    authMethod: 'password' | 'sso' | 'mfa' | 'token';
   442→   442→    metadata?: Record<string, unknown>;
   443→   443→  }): Promise<AuditEvent> {
   444→   444→    return this.log({
   445→   445→      eventType: 'AUTH_SUCCESS',
   446→   446→      userId: params.userId,
   447→   447→      username: params.username,
   448→   448→      userRole: params.userRole,
   449→   449→      sourceIp: params.sourceIp,
   450→   450→      userAgent: params.userAgent,
   451→   451→      sessionId: params.sessionId,
   452→   452→      resource: '/api/auth/login',
   453→   453→      action: `AUTH_${params.authMethod.toUpperCase()}`,
   454→   454→      outcome: 'SUCCESS',
   455→   455→      message: `Successful authentication for user ${params.username} via ${params.authMethod}`,
   456→   456→      metadata: { ...params.metadata, authMethod: params.authMethod },
   457→   457→      dataClassification: 'CONFIDENTIAL',
   458→   458→      requiresReview: false,
   459→   459→    });
   460→   460→  }
   461→   461→
   462→   462→  /** Log failed authentication attempt */
   463→   463→  async logAuthFailure(params: {
   464→   464→    userId?: string;
   465→   465→    username?: string;
   466→   466→    sourceIp: string;
   467→   467→    userAgent?: string;
   468→   468→    resource: string;
   469→   469→    reason: 'invalid_credentials' | 'account_locked' | 'mfa_failed' | 'session_expired' | 'other';
   470→   470→    metadata?: Record<string, unknown>;
   471→   471→  }): Promise<AuditEvent> {
   472→   472→    return this.log({
   473→   473→      eventType: 'AUTH_FAILURE',
   474→   474→      userId: params.userId,

   616→   616→  /**
   617→   617→   * Flush buffered events to SIEM
   618→   618→   */
   619→   619→  async flush(): Promise<void> {
   620→   620→    if (!this.config.siemForwarding?.enabled || this.eventBuffer.length === 0) {
   621→   621→      return;
   622→   622→    }
   623→   623→
   624→   624→    const eventsToSend = [...this.eventBuffer];
   625→   625→    this.eventBuffer = [];
   626→   626→
   627→   627→    try {
   628→   628→      const response = await fetch(this.config.siemForwarding.endpoint, {
   629→   629→        method: 'POST',
   630→   630→        headers: {
   631→   631→          'Content-Type': 'application/json',
   632→   632→          'X-Audit-Source': this.config.applicationId,
   633→   633→          'X-Event-Count': eventsToSend.length.toString(),
   634→   634→        },
   635→   635→        body: JSON.stringify({
   636→   636→          source: this.config.applicationId,
   637→   637→          environment: this.config.environment,
   638→   638→          timestamp: new Date().toISOString(),
   639→   639→          eventCount: eventsToSend.length,
   640→   640→          events: eventsToSend.map(this.sanitizeForSIEM),
   641→   641→        }),
   642→   642→      });
   643→   643→
   644→   644→      if (!response.ok) {
   645→   645→        console.error(`Failed to send audit events to SIEM: ${response.status}`);
   646→   646→        // Re-add failed events to buffer
   647→   647→        this.eventBuffer.unshift(...eventsToSend);
   648→   648→      }
   649→   649→    } catch (error) {
   650→   650→      console.error('Error flushing audit events to SIEM:', error);
   651→   651→      this.eventBuffer.unshift(...eventsToSend);
   652→   652→    }
   653→   653→  }

   669→   669→  /**
   670→   670→   * Create a complete audit event from input
   671→   671→   */
   672→   672→  private async createEvent(eventInput: Omit<AuditEvent, keyof AuditEventBase>): Promise<AuditEvent> {
   673→   673→    const now = new Date();
   674→   674→    const eventType = eventInput.eventType;
   675→   675→    
   676→   676→    // Determine retention period based on data classification
   677→   677→    const classification = eventInput.dataClassification || 'INTERNAL';
   678→   678→    const retentionDays = this.config.retentionPolicies[classification];
   679→   679→
   680→   680→    // Calculate integrity hash
   681→   681→    const eventData = {
   682→   682→      timestamp: now.toISOString(),
   683→   683→      eventType,
   684→   684→      ...eventInput,
   685→   685→    };
   686→   686→    
   687→   687→    const integrityHash = this.calculateIntegrityHash(eventData);
   688→   688→
   689→   689→    const event: AuditEvent = {
   690→   690→      eventId: randomUUID(),
   691→   691→      timestamp: this.formatTimestamp(now),
   692→   692→      eventType,
   693→   693→      severity: eventInput.severity || EVENT_SEVERITY_MAP[eventType],
   694→   694→      category: EVENT_CATEGORY_MAP[eventType],
   695→   695→      sourceIp: this.maskPII('ip', eventInput.sourceIp),
   696→   696→      riskScore: EVENT_RISK_SCORE_MAP[eventType],
   697→   697→      retentionDays,
   698→   698→      requiresReview: eventInput.requiresReview ?? (EVENT_RISK_SCORE_MAP[eventType] >= 50),
   699→   699→      integrityHash,
   700→   700→      previousEventHash: this.previousEventHash || undefined,
   701→   701→      ...eventInput,
   702→   702→    };
   703→   703→
   704→   704→    // Update hash chain
   705→   705→    this.previousEventHash = integrityHash;
   706→   706→
   707→   707→    return event;
   708→   708→  }

   783→   783→   * Calculate HMAC-SHA256 integrity hash for event
   784→   784→   */
   785→   785→  private calculateIntegrityHash(eventData: Record<string, unknown>): string {
   786→   786→    const eventString = JSON.stringify({
   787→   787→      ...eventData,
   788→   788→      previousHash: this.previousEventHash,
   789→   789→    });
   790→   790→

   792→   792→      .update(eventString)
   793→   793→      .digest('hex');
   794→   794→  }

   796→   796→  /**
   797→   797→   * Mask PII in event data
   798→   798→   */
   799→   799→  private maskPII(type: 'ip' | 'email' | 'imsi' | 'msisdn', value: string): string {
   800→   800→    if (!this.config.piiMasking.enabled) {
   801→   801→      return value;
   802→   802→    }
   803→   803→
   804→   804→    switch (type) {
   805→   805→      case 'ip':
   806→   806→        if (this.config.piiMasking.maskIP) {
   807→   807→          // Mask last octet
   808→   808→          return value.replace(/\.\d+$/, '.***');
   809→   809→        }
   810→   810→        return value;
   811→   811→      case 'email':
   812→   812→        if (this.config.piiMasking.maskEmail) {
   813→   813→          // Mask local part
   814→   814→          const [local, domain] = value.split('@');
   815→   815→          if (local && domain) {
   816→   816→            return `${local[0]}***@${domain}`;
   817→   817→          }
   818→   818→        }
   819→   819→        return value;
   820→   820→      case 'imsi':
   821→   821→      case 'msisdn':
   822→   822→        if (this.config.piiMasking.maskIMSIMSISDN) {
   823→   823→          return `***${value.slice(-4)}`;
   824→   824→        }
   825→   825→        return value;
   826→   826→      default:
   827→   827→        return value;
   828→   828→    }
   829→   829→  }
