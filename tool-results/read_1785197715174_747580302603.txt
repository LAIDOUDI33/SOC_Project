    97→
    98→class ProbeConnectionManager {
    99→  private connections: Map<string, ProbeConnection> = new Map();
   100→  private messageBuffer: ParsedMessage[] = [];
   101→  private bufferFlushInterval: NodeJS.Timer;
   102→  private isProcessing = false;
   103→
   104→  constructor() {
   105→    // Flush buffer every 5 seconds or when it reaches 1000 messages
   106→    this.bufferFlushInterval = setInterval(() => this.flushMessageBuffer(), 5000);
   107→  }
   108→
   109→  /**
   110→   * Connect to a probe and start receiving messages
   111→   */
   112→  async connect(config: ProbeConfig): Promise<void> {
   113→    if (!config.enabled) {
   114→      console.log(`Probe ${config.id} (${config.type}) is disabled`);
   115→      return;
   116→    }
   117→
   118→    console.log(`Connecting to ${config.type} probe: ${config.host}:${config.port}`);
   119→
   120→    let connection: ProbeConnection;
   121→
   122→    switch (config.type) {
   123→      case 'ss7':
   124→        connection = new SS7ProbeConnection(config);
   125→        break;
   126→      case 'gtp':
   127→        connection = new GTPProbeConnection(config);
   128→        break;
   129→      case 'sip':
   130→        connection = new SIPProbeConnection(config);
   131→        break;
   132→      case 'diameter':
   133→        connection = new DiameterProbeConnection(config);
   134→        break;
   135→      default:
   136→        throw new Error(`Unknown probe type: ${config.type}`);
   137→    }
   138→
   139→    await connection.connect();
   140→    
   141→    // Set up message handler
   142→    connection.onMessage(async (message) => {
   143→      await this.handleMessage(message);
   144→    });
   145→
   146→    this.connections.set(config.id, connection);
   147→    console.log(`Connected to probe: ${config.name}`);
   148→  }
   149→
   150→  /**
   151→   * Disconnect from a probe
   152→   */
   153→  async disconnect(probeId: string): Promise<void> {
   154→    const connection = this.connections.get(probeId);
   155→    if (connection) {
   156→      await connection.disconnect();
   157→      this.connections.delete(probeId);
   158→      console.log(`Disconnected from probe: ${probeId}`);
   159→    }
   160→  }
   161→
   162→  /**
   163→   * Handle incoming message from any probe
   164→   */
   165→  private async handleMessage(message: ParsedMessage): Promise<void> {
   166→    try {
   167→      // Parse and enrich the message
   168→      const enrichedMessage = await this.enrichMessage(message);
   169→      
   170→      // Check for fraud indicators
   171→      const fraudIndicators = await this.detectFraud(enrichedMessage);
   172→      
   173→      if (fraudIndicators.length > 0) {
   174→        enrichedMessage.alerts = fraudIndicators.map(f => f.type);
   175→        
   176→        // Create alerts for high-severity fraud
   177→        for (const indicator of fraudIndicators) {
   178→          if (indicator.severity === 'high' || indicator.severity === 'critical') {
   179→            await this.createFraudAlert(indicator, enrichedMessage);
   180→          }
   181→        }
   182→      }
   183→
   184→      // Add to buffer for batch processing
   185→      this.messageBuffer.push(enrichedMessage);
   186→
   187→      // Flush if buffer is full
   188→      if (this.messageBuffer.length >= 1000) {
   189→        await this.flushMessageBuffer();
   190→      }
   191→
   192→    } catch (error) {
   193→      console.error('Error handling message:', error);
   194→    }
   195→  }
   196→
   197→  /**
   198→   * Enrich message with additional context
   199→   */
   200→  private async enrichMessage(message: ParsedMessage): Promise<ParsedMessage> {
   201→    const { parsedData } = message;
   202→
   203→    // Look up subscriber info if IMSI/MSISDN present
   204→    if (parsedData.imsi || parsedData.msisdn) {
   205→      const subscriber = await db.subscriber.findFirst({
   206→        where: {
   207→          OR: [
   208→            { imsi: parsedData.imsi },
   209→            { msisdn: parsedData.msisdn }
   210→          ]
   211→        }
   212→      });
   213→
   214→      if (subscriber) {
   215→        parsedData.subscriberInfo = {
   216→          riskScore: subscriber.riskScore,
   217→          status: subscriber.subscriberStatus,
   218→          isRoaming: subscriber.isRoaming,
   219→          homeRegion: subscriber.region,
   220→        };
   221→
   222→        // Update risk score based on this activity
   223→        await this.updateSubscriberRisk(subscriber.id, message);
   224→      }
   225→    }
   226→
   227→    return message;
   228→  }
   229→
   230→  /**
   231→   * Detect fraud patterns in message
   232→   */
   233→  private async detectFraud(message: ParsedMessage): Promise<FraudIndicator[]> {
   234→    const indicators: FraudIndicator[] = [];
   235→    const { parsedData, probeId } = message;
   236→
   237→    switch (probeId.split('-')[0]) {
   238→      case 'ss7':
   239→        indicators.push(...await this.detectSS7Fraud(parsedData));
   240→        break;
   241→      case 'gtp':
   242→        indicators.push(...await this.detectGTPFraud(parsedData));
   243→        break;
   244→      case 'sip':
   245→        indicators.push(...await this.detectSIPFraud(parsedData));
   246→        break;
   247→      case 'diameter':
   248→        indicators.push(...await this.detectDiameterFraud(parsedData));
   249→        break;
   250→    }
   251→
   252→    return indicators;
   253→  }

   255→  /**
   256→   * SS7-specific fraud detection
   257→   */
   258→  private async detectSS7Fraud(data: Record<string, any>): Promise<FraudIndicator[]> {
   259→    const indicators: FraudIndicator[] = [];
   260→
   261→    // SIM Swap Detection
   262→    if (data.operationCode === 'UpdateLocation' && data.imsi) {
   263→      const recentUpdates = await this.countRecentOperations(data.imsi, 'UpdateLocation', 3600000); // 1 hour
   264→      if (recentUpdates > 3) {
   265→        indicators.push({
   266→          id: `sim-swap-${Date.now()}`,
   267→          type: 'SIM_SWAP_SUSPECTED',
   268→          severity: 'critical',
   269→          confidence: Math.min(recentUpdates * 20, 100),
   270→          description: `Multiple location updates for IMSI ${data.imsi} in last hour`,
   271→          evidence: { imsi: data.imsi, updateCount: recentUpdates },
   272→          recommendedAction: 'BLOCK_SIM_AND_INVESTIGATE',
   273→          artpReportable: true
   274→        });
   275→      }
   276→    }
   277→
   278→    // IRSF (International Revenue Share Fraud) Detection
   279→    if (data.calledParty?.startsWith('+') && !data.calledParty?.startsWith('+213')) {
   280→      // International call from potentially compromised subscriber
   281→      const subscriber = await this.getSubscriberByIMSI(data.imsi);
   282→      if (subscriber && subscriber.riskScore > 60) {
   283→        indicators.push({
   284→          id: `irsf-${Date.now()}`,
   285→          type: 'IRSF_SUSPECTED',
   286→          severity: 'high',
   287→          confidence: subscriber.riskScore,
   288→          description: `High-risk subscriber making international calls`,
   289→          evidence: {
   290→            imsi: data.imsi,
   291→            calledParty: data.calledParty,
   292→            callerRiskScore: subscriber.riskScore
   293→          },
   294→          recommendedAction: 'MONITOR_AND_LIMIT',
   295→          artpReportable: true
   296→        });
   297→      }
   298→    }
   299→
   300→    // Wangiri Detection (short repeated calls)
   301→    if (data.callDuration < 3 && data.calledParty) {
   302→      const recentShortCalls = await this.countShortCalls(data.callingParty, data.calledParty, 3600000);
   303→      if (recentShortCalls > 20) {
   304→        indicators.push({
   305→          id: `wangiri-${Date.now()}`,
   306→          type: 'WANGIRI_PATTERN',
   307→          severity: 'medium',
   308→          confidence: Math.min(recentShortCalls * 4, 100),
   309→          description: `Wangiri pattern detected: ${recentShortCalls} short calls to same number`,
   310→          evidence: {
   311→            callingParty: data.callingParty,
   312→            calledParty: data.calledParty,
   313→            shortCallCount: recentShortCalls
   314→          },
   315→          recommendedAction: 'INVESTIGATE_AND_BLOCK',
   316→          artpReportable: false
   317→        });
   318→      }
   319→    }
   320→
   321→    return indicators;
   322→  }

   324→  /**
   325→   * GTP-specific fraud detection
   326→   */
   327→  private async detectGTPFraud(data: Record<string, any>): Promise<FraudIndicator[]> {
   328→    const indicators: FraudIndicator[] = [];
   329→
   330→    // Data Tunneling Detection (unusual APN usage)
   331→    if (data.apn && !data.apn.includes('djezzy') && !data.apn.includes('internet')) {
   332→      indicators.push({
   333→        id: `tunnel-${Date.now()}`,
   334→        type: 'DATA_TUNNELING_SUSPECTED',
   335→        severity: 'high',
   336→        confidence: 75,
   337→        description: `Suspicious APN usage: ${data.apn}`,
   338→        evidence: {
   339→          imsi: data.imsi,
   340→          apn: data.apn,
   341→          ipAddress: data.ipAddress
   342→        },
   343→        recommendedAction: 'INVESTIGATE_APN_ACCESS',
   344→        artpReportable: false
   345→      });
   346→    }
   347→
   348→    // Roaming Abuse Detection
   349→    if (data.isRoaming && data.bytesUp > 1073741824) { // > 1GB while roaming
   350→      indicators.push({
   351→        id: `roaming-abuse-${Date.now()}`,
   352→        type: 'ROAMING_DATA_ABUSE',
   353→        severity: 'medium',
   354→        confidence: 80,
   355→        description: `High data volume while roaming: ${(data.bytesUp / 1073741824).toFixed(2)} GB`,
   356→        evidence: {
   357→          imsi: data.imsi,
   358→          bytesUp: data.bytesUp,
   359→          visitedNetwork: data.visitedPLMN
   360→        },
   361→        recommendedAction: 'NOTIFY_SUBSCRIBER_AND_MONITOR',
   362→        artpReportable: false
   363→      });
   364→    }
   365→
   366→    // IMEI Cloning Detection
   367→    if (data.imei) {
   368→      const sessionsWithIMEI = await this.countActiveSessionsByIMEI(data.imei);
   369→      if (sessionsWithIMEI > 1) {
   370→        indicators.push({
   371→          id: `imei-clone-${Date.now()}`,
   372→          type: 'IMEI_CLONING_DETECTED',
   373→          severity: 'critical',
   374→          confidence: 95,
   375→          description: `Same IMEI active on multiple sessions: ${data.imei}`,
   376→          evidence: {
   377→            imei: data.imei,
   378→            sessionCount: sessionsWithIMEI,
   379→            imsi: data.imsi
   380→          },
   381→          recommendedAction: 'IMMEDIATE_BLOCK_AND_INVESTIGATE',
   382→          artpReportable: true
   383→        });
   384→      }
   385→    }
   386→
   387→    return indicators;
   388→  }

   390→  /**
   391→   * SIP-specific fraud detection
   392→   */
   393→  private async detectSIPFraud(data: Record<string, any>): Promise<FraudIndicator[]> {
   394→    const indicators: FraudIndicator[] = [];
   395→
   396→    // PBX Hijacking Detection
   397→    if (data.userAgent?.toLowerCase().includes('asterisk') || 
   398→        data.userAgent?.toLowerCase().includes('freepbx')) {
   399→      // Check if this is expected behavior for the subscriber
   400→      const subscriber = await this.getSubscriberByMSISDN(data.callingParty);
   401→      if (subscriber && !subscriber.metadata?.pbxAllowed) {
   402→        indicators.push({
   403→          id: `pbx-hijack-${Date.now()}`,
   404→          type: 'PBX_HIJACKING_SUSPECTED',
   405→          severity: 'high',
   406→          confidence: 70,
   407→          description: `PBX software detected from unexpected source`,
   408→          evidence: {
   409→            callingParty: data.callingParty,
   410→            userAgent: data.userAgent,
   411→            subscriberType: subscriber.profileData?.type
   412→          },
   413→          recommendedAction: 'VERIFY_WITH_SUBSCRIBER',
   414→          artpReportable: false
   415→        });
   416→      }
   417→    }
   418→
   419→    // Call Duration Anomaly (very long calls - potential premium rate abuse)
   420→    if (data.duration > 7200) { // > 2 hours
   421→      indicators.push({
   422→        id: `long-call-${Date.now()}`,
   423→        type: 'PREMIUM_RATE_ABUSE_SUSPECTED',
   424→        severity: 'medium',
   425→        confidence: 60,
   426→        description: `Unusually long call duration: ${Math.floor(data.duration / 3600)}h ${Math.floor((data.duration % 3600) / 60)}m`,
   427→        evidence: {
   428→          callingParty: data.callingParty,
   429→          calledParty: data.calledParty,
   430→          duration: data.duration
   431→        },
   432→        recommendedAction: 'REVIEW_CALL_RECORDING',
   433→        artpReportable: false
   434→      });
   435→    }
   436→
   437→    // Call Spam Detection (high call volume in short time)
   438→    const recentCalls = await this.countRecentCallsFromNumber(data.callingParty, 300000); // 5 minutes
   439→    if (recentCalls > 50) {
   440→      indicators.push({
   441→        id: `call-spam-${Date.now()}`,
   442→        type: 'CALL_SPAM_DETECTED',
   443→        severity: 'high',
   444→        confidence: Math.min(recentCalls, 100),
   445→        description: `Abnormally high call volume: ${recentCalls} calls in 5 minutes`,
   446→        evidence: {
   447→          callingParty: data.callingParty,
   448→          callCount: recentCalls
   449→        },
   450→        recommendedAction: 'RATE_LIMIT_OR_BLOCK',
   451→        artpReportable: true
   452→      });
   453→    }
   454→
   455→    return indicators;
   456→  }

   458→  /**
   459→   * Diameter-specific fraud detection
   460→   */
   461→  private async detectDiameterFraud(data: Record<string, any>): Promise<FraudIndicator[]> {
   462→    const indicators: FraudIndicator[] = [];
   463→
   464→    // Authentication Storm Detection
   465→    if (data.commandCode === 'Authentication-Information') {
   466→      const authAttempts = await this.countRecentAuthAttempts(data.imsi, 60000); // 1 minute
   467→      if (authAttempts > 10) {
   468→        indicators.push({
   469→          id: `auth-storm-${Date.now()}`,
   470→          type: 'AUTHENTICATION_STORM',
   471→          severity: 'critical',
   472→          confidence: Math.min(authAttempts * 8, 100),
   473→          description: `Excessive authentication attempts for IMSI: ${authAttempts} in 1 minute`,
   474→          evidence: {
   475→            imsi: data.imsi,
   476→            attemptCount: authAttempts
   477→          },
   478→          recommendedAction: 'BLOCK_IMSI_TEMPORARILY',
   479→          artpReportable: true
   480→        });
   481→      }
   482→    }
   483→
   484→    // Location Update Storm (possible tracking avoidance)
   485→    if (data.commandCode === 'Update-Location') {
   486→      const locationUpdates = await this.countLocationUpdates(data.imsi, 600000); // 10 minutes
   487→      if (locationUpdates > 20) {
   488→        indicators.push({
   489→          id: `location-storm-${Date.now()}`,
   490→          type: 'LOCATION_UPDATE_ABUSE',
   491→          severity: 'medium',
   492→          confidence: Math.min(locationUpdates * 4, 100),
   493→          description: `Excessive location updates: ${locationUpdates} in 10 minutes`,
   494→          evidence: {
   495→            imsi: data.imsi,
   496→            updateCount: locationUpdates
   497→          },
   498→          recommendedAction: 'MONITOR_FOR_TRACKING_AVOIDANCE',
   499→          artpReportable: false
   500→        });
   501→      }
   502→    }
   503→
   504→    return indicators;
   505→  }

   507→  /**
   508→   * Create security alert from fraud indicator
   509→   */
   510→  private async createFraudAlert(indicator: FraudIndicator, message: ParsedMessage): Promise<void> {
   511→    try {
   512→      await db.alert.create({
   513→        data: {
   514→          title: `[FRAUD] ${indicator.type.replace(/_/g, ' ')}`,
   515→          description: indicator.description,
   516→          severity: indicator.severity.toUpperCase() as any,
   517→          status: 'NEW',
   518→          source: `TELECOM_${message.probeId.split('-')[0].toUpperCase()}_PROBE`,
   519→          category: 'FRAUD_DETECTION',
   520→          rawEvent: {
   521→            ...indicator.evidence,
   522→            messageId: message.id,
   523→            probeId: message.probeId,
   524→            timestamp: message.timestamp,
   525→            confidence: indicator.confidence
   526→          },
   527→          mitreTechniques: ['T1589'], // Steal or Forge Authentication Information
   528→          tags: ['fraud', 'telecom', indicator.type.toLowerCase(), indicator.severity],
   529→          context: {
   530→            recommendedAction: indicator.recommendedAction,
   531→            artpReportable: indicator.artpReportable,
   532→            confidence: indicator.confidence
   533→          }
   534→        }
   535→      });
   536→
   537→      // If ARTP reportable, create incident
   538→      if (indicator.artpReportable) {
   539→        await this.createARTPIncident(indicator, message);
   540→      }
   541→
   542→    } catch (error) {
   543→      console.error('Failed to create fraud alert:', error);
   544→    }
   545→  }

   547→  /**
   548→   * Create ARTP-reportable incident
   549→   */
   550→  private async createARTPIncident(indicator: FraudIndicator, message: ParsedMessage): Promise<void> {
   551→    // Check if similar incident already exists

   559→    if (!existingIncident) {
   560→      await db.incident.create({
   561→        data: {
   562→          tatcCode: `TATC-${Date.now().toString(36).toUpperCase()}`,
   563→          title: `[ARTP] ${indicator.type.replace(/_/g, ' ')}`,
   564→          description: indicator.description,
   565→          severity: indicator.severity.toUpperCase() as any,
   566→          phase: 'DETECTION',
   567→          source: 'AUTOMATED_FRAUD_DETECTION',
   568→          impactAssessment: `Potential financial loss and regulatory non-compliance. Confidence: ${indicator.confidence}%`,
   569→          affectedAssets: [message.probeId],
   570→          artifacts: {
   571→            fraudIndicator: indicator,
   572→            sourceMessage: message.parsedData,
   573→            evidence: indicator.evidence
   574→          },
   575→          tags: ['artp', 'fraud', 'telecom', 'regulatory']
   576→        }
   577→      });
   578→    }
   579→  }

   581→  /**
   582→   * Flush message buffer to database
   583→   */
   584→  private async flushMessageBuffer(): Promise<void> {
   585→    if (this.isProcessing || this.messageBuffer.length === 0) return;
   586→
   587→    this.isProcessing = true;
   588→    const messagesToProcess = [...this.messageBuffer];
   589→    this.messageBuffer = [];
   590→
   591→    try {
   592→      // Batch insert into appropriate tables based on probe type
   593→      for (const message of messagesToProcess) {
   594→        const probeType = message.probeId.split('-')[0];
   595→
   596→        switch (probeType) {
   597→          case 'ss7':
   598→            await this.insertSS7Message(message);
   599→            break;
   600→          case 'gtp':
   601→            await this.insertGTPSession(message);
   602→            break;
   603→          case 'sip':
   604→            await this.insertSIPSession(message);
   605→            break;
   606→          case 'diameter':
   607→            await this.insertDiameterSession(message);
   608→            break;
   609→        }
   610→      }
   611→
   612→      console.log(`Processed ${messagesToProcess.length} messages`);
   613→
   614→    } catch (error) {
   615→      console.error('Error flushing message buffer:', error);
   616→      // Re-add failed messages back to buffer
   617→      this.messageBuffer.unshift(...messagesToProcess);
   618→    } finally {
   619→      this.isProcessing = false;
   620→    }
   621→  }
   622→
   623→  /**

   933→// ============================================================
   934→// Base Probe Connection Class
   935→// ============================================================
   936→
   937→abstract class ProbeConnection {
   938→  config: ProbeConfig;
   939→  isConnected = false;
   940→  messagesProcessed = 0;
   941→  private messageHandlers: ((message: ParsedMessage) => void)[] = [];
   942→
   943→  constructor(config: ProbeConfig) {
   944→    this.config = config;
   945→  }
   946→
   947→  abstract connect(): Promise<void>;
   948→  abstract disconnect(): Promise<void>;
   949→
   950→  onMessage(handler: (message: ParsedMessage) => void): void {
   951→    this.messageHandlers.push(handler);
   952→  }
   953→
   954→  protected emitMessage(message: ParsedMessage): void {
   955→    this.messagesProcessed++;
   956→    this.messageHandlers.forEach(handler => handler(message));
   957→  }
   958→}
