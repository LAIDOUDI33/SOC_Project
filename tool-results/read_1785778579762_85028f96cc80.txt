   107→@dataclass
   108→class DetectionRule:
   109→    """A single detection rule definition."""
   110→    name: str
   111→    description: str
   112→    severity: AlertSeverity
   113→    condition: Dict[str, Any]
   114→    action: List[str]
   115→    tags: List[str] = field(default_factory=list)
   116→    mitre_technique: Optional[str] = None
   117→    mitre_tactic: Optional[str] = None
   118→    enabled: bool = True
   119→    
   120→    def matches(self, message: Dict) -> bool:
   121→        """Check if a message matches this rule's conditions."""
   122→        return self._evaluate_condition(self.condition, message)
   123→    
   124→    def _evaluate_condition(self, condition: Dict, message: Dict) -> bool:
   125→        """Recursively evaluate condition tree."""
   126→        if not condition:
   127→            return True
   128→        
   129→        op = condition.get("op", "and")
   130→        
   131→        if op == "and":
   132→            return all(
   133→                self._evaluate_condition(sub_cond, message)
   134→                for sub_cond in condition.get("conditions", [])
   135→            )
   136→        elif op == "or":
   137→            return any(
   138→                self._evaluate_condition(sub_cond, message)
   139→                for sub_cond in condition.get("conditions", [])
   140→            )
   141→        elif op == "not":
   142→            return not self._evaluate_condition(condition.get("condition", {}), message)
   143→        else:
   144→            # Leaf condition - field comparison
   145→            field_path = condition.get("field")
   146→            operator = condition.get("operator", "equals")
   147→            value = condition.get("value")
   148→            
   149→            actual_value = self._get_field_value(message, field_path)
   150→            
   151→            if actual_value is None:
   152→                return False
   153→            
   154→            return self._compare(actual_value, value, operator)
   155→    
   156→    def _get_field_value(self, obj: Dict, path: str) -> Any:
   157→        """Get nested field value using dot notation."""
   158→        keys = path.split(".")
   159→        current = obj
   160→        for key in keys:
   161→            if isinstance(current, dict):
   162→                current = current.get(key)
   163→                if current is None:
   164→                    return None
   165→            else:
   166→                return None
   167→        return current
   168→    
   169→    def _compare(self, actual: Any, expected: Any, operator: str) -> bool:
   170→        """Compare values using specified operator."""
   171→        try:
   172→            if operator == "equals":
   173→                return actual == expected
   174→            elif operator == "not_equals":
   175→                return actual != expected
   176→            elif operator == "contains":
   177→                return expected in str(actual)
   178→            elif operator == "matches":
   179→                return bool(re.match(expected, str(actual)))
   180→            elif operator == "greater_than":
   181→                return float(actual) > float(expected)
   182→            elif operator == "less_than":
   183→                return float(actual) < float(expected)
   184→            elif operator == "in":
   185→                return actual in expected if isinstance(expected, list) else False
   186→            elif operator == "exists":
   187→                return actual is not None
   188→            else:
   189→                logger.warning("unknown_operator", operator=operator)
   190→                return False
   191→        except (ValueError, TypeError) as e:
   192→            logger.debug("comparison_error", error=str(e))
   193→            return False

   229→# ============== State Trackers ==============

   231→class SlidingWindowCounter:
   232→    """Thread-safe sliding window counter for rate-based detection."""
   233→    
   234→    def __init__(self, window_size_seconds: int = 60, max_windows: int = 10):
   235→        self.window_size = window_size_seconds
   236→        self.max_windows = max_windows
   237→        self.windows: Dict[str, List[float]] = defaultdict(list)
   238→        self.lock = asyncio.Lock()
   239→    
   240→    async def increment(self, key: str, timestamp: float = None):
   241→        """Increment counter for key at timestamp."""
   242→        async with self.lock:
   243→            ts = timestamp or time.time()
   244→            self.windows[key].append(ts)
   245→            await self._cleanup(key, ts)
   246→    
   247→    async def get_count(self, key: str, since: float = None) -> int:
   248→        """Get count of events in current window."""
   249→        async with self.lock:
   250→            ts = since or (time.time() - self.window_size)
   251→            return sum(1 for t in self.windows[key] if t >= ts)
   252→    
   253→    async def _cleanup(self, key: str, current_time: float):
   254→        """Remove old entries outside window."""
   255→        cutoff = current_time - (self.window_size * self.max_windows)
   256→        self.windows[key] = [t for t in self.windows[key] if t >= cutoff]

   259→# ============== Rules Engine ==============
   260→
   261→class RulesEngine:
   262→    """Load and manage detection rules from YAML files."""
   263→    
   264→    def __init__(self, rules_dir: Path):
   265→        self.rules_dir = rules_dir
   266→        self.rules: Dict[str, DetectionRule] = {}
   267→        self.last_load_time: float = 0
   268→        self.load_rules()
   269→    
   270→    def load_rules(self):
   271→        """Load all YAML rule files from rules directory."""
   272→        loaded_count = 0
   273→        error_count = 0
   274→        
   275→        if not self.rules_dir.exists():
   276→            logger.warning("rules_directory_not_found", path=str(self.rules_dir))
   277→            return
   278→        
   279→        for rule_file in self.rules_dir.glob("**/*.yaml"):
   280→            try:
   281→                with open(rule_file) as f:
   282→                    rule_data = yaml.safe_load(f)
   283→                
   284→                if isinstance(rule_data, list):
   285→                    for rd in rule_data:
   286→                        rule = self._parse_rule(rd)
   287→                        if rule:
   288→                            self.rules[rule.name] = rule
   289→                            loaded_count += 1
   290→                elif isinstance(rule_data, dict):
   291→                    rule = self._parse_rule(rule_data)
   292→                    if rule:
   293→                        self.rules[rule.name] = rule
   294→                        loaded_count += 1
   295→                        
   296→            except Exception as e:
   297→                error_count += 1
   298→                logger.error("rule_load_error", file=str(rule_file), error=str(e))
   299→        
   300→        self.last_load_time = time.time()
   301→        logger.info("rules_loaded", total=len(self.rules), errors=error_count)
   302→    
   303→    def _parse_rule(self, data: Dict) -> Optional[DetectionRule]:
   304→        """Parse rule dict into DetectionRule object."""
   305→        try:
   306→            required_fields = ["name", "description", "severity", "condition"]
   307→            for f in required_fields:
   308→                if f not in data:
   309→                    logger.warning("rule_missing_field", rule=data.get("name"), field=f)
   310→                    return None
   311→            
   312→            return DetectionRule(
   313→                name=data["name"],
   314→                description=data["description"],
   315→                severity=AlertSeverity(data.get("severity", "MEDIUM").upper()),
   316→                condition=data.get("condition", {}),
   317→                action=data.get("action", ["log"]),
   318→                tags=data.get("tags", []),
   319→                mitre_technique=data.get("mitre_technique"),
   320→                mitre_tactic=data.get("mitre_tactic"),
   321→                enabled=data.get("enabled", True),
   322→            )
   323→        except Exception as e:
   324→            logger.error("rule_parse_error", error=str(e))
   325→            return None
   326→    
   327→    def evaluate(self, message: Dict) -> List[DetectionRule]:
   328→        """Evaluate message against all enabled rules."""
   329→        matches = []
   330→        for rule_name, rule in self.rules.items():
   331→            if rule.enabled and rule.matches(message):
   332→                matches.append(rule)
   333→        return matches
   334→    
   335→    def reload_if_needed(self):
   336→        """Reload rules if modification time has changed."""
   337→        if time.time() - self.last_load_time > settings.rules_reload_interval_seconds:
   338→            self.load_rules()

   341→# ============== TheHive Integration ==============
   342→
   343→class TheHiveClient:
   344→    """Client for creating cases in TheHive platform."""
   345→    
   346→    def __init__(self, base_url: str, api_key: str):
   347→        self.base_url = base_url.rstrip("/")
   348→        self.api_key = api_key
   349→        self.headers = {
   350→            "Authorization": f"Bearer {api_key}",
   351→            "Content-Type": "application/json",
   352→        }
   353→    
   354→    async def create_case_from_alert(self, alert: SecurityAlert) -> Optional[str]:
   355→        """Create a new case in TheHive from security alert."""
   356→        if not settings.auto_create_cases:
   357→            return None
   358→        
   359→        try:
   360→            import httpx
   361→            
   362→            case_payload = {
   363→                "title": alert.title[:200],
   364→                "description": self._build_case_description(alert),
   365→                "severity": settings.case_severity_map.get(alert.severity.value, 2),
   366→                "tlp": 2,  # Amber
   367→                "pap": 2,  # Amber
   368→                "tags": ["ss7", "auto-created"] + [alert.rule_name] + alert.indicators,
   369→                "observables": self._build_observables(alert),
   370→            }
   371→            
   372→            async with httpx.AsyncClient(timeout=30) as client:
   373→                response = await client.post(
   374→                    f"{self.base_url}/api/case",
   375→                    headers=self.headers,
   376→                    json=case_payload,
   377→                )
   378→                
   379→                if response.status_code == 201:
   380→                    case_id = response.json().get("_id") or response.json().get("id")
   381→                    logger.info("thehive_case_created", 
   382→                               case_id=case_id, 
   383→                               rule=alert.rule_name,
   384→                               severity=alert.severity.value)
   385→                    return case_id
   386→                else:
   387→                    logger.error("thehive_create_error", 
   388→                                status=response.status_code,
   389→                                body=response.text)
   390→                    return None
   391→                    
   392→        except Exception as e:
   393→            logger.exception("thehive_client_error")
   394→            return None
   395→    
   396→    def _build_case_description(self, alert: SecurityAlert) -> str:
   397→        """Build detailed case description."""
   398→        msg = alert.source_message.get("message_type", "Unknown")
   399→        protocol = alert.source_message.get("protocol", "Unknown")
   400→        calling = alert.source_message.get("calling_party_gt", {}).get("address", "N/A")
   401→        called = alert.source_message.get("called_party_gt", {}).get("address", "N/A")
   402→        imsi = alert.source_message.get("imsi", "N/A")
   403→        
   404→        return f"""## SS7 Security Alert Details
   405→
   406→**Rule:** {alert.rule_name}
   407→**Severity:** {alert.severity.value}
   408→**Risk Score:** {alert.risk_score:.2f}
   409→
   410→### Source Message Information
   411→- **Protocol:** {protocol}
   412→- **Message Type:** {msg}
   413→- **Calling Party GT:** {calling}
   414→- **Called Party GT:** {called}
   415→- **IMSI:** {imsi}
   416→
   417→### Detection Indicators
   418→{chr(10).join(f'- {ind}' for ind in alert.indicators)}
   419→
   420→### IOCs Identified
   421→{chr(10).join(f'- **{ioc.get("type", "unknown")}**: {ioc.get("value", "N/A")}' for ioc in alert.iocs)}
   422→
   423→### Raw Message Context
   424→```json
   425→{orjson.dumps(alert.source_message, option=orjson.OPT_INDENT_2).decode()}
   426→```
   427→
   428→---
   429→*Automatically created by Djezzy SOC SS7 Analyzer*
   430→*Detection Time: {alert.timestamp.isoformat()}*
   431→"""
   432→    
   433→    def _build_observables(self, alert: SecurityAlert) -> List[Dict]:
   434→        """Build observables list for TheHive case."""
   435→        observables = []
   436→        
   437→        # Add IOCs as observables
   438→        for ioc in alert.iocs:
   439→            obs = {
   440→                "dataType": ioc.get("type", "other"),
   441→                "data": ioc.get("value", ""),
   442→                "tlp": 2,
   443→                "tags": ["ss7-ioc"],
   444→            }
   445→            observables.append(obs)
   446→        
   447→        # Add source/destination addresses
   448→        calling_gt = alert.source_message.get("calling_party_gt", {})
   449→        if calling_gt.get("address"):
   450→            observables.append({
   451→                "dataType": "phone-number",
   452→                "data": calling_gt["address"],
   453→                "tlp": 2,
   454→                "tags": ["calling-party-gt", "ss7"],
   455→            })
   456→        
   457→        called_gt = alert.source_message.get("called_party_gt", {})
   458→        if called_gt.get("address"):
   459→            observables.append({
   460→                "dataType": "phone-number",
   461→                "data": called_gt["address"],
   462→                "tlp": 2,
   463→                "tags": ["called-party-gt", "ss7"],
   464→            })
   465→        
   466→        return observables

   605→        'max.poll.interval.ms': 300000,
   606→        'fetch.max.bytes': 52428800,  # 50MB
   607→        'max.partition.fetch.bytes': 1048576,  # 1MB per partition
   608→    }
   609→    
   610→    producer_config = {
   611→        'bootstrap.servers': settings.kafka_bootstrap_servers,
   612→        'client.id': 'ss7-analyzer-producer',
   613→        'linger.ms': 10,
   614→        'compression.type': 'lz4',
   615→    }
   616→    
   617→    consumer = Consumer(consumer_config)
   618→    producer = Producer(producer_config)
   619→    
   620→    consumer.subscribe([settings.input_topic])
   621→    
   622→    logger.info("kafka_consumer_started",
   623→                input_topic=settings.input_topic,
   624→                output_topic=settings.output_topic,
   625→                group_id=settings.consumer_group)
   626→    
   627→    try:
   628→        while True:
   629→            msg = consumer.poll(timeout=1.0)
   630→            
   631→            if msg is None:
   632→                continue
   633→            
   634→            if msg.error():
   635→                if msg.error().code() == KafkaError._PARTITION_EOF:
   636→                    continue
   637→                else:
   638→                    logger.error("kafka_consumer_error", error=msg.error())
   639→                    stats["errors"] += 1
   640→                    continue
   641→            
   642→            try:
   643→                # Parse message
   644→                message = orjson.loads(msg.value())
   645→                stats["messages_processed"] += 1
   646→                
   647→                # Evaluate against rules
   648→                matching_rules = rules_engine.evaluate(message)
   649→                
   650→                if matching_rules:
   651→                    for rule in matching_rules:
   652→                        alert = create_alert(rule, message)
   653→                        
   654→                        # Send to output topic
   655→                        alert_json = orjson.dumps(alert.to_dict())
   656→                        producer.produce(
   657→                            topic=settings.output_topic,
   658→                            key=alert.alert_id.encode('utf-8'),
   659→                            value=alert_json,
   660→                        )
   661→                        
   662→                        stats["alerts_generated"] += 1
   663→                        
   664→                        # Create TheHive case for high/critical alerts
   665→                        if alert.severity in [AlertSeverity.HIGH, AlertSeverity.CRITICAL]:
   666→                            if thehive_client:
   667→                                case_id = await thehive_client.create_case_from_alert(alert)
   668→                                if case_id:
   669→                                    stats["cases_created"] += 1
   670→                                    alert.actions_taken.append(f"Created TheHive case: {case_id}")
   671→                        
   672→                        logger.info("alert_generated",
   673→                                    rule=rule.name,
   674→                                    severity=alert.severity.value,
   675→                                    alert_id=alert.alert_id)
   676→                
   677→                # Periodically check for rule reload
   678→                rules_engine.reload_if_needed()
   679→                
   680→            except orjson.JSONDecodeError as e:
   681→                stats["errors"] += 1
   682→                logger.error("json_parse_error", error=str(e))
   683→            except Exception as e:
   684→                stats["errors"] += 1
   685→                logger.exception("message_processing_error")
   686→            
   687→            # Periodic producer flush
   688→            producer.poll(0)
   689→            
   690→    except KeyboardInterrupt:
   691→        pass
   692→    finally:
   693→        logger.info("kafka_consumer_stopping")
   694→        producer.flush()
   695→        consumer.close()

   698→def create_alert(rule: DetectionRule, message: Dict) -> SecurityAlert:
   699→    """Create a SecurityAlert from a matched rule and message."""
   700→    alert_id = f"ss7_alert_{int(time.time()*1000000)}"
   701→    
   702→    # Extract indicators from message
   703→    indicators = extract_indicators(rule, message)
   704→    
   705→    # Calculate risk score
   706→    risk_score = calculate_risk_score(rule, message, indicators)
   707→    
   708→    # Build title and description
   709→    title = build_alert_title(rule, message)
   710→    description = build_alert_description(rule, message, indicators)
   711→    
   712→    # Extract IOCs
   713→    iocs = extract_iocs(message)
   714→    
   715→    return SecurityAlert(
   716→        alert_id=alert_id,
   717→        rule_name=rule.name,
   718→        severity=rule.severity,
   719→        title=title,
   720→        description=description,
   721→        source_message=message,
   722→        indicators=indicators,
   723→        risk_score=risk_score,
   724→        iocs=iocs,
   725→        actions_taken=[],
   726→    )

   729→def extract_indicators(rule: DetectionRule, message: Dict) -> List[str]:
   730→    """Extract human-readable indicators from message."""
   731→    indicators = []
   732→    
   733→    # Protocol/message type
   734→    protocol = message.get("protocol", "Unknown")
   735→    msg_type = message.get("message_type", "Unknown")
   736→    indicators.append(f"Protocol: {protocol}, Type: {msg_type}")
   737→    
   738→    # Addresses
   739→    calling = message.get("calling_party_gt", {})
   740→    if calling.get("address"):
   741→        indicators.append(f"Calling GT: {calling['address']}")
   742→    
   743→    called = message.get("called_party_gt", {})
   744→    if called.get("address"):
   745→        indicators.append(f"Called GT: {called['address']}")
   746→    
   747→    # Subscriber info
   748→    if message.get("imsi"):
   749→        indicators.append(f"IMSI present: {message['imsi'][:6]}******")
   750→    if message.get("msisdn"):
   751→        indicators.append(f"MSISDN present: {message['msisdn'][-4:]}****")
   752→    
   753→    # Location
   754→    if message.get("location_area"):
   755→        indicators.append("Location info present")
   756→    
   757→    # Rule-specific indicators
   758→    if "location_tracking" in rule.name.lower():
   759→        indicators.append("Potential location tracking pattern detected")
   760→    if "fraud" in rule.name.lower() or "irsf" in rule.name.lower():
   761→        indicators.append("Financial fraud indicator")
   762→    if "brute_force" in rule.name.lower() or "flooding" in rule.name.lower():
   763→        indicators.append("Rate anomaly detected")
   764→    
   765→    return indicators

   768→def calculate_risk_score(rule: DetectionRule, message: Dict, indicators: List[str]) -> float:
   769→    """Calculate risk score (0.0 to 1.0) for an alert."""
   770→    score = 0.0
   771→    
   772→    # Base score from severity
   773→    severity_scores = {
   774→        AlertSeverity.CRITICAL: 0.8,
   775→        AlertSeverity.HIGH: 0.6,
   776→        AlertSeverity.MEDIUM: 0.4,
   777→        AlertSeverity.LOW: 0.2,
   778→    }
   779→    score = severity_scores.get(rule.severity, 0.3)
   780→    
   781→    # Adjust based on subscriber presence
   782→    if message.get("imsi"):
   783→        score += 0.05
   784→    if message.get("imei"):
   785→        score += 0.03
   786→    
   787→    # Adjust based on sensitive operations
   788→    sensitive_ops = [
   789→        "sendRoutingInfoForSM",
   790→        "provideRoamingNumber",
   791→        "anyTimeInterrogation",
   792→        "cancelLocation",
   793→    ]
   794→    if message.get("message_type") in sensitive_ops:
   795→        score += 0.1
   796→    
   797→    # MITRE technique adjustment
   798→    if rule.mitre_technique:
   799→        score += 0.05
   800→    
   801→    # Clamp to valid range
   802→    return min(1.0, max(0.0, score))

   851→def extract_iocs(message: Dict) -> List[Dict]:
   852→    """Extract Indicators of Compromise from message."""
   853→    iocs = []
   854→    
   855→    # Phone numbers/GTs
   856→    for party_type in ["calling_party_gt", "called_party_gt"]:
   857→        gt = message.get(party_type, {})
   858→        if gt.get("address"):
   859→            iocs.append({
   860→                "type": "phone-number",
   861→                "value": gt["address"],
   862→                "source": party_type,
   863→            })
   864→    
   865→    # IMSI (partial, masked)
   866→    if message.get("imsi"):
   867→        iocs.append({
   868→            "type": "imsi",
   869→            "value": message["imsi"],
   870→            "source": "subscriber_id",
   871→        })
   872→    
   873→    # IMEI
   874→    if message.get("imei"):
   875→        iocs.append({
   876→            "type": "imei",
   877→            "value": message["imei"],
   878→            "source": "device_id",
   879→        })
   880→    
   881→    # MSISDN
   882→    if message.get("msisdn"):
   883→        iocs.append({
   884→            "type": "msisdn",
   885→            "value": message["msisdn"],
   886→            "source": "subscriber_number",
   887→        })
   888→    
   889→    return iocs
