   292→function RequirementStatusBadge({ status }: { status: ANRTRequirement['status'] }) {
   293→  const styles = {
   294→    compliant: 'bg-green-500/20 text-green-400 border-green-500/30',
   295→    partial: 'bg-yellow-500/20 text-yellow-400 border-yellow-500/30',
   296→    'non-compliant': 'bg-red-500/20 text-red-400 border-red-500/30',
   297→    'not-assessed': 'bg-slate-500/20 text-slate-400 border-slate-500/30'
   298→  }
   299→
   300→  return (
   301→    <Badge variant="outline" className={`text-xs ${styles[status]}`}>
   302→      {status.replace('-', ' ').toUpperCase()}
   303→    </Badge>
   304→  )
   305→}

   325→// ANRT Requirements Matrix Component
   326→function ANRTRequirementsMatrix({ requirements }: { requirements: ANRTRequirement[] }) {
   327→  const [filterCategory, setFilterCategory] = useState('all')

   329→  const filtered = filterCategory === 'all'

   335→  const complianceScore = Math.round(
   336→    (requirements.filter(r => r.status === 'compliant').length / requirements.length) * 100

   338→
   339→  return (
   340→    <Card className="bg-slate-900 border-slate-700">
   341→      <CardHeader className="pb-3">

   347→            </CardTitle>
   348→            <CardDescription className="text-slate-400 mt-1">
   349→              Overall Compliance Score: <span className="text-green-400 font-bold">{complianceScore}%</span>

   424→// Audit Findings Lifecycle Component
   425→function AuditFindingsLifecycle({ findings }: { findings: AuditFinding[] }) {
   426→  const getStatusColor = (status: AuditFinding['status']) => {
   427→    switch (status) {
   428→      case 'open': return 'border-red-500/30 bg-red-500/5'
   429→      case 'in-progress': return 'border-yellow-500/30 bg-yellow-500/5'
   430→      case 'remediated': return 'border-blue-500/30 bg-blue-500/5'
   431→      case 'closed': return 'border-green-500/30 bg-green-500/5'
   432→      case 'accepted-risk': return 'border-orange-500/30 bg-orange-500/5'
   433→    }
   434→  }
   435→
   436→  const getDaysUntilDeadline = (date: Date) => {
   437→    const diff = date.getTime() - Date.now()
   438→    return Math.ceil(diff / 86400000)
   439→  }
   440→
   441→  return (
   442→    <Card className="bg-slate-900 border-slate-700">
   443→      <CardHeader className="pb-3">
   444→        <div className="flex items-center justify-between">
   445→          <CardTitle className="text-white text-lg flex items-center gap-2">
   446→            <ClipboardCheck className="h-5 w-5 text-orange-400" />
   447→            Audit Findings Lifecycle
   448→          </CardTitle>
   449→          <Badge variant="outline" className="border-red-500/50 text-red-400">
   450→            {findings.filter(f => f.status === 'open').length} Open
   451→          </Badge>
   452→        </div>
   453→      </CardHeader>
   454→      <CardContent>
   455→        <div className="space-y-3 max-h-[380px] overflow-y-auto">
   456→          {findings.map(finding => {
   457→            const daysLeft = getDaysUntilDeadline(finding.targetResolution)
   458→            
   459→            return (
   460→              <div key={finding.id} className={`p-4 rounded-lg border ${getStatusColor(finding.status)}`}>
   461→                <div className="flex items-start justify-between mb-2">
   462→                  <div className="flex items-center gap-2">
   463→                    <SeverityBadge severity={finding.severity} />
   464→                    <Badge variant="outline" className={`text-xs ${
   465→                      finding.status === 'open' ? 'border-red-500 text-red-400' :
   466→                      finding.status === 'in-progress' ? 'border-yellow-500 text-yellow-400' :
   467→                      finding.status === 'remediated' ? 'border-blue-500 text-blue-400' :
   468→                      finding.status === 'closed' ? 'border-green-500 text-green-400' :
   469→                      'border-orange-500 text-orange-400'
   470→                    }`}>
   471→                      {finding.status.replace('-', ' ').toUpperCase()}
   472→                    </Badge>
   473→                  </div>
   474→                  <span className={`text-xs font-mono ${
   475→                    daysLeft < 0 ? 'text-red-400' : 
   476→                    daysLeft <= 7 ? 'text-yellow-400' : 
   477→                    'text-slate-400'
   478→                  }`}>
   479→                    {daysLeft < 0 ? `${Math.abs(daysLeft)}d overdue` : `${daysLeft}d remaining`}
   480→                  </span>
   481→                </div>

   627→// Regulatory Deadline Countdown Component
   628→function RegulatoryDeadlines({ deadlines }: { deadlines: RegulatoryDeadline[] }) {
   629→  const getDaysRemaining = (date: Date) => {
   630→    const diff = date.getTime() - Date.now()
   631→    return Math.ceil(diff / 86400000)
   632→  }
   633→
   634→  const getStatusStyle = (status: RegulatoryDeadline['status']) => {
   635→    switch (status) {
   636→      case 'on-track': return 'border-green-500/30 bg-green-500/5'
   637→      case 'at-risk': return 'border-yellow-500/30 bg-yellow-500/5'
   638→      case 'overdue': return 'border-red-500/30 bg-red-500/5'
   639→      case 'completed': return 'border-blue-500/30 bg-blue-500/5'
   640→    }
   641→  }
   642→
   643→  return (
   644→    <Card className="bg-slate-900 border-slate-700">
   645→      <CardHeader className="pb-3">
   646→        <CardTitle className="text-white text-lg flex items-center gap-2">
   647→          <Calendar className="h-5 w-5 text-indigo-400" />
   648→          Regulatory Deadline Countdown
   649→        </CardTitle>
   650→      </CardHeader>
   651→      <CardContent>
   652→        <div className="space-y-3">
   653→          {deadlines.map((deadline, index) => {
   654→            const daysRemaining = getDaysRemaining(deadline.deadline)
   655→            
   656→            return (
   657→              <div key={index} className={`p-4 rounded-lg border ${getStatusStyle(deadline.status)}`}>
   658→                <div className="flex items-start justify-between mb-2">
   659→                  <div className="flex items-center gap-2">
   660→                    <StatusIndicator 
   661→                      status={
   662→                        deadline.status === 'completed' ? 'excellent' :
   663→                        deadline.status === 'on-track' ? 'good' :
   664→                        deadline.status === 'at-risk' ? 'warning' : 'critical'
   665→                      } 
   666→                      size="sm"
   667→                    />
   668→                    <h4 className="font-medium text-white text-sm">{deadline.regulation}</h4>
   669→                  </div>
   670→                  <Badge variant="outline" className={`text-xs ${
   671→                    deadline.status === 'completed' ? 'border-blue-500 text-blue-400' :
   672→                    deadline.status === 'on-track' ? 'border-green-500 text-green-400' :
   673→                    deadline.status === 'at-risk' ? 'border-yellow-500 text-yellow-400' :
   674→                    'border-red-500 text-red-400'
   675→                  }`}>
   676→                    {deadline.status.replace('-', ' ').toUpperCase()}
   677→                  </Badge>
   678→                </div>
   679→
   680→                <p className="text-xs text-slate-400 mb-3">{deadline.description}</p>
   681→
   682→                <div className="grid grid-cols-3 gap-3">
   683→                  <div>
   684→                    <p className="text-xs text-slate-500">Days Remaining</p>
   685→                    <p className={`text-lg font-bold ${
   686→                      daysRemaining < 0 ? 'text-red-400' :
   687→                      daysRemaining <= 15 ? 'text-yellow-400' :
   688→                      'text-green-400'
   689→                    }`}>
   690→                      {daysRemaining < 0 ? `+${Math.abs(daysRemaining)}` : daysRemaining}
   691→                    </p>
   692→                  </div>
   693→                  <div>
   694→                    <p className="text-xs text-slate-500">Progress</p>
   695→                    <div className="flex items-center gap-2 mt-1">
   696→                      <Progress value={deadline.progress} className="flex-1 h-2 bg-slate-700" />
   697→                      <span className="text-sm text-white">{deadline.progress}%</span>
   698→                    </div>
   699→                  </div>
   700→                  <div>
   701→                    <p className="text-xs text-slate-500">Owner</p>
   702→                    <p className="text-sm text-white">{deadline.owner}</p>
   703→                  </div>
   704→                </div>
   705→              </div>
   706→            )
   707→          })}
