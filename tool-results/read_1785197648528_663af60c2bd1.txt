   143→/**
   144→ * Analyze user behavior and generate risk assessment
   145→ */
   146→export function analyzeBehavior(
   147→  currentActivities: ActionEvent[],
   148→  baseline: BehaviorBaseline,
   149→  context: {
   150→    currentTime: Date;
   151→    entityInfo: { name: string; department: string; role: string };
   152→    peerGroupAvg?: BehaviorBaseline;
   153→  }
   154→): BehaviorProfile {
   155→  const session = buildSession(currentActivities);
   156→  const anomalies = detectBehavioralAnomalies(currentActivities, baseline, context.currentTime);
   157→  const riskFactors = calculateRiskFactors(currentActivities, anomalies, baseline, context);
   158→  const riskScore = calculateOverallRiskScore(riskFactors, anomalies);
   159→
   160→  return {
   161→    entityId: context.entityInfo.name.toLowerCase().replace(/\s+/g, '.'),
   162→    entityType: inferEntityType(context.entityInfo.role),
   163→    entityName: context.entityInfo.name,
   164→    baseline,
   165→    currentSession: {
   166→      ...session,
   167→      anomalies,
   168→      anomalyCount: anomalies.length,
   169→      maxAnomalySeverity: anomalies.length > 0 
   170→        ? anomalies.sort((a, b) => getSeverityWeight(b.severity) - getSeverityWeight(a.severity))[0].severity 
   171→        : 'low',
   172→    },
   173→    riskScore,
   174→    riskLevel: getRiskLevel(riskScore),
   175→    riskFactors,
   176→    lastUpdated: new Date(),
   177→    profileAgeDays: Math.floor(Math.random() * 90) + 30, // Simulated profile age
   178→  };
   179→}

   181→/**
   182→ * Build activity session from raw events
   183→ */
   184→function buildSession(events: ActionEvent[]): Omit<ActivitySession, 'anomalies' | 'anomalyCount' | 'maxAnomalySeverity'> {
   185→  return {
   186→    sessionId: `sess-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
   187→    startTime: events.length > 0 ? events[0].timestamp : new Date(),
   188→    endTime: events.length > 0 ? events[events.length - 1].timestamp : new Date(),
   189→    actions: events,
   190→    totalActions: events.length,
   191→    uniqueSystemsAccessed: new Set(events.map(e => e.endpoint).filter(Boolean) as string[]),
   192→    uniqueResourcesAccessed: new Set(events.map(e => e.resource)),
   193→  };
   194→}

   196→/**
   197→ * Detect behavioral anomalies compared to baseline
   198→ */
   199→function detectBehavioralAnomalies(
   200→  events: ActionEvent[],
   201→  baseline: BehaviorBaseline,
   202→  currentTime: Date
   203→): BehaviorAnomaly[] {
   204→  const anomalies: BehaviorAnomaly[] = [];
   205→
   206→  // 1. Check for impossible travel
   207→  const travelAnomaly = detectImpossibleTravel(events, baseline);
   208→  if (travelAnomaly) anomalies.push(travelAnomaly);
   209→
   210→  // 2. Check for after-hours access
   211→  const hoursAnomaly = detectAfterHoursAccess(events, baseline, currentTime);
   212→  if (hoursAnomaly) anomalies.push(hoursAnomaly);
   213→
   214→  // 3. Check for unusual data volume
   215→  const volumeAnomaly = detectUnusualDataVolume(events, baseline);
   216→  if (volumeAnomaly) anomalies.push(volumeAnomaly);
   217→
   218→  // 4. Check for unusual location
   219→  const locationAnomaly = detectUnusualLocation(events, baseline);
   220→  if (locationAnomaly) anomalies.push(locationAnomaly);
   221→
   222→  // 5. Check for privilege abuse
   223→  const privilegeAnomaly = detectPrivilegeAbuse(events, baseline);
   224→  if (privilegeAnomaly) anomalies.push(privilegeAnomaly);
   225→
   226→  // 6. Check for lateral movement
   227→  const lateralAnomaly = detectLateralMovement(events, baseline);
   228→  if (lateralAnomaly) anomalies.push(lateralAnomaly);
   229→
   230→  // 7. Check for data exfiltration signals
   231→  const exfilAnomaly = detectDataExfiltration(events, baseline);
   232→  if (exfilAnomaly) anomalies.push(exfilAnomaly);
   233→
   234→  // 8. Check for credential stuffing
   235→  const credAnomaly = detectCredentialStuffing(events, baseline);
   236→  if (credAnomaly) anomalies.push(credAnomaly);
   237→
   238→  return anomalies.sort((a, b) => b.score - a.score);
   239→}

   245→function detectImpossibleTravel(events: ActionEvent[], _baseline: BehaviorBaseline): BehaviorAnomaly | null {
   246→  const locationEvents = events.filter(e => e.location && e.actionType === 'login');
   247→  
   248→  if (locationEvents.length < 2) return null;
   249→
   250→  for (let i = 1; i < locationEvents.length; i++) {
   251→    const prev = locationEvents[i - 1];
   252→    const curr = locationEvents[i];
   253→    
   254→    const timeDiffHours = (curr.timestamp.getTime() - prev.timestamp.getTime()) / (1000 * 60 * 60);
   255→    const distanceKm = estimateDistance(prev.location!, curr.location!);
   256→    const speedKmh = distanceKm / timeDiffHours;
   257→
   258→    // Impossible travel: >900 km/h (commercial flight speed)
   259→    if (speedKmh > 900 && timeDiffHours < 24) {
   260→      return {
   261→        id: `anom-${Date.now()}-travel`,
   262→        timestamp: curr.timestamp,
   263→        anomalyType: 'impossible_travel',
   264→        severity: 'critical',
   265→        score: 95,
   266→        description: `Impossible travel detected: ${prev.location} to ${curr.location} in ${timeDiffHours.toFixed(1)} hours (${speedKmh.toFixed(0)} km/h)`,
   267→        baselineValue: 0,
   268→        observedValue: speedKmh,
   269→        deviationFactor: speedKmh / 900,
   270→        relatedEvents: [prev.timestamp.toISOString(), curr.timestamp.toISOString()],
   271→        mitreTechnique: 'T1021.003', // Remote Services: Remote Desktop Protocol
   272→        recommendation: 'Immediate investigation required. Possible account compromise or credential sharing.'
   273→      };
   274→    }
   275→  }
   276→
   277→  return null;
   278→}

   280→function detectAfterHoursAccess(events: ActionEvent[], baseline: BehaviorBaseline, currentTime: Date): BehaviorAnomaly | null {
   281→  const hour = currentTime.getHours();
   282→  const dayOfWeek = currentTime.getDay();
   283→  
   284→  // Check if this is outside normal hours
   285→  const isWeekend = dayOfWeek === 0 || dayOfWeek === 6;
   286→  const isAfterHours = hour < 6 || hour > 22 || isWeekend;
   287→  
   288→  if (!isAfterHours) return null;
   289→
   290→  // Calculate how unusual this is based on baseline
   291→  const hourProbability = baseline.typicalLoginHours[hour] || 0;
   292→  const dayProbability = baseline.typicalWorkdays[dayOfWeek] || 0;
   293→  const combinedProbability = hourProbability * dayProbability;
   294→
   295→  // If very unlikely based on baseline
   296→  if (combinedProbability < 0.05 && events.length > 0) {
   297→    return {
   298→      id: `anom-${Date.now()}-afterhours`,
   299→      timestamp: currentTime,
   300→      anomalyType: 'after_hours_access',
   301→      severity: combinedProbability < 0.01 ? 'high' : 'medium',
   302→      score: Math.round((1 - combinedProbability) * 70),
   303→      description: `After-hours access at ${hour}:00 on ${['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'][dayOfWeek]} (${events.length} actions)`,
   304→      baselineValue: combinedProbability * 100,
   305→      observedValue: 100,
   306→      deviationFactor: 1 / (combinedProbability + 0.01),
   307→      relatedEvents: events.slice(0, 3).map(e => e.timestamp.toISOString()),
   308→      recommendation: 'Verify legitimate business reason. Monitor for additional suspicious activity.'
   309→    };
   310→  }
   311→
   312→  return null;
   313→}

   315→function detectUnusualDataVolume(events: ActionEvent[], baseline: BehaviorBaseline): BehaviorAnomaly | null {
   316→  const dataEvents = events.filter(e => 
   317→    ['file_download', 'file_upload', 'data_export', 'database_query'].includes(e.actionType)
   318→  );
   319→  
   320→  if (dataEvents.length === 0) return null;
   321→
   322→  // Estimate total data volume (simplified)
   323→  const estimatedVolumeMB = dataEvents.reduce((sum, e) => {
   324→    switch (e.actionType) {
   325→      case 'file_download': return sum + (e.details?.sizeMB || 5);
   326→      case 'file_upload': return sum + (e.details?.sizeMB || 3);
   327→      case 'data_export': return sum + (e.details?.sizeMB || 10);
   328→      case 'database_query': return sum + (e.details?.rowCount ? e.details.rowCount / 1000 : 2);
   329→      default: return sum;
   330→    }
   331→  }, 0);
   332→
   333→  const volumeRatio = estimatedVolumeMB / (baseline.avgDataAccessedMB || 1);
   334→
   335→  if (volumeRatio > 5) {
   336→    return {
   337→      id: `anom-${Date.now()}-volume`,
   338→      timestamp: new Date(),
   339→      anomalyType: 'unusual_data_volume',
   340→      severity: volumeRatio > 20 ? 'critical' : volumeRatio > 10 ? 'high' : 'medium',
   341→      score: Math.min(95, Math.round(volumeRatio * 4)),
   342→      description: `Unusual data volume: ${estimatedVolumeMB.toFixed(1)}MB accessed (baseline: ${baseline.avgDataAccessedMB?.toFixed(1) || 'N/A'}MB, ${(volumeRatio).toFixed(1)}x normal)`,
   343→      baselineValue: baseline.avgDataAccessedMB || 0,
   344→      observedValue: estimatedVolumeMB,
   345→      deviationFactor: volumeRatio,
   346→      relatedEvents: dataEvents.map(e => e.timestamp.toISOString()),
   347→      mitreTechnique: 'T1530', // Data from Cloud Storage Object
   348→      recommendation: 'Review data access purpose. Check for bulk downloads or unauthorized exports.'
   349→    };
   350→  }
   351→
   352→  return null;
   353→}

   355→function detectUnusualLocation(events: ActionEvent[], baseline: BehaviorBaseline): BehaviorAnomaly | null {
   356→  const locations = [...new Set(events.map(e => e.location).filter(Boolean))];
   357→  const unusualLocations = locations.filter(l => !baseline.typicalLocations.includes(l!));
   358→
   359→  if (unusualLocations.length > 0) {
   360→    return {
   361→      id: `anom-${Date.now()}-location`,
   362→      timestamp: new Date(),
   363→      anomalyType: 'unusual_location',
   364→      severity: 'medium',
   365→      score: Math.min(70, unusualLocations.length * 25),
   366→      description: `Access from unusual location(s): ${unusualLocations.join(', ')}`,
   367→      baselineValue: baseline.typicalLocations.length,
   368→      observedValue: locations.length,
   369→      deviationFactor: locations.length / (baseline.typicalLocations.length || 1),
   370→      relatedEvents: events.filter(e => unusualLocations.includes(e.location!)).map(e => e.timestamp.toISOString()),
   371→      recommendation: 'Verify location legitimacy. Consider geo-blocking if unauthorized region.'
   372→    };
   373→  }
   374→
   375→  return null;
   376→}

   378→function detectPrivilegeAbuse(events: ActionEvent[], _baseline: BehaviorBaseline): BehaviorAnomaly | null {
   379→  const privEvents = events.filter(e => e.actionType === 'privilege_escalation');
   380→  
   381→  if (privEvents.length > 0) {
   382→    const failedPrivEvents = privEvents.filter(e => e.outcome !== 'success');
   383→    
   384→    if (failedPrivEvents.length > 0) {
   385→      return {
   386→        id: `anom-${Date.now()}-privilege`,
   387→        timestamp: failedPrivEvents[0].timestamp,
   388→        anomalyType: 'privilege_abuse',
   389→        severity: 'high',
   390→        score: 85,
   391→        description: `${failedPrivEvents.length} failed privilege escalation attempt(s) detected`,
   392→        baselineValue: 0,
   393→        observedValue: failedPrivEvents.length,
   394→        deviationFactor: failedPrivEvents.length + 1,
   395→        relatedEvents: failedPrivEvents.map(e => e.timestamp.toISOString()),
   396→        mitreTechnique: 'T1548', // Abuse Elevation Control Mechanism
   397→        recommendation: 'Immediate security review. Check for account compromise or insider threat.'
   398→      };
   399→    }
   400→
   401→    if (privEvents.length > 3) {
   402→      return {
   403→        id: `anom-${Date.now()}-privilege-multi`,
   404→        timestamp: new Date(),
   405→        anomalyType: 'privilege_abuse',
   406→        severity: 'medium',
   407→        score: 60,
   408→        description: `Multiple privilege escalations (${privEvents.length}) in single session`,
   409→        baselineValue: 1,
   410→        observedValue: privEvents.length,
   411→        deviationFactor: privEvents.length,
   412→        relatedEvents: privEvents.map(e => e.timestamp.toISOString()),
   413→        recommendation: 'Review privilege escalation justifications. Verify change management approval.'
   414→      };
   415→    }
   416→  }
   417→
   418→  return null;
   419→}

   421→function detectLateralMovement(events: ActionEvent[], baseline: BehaviorBaseline): BehaviorAnomaly | null {
   422→  const endpoints = [...new Set(events.map(e => e.endpoint).filter(Boolean))];
   423→  const newEndpoints = endpoints.filter(e => !baseline.typicalEndpoints.includes(e!));
   424→
   425→  if (newEndpoints.length > baseline.typicalSystemsAccessed * 2) {
   426→    return {
   427→      id: `anom-${Date.now()}-lateral`,
   428→      timestamp: new Date(),
   429→      anomalyType: 'lateral_movement',
   430→      severity: newEndpoints.length > 5 ? 'high' : 'medium',
   431→      score: Math.min(80, newEndpoints.length * 15),
   432→      description: `Potential lateral movement: ${newEndpoints.length} new system(s) accessed`,
   433→      baselineValue: baseline.typicalSystemsAccessed,
   434→      observedValue: endpoints.length,
   435→      deviationFactor: endpoints.length / (baseline.typicalSystemsAccessed || 1),
   436→      relatedEvents: events.filter(e => newEndpoints.includes(e.endpoint!)).map(e => e.timestamp.toISOString()),
   437→      mitreTechnique: 'T1021', // Remote Services
   438→      recommendation: 'Investigate access pattern. Check for reconnaissance or pivot attempts.'
   439→    };
   440→  }
   441→
   442→  return null;
   443→}

   445→function detectDataExfiltration(events: ActionEvent[], baseline: BehaviorBaseline): BehaviorAnomaly | null {
   446→  const exfilSignals = events.filter(e => 
   447→    e.actionType === 'data_export' || 
   448→    (e.actionType === 'file_download' && (e.details?.sizeMB || 0) > 50) ||
   449→    e.actionType === 'email_sent' && e.resource.includes('.zip')
   450→  );
   451→
   452→  if (exfilSignals.length >= 2) {
   453→    const totalVolume = exfilSignals.reduce((sum, e) => sum + (e.details?.sizeMB || 20), 0);
   454→    
   455→    return {
   456→      id: `anom-${Date.now()}-exfil`,
   457→      timestamp: new Date(),
   458→      anomalyType: 'data_exfiltration',
   459→      severity: totalVolume > 500 ? 'critical' : totalVolume > 100 ? 'high' : 'medium',
   460→      score: Math.min(95, exfilSignals.length * 30 + (totalVolume > 100 ? 20 : 0)),
   461→      description: `Potential data exfiltration: ${exfilSignals.length} export(s), ~${totalVolume}MB total`,
   462→      baselineValue: baseline.avgDataAccessedMB * 0.1, // 10% of normal as export baseline
   463→      observedValue: totalVolume,
   464→      deviationFactor: totalVolume / (baseline.avgDataAccessedMB * 0.1 + 1),
   465→      relatedEvents: exfilSignals.map(e => e.timestamp.toISOString()),
   466→      mitreTechnique: 'T1048', // Exfiltration Over Alternative Protocol
   467→      recommendation: 'URGENT: Review data exports. Check DLP alerts. Interview user if needed.'
   468→    };
   469→  }
   470→
   471→  return null;
   472→}

   474→function detectCredentialStuffing(events: ActionEvent[], _baseline: BehaviorBaseline): BehaviorAnomaly | null {
   475→  const loginFailures = events.filter(e => e.actionType === 'login' && e.outcome === 'failure');
   476→
   477→  if (loginFailures.length >= 5) {
   478→    const timeSpanMinutes = (events[events.length - 1].timestamp.getTime() - events[0].timestamp.getTime()) / (1000 * 60);
   479→    const failureRate = loginFailures.length / (timeSpanMinutes || 1);
   480→
   481→    if (failureRate > 2) { // More than 2 failures per minute
   482→      return {
   483→        id: `anom-${Date.now()}-creds`,
   484→        timestamp: loginFailures[0].timestamp,
   485→        anomalyType: 'credential_stuffing',
   486→        severity: failureRate > 10 ? 'critical' : 'high',
   487→        score: Math.min(90, loginFailures.length * 12),
   488→        description: `Possible credential stuffing: ${loginFailures.length} failures in ${timeSpanMinutes.toFixed(0)}min (${failureRate.toFixed(1)}/min)`,
   489→        baselineValue: 0.5, // Normal: occasional typo
   490→        observedValue: failureRate,
   491→        deviationFactor: failureRate / 0.5,
   492→        relatedEvents: loginFailures.map(e => e.timestamp.toISOString()),
   493→        mitreTechnique: 'T1110', // Brute Force
   494→        recommendation: 'Lock account if not already. Check source IPs. Enable MFA enforcement.'
   495→      };
   496→    }
   497→  }
   498→
   499→  return null;
   500→}

   502→// ============================================================
   503→// RISK CALCULATION
   504→// ============================================================
   505→
   506→function calculateRiskFactors(
   507→  events: ActionEvent[],
   508→  anomalies: BehaviorAnomaly[],
   509→  baseline: BehaviorBaseline,
   510→  context: { currentTime: Date; entityInfo: { name: string; department: string; role: string } }
   511→): RiskFactor[] {
   512→  const factors: RiskFactor[] = [];
   513→
   514→  // Anomaly-based risk
   515→  if (anomalies.length > 0) {
   516→    factors.push({
   517→      name: 'Behavioral Anomalies',
   518→      weight: 0.35,
   519→      value: anomalies.reduce((sum, a) => sum + a.score, 0) / anomalies.length,
   520→      severity: anomalies.some(a => a.severity === 'critical') ? 'critical' : 
   521→              anomalies.some(a => a.severity === 'high') ? 'high' : 'medium',
   522→      description: `${anomalies.length} anomaly(ies) detected`
   523→    });
   524→  }
   525→
   526→  // Volume-based risk
   527→  const volumeRatio = events.length / (baseline.avgDailyActions / 24 || 1); // Hourly baseline
   528→  if (volumeRatio > 2) {
   529→    factors.push({
   530→      name: 'Activity Volume',
   531→      weight: 0.2,
   532→      value: Math.min(1, volumeRatio / 10),
   533→      severity: volumeRatio > 5 ? 'high' : 'medium',
   534→      description: `${events.length} actions (${volumeRatio.toFixed(1)}x normal hourly rate)`
   535→    });
   536→  }
   537→
   538→  // Sensitive resource access
   539→  const sensitiveAccesses = events.filter(e => 
   540→    e.resource.includes('admin') || 
   541→    e.resource.includes('config') ||
   542→    e.resource.includes('password') ||
   543→    e.resource.includes('secret')
   544→  ).length;
   545→  if (sensitiveAccesses > 0) {
   546→    factors.push({
   547→      name: 'Sensitive Resource Access',
   548→      weight: 0.25,
   549→      value: sensitiveAccesses / events.length,
   550→      severity: sensitiveAccesses > 3 ? 'high' : 'medium',
   551→      description: `${sensitiveAccesses} sensitive resource(s) accessed`
   552→    });
   553→  }
   554→
   555→  // Failure rate
   556→  const failures = events.filter(e => e.outcome !== 'success').length;
   557→  const failureRate = failures / events.length;
   558→  if (failureRate > 0.1) {
   559→    factors.push({
   560→      name: 'Failure Rate',
   561→      weight: 0.1,
   562→      value: failureRate,
   563→      severity: failureRate > 0.3 ? 'high' : 'medium',
   564→      description: `${((failureRate) * 100).toFixed(0)}% action failure rate`
   565→    });
   566→  }
   567→
   568→  // Time-based risk (after hours, weekend)
   569→  const hour = context.currentTime.getHours();
   570→  const dayOfWeek = context.currentTime.getDay();
   571→  if (hour < 7 || hour > 20 || dayOfWeek === 0 || dayOfWeek === 6) {
   572→    factors.push({
   573→      name: 'Temporal Anomaly',
   574→      weight: 0.1,
   575→      value: 0.7,
   576→      severity: 'low',
   577→      description: `Access during non-standard hours`
   578→    });
   579→  }
   580→
   581→  return factors.sort((a, b) => b.weight - a.weight);
   582→}

   584→function calculateOverallRiskScore(factors: RiskFactor[], anomalies: BehaviorAnomaly[]): number {
   585→  let weightedScore = 0;
   586→  
   587→  factors.forEach(factor => {
   588→    weightedScore += factor.value * factor.weight * 100;
   589→  });
   590→
   591→  // Boost from critical anomalies
   592→  const criticalBoost = anomalies.filter(a => a.severity === 'critical').length * 15;
   593→  const highBoost = anomalies.filter(a => a.severity === 'high').length * 8;
   594→
   595→  return Math.min(100, Math.round(weightedScore + criticalBoost + highBoost));
   596→}
