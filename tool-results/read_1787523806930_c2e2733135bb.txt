     6→export interface AITask {
     7→  id: string
     8→  name: string
     9→  type: 'detection' | 'response' | 'analysis' | 'prediction' | 'remediation' | 'learning'
    10→  status: 'pending' | 'running' | 'completed' | 'failed' | 'skipped'
    11→  priority: 'critical' | 'high' | 'medium' | 'low'
    12→  confidence: number // 0-1
    13→  createdAt: Date
    14→  startedAt?: Date
    15→  completedAt?: Date
    16→  result?: any
    17→  error?: string
    18→  dependencies: string[]
    19→  estimatedDuration: number // seconds
    20→  actualDuration?: number
    21→  config?: Record<string, any> // Optional configuration for task execution
    22→}

    24→export interface AIModel {
    25→  id: string
    26→  name: string
    27→  version: string
    28→  type: 'classification' | 'regression' | 'clustering' | 'nlp' | 'anomaly' | 'reinforcement' | 'transformer'
    29→  status: 'training' | 'ready' | 'deployed' | 'deprecated' | 'error'
    30→  accuracy: number
    31→  precision: number
    32→  recall: number
    33→  f1Score: number
    34→  lastTrained: Date
    35→  trainingDataSize: number
    36→  inferenceTime: number // ms
    37→  description: string
    38→  inputFeatures: string[]
    39→  outputLabels: string[]
    40→}
    41→
    42→export interface AutomatedPlaybook {
    43→  id: string
    44→  name: string
    45→  description: string
    46→  triggerConditions: PlaybookTrigger[]
    47→  actions: PlaybookAction[]
    48→  status: 'active' | 'inactive' | 'testing' | 'deprecated'
    49→  executionCount: number
    50→  successRate: number
    51→  avgExecutionTime: number // seconds
    52→  lastExecuted?: Date
    53→  createdBy: string
    54→  version: number
    55→}
    56→
    57→export interface PlaybookTrigger {
    58→  type: 'alert' | 'schedule' | 'event' | 'threshold' | 'ml_prediction' | 'manual'
    59→  conditions: {
    60→    field: string
    61→    operator: 'equals' | 'contains' | 'greater_than' | 'less_than' | 'in_range' | 'regex'
    62→    value: any
    63→  }[]
    64→  severityFilter?: ('info' | 'low' | 'medium' | 'high' | 'critical')[]
    65→}
    66→
    67→export interface PlaybookAction {
    68→  id: string
    69→  order: number
    70→  type: 'notification' | 'containment' | 'remediation' | 'enrichment' | 'escalation' | 'forensics' | 'integration'
    71→  target: string
    72→  config: Record<string, any>
    73→  timeout: number // seconds
    74→  retryCount: number
    75→  onFailure: 'continue' | 'abort' | 'retry' | 'escalate'
    76→}
    77→
    78→export interface PredictionResult {
    79→  id: string
    80→  modelId: string
    81→  modelName: string
    82→  input: Record<string, any>
    83→  output: {
    84→    label: string
    85→    confidence: number
    86→    probabilities: Record<string, number>
    87→  }
    88→  featuresUsed: string[]
    89→  processingTime: number
    90→  timestamp: Date
    91→  explanation?: string
    92→  recommendations: string[]
    93→}

    95→export interface AnomalyRecord {
    96→  id: string
    97→  timestamp: Date
    98→  source: string
    99→  anomalyType: 'statistical' | 'behavioral' | 'network' | 'endpoint' | 'telecom' | 'spatial'
   100→  severity: 'info' | 'low' | 'medium' | 'high' | 'critical'
   101→  score: number // 0-100, how anomalous
   102→  baseline: number
   103→  deviation: number // standard deviations from baseline
   104→  context: Record<string, any>
   105→  rootCauseAnalysis?: string
   106→  predictedImpact: string
   107→  autoRemediated: boolean
   108→  remediationAction?: string
   109→  confirmed: boolean
   110→  falsePositive: boolean
   111→}
