   442→// User Management Component
   443→function UserManagement() {
   444→  const [users, setUsers] = useState<User[]>([])
   445→  const [loading, setLoading] = useState(true)
   446→  const [searchQuery, setSearchQuery] = useState('')
   447→  const [statusFilter, setStatusFilter] = useState('all')
   448→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   449→  const [selectedUser, setSelectedUser] = useState<User | null>(null)
   450→  const [pagination, setPagination] = useState({ page: 1, total: 0, pages: 0 })
   451→
   452→  const fetchUsers = useCallback(async () => {
   453→    setLoading(true)
   454→    try {
   455→      const params = new URLSearchParams({
   456→        page: pagination.page.toString(),
   457→        limit: '20',
   458→        ...(searchQuery && { search: searchQuery }),
   459→        ...(statusFilter !== 'all' && { status: statusFilter })
   460→      })
   461→      
   462→      const res = await fetch(`/api/admin/users?${params}`)
   463→      if (res.ok) {
   464→        const data = await res.json()
   465→        setUsers(data.data.users)
   466→        setPagination(data.data.pagination)
   467→      }
   468→    } catch (error) {
   469→      toast.error('Failed to fetch users')
   470→    } finally {
   471→      setLoading(false)
   472→    }
   473→  }, [pagination.page, searchQuery, statusFilter])
   474→
   475→  useEffect(() => {
   476→    fetchUsers()
   477→  }, [fetchUsers])
   478→
   479→  const handleCreateUser = async (formData: FormData) => {
   480→    try {
   481→      const res = await fetch('/api/admin/users', {
   482→        method: 'POST',
   483→        headers: { 'Content-Type': 'application/json' },
   484→        body: JSON.stringify(Object.fromEntries(formData))
   485→      })
   486→      
   487→      if (res.ok) {
   488→        toast.success('User created successfully')
   489→        setShowCreateDialog(false)
   490→        fetchUsers()
   491→      } else {
   492→        const data = await res.json()
   493→        toast.error(data.error || 'Failed to create user')
   494→      }
   495→    } catch (error) {
   496→      toast.error('Failed to create user')
   497→    }
   498→  }
   499→
   500→  const handleToggleStatus = async (user: User) => {
   501→    const newStatus = user.status === 'ACTIVE' ? 'SUSPENDED' : 'ACTIVE'
   502→    try {
   503→      const res = await fetch(`/api/admin/users/${user.id}`, {
   504→        method: 'PATCH',
   505→        headers: { 'Content-Type': 'application/json' },
   506→        body: JSON.stringify({ status: newStatus })
   507→      })
   508→      
   509→      if (res.ok) {
   510→        toast.success(`User ${newStatus.toLowerCase()}`)
   511→        fetchUsers()
   512→      }
   513→    } catch (error) {
   514→      toast.error('Failed to update user status')
   515→    }
   516→  }
   517→
   518→  const handleDeleteUser = async (userId: string) => {
   519→    if (!confirm('Are you sure you want to delete this user?')) return
   520→    
   521→    try {
   522→      const res = await fetch(`/api/admin/users/${userId}`, { method: 'DELETE' })
   523→      if (res.ok) {
   524→        toast.success('User deleted')
   525→        fetchUsers()
   526→      }
   527→    } catch (error) {
   528→      toast.error('Failed to delete user')
   529→    }
   530→  }
   531→
   532→  const getStatusColor = (status: string) => {
   533→    switch (status) {
   534→      case 'ACTIVE': return 'bg-green-500/10 text-green-400 border-green-500/20'
   535→      case 'INACTIVE': return 'bg-gray-500/10 text-gray-400 border-gray-500/20'
   536→      case 'SUSPENDED': return 'bg-yellow-500/10 text-yellow-400 border-yellow-500/20'
   537→      case 'LOCKED': return 'bg-red-500/10 text-red-400 border-red-500/20'
   538→      default: return 'bg-gray-500/10 text-gray-400 border-gray-500/20'
   539→    }
   540→  }
   541→
   542→  return (
   543→    <div className="space-y-6">

   893→function SessionManagement() {
   894→  const [sessions, setSessions] = useState<Session[]>([])
   895→  const [loading, setLoading] = useState(true)
   896→  const [stats, setStats] = useState<any>(null)
   897→  const [filter, setFilter] = useState('all')
   898→
   899→  useEffect(() => {
   900→    fetchSessions()
   901→  }, [filter])
   902→
   903→  const fetchSessions = async () => {
   904→    setLoading(true)
   905→    try {
   906→      const params = new URLSearchParams({ limit: '50', status: filter })
   907→      const res = await fetch(`/api/admin/sessions?${params}`)
   908→      if (res.ok) {
   909→        const data = await res.json()
   910→        setSessions(data.data.sessions)
   911→        setStats(data.data.statistics)
   912→      }
   913→    } catch (error) {
   914→      toast.error('Failed to fetch sessions')
   915→    } finally {
   916→      setLoading(false)
   917→    }
   918→  }
   919→
   920→  const terminateSession = async (sessionId: string) => {
   921→    if (!confirm('Terminate this session?')) return
   922→    
   923→    try {
   924→      const res = await fetch('/api/admin/sessions', {
   925→        method: 'POST',
   926→        headers: { 'Content-Type': 'application/json' },
   927→        body: JSON.stringify({ action: 'force-logout', sessionId })
   928→      })
   929→      
   930→      if (res.ok) {
   931→        toast.success('Session terminated')
   932→        fetchSessions()
   933→      }
   934→    } catch (error) {
   935→      toast.error('Failed to terminate session')
   936→    }
   937→  }
   938→
   939→  const terminateStaleSessions = async () => {
   940→    if (!confirm('Terminate all stale sessions (inactive for 2+ hours)?')) return
   941→    
   942→    try {
   943→      const res = await fetch('/api/admin/sessions', {
   944→        method: 'DELETE',
   945→        headers: { 'Content-Type': 'application/json' },
   946→        body: JSON.stringify({ terminateAllStale: true })
   947→      })
   948→      
   949→      if (res.ok) {
   950→        const data = await res.json()
   951→        toast.success(data.message)
   952→        fetchSessions()
   953→      }
   954→    } catch (error) {
   955→      toast.error('Failed to terminate stale sessions')
   956→    }
   957→  }
   958→
   959→  const formatTimeAgo = (dateString: string) => {
   960→    const diff = Date.now() - new Date(dateString).getTime()
   961→    const minutes = Math.floor(diff / 60000)
   962→    if (minutes < 1) return 'Just now'
   963→    if (minutes < 60) return `${minutes}m ago`
   964→    const hours = Math.floor(minutes / 60)
   965→    if (hours < 24) return `${hours}h ago`
   966→    return `${Math.floor(hours / 24)}d ago`
   967→  }
   968→
   969→  return (
   970→    <div className="space-y-6">
   971→      <div className="flex items-center justify-between">
   972→        <div>
   973→          <h2 className="text-2xl font-bold">Session Management</h2>
   974→          <p className="text-slate-400 mt-1">Monitor and manage active user sessions</p>
   975→        </div>
   976→        <Button variant="destructive" onClick={terminateStaleSessions}>
   977→          <Trash2 className="h-4 w-4 mr-2" /> Clean Stale Sessions
   978→        </Button>
   979→      </div>
   980→
   981→      {/* Session Stats */}
   982→      {stats && (
   983→        <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   984→          <MiniStatCard label="Active Now" value={stats.activeLast5min} color="green" />
   985→          <MiniStatCard label="Last 30 min" value={stats.activeLast30min} color="blue" />
   986→          <MiniStatCard label="Last Hour" value={stats.activeLast1hour} color="yellow" />
   987→          <MiniStatCard label="Stale (>2h)" value={stats.staleSessions} color="red" />
   988→        </div>
   989→      )}
   990→
   991→      {/* Filters */}
   992→      <div className="flex gap-2">
   993→        {['all', 'active', 'inactive'].map(f => (
   994→          <Button
   995→            key={f}
   996→            variant={filter === f ? 'default' : 'outline'}
   997→            size="sm"
   998→            onClick={() => setFilter(f)}
   999→            className="capitalize"
  1000→          >
  1001→            {f}
  1002→          </Button>
  1003→        ))}
  1004→      </div>
  1005→
  1006→      {/* Sessions List */}
  1007→      <Card className="bg-slate-900/50 border-slate-800">
  1008→        <CardContent className="p-0">
  1009→          {loading ? (
  1010→            <div className="flex items-center justify-center py-12">
  1011→              <RefreshCw className="h-8 w-8 animate-spin text-slate-400" />
  1012→            </div>
  1013→          ) : (
  1014→            <Table>
  1015→              <TableHeader>
  1016→                <TableRow className="border-slate-700 hover:bg-transparent">
  1017→                  <TableHead>User</TableHead>
  1018→                  <TableHead>Status</TableHead>
  1019→                  <TableHead>IP Address</TableHead>
  1020→                  <TableHead>Last Active</TableHead>
  1021→                  <TableHead>Created</TableHead>
  1022→                  <TableHead className="text-right">Actions</TableHead>
  1023→                </TableRow>
  1024→              </TableHeader>
  1025→              <TableBody>
  1026→                {sessions.map((session) => (
  1027→                  <TableRow key={session.id} className="border-slate-700">
  1028→                    <TableCell>
  1029→                      <div>
  1030→                        <p className="font-medium">{session.user?.name || 'Unknown'}</p>
  1031→                        <p className="text-sm text-slate-400">{session.user?.email}</p>
