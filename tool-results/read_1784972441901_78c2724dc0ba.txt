   123→  private async request<T>(
   124→    endpoint: string,
   125→    options: RequestInit = {}
   126→  ): Promise<T> {
   127→    const url = `${this.config.url.replace(/\/$/, '')}${endpoint}`;
   128→    const headers: Record<string, string> = {
   129→      'Authorization': this.authHeader,
   130→      'Content-Type': 'application/json',
   131→      'Accept': 'application/json',
   132→      ...options.headers as Record<string, string>,
   133→    };
   134→
   135→    const controller = new AbortController();
   136→    const timeoutId = setTimeout(() => controller.abort(), this.config.timeout);
   137→
   138→    try {
   139→      const response = await fetch(url, {
   140→        ...options,
   141→        headers,
   142→        signal: controller.signal,
   143→      });
   144→
   145→      clearTimeout(timeoutId);
   146→
   147→      // Handle rate limiting
   148→      if (response.status === 429) {
   149→        const retryAfter = parseInt(response.headers.get('Retry-After') || '60', 10);
   150→        throw new MISPRateLimitError(
   151→          `Rate limited. Retry after ${retryAfter}s`,
   152→          retryAfter
   153→        );
   154→      }
   155→
   156→      // Handle auth errors
   157→      if (response.status === 401 || response.status === 403) {
   158→        throw new MISPAuthenticationError();
   159→      }
   160→
   161→      // Parse response
   162→      const data = await response.json();
   163→
   164→      if (!response.ok) {
   165→        throw new MISPError(
   166→          data.message || data.error?.message || `Request failed with status ${response.status}`,
   167→          response.status,
   168→          data.error?.code,
   169→          data.error
   170→        );
   171→      }
   172→
   173→      return data as T;
   174→    } catch (error) {
   175→      clearTimeout(timeoutId);
   176→      
   177→      if (error instanceof MISPError) {
   178→        throw error;
   179→      }
   180→      
   181→      if (error instanceof Error && error.name === 'AbortError') {
   182→        throw new MISPError(`Request timed out after ${this.config.timeout}ms`, 408, 'TIMEOUT');
   183→      }
   184→      
   185→      throw new MISPError(
   186→        `Network error: ${error instanceof Error ? error.message : 'Unknown error'}`,
   187→        0,
   188→        'NETWORK_ERROR'
   189→      );
   190→    }
   191→  }

   195→      console.log(`[MISP] [${method}] ${message}`, data ?? '');
   196→    }

   285→  /**
   286→   * Create a new MISP event

   288→  async createEvent(params: {
   289→    info: string;
   290→    threat_level_id?: number;
   291→    analysis?: AnalysisStatus;
   292→    distribution?: DistributionLevel;
   293→    date?: string;  // YYYY-MM-DD format
   294→    tags?: string[];
   295→  }): Promise<MISPEvent> {
   296→    this.log('createEvent', `Creating event: ${params.info}`);
   297→
   298→    const payload: Record<string, unknown> = {
   299→      info: params.info,
   300→      threat_level_id: params.threat_level_id || 4,
   301→      analysis: params.analysis || AnalysisStatus.INITIAL,
   302→      distribution: params.distribution || DistributionLevel.YOUR_ORG_ONLY,
   303→    };
   304→
   305→    if (params.date) payload.date = params.date;
   306→    if (params.tags) payload.Tag = params.tags.map(name => ({ name }));
   307→
   308→    const response = await this.request<{
   309→      response: Array<{ Event: MISPEvent }>;
   310→    }>('/events', {
   311→      method: 'POST',
   312→      body: JSON.stringify(payload),
   313→    });
   314→
   315→    return response.response[0].Event;
   316→  }

   318→  /**
   319→   * Get event by ID or UUID
   320→   */
   321→  async getEvent(eventId: string, options?: {
   322→    includeAttributes?: boolean;
   323→    includeObjects?: boolean;
   324→    includeCorrelations?: boolean;
   325→    includeNotes?: boolean;
   326→    includeRelations?: boolean;
   327→  }): Promise<MISPEvent> {
   328→    this.log('getEvent', `Fetching event: ${eventId}`);
   329→
   330→    let query = '';
   331→    if (options) {
   332→      const params: string[] = [];
   333→      if (options.includeAttributes !== undefined) params.push(`includeAttribute=${Number(options.includeAttributes)}`);
   334→      if (options.includeObjects !== undefined) params.push(`includeObject=${Number(options.includeObjects)}`);
   335→      if (options.includeCorrelations) params.push('includeEventCorrelations=1');
   336→      if (options.includeNotes) params.push('includeEventReports=1');
   337→      if (options.includeRelations) params.push('includeRelationships=1');
   338→      query = params.length > 0 ? `?${params.join('&')}` : '';
   339→    }
   340→
   341→    const response = await this.request<{
   342→      response: Array<{ Event: MISPEvent }>;
   343→    }>(`/events/${eventId}${query}`);
   344→
   345→    return response.response[0].Event;
   346→  }

   399→  /**
   400→   * Search events with comprehensive filtering
   401→   */
   402→  async searchEvents(params: MISPEventSearchParams): Promise<MISPEventSearchResponse> {
   403→    this.log('searchEvents', `Searching events`, params);
   404→
   405→    // Build search payload based on MISP REST API format
   406→    const searchPayload: Record<string, unknown> = {};
   407→
   408→    if (params.value) searchPayload.value = params.value;
   409→    if (params.type) searchPayload.type = params.type;
   410→    if (params.category) searchPayload.category = params.category;
   411→    if (params.tag?.length) searchPayload.tags = params.tag.join('&&');
   412→    if (params.from) searchPayload.from = params.from;
   413→    if (params.to) searchPayload.to = params.to;
   414→    if (params.last) searchPayload.last = params.last;
   415→    if (params.eventid) searchPayload.eventid = params.eventid;
   416→    if (params.threat_level_id) searchPayload.threat_level_id = params.threat_level_id;
   417→    if (params.distribution) searchPayload.distribution = params.distribution;
   418→    if (params.analysis) searchPayload.analysis = params.analysis;
   419→    if (params.org) searchPayload.org = params.org;
   420→    if (params.published !== undefined) searchPayload.published = params.published;
   421→    if (params.date_from) searchPayload.datefrom = params.date_from;
   422→    if (params.date_to) searchPayload.dateto = params.date_to;
   423→    if (params.searchall !== undefined) searchPayload.searchall = params.searchall;
   424→    if (params.metadata) searchPayload.metadata = params.metadata;
   425→    if (params.include_context) searchPayload.includeContext = true;
   426→    if (params.include_correlations) searchPayload.includeEventCorrelations = 1;
   427→    if (params.includeWarnings) searchPayload.includeWarninglistHits = true;
   428→    if (params.decaying) searchPayload.decaying = true;
   429→    if (params.exclude_local_tags) searchPayload.excludeLocalTags = true;
   430→    if (params.enforceWarninglist) searchPayload.enforceWarninglist = true;
   431→    if (params.returnFormat) searchPayload.returnFormat = params.returnFormat;
   432→
   433→    // Pagination
   434→    if (params.limit) searchPayload.limit = params.limit;
   435→    if (params.page) searchPayload.page = params.page;
   436→
   437→    const response = await this.request<MISPEventSearchResponse>(
   438→      '/events/restSearch',
   439→      {
   440→        method: 'POST',
   441→        body: JSON.stringify(searchPayload),
   442→      }
   443→    );
   444→
   445→    return response;
   446→  }

   535→        const result = await this.addAttribute(eventId, attr);

   541→        });
   542→      }
   543→    }
   544→
   545→    return {
   546→      success: errors.length === 0,
   547→      processed: results.length,
   548→      failed: errors.length,
   549→      skipped: 0,
   550→      errors,
   551→    };
   552→  }

  1054→  /**
  1055→   * Generate YARA rule from event attributes
  1056→   */
  1057→  async generateYARAFromEvent(eventId: string, options?: {
  1058→    includeHashes?: boolean;
  1059→    includeStrings?: boolean;
  1060→    includeIPs?: boolean;
  1061→    includeDomains?: boolean;
  1062→    maxSize?: number;
  1063→  }): Promise<YARARule> {
  1064→    this.log('generateYARAFromEvent', `Generating YARA for event: ${eventId}`);
  1065→
  1066→    const event = await this.getEvent(eventId);
  1067→
  1068→    // Filter IOCs based on options
  1069→    const iocs = event.Attribute.filter(a => a.to_ids);
  1070→    
  1071→    const strings: YARAString[] = [];
  1072→    const meta: YARAMeta = {
  1073→      description: event.info,
  1074→      date: event.date,
  1075→      misp_event_id: event.id,
  1076→      misp_event_uuid: event.uuid,
  1077→      author: event.orgc?.name || 'Algeria SOC',
  1078→    };
  1079→
  1080→    // Extract hash-based strings
  1081→    if (options?.includeHashes !== false) {
  1082→      for (const attr of iocs.filter(a => ['md5', 'sha1', 'sha256'].includes(a.type))) {
  1083→        strings.push({
  1084→          name: `$hash_${attr.type}`,
  1085→          value: attr.value,
  1086→          type: 'text',
  1087→          fullword: true,
  1088→          ascii: true,
  1089→        });
  1090→      }
  1091→    }
  1092→
  1093→    // Extract domain/hostname strings
  1094→    if (options?.includeDomains !== false) {
  1095→      for (const attr of iocs.filter(a => ['domain', 'hostname'].includes(a.type))) {
  1096→        strings.push({
  1097→          name: `$domain_${strings.filter(s => s.name.startsWith('$domain')).length}`,
  1098→          value: attr.value,
  1099→          type: 'text',
  1100→          nocase: true,
  1101→          fullword: true,
  1102→          ascii: true,
  1103→        });
  1104→      }
  1105→    }
  1106→
  1107→    // Extract URL patterns
  1108→    for (const attr of iocs.filter(a => a.type === 'url')) {
  1109→      strings.push({
  1110→        name: `$url_${strings.filter(s => s.name.startsWith('$url')).length}`,
  1111→        value: attr.value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'),
  1112→        type: 'regex',
  1113→        nocase: true,
  1114→        ascii: true,
  1115→      });
  1116→    }
  1117→
  1118→    // Build condition
  1119→    const hashNames = strings.filter(s => s.name.startsWith('$hash')).map(s => s.name).join(' or ');
  1120→    const domainNames = strings.filter(s => s.name.startsWith('$domain')).map(s => s.name).join(' or ');
  1121→    const urlNames = strings.filter(s => s.name.startsWith('$url')).map(s => s.name).join(' or ');
  1122→
  1123→    const conditions: string[] = [];
  1124→    if (hashNames) conditions.push(`any of (${hashNames})`);
  1125→    if (domainNames) conditions.push(`any of (${domainNames})`);
  1126→    if (urlNames) conditions.push(`any of (${urlNames})`);
  1127→
  1128→    const condition = conditions.join(' or ') || 'false';
  1129→
  1130→    // Generate rule name
  1131→    const safeName = event.info
  1132→      .replace(/[^a-zA-Z0-9_]/g, '_')
  1133→      .substring(0, 50)
  1134→      .replace(/_+$/, '');
  1135→
  1136→    const rule: YARARule = {
  1137→      name: `misp_${safeName}_${event.id}`,
  1138→      condition,
  1139→      strings,
  1140→      meta,
  1141→      scopes: ['file'],
  1142→      source: this.formatYARARule(rule),
  1143→    };
  1144→
  1145→    return rule;
  1146→  }
  1147→
  1148→  /**
  1149→   * Format YARA rule as text
  1150→   */
  1151→  private formatYARARule(rule: YARARule): string {
  1152→    const lines: string[] = [
  1153→      `rule ${rule.name} {`,
  1154→      '    meta:',
  1155→    ];
  1156→
  1157→    // Meta section
  1158→    for (const [key, value] of Object.entries(rule.meta)) {
  1159→      if (Array.isArray(value)) {
  1160→        lines.push(`        ${key} = "${value.join(', ')}"`);
  1161→      } else if (value) {
  1162→        lines.push(`        ${key} = "${value}"`);
  1163→      }
  1164→    }
  1165→
  1166→    // Strings section
  1167→    lines.push('    strings:');
  1168→    for (const str of rule.strings) {
  1169→      let modifiers = '';
  1170→      if (str.nocase) modifiers += ' nocase';
  1171→      if (str.wide) modifiers += ' wide';
  1172→      if (str.fullword) modifiers += ' fullword';
  1173→      if (str.ascii) modifiers += ' ascii';
  1174→
  1175→      switch (str.type) {
  1176→        case 'text':
  1177→          lines.push(`        ${str.name} = "${str.value}"${modifiers}`);
  1178→          break;
  1179→        case 'hex':
  1180→          lines.push(`        ${str.name} = { ${str.value} }${modifiers}`);
  1181→          break;
  1182→        case 'regex':
  1183→          lines.push(`        ${str.name} = /${str.value}/${modifiers}`);
  1184→          break;
  1185→      }
  1186→    }
  1187→
  1188→    // Condition
  1189→    lines.push(`    condition:`);
  1190→    lines.push(`        ${rule.condition}`);
  1191→    lines.push('}');
  1192→
  1193→    return lines.join('\n');
  1194→  }

  1279→    // Calculate statistics
  1280→    const stats: MISPStatistics = {
  1281→      events: {
  1282→        total: events.length,
  1283→        published: events.filter(e => e.published).length,
  1284→        unpublished: events.filter(e => !e.published).length,
  1285→        today: events.filter(e => e.date === new Date().toISOString().split('T')[0]).length,
  1286→        this_week: events.filter(e => {
  1287→          const d = new Date(e.date);
  1288→          const weekAgo = new Date();
  1289→          weekAgo.setDate(weekAgo.getDate() - 7);
  1290→          return d >= weekAgo;
  1291→        }).length,
  1292→        this_month: events.filter(e => {
  1293→          const d = new Date(e.date);
  1294→          const monthAgo = new Date();
  1295→          monthAgo.setMonth(monthAgo.getMonth() - 1);
  1296→          return d >= monthAgo;
  1297→        }).length,
  1298→        by_threat_level: {},
  1299→        by_analysis_status: {
  1300→          [AnalysisStatus.INITIAL]: 0,
  1301→          [AnalysisStatus.ONGOING]: 0,
  1302→          [AnalysisStatus.COMPLETED]: 0,
  1303→        },
  1304→      },
  1305→      attributes: {
  1306→        total: attributes.length,
  1307→        with_ids: attributes.filter(a => a.to_ids).length,
  1308→        unique_types: {},
  1309→        unique_categories: {},
  1310→      },
  1311→      correlations: {
  1312→        total: 0,
  1313→        events_with_correlations: 0,
  1314→        avg_correlations_per_event: 0,
  1315→      },
  1316→      organizations: {
  1317→        local: 0,
  1318→        external: 0,
  1319→        contributing: 0,
  1320→      },
  1321→      feeds: {
  1322→        active: 0,
  1323→        total_events_imported: 0,
  1324→        last_fetch_times: {},
  1325→      },
  1326→      threats: {
  1327→        top_actors: [],
  1328→        top_mitres: [],
  1329→        trending_iocs: [],
  1330→      },
  1331→    };
  1332→
  1333→    // Aggregate event statistics
  1334→    for (const event of events) {
  1335→      // By threat level
  1336→      const tl = event.threat_level_id;
  1337→      stats.events.by_threat_level[tl] = (stats.events.by_threat_level[tl] || 0) + 1;
  1338→
  1339→      // By analysis status
  1340→      stats.events.by_analysis_status[event.analysis] =
  1341→        (stats.events.by_analysis_status[event.analysis] || 0) + 1;
  1342→
  1343→      // Track organizations
  1344→      if (event.owner_org?.local) {
  1345→        stats.organizations.local++;
  1346→      } else {
  1347→        stats.organizations.external++;
  1348→      }
  1349→
  1350→      // Extract threat actor tags
  1351→      for (const tag of event.Tag || []) {
  1352→        if (tag.name.startsWith('misp-galaxy:intrusion-set="')) {
  1353→          const actor = tag.name.match(/intrusion-set="([^"]+)"/)?.[1];
  1354→          if (actor) {
  1355→            const existing = stats.threats.top_actors.find(t => t.name === actor);
  1356→            if (existing) {
  1357→              existing.count++;
  1358→            } else {
  1359→              stats.threats.top_actors.push({ name: actor, count: 1 });
  1360→            }
  1361→          }
  1362→        }
  1363→      }
  1364→    }
  1365→
  1366→    // Aggregate attribute statistics
  1367→    for (const attr of attributes) {
  1368→      stats.attributes.unique_types[attr.type] =
  1369→        (stats.attributes.unique_types[attr.type] || 0) + 1;
  1370→      stats.attributes.unique_categories[attr.category] =
  1371→        (stats.attributes.unique_categories[attr.category] || 0) + 1;
  1372→    }
  1373→
  1374→    // Sort threats
  1375→    stats.threats.top_actors.sort((a, b) => b.count - a.count);
  1376→    stats.threats.top_actors = stats.threats.top_actors.slice(0, 10);
  1377→
  1378→    // Top trending IOCs (most common values)
  1379→    const iocCounts: Record<string, { type: string; count: number }> = {};
  1380→    for (const attr of attributes) {
  1381→      const key = `${attr.type}:${attr.value}`;
  1382→      if (!iocCounts[key]) {
  1383→        iocCounts[key] = { type: attr.type, count: 0 };
  1384→      }
  1385→      iocCounts[key].count++;
  1386→    }
  1387→    stats.threats.trending_iocs = Object.entries(iocCounts)
  1388→      .map(([key, val]) => ({ type: val.type, value: key.split(':')[1], count: val.count }))
  1389→      .sort((a, b) => b.count - a.count)
  1390→      .slice(0, 20);
  1391→
  1392→    return stats;
  1393→  }

  1396→   * Get event timeline data
  1397→   */
  1398→  async getTimeline(days: number = 30): Promise<TimelinePoint[]> {
  1399→    this.log('timeline', `Getting timeline for ${days} days`);
  1400→
  1401→    const events = await this.searchEvents({ last: `${days}d`, limit: 5000 });
  1402→    const eventList = events.response.MISPEvent || [];
  1403→
  1404→    // Group by date
  1405→    const dailyData: Record<string, { events: number; attributes: number; published: number }> = {};
  1406→
  1407→    for (let i = 0; i < days; i++) {
  1408→      const date = new Date();
  1409→      date.setDate(date.getDate() - i);
  1410→      const dateStr = date.toISOString().split('T')[0];
  1411→      dailyData[dateStr] = { events: 0, attributes: 0, published: 0 };
  1412→    }
  1413→
  1414→    for (const event of eventList) {
  1415→      const dateStr = event.date;
  1416→      if (dailyData[dateStr]) {
  1417→        dailyData[dateStr].events++;
  1418→        dailyData[dateStr].attributes += (event.Attribute || []).length;
  1419→        if (event.published) {
  1420→          dailyData[dateStr].published++;
  1421→        }
  1422→      }
  1423→    }
  1424→
  1425→    return Object.entries(dailyData)
  1426→      .map(([date, counts]) => ({
  1427→        date,
  1428→        ...counts,
  1429→      }))
  1430→      .sort((a, b) => a.date.localeCompare(b.date));
  1431→  }

  1440→  async exportEvent(eventId: string, format: 'json' | 'xml' | 'stix' | 'stix2' | 'text' | 'csv' | 'yara'): Promise<MISPExportResult> {
  1441→    this.log('exportEvent', `Exporting event ${eventId} as ${format}`);
  1442→
  1443→    const endpointMap: Record<string, string> = {
  1444→      json: `/events/view/${eventId}`,
  1445→      xml: `/events/xml/download/${eventId}`,
  1446→      stix: `/events/stix/download/${eventId}`,
  1447→      stix2: `/events/stix2/download/${eventId}`,
  1448→      text: `/events/text/download/${eventId}`,
  1449→      csv: `/events/csv/download/${eventId}`,
  1450→    };
  1451→
  1452→    let content: string;
  1453→
  1454→    if (format === 'yara') {
  1455→      const rule = await this.generateYARAFromEvent(eventId);
  1456→      content = rule.source;
  1457→    } else {
  1458→      const response = await fetch(`${this.config.url}${endpointMap[format]}`, {
  1459→        headers: { Authorization: this.authHeader },
  1460→      });
  1461→      content = await response.text();
  1462→    }
  1463→
  1464→    return {
  1465→      success: true,
  1466→      format,
  1467→      filename: `misp_event_${eventId}.${format}`,
  1468→      size: content.length,
  1469→      content,
  1470→      event_count: 1,
  1471→      attribute_count: 0,
  1472→    };
  1473→  }

  1570→  /**
  1571→   * Create alert-to-event conversion
  1572→   * Convert security alert into structured MISP event
  1573→   */
  1574→  async createEventFromAlert(alert: {
  1575→    title: string;
  1576→    description: string;
  1577→    severity: 'critical' | 'high' | 'medium' | 'low';
  1578→    source: string;
  1579→    iocs: Array<{
  1580→      type: AttributeType;
  1581→      value: string;
  1582→      category: string;
  1583→    }>;
  1584→    tags?: string[];
  1585→    references?: string[];
  1586→  }): Promise<MISPEvent> {
  1587→    this.log('createEventFromAlert', `Creating event from alert: ${alert.title}`);
  1588→
  1589→    // Map severity to threat level
  1590→    const severityToThreatLevel: Record<string, number> = {
  1591→      critical: 1,
  1592→      high: 2,
  1593→      medium: 3,
  1594→      low: 4,
  1595→    };
  1596→
  1597→    const event = await this.createEvent({
  1598→      info: `[ALERT] ${alert.title}`,
  1599→      threat_level_id: severityToThreatLevel[alert.severity] || 4,
  1600→      analysis: AnalysisStatus.ONGOING,
  1601→      distribution: DistributionLevel.CONNECTED_COMMUNITIES,
  1602→      tags: alert.tags || [],
  1603→    });
  1604→
  1605→    // Add IOCs as attributes
  1606→    if (alert.iocs.length > 0) {
  1607→      await this.addAttributesBatch(event.id, alert.iocs.map(ioc => ({
  1608→        type: ioc.type,
  1609→        value: ioc.value,
  1610→        category: ioc.category,
  1611→        to_ids: true,
  1612→        comment: `Source: ${alert.source}`,
  1613→      })));
  1614→    }
  1615→
  1616→    // Add description as comment attribute
  1617→    if (alert.description) {
  1618→      await this.addAttribute(event.id, {
  1619→        type: 'comment',
  1620→        value: alert.description.substring(0, 255),
  1621→        to_ids: false,
  1622→      });
  1623→    }
  1624→
  1625→    return event;
  1626→  }

  1628→  /**
  1629→   * Find correlated events for given IOCs
  1630→   */
  1631→  async findCorrelatedEvents(iocs: Array<{ type: AttributeType; value: string }>): Promise<Map<string, MISPEvent[]>> {
  1632→    this.log('findCorrelatedEvents', `Finding correlations for ${iocs.length} IOCs`);
  1633→
  1634→    const correlations = new Map<string, MISPEvent[]>();
  1635→
  1636→    for (const ioc of iocs) {
  1637→      try {
  1638→        const result = await this.searchAttributes({
  1639→          value: ioc.value,
  1640→          type: ioc.type,
  1641→          includeCorrelations: true,
  1642→          limit: 50,
  1643→        });
  1644→
  1645→        const events: MISPEvent[] = [];
  1646→        for (const attr of result.response.Attribute || []) {
  1647→          if (attr.correlating_value) {
  1648→            for (const corr of attr.correlating_value) {
  1649→              // We'd need to fetch full events here, but for now just track IDs
  1650→              events.push({ id: corr.event_id, info: corr.event_info } as MISPEvent);
  1651→            }
  1652→          }
  1653→        }
  1654→
  1655→        correlations.set(`${ioc.type}:${ioc.value}`, events);
  1656→      } catch (error) {
  1657→        this.log('findCorrelatedEvents', `Failed to correlate ${ioc.type}:${ioc.value}`, error);
  1658→        correlations.set(`${ioc.type}:${ioc.value}`, []);
  1659→      }
  1660→    }
  1661→
  1662→    return correlations;
  1663→  }
