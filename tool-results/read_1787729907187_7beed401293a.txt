   214→def create_section_table(data, col_widths, styles):
   215→    """Create a styled table with CyberSOC theme"""
   216→    table = Table(data, colWidths=col_widths, repeatRows=1)
   217→    
   218→    style_commands = [
   219→        ('BACKGROUND', (0, 0), (-1, 0), HEADER_FILL),
   220→        ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
   221→        ('FONTNAME', (0, 0), (-1, 0), 'NotoSerifSC-Bold'),
   222→        ('FONTSIZE', (0, 0), (-1, 0), 9),
   223→        ('ALIGN', (0, 0), (-1, 0), 'CENTER'),
   224→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   225→        ('FONTNAME', (0, 1), (-1, -1), 'NotoSerifSC'),
   226→        ('FONTSIZE', (0, 1), (-1, -1), 8),
   227→        ('ROWBACKGROUNDS', (0, 1), (-1, -1), [CARD_BG, TABLE_STRIPE]),
   228→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   229→        ('TOPPADDING', (0, 0), (-1, -1), 6),
   230→        ('BOTTOMPADDING', (0, 0), (-1, -1), 6),
   231→        ('LEFTPADDING', (0, 0), (-1, -1), 8),
   232→        ('RIGHTPADDING', (0, 0), (-1, -1), 8),
   233→    ]
   234→    
   235→    table.setStyle(TableStyle(style_commands))
