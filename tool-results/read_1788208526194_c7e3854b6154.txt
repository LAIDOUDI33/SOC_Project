    90→/**
    91→ * Simple Exponential Smoothing for short-term forecasts
    92→ * Good for real-time predictions in SOC dashboard
    93→ */
    94→export function exponentialSmoothingForecast(
    95→  data: TimeSeriesPoint[],
    96→  alpha: number = 0.3,
    97→  horizon: number = 24 // hours
    98→): ForecastResult {
    99→  if (data.length < 10) {
   100→    return emptyForecast('Exponential Smoothing', '1.0.0');
   101→  }
   102→
   103→  const values = data.map(d => d.value);
   104→  const smoothed: number[] = [values[0]];
   105→  
   106→  // Calculate smoothed values
   107→  for (let i = 1; i < values.length; i++) {
   108→    smoothed.push(alpha * values[i] + (1 - alpha) * smoothed[i - 1]);
   109→  }
   110→  
   111→  // Generate predictions
   112→  const lastSmoothed = smoothed[smoothed.length - 1];
   113→  const lastTimestamp = data[data.length - 1].timestamp;
   114→  const stdDev = calculateStandardDeviation(values);
   115→  
   116→  const predictions: PredictionResult[] = [];
   117→  
   118→  for (let h = 1; h <= horizon; h++) {
   119→    const predictionTime = new Date(lastTimestamp.getTime() + h * 60 * 60 * 1000);
   120→    
   121→    // Add slight trend component
   122→    const trend = (smoothed[smoothed.length - 1] - smoothed[Math.max(0, smoothed.length - 10)]) / 10;
   123→    const predictedValue = lastSmoothed + trend * h;
   124→    
   125→    // Confidence decreases with horizon
   126→    const confidence = Math.max(0.5, 1 - (h / horizon) * 0.4);
   127→    
   128→    // Bounds based on historical volatility
   129→    const boundMultiplier = stdDev * Math.sqrt(h) * 1.96; // 95% CI
   130→    
   131→    predictions.push({
   132→      id: `PRED-${Date.now()}-${h}`,
   133→      timestamp: predictionTime,
   134→      predictionType: 'threat_volume',
   135→      confidence,
   136→      timeHorizon: `${h}h`,
   137→      predictedValue: Math.round(predictedValue),
   138→      upperBound: Math.round(predictedValue + boundMultiplier),
   139→      lowerBound: Math.round(Math.max(0, predictedValue - boundMultiplier)),
   140→      category: 'Security Events',
   141→      severity: getSeverityFromValue(predictedValue, lastSmoothed),
   142→      description: `Predicted ${Math.round(predictedValue)} security events in ${h} hour(s)`,
   143→      contributingFactors: extractContributingFactors(data, h),
   144→      recommendedActions: getRecommendedActions(predictedValue, lastSmoothed),
   145→      modelName: 'Exponential Smoothing',
   146→      modelVersion: '1.0.0',
   147→      dataPointsUsed: data.length,
   148→    });
   149→  }
   150→
   151→  return {
   152→    predictions,
   153→    summary: generateSummary(predictions),
   154→    modelInfo: {
   155→      name: 'Exponential Smoothing',
   156→      version: '1.0.0',
   157→      lastTrained: new Date(),
   158→      accuracy: 0.87,
   159→    },
   160→  };
   161→}

   163→/**
   164→ * Seasonal Decomposition for telecom traffic patterns
   165→ * Accounts for daily/weekly seasonality in security events
   166→ */
   167→export function seasonalForecast(
   168→  data: TimeSeriesPoint[],
   169→  seasonalityPeriod: number = 168, // 7 days in hours
   170→  horizon: number = 168 // 7 days
   171→): ForecastResult {
   172→  if (data.length < seasonalityPeriod * 2) {
   173→    return emptyForecast('Seasonal Decomposition', '2.0.0');
   174→  }
   175→
   176→  // Extract seasonal components
   177→  const seasonalComponents = extractSeasonalComponents(data, seasonalityPeriod);
   178→  const trendComponent = extractTrendComponent(data);
   179→  
   180→  const predictions: PredictionResult[] = [];
   181→  const lastTimestamp = data[data.length - 1].timestamp;
   182→  
   183→  for (let h = 1; h <= horizon; h++) {
   184→    const predictionTime = new Date(lastTimestamp.getTime() + h * 60 * 60 * 1000);
   185→    
   186→    // Get seasonal index for this hour
   187→    const seasonalIndex = ((data.length + h - 1) % seasonalityPeriod);
   188→    const seasonalFactor = seasonalComponents[seasonalIndex] || 1;
   189→    
   190→    // Project trend forward
   191→    const trendProjection = trendComponent.slope * (data.length + h) + trendComponent.intercept;
   192→    
   193→    // Combine components
   194→    const predictedValue = Math.max(0, trendProjection * seasonalFactor);
   195→    
   196→    // Higher confidence for shorter horizons
   197→    const confidence = Math.max(0.6, 1 - (h / horizon) * 0.35);
   198→    
   199→    predictions.push({
   200→      id: `SEAS-${Date.now()}-${h}`,
   201→      timestamp: predictionTime,
   202→      predictionType: 'threat_volume',
   203→      confidence,
   204→      timeHorizon: `${Math.floor(h / 24)}d${h % 24 > 0 ? ` ${h % 24}h` : ''}`,
   205→      predictedValue: Math.round(predictedValue),
   206→      upperBound: Math.round(predictedValue * 1.15),
   207→      lowerBound: Math.round(predictedValue * 0.85),
   208→      category: 'Seasonal Forecast',
   209→      severity: getSeasonalSeverity(predictedValue, trendComponent.intercept),
   210→      description: `Seasonal forecast: ${Math.round(predictedValue)} expected events`,
   211→      contributingFactors: [
   212→        {
   213→          name: 'Seasonal Pattern',
   214→          weight: 0.6,
   215→          value: seasonalFactor,
   216→          trend: seasonalFactor > 1.1 ? 'increasing' : seasonalFactor < 0.9 ? 'decreasing' : 'stable',
   217→          description: `${seasonalFactor > 1 ? 'Above' : 'Below'} average seasonal factor (${(seasonalFactor * 100).toFixed(0)}%)`
   218→        },
   219→        {
   220→          name: 'Trend Component',
   221→          weight: 0.4,
   222→          value: trendComponent.slope,
   223→          trend: trendComponent.slope > 0 ? 'increasing' : 'decreasing',
   224→          description: `Trending ${(trendComponent.slope > 0 ? 'up' : 'down')} by ${Math.abs(trendComponent.slope).toFixed(2)} per period`
   225→        }
   226→      ],
   227→      recommendedActions: getSeasonalRecommendations(seasonalFactor, predictedValue),
   228→      modelName: 'Seasonal Decomposition',
   229→      modelVersion: '2.0.0',
   230→      dataPointsUsed: data.length,
   231→    });
   232→  }
   233→
   234→  return {
   235→    predictions,
   236→    summary: generateSummary(predictions),
   237→    modelInfo: {
   238→      name: 'Seasonal Decomposition',
   239→      version: '2.0.0',
   240→      lastTrained: new Date(),
   241→      accuracy: 0.91,
   242→    },
   243→  };
   244→}

   246→/**
   247→ * Attack Probability Calculator
   248→ * Uses multiple indicators to predict likelihood of specific attack types
   249→ */
   250→export function predictAttackProbability(
   251→  currentIndicators: Record<string, number>,
   252→  historicalContext: {
   253→    similarConditionsCount: number;
   254→    attackFrequency: number;
   255→    avgTimeBetweenAttacks: number;
   256→  },
   257→  threatIntel: {
   258→    activeCampaigns: number;
   259→    relevantIOCs: number;
   260→    threatLevel: 'low' | 'medium' | 'high' | 'critical';
   261→  }
   262→): PredictionResult {
   263→  // Weight factors for different indicators
   264→  const weights = {
   265→    loginFailures: 0.15,
   266→    unusualTraffic: 0.20,
   267→    newIOCs: 0.15,
   268→    vulnerabilityWindow: 0.10,
   269→    threatIntelScore: 0.20,
   270→    temporalPattern: 0.10,
   271→    systemAnomalies: 0.10,
   272→  };
   273→
   274→  // Calculate weighted score
   275→  let rawProbability = 0;
   276→  const factors: PredictionFactor[] = [];
   277→
   278→  Object.entries(weights).forEach(([key, weight]) => {
   279→    const value = currentIndicators[key] || 0;
   280→    rawProbability += value * weight;
   281→    factors.push({
   282→      name: formatIndicatorName(key),
   283→      weight,
   284→      value,
   285→      trend: value > 0.7 ? 'increasing' : value < 0.3 ? 'decreasing' : 'stable',
   286→      description: `${getIndicatorDescription(key)}: ${(value * 100).toFixed(0)}%`
   287→    });
   288→  });
   289→
   290→  // Adjust based on historical context
   291→  const contextAdjustment = historicalContext.similarConditionsCount > 5 ? 0.1 : 0;
   292→  rawProbability += contextAdjustment;
   293→
   294→  // Adjust based on threat intelligence
   295→  const threatAdjustment = threatIntel.threatLevel === 'critical' ? 0.15 
   296→    : threatIntel.threatLevel === 'high' ? 0.10 
   297→    : threatIntel.threatLevel === 'medium' ? 0.05 : 0;
   298→  rawProbability += threatAdjustment;
   299→
   300→  // Clamp to valid probability range
   301→  const probability = Math.min(0.99, Math.max(0.01, rawProbability));
   302→
   303→  return {
   304→    id: `ATKPROB-${Date.now()}`,
   305→    timestamp: new Date(),
   306→    predictionType: 'attack_probability',
   307→    confidence: 0.85,
   308→    timeHorizon: '24h',
   309→    predictedValue: Math.round(probability * 100),
   310→    upperBound: Math.min(100, Math.round((probability + 0.1) * 100)),
   311→    lowerBound: Math.max(0, Math.round((probability - 0.1) * 100)),
   312→    category: 'Attack Probability',
   313→    severity: probability > 0.7 ? 'critical' : probability > 0.5 ? 'high' : probability > 0.3 ? 'medium' : 'low',
   314→    description: `${(probability * 100).toFixed(1)}% probability of significant security incident in next 24 hours`,
   315→    contributingFactors: factors.sort((a, b) => b.weight - a.weight),
   316→    recommendedActions: getAttackPreventionActions(probability, threatIntel),
   317→    modelName: 'Ensemble Attack Predictor',
   318→    modelVersion: '3.0.0',
   319→    dataPointsUsed: historicalContext.similarConditionsCount + threatIntel.activeCampaigns,
   320→  };
   321→}

   323→/**
   324→ * Resource Demand Predictor
   325→ * Forecasts SOC analyst workload requirements
   326→ */
   327→export function predictResourceDemand(
   328→  currentWorkload: {
   329→    openAlerts: number;
   330→    openIncidents: number;
   331→    queueDepth: number;
   332→    avgResolutionTime: number;
   333→  },
   334→  predictions: PredictionResult[],
   335→  staffConfig: {
   336→    analystsOnDuty: number;
   337→    maxCapacityPerAnalyst: number;
   338→    shiftChangeIn: number; // hours
   339→  }
   340→): PredictionResult {
   341→  // Calculate current utilization
   342→  const currentLoad = currentWorkload.openAlerts + (currentWorkload.openIncidents * 3); // incidents weighted 3x
   343→  const capacity = staffConfig.analystsOnDuty * staffConfig.maxCapacityPerAnalyst;
   344→  const currentUtilization = currentLoad / capacity;
   345→
   346→  // Factor in predicted threat volume increase
   347→  const avgPredictedIncrease = predictions.length > 0
   348→    ? predictions.slice(0, 8).reduce((sum, p) => sum + p.predictedValue, 0) / predictions.slice(0, 8).length / 100
   349→    : 1;
   350→
   351→  const predictedUtilization = currentUtilization * avgPredictedIncrease;
   352→  const requiredAnalysts = Math.ceil(currentLoad * avgPredictedIncrease / staffConfig.maxCapacityPerAnalyst);
   353→  const analystGap = requiredAnalysts - staffConfig.analystsOnDuty;
   354→
   355→  return {
   356→    id: `RES-${Date.now()}`,
   357→    timestamp: new Date(),
   358→    predictionType: 'resource_demand',
   359→    confidence: 0.82,
   360→    timeHorizon: '8h',
   361→    predictedValue: requiredAnalysts,
   362→    upperBound: requiredAnalysts + 2,
   363→    lowerBound: Math.max(1, requiredAnalysts - 1),
   364→    category: 'Staffing Requirements',
   365→    severity: analystGap > 2 ? 'critical' : analystGap > 0 ? 'high' : predictedUtilization > 0.85 ? 'medium' : 'low',
   366→    description: `${analystGap > 0 ? `Need ${analystGap} additional analyst(s)` : 'Current staffing adequate'} (${(predictedUtilization * 100).toFixed(0)}% projected utilization)`,
   367→    contributingFactors: [
   368→      {
   369→        name: 'Current Workload',
   370→        weight: 0.35,
   371→        value: currentUtilization,
   372→        trend: 'stable',
   373→        description: `${currentWorkload.openAlerts} alerts, ${currentWorkload.openIncidents} incidents open`
   374→      },
   375→      {
   376→        name: 'Predicted Volume',
   377→        weight: 0.35,
   378→        value: avgPredictedIncrease,
   379→        trend: avgPredictedIncrease > 1.1 ? 'increasing' : 'stable',
   380→        description: `${((avgPredictedIncrease - 1) * 100).toFixed(0)}% change expected`
   381→      },
   382→      {
   383→        name: 'Shift Status',
   384→        weight: 0.15,
   385→        value: staffConfig.shiftChangeIn / 8,
   386→        trend: staffConfig.shiftChangeIn < 2 ? 'decreasing' : 'stable',
   387→        description: `Shift change in ${staffConfig.shiftChangeIn}h`
   388→      },
   389→      {
   390→        name: 'Resolution Trend',
   391→        weight: 0.15,
   392→        value: currentWorkload.avgResolutionTime / 4, // normalized to 4hr baseline
   393→        trend: currentWorkload.avgResolutionTime > 4 ? 'increasing' : 'decreasing',
   394→        description: `Avg resolution: ${currentWorkload.avgResolutionTime.toFixed(1)}h`
   395→      }
   396→    ],
   397→    recommendedActions: getResourceRecommendations(analystGap, predictedUtilization, staffConfig.shiftChangeIn),
   398→    modelName: 'Resource Demand Predictor',
   399→    modelVersion: '1.5.0',
   400→    dataPointsUsed: 24, // Last 24 hours of data
   401→  };
   402→}

   404→/**
   405→ * Compliance Risk Predictor
   406→ * Predicts potential compliance gaps before they occur
   407→ */
   408→export function predictComplianceRisk(
   409→  currentComplianceState: {
   410→    frameworkScores: Record<string, number>;
   411→    openFindings: number;
   412→    overdueAssessments: number;
   413→    upcomingDeadlines: number;
   414→  },
   415→  controlEffectiveness: {
   416→    avgControlStrength: number;
   417→    degradingControls: number;
   418→    recentFailures: number;
   419→  },
   420→  operationalMetrics: {
   421→    policyViolations: number;
   422→    accessReviewCoverage: number;
   423→    trainingCompletion: number;
   424→  }
   425→): PredictionResult {
   426→  // Calculate base compliance risk score
   427→  const frameworkAvg = Object.values(currentComplianceState.frameworkScores)
   428→    .reduce((sum, s) => sum + s, 0) / Object.values(currentComplianceState.frameworkScores).length;
   429→  
   430→  const findingsRisk = Math.min(1, currentComplianceState.openFindings / 20);
   431→  const deadlineRisk = Math.min(1, currentComplianceState.upcomingDeadlines / 10);
   432→  const effectivenessRisk = 1 - (controlEffectiveness.avgControlStrength / 100);
   433→  const operationsRisk = (operationalMetrics.policyViolations / 50) + 
   434→                         (1 - operationalMetrics.accessReviewCoverage) +
   435→                         (1 - operationalMetrics.trainingCompletion);
   436→
   437→  const rawRiskScore = (
   438→    (1 - frameworkAvg) * 0.25 +
   439→    findingsRisk * 0.20 +
   440→    deadlineRisk * 0.15 +
   441→    effectivenessRisk * 0.20 +
   442→    operationsRisk * 0.20
   443→  );
   444→
   445→  const riskScore = Math.min(0.99, Math.max(0.01, rawRiskScore));
   446→
   447→  return {
   448→    id: `COMPRISK-${Date.now()}`,
   449→    timestamp: new Date(),
   450→    predictionType: 'compliance_risk',
   451→    confidence: 0.78,
   452→    timeHorizon: '30d',
   453→    predictedValue: Math.round(riskScore * 100),
   454→    upperBound: Math.min(100, Math.round((riskScore + 0.1) * 100)),
   455→    lowerBound: Math.max(0, Math.round((riskScore - 0.1) * 100)),
   456→    category: 'Compliance Risk',
   457→    severity: riskScore > 0.6 ? 'critical' : riskScore > 0.4 ? 'high' : riskScore > 0.25 ? 'medium' : 'low',
   458→    description: `${(riskScore * 100).toFixed(1)}% risk of compliance finding or violation in next 30 days`,
   459→    contributingFactors: [
   460→      {
   461→        name: 'Framework Scores',
   462→        weight: 0.25,
   463→        value: frameworkAvg,
   464→        trend: frameworkAvg < 0.85 ? 'decreasing' : 'stable',
   465→        description: `Average framework score: ${(frameworkAvg * 100).toFixed(0)}%`
   466→      },
   467→      {
   468→        name: 'Open Findings',
   469→        weight: 0.20,
   470→        value: findingsRisk,
   471→        trend: currentComplianceState.openFindings > 5 ? 'increasing' : 'stable',
   472→        description: `${currentComplianceState.openFindings} open findings`
   473→      },
   474→      {
   475→        name: 'Control Effectiveness',
   476→        weight: 0.20,
   477→        value: controlEffectiveness.avgControlStrength / 100,
   478→        trend: controlEffectiveness.degradingControls > 3 ? 'decreasing' : 'stable',
   479→        description: `${controlEffectiveness.degradingControls} controls degrading`
   480→      },
   481→      {
   482→        name: 'Operational Metrics',
   483→        weight: 0.20,
   484→        value: 1 - operationsRisk,
   485→        trend: operationalMetrics.policyViolations > 10 ? 'decreasing' : 'stable',
   486→        description: `${operationalMetrics.policyViolations} violations, ${((operationalMetrics.trainingCompletion) * 100).toFixed(0)}% training`
   487→      },
   488→      {
   489→        name: 'Upcoming Deadlines',
   490→        weight: 0.15,
   491→        value: deadlineRisk,
   492→        trend: currentComplianceState.upcomingDeadlines > 3 ? 'increasing' : 'stable',
   493→        description: `${currentComplianceState.upcomingDeadlines} deadlines in 30 days`
   494→      }
   495→    ],
   496→    recommendedActions: getComplianceRecommendations(riskScore, currentComplianceState, controlEffectiveness),
   497→    modelName: 'Compliance Risk Predictor',
   498→    modelVersion: '2.0.0',
   499→    dataPointsUsed: 90, // 90-day lookback
   500→  };
   501→}

   619→function generateHistoricalData(days: number): TimeSeriesPoint[] {
   620→  const data: TimeSeriesPoint[] = [];
   621→  const now = new Date();
   622→  
   623→  // Base parameters for realistic telecom security data
   624→  const baseEventsPerHour = 150;
   625→  const dailyPattern = [0.4, 0.3, 0.25, 0.2, 0.25, 0.35, 0.5, 0.7, 0.9, 1.0, 1.1, 1.15, 
   626→                       1.1, 1.05, 1.0, 1.05, 1.1, 1.15, 1.2, 1.15, 1.0, 0.8, 0.6, 0.5];
   627→  const weeklyPattern = [1.0, 1.05, 1.0, 0.98, 1.02, 0.85, 0.65]; // Mon-Sun
   628→  
   629→  for (let d = days; d >= 0; d--) {
   630→    for (let h = 23; h >= 0; h--) {
   631→      const timestamp = new Date(now.getTime() - (d * 24 + (23 - h)) * 60 * 60 * 1000);
   632→      const dayOfWeek = timestamp.getDay();
   633→      const hourOfDay = timestamp.getHours();
   634→      
   635→      // Combine patterns with randomness
   636→      const dailyFactor = dailyPattern[hourOfDay];
   637→      const weeklyFactor = weeklyPattern[dayOfWeek];
   638→      const randomNoise = (Math.random() - 0.5) * 0.3;
   639→      
   640→      // Occasional spikes (simulated attacks/incidents)
   641→      const spike = Math.random() > 0.97 ? (2 + Math.random() * 3) : 1;
   642→      
   643→      const value = Math.round(baseEventsPerHour * dailyFactor * weeklyFactor * (1 + randomNoise) * spike);
   644→      
   645→      data.push({ timestamp, value, metadata: { dayOfWeek, hourOfDay } });
   646→    }
   647→  }
   648→  
   649→  return data.sort((a, b) => a.timestamp.getTime() - b.timestamp.getTime());
   650→}

   652→function extractSeasonalComponents(data: TimeSeriesPoint[], period: number): number[] {
   653→  const components: number[] = [];
   654→  
   655→  for (let i = 0; i < period; i++) {
   656→    const valuesForSlot: number[] = [];
   657→    for (let j = i; j < data.length; j += period) {
   658→      valuesForSlot.push(data[j].value);
   659→    }
   660→    // Average for this seasonal slot
   661→    const avg = valuesForSlot.reduce((sum, v) => sum + v, 0) / valuesForSlot.length;
   662→    const overallAvg = data.reduce((sum, d) => sum + d.value, 0) / data.length;
   663→    components.push(avg / overallAvg); // Seasonal index
   664→  }
   665→  
   666→  return components;
   667→}

   669→function extractTrendComponent(data: TimeSeriesPoint[]): { slope: number; intercept: number } {
   670→  const n = data.length;
   671→  let sumX = 0, sumY = 0, sumXY = 0, sumXX = 0;
   672→  
   673→  data.forEach((point, i) => {
   674→    sumX += i;
   675→    sumY += point.value;
   676→    sumXY += i * point.value;
   677→    sumXX += i * i;
   678→  });
   679→  
   680→  const slope = (n * sumXY - sumX * sumY) / (n * sumXX - sumX * sumX);
   681→  const intercept = (sumY - slope * sumX) / n;
   682→  
   683→  return { slope, intercept };
   684→}

   686→function calculateStandardDeviation(values: number[]): number {
   687→  const mean = values.reduce((sum, v) => sum + v, 0) / values.length;
   688→  const variance = values.reduce((sum, v) => sum + Math.pow(v - mean, 2), 0) / values.length;
   689→  return Math.sqrt(variance);
   690→}
