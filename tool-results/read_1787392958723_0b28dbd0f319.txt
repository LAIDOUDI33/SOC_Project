   436→// Utility function
   437→function formatTimeAgo(date: Date): string {
   438→  const seconds = Math.floor((Date.now() - date.getTime()) / 1000)
   439→  
   440→  if (seconds < 60) return `${seconds}s ago`
   441→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`
   442→  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`
   443→  return `${Math.floor(seconds / 86400)}d ago`
   444→}

   459→// Main Analyst Workspace Component
   460→export default function AnalystWorkspace() {
   461→  const [alerts, setAlerts] = useState<Alert[]>([])
   462→  const [filteredAlerts, setFilteredAlerts] = useState<Alert[]>([])
   463→  const [selectedAlertId, setSelectedAlertId] = useState<string | null>(null)
   464→  const [selectedAlerts, setSelectedAlerts] = useState<Set<string>>(new Set())
   465→  const [viewMode, setViewMode] = useState<'grid' | 'list'>('list')
   466→  const [showShortcuts, setShowShortcuts] = useState(false)
   467→  const [showNotifications, setShowNotifications] = useState(false)
   468→  const [isLoading, setIsLoading] = useState(true)
   469→
   470→  // Initialize data
   471→  useEffect(() => {
   472→    setIsLoading(true)
   473→    const data = generateAlerts()
   474→    setAlerts(data)
   475→    setFilteredAlerts(data)
   476→    setTimeout(() => setIsLoading(false), 300)
   477→  }, [])
   478→
   479→  // Handle filter changes
   480→  const handleFilterChange = useCallback((filters: Partial<FilterState>) => {
   481→    let filtered = [...alerts]
   482→
   483→    if (filters.severity?.length) {
   484→      filtered = filtered.filter(a => filters.severity!.includes(a.severity))
   485→    }
   486→
   487→    if (filters.status?.length) {
   488→      filtered = filtered.filter(a => filters.status!.includes(a.status))
   489→    }
   490→
   491→    if (filters.searchQuery) {
   492→      const query = filters.searchQuery.toLowerCase()
   493→      filtered = filtered.filter(a => 
   494→        a.title.toLowerCase().includes(query) ||
   495→        a.description.toLowerCase().includes(query) ||
   496→        a.id.toLowerCase().includes(query)
   497→      )
   498→    }
   499→
   500→    setFilteredAlerts(filtered)
   501→  }, [alerts])
   502→
   503→  // Handle quick actions
   504→  const handleQuickAction = (alertId: string, action: string) => {
   505→    setAlerts(prev => prev.map(alert => {
   506→      if (alert.id === alertId) {
   507→        switch (action) {
   508→          case 'acknowledge':
   509→            return { ...alert, status: 'acknowledged' as const }
   510→          case 'escalate':
   511→            return { ...alert, status: 'investigating' as const }
   512→          case 'close':
   513→            return { ...alert, status: 'closed' as const }
   514→          default:
   515→            return alert
   516→        }
   517→      }
   518→      return alert
   519→    }))
   520→  }
   521→
   522→  // Toggle bookmark
   523→  const toggleBookmark = (alertId: string) => {
   524→    setAlerts(prev => prev.map(alert =>
   525→      alert.id === alertId ? { ...alert, isBookmarked: !alert.isBookmarked } : alert
   526→    ))
   527→  }
   528→
   529→  // Bulk actions
   530→  const handleBulkAction = (action: string) => {
   531→    selectedAlerts.forEach(id => {
   532→      handleQuickAction(id, action)
   533→    })
   534→    setSelectedAlerts(new Set())
   535→  }
   536→
   537→  // Keyboard shortcuts handler
   538→  useEffect(() => {
   539→    const handleKeyDown = (e: KeyboardEvent) => {
   540→      // Ignore if typing in input
   541→      if ((e.target as HTMLElement).tagName === 'INPUT' || (e.target as HTMLElement).tagName === 'TEXTAREA') {
   542→        return
   543→      }
   544→
   545→      if (e.key === '?' && !e.metaKey && !e.ctrlKey) {
   546→        e.preventDefault()
   547→        setShowShortcuts(prev => !prev)
   548→        return
   549→      }
   550→
   551→      if (e.key === '/') {
   552→        e.preventDefault()
   553→        document.getElementById('alert-search')?.focus()
   554→        return
   555→      }
   556→
   557→      if (selectedAlertId) {
   558→        if (e.key === 'a') handleQuickAction(selectedAlertId, 'acknowledge')
   559→        if (e.key === 'e') handleQuickAction(selectedAlertId, 'escalate')
   560→        if (e.key === 'c') handleQuickAction(selectedAlertId, 'close')
   561→        if (e.key === 'b') toggleBookmark(selectedAlertId)
   562→      }
   563→    }
   564→
   565→    window.addEventListener('keydown', handleKeyDown)
   566→    return () => window.removeEventListener('keydown', handleKeyDown)
   567→  }, [selectedAlertId])
   568→
   569→  const selectedAlert = alerts.find(a => a.id === selectedAlertId) || null
   570→
   571→  if (isLoading) {
   572→    return (
   573→      <div className="min-h-screen bg-slate-950 p-6 flex items-center justify-center">
   574→        <RefreshCw className="h-12 w-12 animate-spin text-blue-500" />
   575→      </div>
   576→    )
   577→  }
   578→
   579→  return (
   580→    <div className="min-h-screen bg-slate-950">
   581→      {/* Header */}
   582→      <header className="border-b border-slate-800 bg-slate-900/50 backdrop-blur-sm sticky top-0 z-40">
   583→        <div className="px-6 py-4">
   584→          <div className="flex items-center justify-between">
   585→            <div>
   586→              <h1 className="text-2xl font-bold text-white flex items-center gap-3">
   587→                <Zap className="h-7 w-7 text-yellow-400" />
   588→                Analyst Workspace
   589→              </h1>
   590→              <p className="text-slate-400 text-sm mt-1">
   591→                Unified Security Operations Center
   592→              </p>
   593→            </div>
   594→            
   595→            <div className="flex items-center gap-3">
   596→              {/* View Mode Toggle */}
   597→              <div className="flex bg-slate-800 rounded-lg p-1">
   598→                <Button
   599→                  size="sm"
   600→                  variant={viewMode === 'list' ? 'default' : 'ghost'}
   601→                  className={viewMode === 'list' ? 'bg-slate-700' : 'text-slate-400'}
   602→                  onClick={() => setViewMode('list')}
   603→                >
   604→                  <List className="h-4 w-4" />
   605→                </Button>
   606→                <Button
   607→                  size="sm"
   608→                  variant={viewMode === 'grid' ? 'default' : 'ghost'}
   609→                  className={viewMode === 'grid' ? 'bg-slate-700' : 'text-slate-400'}
   610→                  onClick={() => setViewMode('grid')}
   611→                >
   612→                  <LayoutGrid className="h-4 w-4" />
   613→                </Button>
   614→              </div>
   615→
   616→              {/* Notifications */}
   617→              <Button
   618→                variant="outline"
   619→                size="icon"
   620→                className="relative border-slate-600 text-slate-300"
   621→                onClick={() => setShowNotifications(true)}
   622→              >
   623→                <Bell className="h-4 w-4" />
   624→                <span className="absolute -top-1 -right-1 h-4 w-4 bg-red-500 rounded-full text-[10px] flex items-center justify-center">
   625→                  3
   626→                </span>
   627→              </Button>
   628→
   629→              {/* Shortcuts Help */}
   630→              <Button
   631→                variant="outline"
   632→                size="icon"
   633→                className="border-slate-600 text-slate-300"
   634→                onClick={() => setShowShortcuts(true)}
   635→              >
   636→                <Keyboard className="h-4 w-4" />
   637→              </Button>
   638→
   639→              <Button 
   640→                variant="outline" 
   641→                size="icon" 
   642→                className="border-slate-600 text-slate-300"
   643→                onClick={() => window.location.reload()}
   644→              >
   645→                <RefreshCw className="h-4 w-4" />
   646→              </Button>
   647→            </div>
   648→          </div>
   649→        </div>
   650→      </header>
   651→
   652→      {/* Main Content */}
   653→      <main className="flex h-[calc(100vh-73px)]">
   654→        {/* Alerts Panel */}
   655→        <div className="w-[480px] flex-shrink-0 border-r border-slate-800 flex flex-col">
   656→          {/* Filters and Search */}
   657→          <div className="p-4 border-b border-slate-800 space-y-3">
   658→            <SmartFilter 
   659→              onFilterChange={handleFilterChange}
   660→              placeholder="Search alerts..."
   661→              showSeverityFilter
   662→              showStatusFilter
   663→              showSourceFilter
   664→            />
   665→
   666→            {/* Bulk Actions */}
   667→            {selectedAlerts.size > 0 && (
   668→              <div className="flex items-center gap-2 p-2 bg-blue-500/10 border border-blue-500/30 rounded-lg">
   669→                <span className="text-sm text-blue-400">
   670→                  {selectedAlerts.size} selected
   671→                </span>
   672→                <div className="flex gap-1 ml-auto">
   673→                  <Button size="sm" variant="ghost" className="h-7 text-xs text-blue-400" onClick={() => handleBulkAction('acknowledge')}>
   674→                    Ack All
   675→                  </Button>
   676→                  <Button size="sm" variant="ghost" className="h-7 text-xs text-orange-400" onClick={() => handleBulkAction('escalate')}>
   677→                    Escalate All
   678→                  </Button>
   679→                  <Button size="sm" variant="ghost" className="h-7 text-xs text-slate-400" onClick={() => setSelectedAlerts(new Set())}>
   680→                    Clear
   681→                  </Button>
   682→                </div>
   683→              </div>
   684→            )}
   685→          </div>
   686→
   687→          {/* Alerts List */}
   688→          <ScrollArea className="flex-1">
   689→            <div className="p-4 space-y-3">
   690→              {filteredAlerts.length === 0 ? (
   691→                <div className="text-center py-12 text-slate-500">
   692→                  <Search className="h-12 w-12 mx-auto mb-3 opacity-50" />
   693→                  <p>No alerts match your filters</p>
   694→                </div>
   695→              ) : (
   696→                filteredAlerts.map(alert => (
   697→                  <AlertItem
   698→                    key={alert.id}
   699→                    alert={alert}
   700→                    isSelected={selectedAlerts.has(alert.id)}
   701→                    onSelect={(selected) => {
   702→                      setSelectedAlerts(prev => {
   703→                        const next = new Set(prev)
   704→                        if (selected) next.add(alert.id)
   705→                        else next.delete(alert.id)
   706→                        return next
   707→                      })
   708→                    }}
   709→                    onClick={() => setSelectedAlertId(alert.id)}
   710→                    onBookmark={() => toggleBookmark(alert.id)}
   711→                    onQuickAction={(action) => handleQuickAction(alert.id, action)}
   712→                  />
   713→                ))
   714→              )}
   715→            </div>
   716→          </ScrollArea>
   717→
   718→          {/* Footer Stats */}
   719→          <div className="p-4 border-t border-slate-800 bg-slate-900/50">
   720→            <div className="flex items-center justify-between text-xs text-slate-500">
   721→              <span>Showing {filteredAlerts.length} of {alerts.length} alerts</span>
   722→              <div className="flex items-center gap-3">
   723→                <span className="flex items-center gap-1">
   724→                  <span className="w-2 h-2 rounded-full bg-red-500"></span>
   725→                  {alerts.filter(a => a.severity === 'critical').length} Critical
   726→                </span>
   727→                <span className="flex items-center gap-1">
   728→                  <span className="w-2 h-2 rounded-full bg-orange-500"></span>
   729→                  {alerts.filter(a => a.severity === 'high').length} High
   730→                </span>
   731→              </div>
   732→            </div>
   733→          </div>
   734→        </div>
   735→
   736→        {/* Context Panel */}
   737→        <div className="flex-1 overflow-hidden">
   738→          <AlertContextPanel alert={selectedAlert} />
   739→        </div>
   740→      </main>
   741→
   742→      {/* Modals */}
   743→      {showShortcuts && (
   744→        <KeyboardShortcutsHelp 
   745→          shortcuts={keyboardShortcuts}
   746→          onClose={() => setShowShortcuts(false)}
   747→        />
   748→      )}
   749→
   750→      {showNotifications && (
   751→        <NotificationPanel onClose={() => setShowNotifications(false)} />
   752→      )}
   753→    </div>
   754→  )
   755→}
