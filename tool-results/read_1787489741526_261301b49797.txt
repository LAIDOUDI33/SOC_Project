   566→
   567→/**
   568→ * ThreatTimeline - Interactive threat intelligence timeline visualization
   569→ *
   570→ * Features:
   571→ * - Chronological display of security events
   572→ * - MITRE ATT&CK framework integration
   573→ * - Severity-based color coding
   574→ * - Threat actor attribution
   575→ * - IOC extraction and display
   576→ * - Filtering by category/severity/status
   577→ * - Zoom controls for time range adjustment
   578→ *
   579→ * @example
   580→ * ```tsx
   581→ * <ThreatTimeline
   582→ *   showMitreInfo={true}
   583→ *   onEventClick={(event) => showEventDetails(event)}
   584→ * />
   585→ * ```
   586→ */
   587→export function ThreatTimeline({
   588→  initialEvents,
   589→  timeRangeHours = 24,
   590→  onEventClick,
   591→  autoScroll = false,
   592→  showMitreInfo = true,
   593→  className,
   594→}: ThreatTimelineProps) {
   595→  // State management
   596→  const [events] = useState<TimelineEvent[]>(
   597→    initialEvents || generateSampleTimeline()
   598→  );
   599→  const [selectedEventId, setSelectedEventId] = useState<string | null>(null);
   600→  const [severityFilter, setSeverityFilter] = useState<string>('all');
   601→  const [categoryFilter, setCategoryFilter] = useState<string>('all');
   602→  const [statusFilter, setStatusFilter] = useState<string>('all');
   603→  const [zoomLevel, setZoomLevel] = useState(1);
   604→
   605→  // Filter events
   606→  const filteredEvents = useMemo(() => {
   607→    return events.filter((event) => {
   608→      if (severityFilter !== 'all' && event.severity !== severityFilter)
   609→        return false;
   610→      if (categoryFilter !== 'all' && event.category !== categoryFilter)
   611→        return false;
   612→      if (statusFilter !== 'all' && event.status !== statusFilter)
   613→        return false;
   614→      return true;
   615→    });
   616→  }, [events, severityFilter, categoryFilter, statusFilter]);
   617→

   672→  };
   673→
   674→  return (

   878→                            )}

   880→                            {event.actorType && (
   881→                              <span
   882→                                className={cn(
   883→                                  'flex items-center gap-1',
   884→                                  ACTOR_CONFIG[event.actorType]?.color

   888→                                {ACTOR_CONFIG[event.actorType]?.label}

   892→                            <span
   893→                              className="flex items-center gap-1"

   901→                              <span className="flex items-center gap-1">
   902→                                <Eye className="h-3 w-3" />
   903→                                {event.source}
   904→                              </span>
   905→                            )}
   906→                          </div>
   907→
   908→                          {/* IOCs Preview */}
   909→                          {event.iocs && event.iocs.length > 0 && (
   910→                            <div className="mt-2 flex flex-wrap gap-1">
   911→                              {event.iocs.slice(0, 3).map((ioc, i) => (
   912→                                <code
   913→                                  key={i}
   914→                                  className="text-[10px] px-1.5 py-0.5 rounded bg-slate-800 text-cyan-400 font-mono"
   915→                                >
   916→                                  {ioc.length > 25 ? ioc.substring(0, 24) + '...' : ioc}

   921→                                  +{event.iocs.length - 3} more

   926→                        </div>
   927→                      </div>
   928→                    );
   929→                  })}
   930→
   931→                  {filteredEvents.length === 0 && (
   932→                    <div className="text-center py-12 text-slate-500">
   933→                      <Search className="h-12 w-12 mx-auto mb-3 opacity-50" />
   934→                      <p>No events match your filters</p>
   935→                    </div>
   936→                  )}
   937→                </div>
   938→              </div>
   939→            </ScrollArea>
   940→
   941→            {/* Selected Event Details Panel */}
   942→            {selectedEvent && (
   943→              <div className="w-80 border-l border-slate-700 bg-slate-900/50 overflow-y-auto">
   944→                <div className="p-4 space-y-4 sticky top-0 bg-slate-900/95 backdrop-blur-sm">
   945→                  {/* Header */}
   946→                  <div className="flex items-start justify-between">
   947→                    <div className="flex items-center gap-2">
   948→                      {React.createElement(
   949→                        SEVERITY_CONFIG[selectedEvent.severity].icon,
   950→                        {
   951→                          className: cn(
   952→                            'h-5 w-5',
   953→                            SEVERITY_CONFIG[selectedEvent.severity].color
   954→                          ),
   955→                        }
   956→                      )}
   957→                      <h3 className="font-semibold text-white text-sm">
   958→                        {selectedEvent.title}
   959→                      </h3>
   960→                    </div>
   961→                    <Button
   962→                      variant="ghost"
   963→                      size="icon"
   964→                      className="h-6 w-6 text-slate-500"
   965→                      onClick={() => setSelectedEventId(null)}
   966→                    >
   967→                      ×
   968→                    </Button>
   969→                  </div>
   970→
   971→                  {/* Status & Severity Badges */}
   972→                  <div className="flex items-center gap-2">
   973→                    <Badge variant={STATUS_CONFIG[selectedEvent.status].variant}>
   974→                      {STATUS_CONFIG[selectedEvent.status].label}
   975→                    </Badge>
   976→                    <Badge
   977→                      variant="outline"
   978→                      className={cn(
   979→                        SEVERITY_CONFIG[selectedEvent.severity].bgColor,
   980→                        SEVERITY_CONFIG[selectedEvent.severity].color,
   981→                        'border-current'
   982→                      )}
   983→                    >
   984→                      {SEVERITY_CONFIG[selectedEvent.severity].label}
   985→                    </Badge>
   986→                  </div>
   987→
   988→                  <Separator className="bg-slate-700" />
   989→
   990→                  {/* Description */}
   991→                  <div>
   992→                    <h4 className="text-xs font-medium text-slate-400 mb-1">
   993→                      Description

   995→                    <p className="text-sm text-slate-300">
   996→                      {selectedEvent.description}
   997→                    </p>
   998→                  </div>
   999→
  1000→                  {/* Timestamp */}
  1001→                  <div>
  1002→                    <h4 className="text-xs font-medium text-slate-400 mb-1">

  1012→

  1017→                        <Target className="h-3 w-3 text-purple-400" />
