   222→// ============================================================
   223→// Schema Registry Client (for Avro schema management)
   224→// ============================================================
   225→
   226→export class SchemaRegistryClient {
   227→  private config: SchemaRegistryConfig;
   228→  private cache: Map<string, { schema: any; version: number }> = new Map();
   229→
   230→  constructor(config: SchemaRegistryConfig) {
   231→    this.config = config;
   232→  }
   233→
   234→  /**
   235→   * Register a new schema version
   236→   */
   237→  async registerSchema(subject: string, schema: object): Promise<number> {
   238→    try {
   239→      const response = await fetch(`${this.config.url}/subjects/${subject}/versions`, {
   240→        method: 'POST',
   241→        headers: this.getAuthHeaders(),
   242→        body: JSON.stringify({ schema: JSON.stringify(schema) }),
   243→      });
   244→
   245→      if (!response.ok) {
   246→        const error = await response.json();
   247→        throw new SchemaRegistryError(
   248→          `Failed to register schema: ${error.message || response.statusText}`,
   249→          undefined,
   250→          error
   251→        );
   252→      }
   253→
   254→      const result = await response.json();
   255→      
   256→      // Cache the registered schema
   257→      this.cache.set(subject, { schema, version: result.id });
   258→      
   259→      return result.id;
   260→    } catch (error) {
   261→      if (error instanceof SchemaRegistryError || error instanceof KafkaIntegrationError) throw error;
   262→      throw new SchemaRegistryError(
   263→        `Schema registry request failed: ${error instanceof Error ? error.message : String(error)}`,
   264→        error as Error
   265→      );
   266→    }
   267→  }
   268→
   269→  /**
   270→   * Get latest schema for a subject
   271→   */
   272→  async getLatestSchema(subject: string): Promise<{ schema: any; version: number }> {
   273→    // Check cache first
   274→    const cached = this.cache.get(subject);
   275→    if (cached) return cached;
   276→
   277→    try {
   278→      const response = await fetch(`${this.config.url}/subjects/${subject}/versions/latest`, {
   279→        headers: this.getAuthHeaders(),
   280→      });
   281→
   282→      if (!response.ok) {
   283→        if (response.status === 404) {
   284→          throw new SchemaRegistryError(`Schema not found: ${subject}`);
   285→        }
   286→        throw new SchemaRegistryError(`Failed to get schema: ${response.statusText}`);
   287→      }
   288→
   289→      const result = await response.json();
   290→      const schemaData = {
   291→        schema: JSON.parse(result.schema),
   292→        version: result.version,
   293→      };
   294→
   295→      // Cache it
   296→      this.cache.set(subject, schemaData);
   297→
   298→      return schemaData;
   299→    } catch (error) {
   300→      if (error instanceof SchemaRegistryError || error instanceof KafkaIntegrationError) throw error;
   301→      throw new SchemaRegistryError(
   302→        `Failed to retrieve schema: ${error instanceof Error ? error.message : String(error)}`,
   303→        error as Error
   304→      );
   305→    }
   306→  }
   307→
   308→  /**
   309→   * Validate an event against a schema
   310→   */
   311→  async validateEvent(subject: string, event: any): Promise<{
   312→    isValid: boolean;
   313→    errors?: string[];
   314→  }> {
   315→    try {
   316→      const { schema } = await this.getLatestSchema(subject);
   317→      
   318→      // Basic validation (in production, use proper Avro validation library)
   319→      const requiredFields = Object.keys(schema).filter(key => !schema[key].optional);
   320→      const missingFields = requiredFields.filter(field => !(field in event));
   321→      
   322→      if (missingFields.length > 0) {
   323→        return {
   324→          isValid: false,
   325→          errors: [`Missing required fields: ${missingFields.join(', ')}`],
   326→        };
   327→      }
   328→
   329→      return { isValid: true };
   330→    } catch (error) {
   331→      return {
   332→        isValid: false,
   333→        errors: [error instanceof Error ? error.message : String(error)],
   334→      };
   335→    }
   336→  }
   337→
   338→  private getAuthHeaders(): Record<string, string> {
   339→    const headers: Record<string, string> = {
   340→      'Content-Type': 'application/json',
   341→    };
   342→
   343→    if (this.config.username && this.config.password) {
   344→      headers['Authorization'] = `Basic ${Buffer.from(`${this.config.username}:${this.config.password}`).toString('base64')}`;
   345→    }
   346→
   347→    return headers;
   348→  }
   349→}

   428→  /**
   429→   * Produce a single event to Kafka
   430→   */
   431→  async produce(
   432→    event: SocEventEnvelope | SecurityAlertEvent | CdrEvent | ThreatIntelEvent,
   433→    options?: {
   434→      key?: string;
   435→      partition?: number;
   436→      headers?: Record<string, string>;
   437→    }
   438→  ): Promise<void> {
   439→    if (!this.isConnected) {
   440→      throw new ProducerError('Producer is not connected');
   441→    }
   442→
   443→    const startTime = Date.now();
   444→    const value = JSON.stringify(event);
   445→    const key = options?.key || this.generatePartitionKey(event);
   446→
   447→    try {
   448→      // Add to queue for batching
   449→      this.messageQueue.push({
   450→        key,
   451→        value: event,
   452→        partition: options?.partition,
   453→        headers: options?.headers,
   454→      });
   455→
   456→      // Update metrics
   457→      this.metrics.totalMessagesProduced++;
   458→      this.metrics.bytesProduced += value.length;
   459→      this.metrics.currentQueueSize = this.messageQueue.length;
   460→      
   461→      // Calculate running average latency
   462→      const latency = Date.now() - startTime;
   463→      this.metrics.averageLatencyMs = (
   464→        (this.metrics.averageLatencyMs * (this.metrics.totalMessagesProduced - 1) + latency) /
   465→        this.metrics.totalMessagesProduced
   466→      );
   467→
   468→      // Auto-flush if queue size exceeds batch size
   469→      if (this.messageQueue.length >= this.config.batchSize!) {
   470→        await this.flush();
   471→      }
   472→
   473→      this.emit('message_produced', { eventId: (event as any).eventId || (event as any).alertId, size: value.length });
   474→    } catch (error) {
   475→      this.metrics.produceErrorCount++;
   476→      this.emit('produce_error', { event, error });
   477→      throw new ProducerError(
   478→        `Failed to produce message: ${error instanceof Error ? error.message : String(error)}`,
   479→        error as Error
   480→      );
   481→    }
   482→  }
   483→
   484→  /**
   485→   * Produce multiple events in batch
   486→   */
   487→  async produceBatch(
   488→    events: Array<SocEventEnvelope | SecurityAlertEvent | CdrEvent | ThreatIntelEvent>,
   489→    options?: {
   490→      keyPrefix?: string;
   491→      partition?: number;
   492→    }
   493→  ): Promise<{ success: number; failed: number }> {
   494→    let success = 0;
   495→    let failed = 0;
   496→
   497→    for (const event of events) {
   498→      try {
   499→        await this.produce(event, {
   500→          key: options?.keyPrefix ? `${options.keyPrefix}-${(event as any).eventId}` : undefined,
   501→          partition: options?.partition,
   502→        });
   503→        success++;
   504→      } catch (error) {
   505→        failed++;
   506→        console.error(`[Kafka-Producer] Failed to produce event:`, error);
   507→      }
   508→    }
   509→
   510→    // Final flush
   511→    await this.flush();
   512→
   513→    return { success, failed };
   514→  }
   515→
   516→  /**
   517→   * Flush queued messages to Kafka
   518→   */
   519→  async flush(): Promise<void> {
   520→    if (this.messageQueue.length === 0) return;
   521→
   522→    const batch = [...this.messageQueue];
   523→    this.messageQueue = [];
   524→
   525→    try {
   526→      // In production, this would call producer.sendBatch()
   527→      console.log(`[Kafka-Producer] Flushing ${batch.length} messages to topic '${this.config.topic}'`);
   528→      
   529→      // Simulate async send
   530→      await new Promise(resolve => setTimeout(resolve, 10));
   531→      
   532→      this.metrics.batchCount++;
   533→      this.metrics.currentQueueSize = 0;
   534→      
   535→      this.emit('batch_flushed', { count: batch.length });
   536→    } catch (error) {
   537→      // Re-queue on failure
   538→      this.messageQueue.unshift(...batch);
   539→      this.metrics.produceErrorCount++;
   540→      throw new ProducerError(`Flush failed: ${error instanceof Error ? error.message : String(error)}`, error as Error);
   541→    }
   542→  }
   543→
   544→  /**
   545→   * Get current producer metrics
   546→   */
   547→  getMetrics(): ProducerMetrics {
   548→    return { ...this.metrics };
   549→  }
   550→
   551→  /**
   552→   * Reset metrics counters
   553→   */
   554→  resetMetrics(): void {
   555→    this.metrics = {
   556→      totalMessagesProduced: 0,
   557→      bytesProduced: 0,
   558→      produceErrorCount: 0,
   559→      averageLatencyMs: 0,
   560→      currentQueueSize: 0,
   561→      batchCount: 0,
   562→      compressionRatio: 1,
   563→    };
   564→  }
   565→
   566→  private generatePartitionKey(event: any): string {
   567→    // Generate consistent partition key based on event type
   568→    if ((event as SocEventEnvelope).eventId) return (event as SocEventEnvelope).eventId;
   569→    if ((event as SecurityAlertEvent).alertId) return (event as SecurityAlertEvent).alertId;
   570→    if ((event as CdrEvent).cdrId) return (event as CdrEvent).cdrId;
   571→    if ((event as ThreatIntelEvent).iocId) return (event as ThreatIntelEvent).iocId;
   572→    return `evt-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
   573→  }
   574→
   575→  private startFlushTimer(): void {
   576→    // Auto-flush at configured interval
   577→    this.flushInterval = setInterval(async () => {
   578→      if (this.messageQueue.length > 0) {
   579→        await this.flush();
   580→      }
   581→    }, this.config.lingerMs!);
   582→  }
   583→}

   803→    this.maxRetries = options.maxRetries || 3;

   808→   */

   810→    originalMessage: any,
   811→    error: Error,
   812→    retryCount: number = 0,
   813→    originalTopic?: string
   814→  ): Promise<void> {
   815→    const dlqEntry = {
   816→      originalMessage,

   818→        message: error.message,

   822→      retryCount,
   823→      originalTopic,
   824→      failedAt: new Date().toISOString(),
   825→      willRetry: retryCount < this.maxRetries,
   826→    };
   827→
   828→    if (retryCount < this.maxRetries) {
   829→      // Send to retry topic with delay hint
   830→      console.log(`[DLQ] Message queued for retry (${retryCount + 1}/${this.maxRetries})`);
   831→      await this.producer.produce({
   832→        ...dlqEntry,
   833→        retryAfter: Math.pow(2, retryCount) * 1000, // Exponential backoff

   835→    } else {

   838→      await this.producer.produce(dlqEntry, { topic: this.dlqTopic });
   839→    }
   840→  }
