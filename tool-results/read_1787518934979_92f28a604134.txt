   114→/**
   115→ * Extract client info for audit logging

   117→function extractClientInfo(request: Request): { ipAddress?: string; userAgent?: string } {
   118→  return {
   119→    ipAddress: request.headers.get('x-forwarded-for')?.split(',')[0] || 
   120→               request.headers.get('x-real-ip') || 
   121→               undefined,
   122→    userAgent: request.headers.get('user-agent')?.substring(0, 500) || undefined
   123→  };
   124→}

   199→// ============================================================
   200→// GET /api/threats - Fetch threat intelligence with caching
   201→// ============================================================
   202→
   203→export async function GET(request: Request) {
   204→  const requestId = generateRequestId();
   205→  const startTime = Date.now();
   206→
   207→  // v2.0: Authentication required
   208→  const authResult = await authenticateRequest(request as NextRequest);
   209→  
   210→  if (!authResult.success || !authResult.user) {
   211→    return createErrorResponse(
   212→      401,
   213→      'UNAUTHORIZED',
   214→      authResult.error || 'Authentication required',
   215→      requestId
   216→    );
   217→  }
   218→
   219→  try {
   220→    const { searchParams } = new URL(request.url);
   221→    
   222→    // Validate query parameters using Zod schema
   223→    const queryValidation = ThreatQuerySchema.safeParse(Object.fromEntries(searchParams.entries()));
   224→    
   225→    if (!queryValidation.success) {
   226→      return createErrorResponse(
   227→        400,
   228→        'INVALID_QUERY',
   229→        'Invalid query parameters',
   230→        requestId,
   231→        formatZodError(queryValidation.error)
   232→      );
   233→    }
   234→
   235→    const query = queryValidation.data;
   236→    const limit = query.limit;
   237→    const offset = sanitizeOffset(query.offset);
   238→
   239→    // Build where clause with validated parameters
   240→    const where: any = {};
   241→    
   242→    if (query.type) where.type = query.type;
   243→    
   244→    if (query.threatActor) {
   245→      where.threatActor = { contains: query.threatActor, mode: 'insensitive' };
   246→    }
   247→    
   248→    if (query.active === "true") {
   249→      where.isActive = true;
   250→    } else if (query.active === "false") {
   251→      where.isActive = false;
   252→    }
   253→
   254→    if (query.tlp) where.tlp = query.tlp;
   255→    if (query.threatLevel) where.threatLevel = query.threatLevel;
   256→    
   257→    if (query.validated === "true") {
   258→      where.isValidated = true;
   259→    } else if (query.validated === "false") {
   260→      where.isValidated = false;
   261→    }
   262→
   263→    if (query.source) {
   264→      where.source = { contains: query.source, mode: 'insensitive' };
   265→    }
   266→
   267→    if (query.search) {
   268→      where.OR = [
   269→        { value: { contains: query.search, mode: 'insensitive' } },
   270→        { threatActor: { contains: query.search, mode: 'insensitive' } },
   271→        { malwareFamily: { contains: query.search, mode: 'insensitive' } },
   272→        { tags: { contains: query.search } }
   273→      ];
   274→    }
   275→
   276→    if (query.dateFrom || query.dateTo) {
   277→      where.lastSeen = {};
   278→      if (query.dateFrom) where.lastSeen.gte = new Date(query.dateFrom);
   279→      if (query.dateTo) where.lastSeen.lte = new Date(query.dateTo);
   280→    }
   281→
   282→    if (query.tags) {
   283→      const tagArray = query.tags.split(',').map(t => t.trim()).filter(Boolean);
   284→      if (tagArray.length > 0) {
   285→        // Tags stored as JSON array in DB
   286→        where.OR = tagArray.map(tag => ({
   287→          tags: { contains: tag }
   288→        }));
   289→      }
   290→    }
   291→
   292→    const orderBy: any = {};
   293→    orderBy[query.sortBy] = query.sortOrder;
   294→
   295→    // Execute queries in parallel
   296→    const [indicators, total, campaigns, iocStats] = await Promise.all([
   297→      db.threatIndicator.findMany({
   298→        where,
   299→        orderBy,
   300→        take: limit,
   301→        skip: offset,
   302→      }),
   303→      
   304→      db.threatIndicator.count({ where }),
   305→      
   306→      // Active campaigns (if requested)
   307→      query.includeCampaigns === 'true' ? db.campaign.findMany({
   308→        where: {
   309→          isActive: true,
   310→          status: CampaignStatus.ACTIVE
   311→        },
   312→        take: 10,
   313→        orderBy: { lastSeen: 'desc' }
   314→      }) : [],
   315→      
   316→      // IOC statistics (if requested)
   317→      query.includeStatistics === 'true' ? db.iOC.groupBy({
   318→        by: ['type', 'threatLevel', 'isValidated'],
   319→        _count: { id: true }
   320→      }) : []
   321→    ]);
   322→
   323→    const processingTimeMs = Date.now() - startTime;
   324→
   325→    const response = NextResponse.json({
   326→      success: true,
   327→      data: {
   328→        indicators: indicators.map(ind => ({
   329→          id: ind.id,
   330→          type: ind.type.toLowerCase(),
   331→          value: ind.value,
   332→          confidence: ind.confidence,
   333→          source: ind.source,
   334→          threatActor: ind.threatActor,
   335→          malwareFamily: ind.malwareFamily,
   336→          isActive: ind.isActive,
   337→          firstSeen: ind.firstSeen,
   338→          lastSeen: ind.lastSeen,
   339→          ttl: ind.ttl,
   340→          tags: ind.tags ? JSON.parse(ind.tags) : []
   341→        })),
   342→        campaigns: campaigns.map(camp => ({
   343→          id: camp.id,
   344→          name: camp.name,
   345→          alias: camp.alias,
   346→          description: camp.description,
   347→          threatActor: camp.threatActor,
   348→          attributionConfidence: camp.attributionConfidence,
   349→          status: camp.status.toLowerCase(),
   350→          targetSector: camp.targetSector,
   351→          targetRegion: camp.targetRegion,
   352→          objectives: camp.objectives ? JSON.parse(camp.objectives) : [],
   353→          lastSeen: camp.lastSeen
   354→        })),
   355→        statistics: query.includeStatistics === 'true' ? {
   356→          totalIndicators: total,
   357→          byType: iocStats.reduce((acc, ioc) => {
   358→            const type = ioc.type.toLowerCase();
   359→            if (!acc[type]) acc[type] = { total: 0, validated: 0 };
   360→            acc[type].total += ioc._count.id;
   361→            if (ioc.isValidated) acc[type].validated += ioc._count.id;
   362→            return acc;
   363→          }, {} as Record<string, { total: number; validated: number }>),
   364→          byThreatLevel: iocStats.reduce((acc, ioc) => {
   365→            const level = ioc.threatLevel.toLowerCase();
   366→            acc[level] = (acc[level] || 0) + ioc._count.id;
   367→            return acc;
   368→          }, {} as Record<string, number>),
   369→          activeCampaigns: campaigns.length
   370→        } : undefined,
   371→        
   372→        // Threat actor summary (from actual data, not hardcoded)
   373→        threatActors: [...new Set(indicators
   374→          .map(ind => ind.threatActor)
   375→          .filter(Boolean)
   376→        )].map(actor => ({
   377→          name: actor,
   378→          activity: indicators.filter(i => i.threatActor === actor && i.isActive).length > 3 ? 'High' : 
   379→                   indicators.filter(i => i.threatActor === actor && i.isActive).length > 0 ? 'Medium' : 'Low',
   380→          targets: [], // Would be populated from campaign data
   381→          lastSeen: indicators
   382→            .filter(i => i.threatActor === actor)
   383→            .sort((a, b) => b.lastSeen.getTime() - a.lastSeen.getTime())[0]?.lastSeen?.toRelativeString()
   384→        }))
   385→      },
   386→      pagination: {
   387→        total,
   388→        limit,
   389→        offset,
   390→        hasMore: offset + limit < total
   391→      },
   392→      meta: {
   393→        requestId,
   394→        processingTimeMs,
   395→        cached: false
   396→      },
   397→      timestamp: new Date().toISOString(),
   398→    });
   399→
   400→    // Add performance headers
   401→    response.headers.set('X-Request-ID', requestId);
   402→    response.headers.set('X-Processing-Time', String(processingTimeMs));
   403→    response.headers.set('X-Total-Count', String(total));
   404→
   405→    return response;
   406→
   407→  } catch (error) {
   408→    console.error(`[THREATS][GET] Error (${requestId}):`, error);
   409→    
   410→    return createErrorResponse(
   411→      500,
   412→      'INTERNAL_ERROR',
   413→      'Failed to fetch threat intelligence',
   414→      requestId,
   415→      {
   416→        error: error instanceof Error ? error.message : 'Unknown error',
   417→        stack: process.env.NODE_ENV === 'development' ? 
   418→          (error instanceof Error ? error.stack : undefined) : undefined
   419→      }
   420→    );
   421→  }
   422→}

   424→// ============================================================
   425→// POST /api/threats - Create or manage threat intelligence
   426→// ============================================================
   427→
   428→export async function POST(request: Request) {
   429→  const requestId = generateRequestId();
   430→  const startTime = Date.now();
   431→  const clientInfo = extractClientInfo(request);
   432→
   433→  // v2.0: Authentication required
   434→  const authResult = await authenticateRequest(request as NextRequest);
   435→  
   436→  if (!authResult.success || !authResult.user) {
   437→    return createErrorResponse(
   438→      401,
   439→      'UNAUTHORIZED',
   440→      authResult.error || 'Authentication required',
   441→      requestId
   442→    );
   443→  }
   444→
   445→  const user = authResult.user;
   446→  
   447→  try {
   448→    const body = await request.json();
   449→    const { action, ...data } = body;
   450→
   451→    // Validate action is present
   452→    if (!action || typeof action !== 'string') {
   453→      return createErrorResponse(
   454→        400,
   455→        'MISSING_ACTION',
   456→        'Action is required. Supported actions: addIndicator, addIOC, bulkImport, validateIOC',
   457→        requestId,
   458→        { supportedActions: ['addIndicator', 'addIOC', 'bulkImport', 'validateIOC'] }
   459→      );
   460→    }
   461→
   462→    console.log(`[THREATS][POST] User ${user.userId} (${user.roleName}) performing action: ${action}`, { requestId });
   463→
   464→    // ========================================
   465→    // ADD INDICATOR
   466→    // ========================================
   467→    if (action === "addIndicator") {
   468→      // Validate input using Zod schema
   469→      const validation = AddIndicatorSchema.safeParse(data);
   470→      
   471→      if (!validation.success) {
   472→        return createErrorResponse(
   473→          400,
   474→          'VALIDATION_ERROR',
   475→          'Invalid indicator data',
   476→          requestId,
   477→          formatZodError(validation.error)
   478→        );
   479→      }
   480→
   481→      const indicatorData = validation.data;
   482→
   483→      // Check for existing indicator (unique constraint on type+value)
   484→      const existing = await db.threatIndicator.findUnique({
   485→        where: { type_value: { type: indicatorData.type, value: indicatorData.value } }
   486→      });
   487→
   488→      if (existing) {
   489→        // Update existing indicator's metadata
   490→        const updatedIndicator = await db.threatIndicator.update({
   491→          where: { id: existing.id },
   492→          data: { 
   493→            lastSeen: new Date(),
   494→            confidence: Math.max(existing.confidence, indicatorData.confidence),
   495→            source: indicatorData.source || existing.source,
   496→            threatActor: indicatorData.threatActor || existing.threatActor,
   497→            malwareFamily: indicatorData.malwareFamily || existing.malwareFamily,
   498→            tags: indicatorData.tags.length > 0 ? 
   499→              JSON.stringify([...new Set([
   500→                ...(existing.tags ? JSON.parse(existing.tags) : []),
   501→                ...indicatorData.tags
   502→              ])]) : existing.tags,
   503→            isActive: true
   504→          }
   505→        });
   506→
   507→        // Audit log
   508→        await writeAuditLog({
   509→          action: 'UPDATE_INDICATOR',
   510→          entity: 'ThreatIndicator',
   511→          entityId: existing.id,
   512→          userId: user.userId,
   513→          userName: user.name || user.email,
   514→          newValue: { type: indicatorData.type, value: indicatorData.value, action: 'updated' },
   515→          ...clientInfo,
   516→          timestamp: new Date()
   517→        });
   518→
   519→        // Invalidate cache
   520→        await invalidateIOCCache(indicatorData.type, indicatorData.value);
   521→        await invalidateByTag(CacheTags.THREATS[0]);
   522→
   523→        const processingTimeMs = Date.now() - startTime;
   524→
   525→        return NextResponse.json({
   526→          success: true,
   527→          message: "Existing indicator updated",
   528→          action: 'updated',
   529→          data: updatedIndicator,
   530→          meta: { requestId, processingTimeMs },
   531→          timestamp: new Date().toISOString(),
   532→        });
   533→      }
   534→
   535→      // Create new indicator
   536→      const indicator = await db.threatIndicator.create({
   537→        data: {
   538→          type: indicatorData.type,
   539→          value: indicatorData.value,
   540→          confidence: indicatorData.confidence,
   541→          source: indicatorData.source,
   542→          threatActor: indicatorData.threatActor,
   543→          malwareFamily: indicatorData.malwareFamily,
   544→          tags: indicatorData.tags.length > 0 ? JSON.stringify(indicatorData.tags) : null,
   545→          firstSeen: new Date(),
   546→          lastSeen: new Date(),
   547→          isActive: true,
   548→          ttl: indicatorData.ttlHours ? 
   549→            new Date(Date.now() + indicatorData.ttlHours * 60 * 60 * 1000) : undefined
   550→        }
   551→      });
   552→
   553→      // Set in cache for fast lookups
   554→      await setIOCCache(indicatorData.type, indicatorData.value, indicator);
   555→
   556→      // Audit log
   557→      await writeAuditLog({
   558→        action: 'CREATE_INDICATOR',
   559→        entity: 'ThreatIndicator',
   560→        entityId: indicator.id,
   561→        userId: user.userId,
   562→        userName: user.name || user.email,
   563→        newValue: { type: indicatorData.type, value: indicatorData.value, confidence: indicatorData.confidence },
   564→        ...clientInfo,
   565→        timestamp: new Date()
   566→      });
   567→
   568→      // Invalidate threats cache
   569→      await invalidateByTag(CacheTags.THREATS[0]);
   570→
   571→      const processingTimeMs = Date.now() - startTime;
   572→
   573→      return NextResponse.json({
   574→        success: true,
   575→        message: "Threat indicator created",
   576→        action: 'created',
   577→        data: indicator,
   578→        meta: { requestId, processingTimeMs },
   579→        timestamp: new Date().toISOString(),
   580→      }, { status: 201 });
   581→    }
   582→
   583→    // ========================================
   584→    // ADD IOC
   585→    // ========================================
   586→    if (action === "addIOC") {
   587→      // Validate input using Zod schema
   588→      const validation = AddIOCSchema.safeParse(data);
   589→      
   590→      if (!validation.success) {
   591→        return createErrorResponse(
   592→          400,
   593→          'VALIDATION_ERROR',
   594→          'Invalid IOC data',
   595→          requestId,
   596→          formatZodError(validation.error)
   597→        );
   598→      }
   599→
   600→      const iocData = validation.data;
   601→
   602→      const ioc = await db.iOC.create({
   603→        data: {
   604→          iocId: iocData.iocId || `IOC-${Date.now()}-${crypto.randomUUID().replace(/-/g, '').substring(0, 6).toUpperCase()}`,
   605→          type: iocData.type,
   606→          value: iocData.value,
   607→          threatLevel: iocData.threatLevel,
   608→          description: iocData.description,
   609→          source: iocData.source,
   610→          confidence: iocData.confidence,
   611→          isValidated: iocData.isValidated,
   612→          labels: iocData.labels.length > 0 ? JSON.stringify(iocData.labels) : null,
   613→          tlp: iocData.tlp,
   614→          killChainPhase: iocData.killChainPhase
   615→        }
   616→      });
   617→
   618→      // Set in cache
   619→      await setIOCCache(iocData.type, iocData.value, ioc);
   620→
   621→      // Audit log
   622→      await writeAuditLog({
   623→        action: 'CREATE_IOC',
   624→        entity: 'IOC',
   625→        entityId: ioc.id,
   626→        userId: user.userId,
   627→        userName: user.name || user.email,
   628→        newValue: { type: iocData.type, value: iocData.value, threatLevel: iocData.threatLevel },
   629→        ...clientInfo,
   630→        timestamp: new Date()
   631→      });
   632→
   633→      // Invalidate caches
   634→      await invalidateByTag(CacheTags.THREATS[0]);
   635→
   636→      const processingTimeMs = Date.now() - startTime;
   637→
   638→      return NextResponse.json({
   639→        success: true,
   640→        message: "IOC created successfully",
   641→        data: ioc,
   642→        meta: { requestId, processingTimeMs },
   643→        timestamp: new Date().toISOString(),
   644→      }, { status: 201 });
   645→    }
   646→
   647→    // ========================================
   648→    // BULK IMPORT IOCs
   649→    // ========================================
   650→    if (action === "bulkImport") {
   651→      // Validate input
   652→      const validation = BulkImportIOCSchema.safeParse(data);
   653→      
   654→      if (!validation.success) {
   655→        return createErrorResponse(
   656→          400,
   657→          'VALIDATION_ERROR',
   658→          'Invalid bulk import data',
   659→          requestId,
   660→          formatZodError(validation.error)
   661→        );
   662→      }
   663→
   664→      const importData = validation.data;
   665→      const results = {
   666→        created: 0,
   667→        updated: 0,
   668→        skipped: 0,
   669→        errors: [] as Array<{ index: number; value: string; error: string }>
   670→      };
   671→
   672→      // Process each IOC
   673→      for (let i = 0; i < importData.iocs.length; i++) {
   674→        const iocData = importData.iocs[i];
   675→        
   676→        try {
   677→          // Determine if it's an Indicator or IOC schema
   678→          const isIndicator = 'type' in iocData && 'confidence' in iocData && !('threatLevel' in iocData);
   679→          
   680→          if (isIndicator) {
   681→            // Handle as indicator
   682→            const indicatorData = iocData as typeof importData.iocs[number] & { type: string; value: string };
   683→            
   684→            const existing = await db.threatIndicator.findUnique({
   685→              where: { type_value: { type: indicatorData.type, value: indicatorData.value } }
   686→            });
   687→
   688→            if (existing) {
   689→              if (importData.importOptions.updateExisting) {
   690→                await db.threatIndicator.update({
   691→                  where: { id: existing.id },
   692→                  data: { lastSeen: new Date(), isActive: true }
   693→                });
   694→                results.updated++;
   695→              } else {
   696→                results.skipped++;
   697→              }
   698→            } else {
   699→              await db.threatIndicator.create({
   700→                data: {
   701→                  type: indicatorData.type,
   702→                  value: indicatorData.value,
   703→                  confidence: 50,
   704→                  source: importData.source,
   705→                  firstSeen: new Date(),
   706→                  lastSeen: new Date(),
   707→                  isActive: true,
   708→                  tags: importData.tags.length > 0 ? JSON.stringify(importData.tags) : null
   709→                }
   710→              });
   711→              results.created++;
   712→            }
   713→          } else {
   714→            // Handle as IOC
   715→            const iocInput = iocData as typeof importData.iocs[number];
   716→            
   717→            await db.iOC.create({
   718→              data: {
   719→                iocId: `IOC-BULK-${Date.now()}-${i}`,
   720→                type: iocInput.type || 'IPV4',
   721→                value: iocInput.value,
   722→                threatLevel: iocInput.threatLevel || 'MEDIUM',
   723→                source: importData.source,
   724→                confidence: iocInput.confidence || 50,
   725→                isValidated: false
   726→              }
   727→            });
   728→            results.created++;
   729→          }
   730→        } catch (error) {
   731→          results.errors.push({
   732→            index: i,
   733→            value: ('value' in iocData) ? iocData.value : 'unknown',
   734→            error: error instanceof Error ? error.message : 'Processing failed'
   735→          });
   736→        }
   737→      }
   738→
   739→      // Audit log for bulk operation
   740→      await writeAuditLog({
   741→        action: 'BULK_IMPORT',
   742→        entity: 'IOC/Indicator',
   743→        entityId: `bulk_${Date.now()}`,
   744→        userId: user.userId,
   745→        userName: user.name || user.email,
   746→        newValue: { 
   747→          source: importData.source, 
   748→          totalCount: importData.iocs.length,
   749→          ...results 
   750→        },
   751→        ...clientInfo,
   752→        timestamp: new Date()
   753→      });
   754→
   755→      // Invalidate all threat caches
   756→      await invalidateByTag(CacheTags.THREATS[0]);
   757→
   758→      const processingTimeMs = Date.now() - startTime;
   759→
   760→      return NextResponse.json({
   761→        success: true,
   762→        message: `Bulk import completed: ${results.created} created, ${results.updated} updated, ${results.skipped} skipped`,
   763→        results,
   764→        meta: { requestId, processingTimeMs },
   765→        timestamp: new Date().toISOString(),
   766→      });
   767→    }
   768→
   769→    // ========================================
   770→    // VALIDATE IOC (check against known bad)
   771→    // ========================================
   772→    if (action === "validateIOC") {
   773→      const { type, value } = data;
   774→
   775→      if (!type || !value) {
   776→        return createErrorResponse(
   777→          400,
   778→          'MISSING_FIELDS',
   779→          'Type and value are required for IOC validation',
   780→          requestId
   781→        );
   782→      }
   783→
   784→      // Check cache first
   785→      const cached = await getIOCFromCache(type, value);
   786→      if (cached) {
   787→        return NextResponse.json({
   788→          success: true,
   789→          data: cached,
   790→          meta: { requestId, cached: true, processingTimeMs: Date.now() - startTime },
   791→          timestamp: new Date().toISOString()
   792→        });
   793→      }
   794→
   795→      // Query database
   796→      const indicator = await db.threatIndicator.findUnique({
   797→        where: { type_value: { type: type.toUpperCase(), value } },
   798→        include: {
   799→          relatedIndicators: {
   800→            take: 10,
   801→            where: { isActive: true }
   802→          }
   803→        }
   804→      });
   805→
   806→      const result = {
   807→        found: !!indicator,
   808→        indicator: indicator ? {
   809→          id: indicator.id,
   810→          type: indicator.type.toLowerCase(),
   811→          value: indicator.value,
   812→          confidence: indicator.confidence,
   813→          threatActor: indicator.threatActor,
   814→          malwareFamily: indicator.malwareFamily,
   815→          isActive: indicator.isActive,
   816→          lastSeen: indicator.lastSeen,
   817→          firstSeen: indicator.firstSeen,
   818→          daysSinceLastSeen: Math.floor((Date.now() - indicator.lastSeen.getTime()) / (1000 * 60 * 60 * 24)),
   819→          relatedIndicators: indicator.relatedIndicators.map(rel => ({
   820→            type: rel.type.toLowerCase(),
   821→            value: rel.value,
   822→            confidence: rel.confidence
   823→          }))
   824→        } : null,
   825→        recommendation: indicator ? 
   826→          (indicator.isActive ? 'BLOCK - Known malicious indicator' : 'MONITOR - Previously seen but inactive') :
   827→          'ALLOW - No matches found'
   828→      };
   829→
   830→      // Cache the result
   831→      if (indicator) {
   832→        await setIOCCache(type, value, result);
   833→      }
   834→
   835→      const processingTimeMs = Date.now() - startTime;
   836→
   837→      return NextResponse.json({
   838→        success: true,
   839→        data: result,
   840→        meta: { requestId, cached: false, processingTimeMs },
   841→        timestamp: new Date().toISOString()
   842→      });
   843→    }
   844→
   845→    // ========================================
   846→    // UNKNOWN ACTION
   847→    // ========================================
   848→    return createErrorResponse(
   849→      400,
   850→      'INVALID_ACTION',
   851→      `Unsupported action: ${action}. Supported actions: addIndicator, addIOC, bulkImport, validateIOC`,
   852→      requestId,
   853→      { supportedActions: ['addIndicator', 'addIOC', 'bulkImport', 'validateIOC'] }
   854→    );
   855→
   856→  } catch (error) {
   857→    console.error(`[THREATS][POST] Error (${requestId}):`, error);
   858→    
   859→    let errorMessage = 'Failed to process request';
   860→    let errorCode = 'INTERNAL_ERROR';
   861→    
   862→    if (error && typeof error === 'object' && 'code' in error) {
   863→      const prismaError = error as { code: string; meta?: Record<string, unknown> };
   864→      
   865→      if (prismaError.code === 'P2002') {
   866→        errorMessage = 'Duplicate indicator/IOC';
   867→        errorCode = 'DUPLICATE';
   868→      } else if (prismaError.code === 'P2003') {
   869→        errorMessage = 'Invalid reference';
   870→        errorCode = 'INVALID_REFERENCE';
   871→      }
   872→    }
   873→    
   874→    return createErrorResponse(
   875→      500,
   876→      errorCode,
   877→      errorMessage,
   878→      requestId,
   879→      {
   880→        error: error instanceof Error ? error.message : 'Unknown error',
   881→        action: body.action
   882→      }
   883→    );
   884→  }
   885→}

   887→// ============================================================
   888→// PUT /api/threats - Update existing indicators/IOCs
   889→// ============================================================
   890→
   891→export async function PUT(request: Request) {
   892→  const requestId = generateRequestId();
   893→  const startTime = Date.now();
   894→  const clientInfo = extractClientInfo(request);
   895→
   896→  // Authentication required
   897→  const authResult = await authenticateRequest(request as NextRequest);
   898→  
   899→  if (!authResult.success || !authResult.user) {
   900→    return createErrorResponse(401, 'UNAUTHORIZED', 'Authentication required', requestId);
   901→  }
   902→
   903→  const user = authResult.user;
   904→
   905→  try {
   906→    const body = await request.json();
   907→    const { id, type, ...updateData } = body;
   908→
   909→    if (!id) {
   910→      return createErrorResponse(400, 'MISSING_ID', 'ID is required for updates', requestId);
   911→    }
   912→
   913→    if (!type || !['indicator', 'ioc'].includes(type)) {
   914→      return createErrorResponse(
   915→        400,
   916→        'INVALID_TYPE',
   917→        'Type must be either "indicator" or "ioc"',
   918→        requestId
   919→      );
   920→    }
   921→
   922→    // Get current state for audit
   923→    const existing = type === 'indicator' ?
   924→      await db.threatIndicator.findUnique({ where: { id } }) :
   925→      await db.iOC.findUnique({ where: { id } });
   926→
   927→    if (!existing) {
   928→      return createErrorResponse(404, 'NOT_FOUND', `${type} ${id} not found`, requestId);
   929→    }
   930→
   931→    let updated;
   932→    
   933→    if (type === 'indicator') {
   934→      // Validate allowed update fields
   935→      const allowedUpdates = [
   936→        'confidence', 'source', 'threatActor', 'malwareFamily', 
   937→        'tags', 'isActive', 'ttl'
   938→      ];
   939→      
   940→      const filteredData: Record<string, unknown> = {};
   941→      for (const key of allowedUpdates) {
   942→        if (key in updateData) {
   943→          if (key === 'tags' && Array.isArray(updateData[key])) {
   944→            filteredData[key] = JSON.stringify(updateData[key]);
   945→          } else {
   946→            filteredData[key] = updateData[key];
   947→          }
   948→        }
   949→      }
   950→
   951→      updated = await db.threatIndicator.update({
   952→        where: { id },
   953→        data: filteredData
   954→      });
   955→
   956→      // Invalidate IOC cache
   957→      await invalidateIOCCache(updated.type, updated.value);
   958→    } else {
   959→      // Update IOC
   960→      const allowedUpdates = [
   961→        'threatLevel', 'description', 'source', 'confidence',
   962→        'isValidated', 'labels', 'tlp', 'killChainPhase', 'falsePositiveRate'
   963→      ];
   964→      
   965→      const filteredData: Record<string, unknown> = {};
   966→      for (const key of allowedUpdates) {
   967→        if (key in updateData) {
   968→          if (key === 'labels' && Array.isArray(updateData[key])) {
   969→            filteredData[key] = JSON.stringify(updateData[key]);
   970→          } else {
   971→            filteredData[key] = updateData[key];
   972→          }
   973→        }
   974→      }
   975→
   976→      updated = await db.iOC.update({
   977→        where: { id },
   978→        data: filteredData
   979→      });
   980→
   981→      // Invalidate IOC cache
   982→      await invalidateIOCCache(updated.type, updated.value);
   983→    }
   984→
   985→    // Audit log
   986→    await writeAuditLog({
   987→      action: 'UPDATE',
   988→      entity: type === 'indicator' ? 'ThreatIndicator' : 'IOC',
   989→      entityId: id,
   990→      userId: user.userId,
   991→      userName: user.name || user.email,
   992→      previousValue: existing,
   993→      newValue: updateData,
   994→      ...clientInfo,
   995→      timestamp: new Date()
   996→    });
   997→
   998→    // Invalidate caches
   999→    await invalidateByTag(CacheTags.THREATS[0]);
  1000→
  1001→    const processingTimeMs = Date.now() - startTime;
  1002→
  1003→    return NextResponse.json({
  1004→      success: true,
  1005→      message: `${type} ${id} updated`,
  1006→      data: updated,
  1007→      meta: { requestId, processingTimeMs },
  1008→      timestamp: new Date().toISOString()
  1009→    });
  1010→
  1011→  } catch (error) {
  1012→    console.error(`[THREATS][PUT] Error (${requestId}):`, error);
  1013→    return createErrorResponse(500, 'INTERNAL_ERROR', 'Failed to update', requestId);
  1014→  }
  1015→}

  1018→// DELETE /api/threats - Soft delete indicators/IOCs (admin only)
  1019→// ============================================================
  1020→
  1021→export async function DELETE(request: Request) {
  1022→  const requestId = generateRequestId();
  1023→
  1024→  const authResult = await authenticateRequest(request as NextRequest);
  1025→  
  1026→  if (!authResult.success || !authResult.user) {
  1027→    return createErrorResponse(401, 'UNAUTHORIZED', 'Authentication required', requestId);
  1028→  }
  1029→
  1030→  // Admin only
  1031→  if (!['ADMIN', 'SUPERADMIN', 'THREAT_LEAD'].includes(authResult.user.role)) {
  1032→    return createErrorResponse(403, 'FORBIDDEN', 'Admin or Threat Lead privileges required', requestId);
  1033→  }
  1034→
  1035→  try {
  1036→    const { searchParams } = new URL(request.url);
  1037→    const id = searchParams.get('id');
  1038→    const type = searchParams.get('type'); // 'indicator' or 'ioc'
  1039→
  1040→    if (!id || !type) {
  1041→      return createErrorResponse(
  1042→        400,
  1043→        'MISSING_PARAMS',
  1044→        'Both id and type (indicator/ioc) are required',
  1045→        requestId
  1046→      );
  1047→    }
  1048→
  1049→    if (type === 'indicator') {
  1050→      // Soft delete - mark as inactive
  1051→      const indicator = await db.threatIndicator.findUnique({ where: { id } });
  1052→      if (indicator) {
  1053→        await db.threatIndicator.update({
  1054→          where: { id },
  1055→          data: { isActive: false }
  1056→        });
  1057→        await invalidateIOCCache(indicator.type, indicator.value);
  1058→      }
  1059→    } else if (type === 'ioc') {
  1060→      // Delete IOC
  1061→      const ioc = await db.iOC.findUnique({ where: { id } });
  1062→      if (ioc) {
  1063→        await db.iOC.delete({ where: { id } });
  1064→        await invalidateIOCCache(ioc.type, ioc.value);
  1065→      }
  1066→    } else {
  1067→      return createErrorResponse(400, 'INVALID_TYPE', 'Type must be "indicator" or "ioc"', requestId);
  1068→    }
  1069→
  1070→    // Audit log
  1071→    await writeAuditLog({
  1072→      action: 'DELETE',
  1073→      entity: type === 'indicator' ? 'ThreatIndicator' : 'IOC',
  1074→      entityId: id,
  1075→      userId: authResult.user.userId,
  1076→      userName: authResult.user.name || authResult.user.email,
  1077→      ...extractClientInfo(request),
  1078→      timestamp: new Date()
  1079→    });
  1080→
  1081→    // Invalidate caches
  1082→    await invalidateByTag(CacheTags.THREATS[0]);
  1083→
  1084→    return NextResponse.json({
  1085→      success: true,
  1086→      message: `${type} ${id} deleted/deactivated`,
  1087→      meta: { requestId },
  1088→      timestamp: new Date().toISOString()
  1089→    });
  1090→
  1091→  } catch (error) {
  1092→    console.error(`[THREATS][DELETE] Error (${requestId}):`, error);
  1093→    return createErrorResponse(500, 'INTERNAL_ERROR', 'Failed to delete', requestId);
  1094→  }
  1095→}
