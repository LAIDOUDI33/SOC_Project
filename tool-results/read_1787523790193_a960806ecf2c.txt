   184→class BERDecoder {
   185→  private buffer: Buffer;
   186→  private offset: number;
   187→
   188→  constructor(buffer: Buffer) {
   189→    this.buffer = buffer;
   190→    this.offset = 0;
   191→  }
   192→
   193→  get position(): number {
   194→    return this.offset;
   195→  }
   196→
   197→  set position(value: number) {
   198→    this.offset = value;
   199→  }
   200→
   201→  get remaining(): number {
   202→    return this.buffer.length - this.offset;
   203→  }
   204→
   205→  decodeTag(): { tagClass: number; constructed: boolean; tagNumber: number; length: number } {
   206→    if (this.remaining < 2) throw new Error('Insufficient data for tag');
   207→
   208→    const byte = this.buffer[this.offset++];
   209→    const tagClass = (byte >> 6) & 0x03;
   210→    const constructed = (byte & 0x20) !== 0;
   211→    let tagNumber = byte & 0x1F;
   212→
   213→    // Long form tag
   214→    if (tagNumber === 0x1F) {
   215→      tagNumber = 0;
   216→      let b;
   217→      do {
   218→        if (this.remaining === 0) throw new Error('Truncated tag');
   219→        b = this.buffer[this.offset++];
   220→        tagNumber = (tagNumber << 7) | (b & 0x7F);
   221→      } while (b & 0x80);
   222→    }
   223→
   224→    const length = this.decodeLength();
   225→
   226→    return { tagClass, constructed, tagNumber, length };
   227→  }
   228→
   229→  decodeLength(): number {
   230→    if (this.remaining === 0) throw new Error('Insufficient data for length');
   231→
   232→    const byte = this.buffer[this.offset++];
   233→
   234→    if (byte & 0x80) {
   235→      const numBytes = byte & 0x7F;
   236→      if (numBytes > 4) throw new Error(`Length too long: ${numBytes} bytes`);
   237→      
   238→      if (this.remaining < numBytes) throw new Error('Insufficient data for length');
   239→      
   240→      let length = 0;
   241→      for (let i = 0; i < numBytes; i++) {
   242→        length = (length << 8) | this.buffer[this.offset++];
   243→      }
   244→      return length;
   245→    }
   246→
   247→    return byte;
   248→  }
   249→
   250→  readBytes(length: number): Buffer {
   251→    if (this.remaining < length) throw new Error(`Insufficient data: need ${length}, have ${this.remaining}`);
   252→    const result = this.buffer.slice(this.offset, this.offset + length);
   253→    this.offset += length;
   254→    return result;
   255→  }
   256→
   257→  readOctetString(length: number): string {
   258→    return this.readBytes(length).toString('hex').toUpperCase();
   259→  }
   260→
   261→  readInteger(): number {
   262→    const { length } = this.decodeTag();
   263→    const bytes = this.readBytes(length);
   264→    
   265→    let value = 0;
   266→    for (const byte of bytes) {
   267→      value = (value << 8) | byte;
   268→    }
   269→    
   270→    // Handle negative numbers (two's complement)
   271→    if (bytes.length > 0 && (bytes[0] & 0x80)) {
   272→      value -= (1 << (bytes.length * 8));
   273→    }
   274→    
   275→    return value;
   276→  }
   277→
   278→  readString(length: number): string {
   279→    const bytes = this.readBytes(length);
   280→    // Try to decode as ASCII/UTF-8
   281→    try {
   282→      return bytes.toString('utf-8');
   283→    } catch {
   284→      return bytes.toString('latin1');
   285→    }
   286→  }
   287→
   288→  skip(length: number): void {
   289→    this.offset += length;
   290→  }
   291→
   292→  peek(): number {
   293→    return this.buffer[this.offset];
   294→  }
   295→}

   297→/**
   298→ * Decode BCD-encoded digits from buffer
   299→ */
   300→function decodeBCDDigits(buffer: Buffer): string {
   301→  let digits = '';
   302→  for (const byte of buffer) {
   303→    const lowNibble = byte & 0x0F;
   304→    const highNibble = (byte >> 4) & 0x0F;
   305→    
   306→    if (lowNibble <= 9) digits += lowNibble.toString();
   307→    if (highNibble <= 9) digits += highNibble.toString();
   308→  }
   309→  return digits;
   310→}
   311→
   312→/**
   313→ * Encode address string to/from TBCD format
   314→ */
   315→function decodeTBCDString(buffer: Buffer): string {
   316→  let digits = '';
   317→  for (let i = 0; i < buffer.length; i++) {
   318→    const byte = buffer[i];
   319→    const lowNibble = byte & 0x0F;
   320→    const highNibble = (byte >> 4) & 0x0F;
   321→    
   322→    // Low nibble first in TBCD
   323→    switch (lowNibble) {
   324→      case 0x0A: digits += '*'; break;
   325→      case 0x0B: digits += '#'; break;
   326→      case 0x0C: digits += 'a'; break;
   327→      case 0x0D: digits += 'b'; break;
   328→      case 0x0E: digits += 'c'; break;
   329→      default:
   330→        if (lowNibble <= 9) digits += lowNibble.toString();
   331→    }
   332→    
   333→    // High nibble second
   334→    switch (highNibble) {
   335→      case 0x0A: digits += '*'; break;
   336→      case 0x0B: digits += '#'; break;
   337→      case 0x0C: digits += 'a'; break;
   338→      case 0x0D: digits += 'b'; break;
   339→      case 0x0E: digits += 'c'; break;
   340→      default:
   341→        if (highNibble <= 9) digits += highNibble.toString();
   342→    }
   343→  }
   344→  return digits;
   345→}

   358→  /**
   359→   * Main entry point for decoding an SS7 message
   360→   */
   361→  decode(rawData: Buffer | string, options?: {
   362→    assumedProtocol?: SS7ProtocolLayer;
   363→    opc?: number;
   364→    dpc?: number;
   365→    direction?: 'inbound' | 'outbound';
   366→  }): DecodeResult<SS7Message> {
   367→    try {
   368→      // Convert hex string to buffer if needed
   369→      const buffer = typeof rawData === 'string' ? hexToBuffer(rawData) : rawData;
   370→      
   371→      if (buffer.length === 0) {
   372→        return { success: false, error: 'Empty input data' };
   373→      }
   374→
   375→      // Auto-detect or use specified protocol
   376→      const protocol = options?.assumedProtocol || this.detectProtocol(buffer);
   377→      
   378→      this.log(`Detected protocol: ${protocol}`);
   379→      
   380→      switch (protocol) {
   381→        case SS7ProtocolLayer.SCTP:
   382→          return this.decodeSCTP(buffer, options);
   383→        case SS7ProtocolLayer.M3UA:
   384→          return this.decodeM3UA(buffer, options);
   385→        case SS7ProtocolLayer.SCCP:
   386→          return this.decodeSCCP(buffer, options);
   387→        case SS7ProtocolLayer.TCAP:
   388→          return this.decodeTCAP(buffer, options);
   389→        case SS7ProtocolLayer.MAP:
   390→          return this.decodeMAP(buffer, options);
   391→        case SS7ProtocolLayer.CAP:
   392→          return this.decodeCAP(buffer, options);
   393→        case SS7ProtocolLayer.ISUP:
   394→          return this.decodeISUP(buffer, options);
   395→        default:
   396→          return this.decodeRaw(buffer, options);
   397→      }
   398→    } catch (error) {
   399→      return {
   400→        success: false,
   401→        error: `Decode failed: ${error instanceof Error ? error.message : 'Unknown error'}`,
   402→      };
   403→    }
   404→  }
   405→
   406→  /**
   407→   * Attempt to detect the protocol from raw bytes
   408→   */
   409→  private detectProtocol(buffer: Buffer): SS7ProtocolLayer {
   410→    if (buffer.length < 4) return SS7ProtocolLayer.MTP3;
   411→
   412→    // Check SCTP signature
   413→    if (buffer[0] >= 0x00 && buffer[0] <= 0x84 && buffer.length >= 12) {
   414→      // Could be SCTP chunk
   415→      const chunkType = buffer[0];
   416→      if (Object.values(SCTPChunkType).includes(chunkType as SCTPChunkType)) {
   417→        return SS7ProtocolLayer.SCTP;
   418→      }
   419→    }
   420→
   421→    // Check M3UA header (Version=2, Class, Type)
   422→    if (buffer[0] === 0x02 || buffer[0] === 0x03) {
   423→      const m3uaClass = buffer[1];
   424→      if (m3uaClass >= 0 && m3uaClass <= 5) {
   425→        return SS7ProtocolLayer.M3UA;
   426→      }
   427→    }
   428→
   429→    // Check SCCP message type
   430→    const potentialSCCPMsgType = buffer[0];
   431→    if (Object.values(SCCPMessageType).includes(potentialSCCPMsgType as SCCPMessageType)) {
   432→      return SS7ProtocolLayer.SCCP;
   433→    }
   434→
   435→    // Check TCAP tag (typically starts with 0x60-0x68)
   436→    if ((buffer[0] >= 0x60 && buffer[0] <= 0x68) || buffer[0] === 0x61) {
   437→      return SS7ProtocolLayer.TCAP;
   438→    }
   439→
   440→    // Default to MTP3/ISUP
   441→    return SS7ProtocolLayer.ISUP;
   442→  }

   444→  // ============================================================
   445→  // SCTP DECODING
   446→  // ============================================================
   447→
   448→  private decodeSCTP(buffer: Buffer, options?: any): DecodeResult<SS7Message> {
   449→    const result: SS7Message = {
   450→      id: generateMessageId(),
   451→      timestamp: new Date(),
   452→      protocol: SS7ProtocolLayer.SCTP,
   453→      direction: options?.direction || 'inbound',
   454→      rawData: buffer,
   455→      hexDump: hexDump(buffer),
   456→      packetLength: buffer.length,
   457→    };
   458→
   459→    try {
   460→      const decoder = new BERDecoder(buffer);
   461→      
   462→      // SCTP Common Header (16 bytes)
   463→      const srcPort = decoder.readBytes(2).readUInt16BE(0);
   464→      const dstPort = decoder.readBytes(2).readUInt16BE(0);
   465→      const verificationTag = decoder.readBytes(4).readUInt32BE(0);
   466→      const checksum = decoder.readBytes(4).readUInt32BE(0);
   467→
   468→      result.sourcePort = srcPort;
   469→      result.destPort = dstPort;
   470→      result.decodedFields = {
   471→        sourcePort: srcPort,
   472→        destPort: dstPort,
   473→        verificationTag: `0x${verificationTag.toString(16).toUpperCase()}`,
   474→        checksum: `0x${checksum.toString(16).toUpperCase()}`,
   475→        chunks: [],
   476→      };
   477→
   478→      // Parse chunks
   479→      while (decoder.remaining > 0) {
   480→        const chunkStart = decoder.position;
   481→        
   482→        if (decoder.remaining < 4) break;
   483→        
   484→        const chunkType = decoder.buffer[decoder.position];
   485→        const chunkLength = decoder.buffer.slice(decoder.position + 2, decoder.position + 4).readUInt16BE(0);
   486→        
   487→        const chunkTypeName = SCTP_CHUNK_NAMES[chunkType as SCTPChunkType] || `Unknown (${chunkType})`;
   488→        
   489→        const chunkInfo: Record<string, any> = {
   490→          type: chunkTypeName,
   491→          typeCode: chunkType,
   492→          length: chunkLength,
   493→        };
   494→
   495→        // Parse specific chunk types
   496→        switch (chunkType) {
   497→          case SCTPChunkType.DATA:
   498→            if (decoder.remaining >= 17) {
   499→              decoder.skip(1); // Type
   500→              decoder.skip(2); // Length
   501→              const flags = decoder.buffer[decoder.position];
   502→              const tsn = decoder.readBytes(4).readUInt32BE(0);
   503→              const sid = decoder.readBytes(2).readUInt16BE(0);
   504→              const ssn = decoder.readBytes(2).readUInt16BE(0);
   505→              const ppi = decoder.readBytes(4).readUInt32BE(0);
   506→              
   507→              chunkInfo.data = {
   508→                flags,
   509→                tsn,
   510→                streamId: sid,
   511→                streamSeqNum: ssn,
   512→                payloadProtocolId: ppi,
   513→                payloadSize: chunkLength - 17,
   514→              };
   515→            }
   516→            break;
   517→
   518→          case SCTPChunkType.INIT:
   519→            if (decoder.remaining >= 20) {
   520→              decoder.skip(3); // Type + Length
   521→              const initTag = decoder.readBytes(4).readUInt32BE(0);
   522→              const aRwnd = decoder.readBytes(4).readUInt32BE(0);
   523→              const numOutbound = decoder.readBytes(2).readUInt16BE(0);
   524→              const numInbound = decoder.readBytes(2).readUInt16BE(0);
   525→              
   526→              chunkInfo.init = {
   527→                initTag: `0x${initTag.toString(16)}`,
   528→                aRwnd,
   529→                numOutboundStreams: numOutbound,
   530→                numInboundStreams: numInbound,
   531→              };
   532→            }
   533→            break;
   534→
   535→          case SCTPChunkType.HEARTBEAT:
   536→            // Heartbeat with optional parameter
   537→            break;
   538→        }
   539→
   540→        (result.decodedFields.chunks as Record<string, any>[]).push(chunkInfo);
   541→        
   542→        // Move to next chunk (pad to 4-byte boundary)
   543→        const paddedLength = Math.ceil(chunkLength / 4) * 4;
   544→        decoder.position = chunkStart + paddedLength;
   545→      }
   546→
   547→      return { success: true, data: result };
   548→    } catch (error) {
   549→      return { success: true, data: result, warnings: [`SCTP parse warning: ${error}`] };
   550→    }
   551→  }

   554→  // M3UA DECODING
   555→  // ============================================================
   556→
   557→  private decodeM3UA(buffer: Buffer, options?: any): DecodeResult<SS7Message> {
   558→    const result: SS7Message = {
   559→      id: generateMessageId(),
   560→      timestamp: new Date(),
   561→      protocol: SS7ProtocolLayer.M3UA,
   562→      direction: options?.direction || 'inbound',
   563→      rawData: buffer,
   564→      hexDump: hexDump(buffer),
   565→      packetLength: buffer.length,
   566→    };
   567→
   568→    try {
   569→      // M3UA Common Header (8 bytes)
   570→      if (buffer.length < 8) {
   571→        return { success: false, error: 'Buffer too short for M3UA' };
   572→      }
   573→
   574→      const version = buffer[0];           // Should be 2 or 3
   575→      const m3uaClass = buffer[1];         // Message Class
   576→      const m3uaType = buffer[2];          // Message Type
   577→      const m3uaLength = buffer.slice(6, 8).readUInt32BE(0); // Actually 4-byte length at offset 4
   578→
   579→      const msgName = M3UA_MESSAGE_NAMES[m3uaType as M3UAMessageType] || `Unknown (${m3uaType})`;
   580→
   581→      result.decodedFields = {
   582→        version,
   583→        class: m3uaClass,
   584→        className: this.getM3UAClassName(m3uaClass),
   585→        type: m3uaType,
   586→        typeName: msgName,
   587→        length: m3uaLength,
   588→      };
   589→
   590→      // Parse based on message class/type
   591→      if (m3uaClass === 1 && m3uaType === 1) {
   592→        // TRANSFER - contains MTP3 routing label + payload
   593→        this.parseM3UATransfer(result, buffer.slice(8));
   594→      }
   595→
   596→      return { success: true, data: result };
   597→    } catch (error) {
   598→      return { success: false, error: `M3UA decode error: ${error}`, data: result };
   599→    }
   600→  }
   601→
   602→  private getM3UAClassName(classCode: number): string {
   603→    switch (classCode) {
   604→      case 0: return 'Message Transfer (MMT)';
   605→      case 1: return 'SS7 Signaling Network Management (SSNM)';
   606→      case 2: return 'ASP State Maintenance (ASPSM)';
   607→      case 3: return 'ASP Traffic Maintenance (ASPTM)';
   608→      case 4: return 'Routing Key Management (RKM)';
   609→      case 5: return 'Reserved for IETF';
   610→      default: return `Unknown (${classCode})`;
   611→    }
   612→  }
   613→
   614→  private parseM3UATransfer(result: SS7Message, payload: Buffer): void {
   615→    if (payload.length < 11) return;
   616→
   617→    // Protocol Data (Tag=0x0100)
   618→    // Routing Label (7 bytes)
   619→    const opcRaw = payload.slice(0, 2).readUInt16BE(0);
   620→    const dpcRaw = payload.slice(2, 4).readUInt16BE(0);
   621→    const sls = payload[4];
   622→    const slc = payload[5];
   623→    const ni_ni = payload[6]; // Network Indicator
   624→    
   625→    result.opc = parsePointCode(opcRaw, PointCodeFormat.ITU_14BIT);
   626→    result.dpc = parsePointCode(dpcRaw, PointCodeFormat.ITU_14BIT);
   627→    result.sls = sls;
   628→    result.slc = slc;
   629→
   630→    // SIO (Service Information Octet)
   631→    const sio = payload[7];
   632→    const serviceIndicator = sio & 0x0F;
   633→    const priority = (sio >> 4) & 0x03;
   634→    const networkIndicator = (sio >> 6) & 0x03;
   635→
   636→    result.decodedFields!.routingLabel = {
   637→      opc: result.opc.display,
   638→      dpc: result.dpc.display,
   639→      sls,
   640→      slc,
   641→      serviceIndicator: this.getSIName(serviceIndicator),
   642→      priority,
   643→      networkIndicator: ['international', 'spare', 'national', 'reserved'][networkIndicator],
   644→    };
   645→
   646→    // The rest is the actual payload (SCCP, ISUP, etc.)
   647→    const userPart = payload.slice(8);
   648→    result.decodedFields!.payloadSize = userPart.length;
   649→    result.decodedFields!.payloadHex = bufferToHex(userPart);
   650→  }

   655→      case 1: return 'Signaling Network Testing/Maintenance';
   656→      case 2: return 'SCCP';
   657→      case 3: return 'TUP';
   658→      case 4: return 'ISUP';
   659→      case 5: return 'Duplex';
   660→      case 6: return 'MSC/TUP';
   661→      case 7: return 'DUP (Reserved)';
   662→      case 8: return 'B-ISUP';
   663→      case 9: return 'Satellite ISUP';
   664→      case 10: return 'Broadband ISUP';
   665→      case 11: return 'SS7 Signaling Test';
   666→      case 12: return 'AAL Type 2 Signaling';
   667→      case 13: return 'BICC';
   668→      case 14: return 'Gateway Control';
   669→      case 15: return 'Reserved';

   671→    }
   672→  }

   675→  // SCCP DECODING
   676→  // ============================================================
   677→
   678→  decodeSCCP(buffer: Buffer, options?: any): DecodeResult<DecodedSCCPMessage> {
   679→    const result: DecodedSCCPMessage = {
   680→      id: generateMessageId(),
   681→      timestamp: new Date(),
   682→      protocol: SS7ProtocolLayer.SCCP,
   683→      direction: options?.direction || 'inbound',
   684→      opc: options?.opc ? parsePointCode(options.opc) : parsePointCode(0),
   685→      dpc: options?.dpc ? parsePointCode(options.dpc) : parsePointCode(0),
   686→      rawData: buffer,
   687→      hexDump: hexDump(buffer),
   688→      packetLength: buffer.length,
   689→    };
   690→
   691→    try {
   692→      const decoder = new BERDecoder(buffer);
   693→      
   694→      // Message Type (first octet)
   695→      const msgTypeByte = decoder.readBytes(1)[0];
   696→      result.messageType = msgTypeByte as SCCPMessageType;
   697→      result.messageName = SCCP_MESSAGE_NAMES[msgTypeByte as SCCPMessageType] || `Unknown (${msgTypeByte})`;
   698→
   699→      result.decodedFields = {
   700→        messageType: result.messageType,
   701→        messageName: result.messageName,
   702→      };
   703→
   704→      // Parse based on message type
   705→      switch (result.messageType) {
   706→        case SCCPMessageType.UDT:   // Unitdata - most common for mobile
   707→        case SCCPMessageType.XUDT:  // Extended Unitdata
   708→        case SCCPMessageType.LUDT:  // Long Unitdata
   709→          this.parseSCCPUnitdata(decoder, result);
   710→          break;
   711→          
   712→        case SCCPMessageType.CR:    // Connection Request
   713→        case SCCPMessageType.CC:    // Connection Confirm
   714→          this.parseSCCPConnection(decoder, result);
   715→          break;
   716→          
   717→        case SCCPMessageType.DT1:   // Data Form 1
   718→        case SCCPMessageType.DT2:   // Data Form 2
   719→          this.parseSCCPData(decoder, result);
   720→          break;
   721→          
   722→        default:
   723→          // Skip remaining parsing for unknown types
   724→          break;
   725→      }
   726→
   727→      return { success: true, data: result };
   728→    } catch (error) {
   729→      return { success: false, error: `SCCP decode error: ${error}`, data: result };
   730→    }
   731→  }
   732→
   733→  private parseSCCPUnitdata(decoder: BERDecoder, result: DecodedSCCPMessage): void {
   734→    // UDT fixed part
   735→    if (decoder.remaining < 3) return;
   736→
   737→    const classAndHandling = decoder.readBytes(1)[0];
   738→    const pointerCalledAddr = decoder.readBytes(1)[0];
   739→    const pointerCallingAddr = decoder.readBytes(1)[0];
   740→    const pointerData = decoder.readBytes(1)[0];
   741→
   742→    result.decodedFields!.messageClass = (classAndHandling >> 4) & 0x03;
   743→    result.decodedFields!.handling = classAndHandling & 0x0F;
   744→
   745→    // Called Party Address (at pointer position)
   746→    const currentPos = decoder.position;
   747→    decoder.position = pointerCalledAddr - 1; // Adjust for 0-based indexing
   748→    result.sccp = {
   749→      ...result.sccp!,
   750→      messageType: result.messageType,
   751→      destinationGlobalTitle: this.parseSCCPAddress(decoder),
   752→    };
   753→    result.decodedFields!.calledPartyAddress = result.sccp.destinationGlobalTitle;
   754→
   755→    // Calling Party Address
   756→    decoder.position = pointerCallingAddr - 1;
   757→    result.sccp = {
   758→      ...result.sccp!,
   759→      sourceGlobalTitle: this.parseSCCPAddress(decoder),
   760→    };
   761→    result.decodedFields!.callingPartyAddress = result.sccp.sourceGlobalTitle;
   762→
   763→    // Data (at pointer position)
   764→    decoder.position = pointerData - 1;
   765→    const dataLength = decoder.remaining;
   766→    const data = decoder.readBytes(dataLength);
   767→    
   768→    result.decodedFields!.dataLength = dataLength;
   769→    result.decodedFields!.dataHex = bufferToHex(data);
   770→
   771→    // Try to decode inner protocol (usually TCAP)
   772→    if (dataLength > 0) {
   773→      result.decodedFields!.innerProtocol = this.detectInnerProtocol(data);
   774→    }
   775→  }
   776→
   777→  private parseSCCPConnection(decoder: BERDecoder, result: DecodedSCCPMessage): void {
   778→    // CR/CC have slightly different structure
   779→    if (decoder.remaining < 1) return;
   780→
   781→    const sourceLocalRef = decoder.readBytes(3); // Variable length
   782→    result.decodedFields!.sourceLocalRef = bufferToHex(sourceLocalRef);
   783→
   784→    if (decoder.remaining >= 3) {
   785→      const destLocalRef = decoder.readBytes(3);
   786→      result.decodedFields!.destLocalRef = bufferToHex(destLocalRef);
   787→    }
   788→
   789→    // Parse optional addresses if present
   790→    if (decoder.remaining > 0) {
   791→      result.sccp = {
   792→        ...result.sccp!,
   793→        messageType: result.messageType,
   794→        sourceGlobalTitle: this.parseSCCPAddress(decoder),
   795→      };
   796→    }
   797→  }
   798→
   799→  private parseSCCPData(decoder: BERDecoder, result: DecodedSCCPMessage): void {
   800→    // DT1/DT2 contain just local reference and data
   801→    if (decoder.remaining < 1) return;
   802→
   803→    const localRef = decoder.readBytes(3);
   804→    result.decodedFields!.localRef = bufferToHex(localRef);
   805→
   806→    if (decoder.remaining > 0) {
   807→      const data = decoder.readBytes(decoder.remaining);
   808→      result.decodedFields!.dataHex = bufferToHex(data);
   809→    }
   810→  }
   811→
   812→  private parseSCCPAddress(decoder: BERDecoder): GlobalTitle | undefined {
   813→    if (decoder.remaining < 2) return undefined;
   814→
   815→    const addrHeader = decoder.readBytes(1)[0];
   816→    const addrLength = decoder.readBytes(1)[0];
   817→
   818→    if (addrLength === 0 || decoder.remaining < addrLength) return undefined;
   819→
   820→    const addrData = decoder.readBytes(addrLength);
   821→
   822→    // Parse address indicators
   823→    const pointCodeIndicator = (addrHeader >> 7) & 0x01;
   824→    const ssnIndicator = (addrHeader >> 6) & 0x01;
   825→    const gtIndicator = (addrHeader >> 5) & 0x01;
   826→    const gtTranslationType = addrHeader & 0x1F;
   827→
   828→    let offset = 0;
   829→    let globalTitle: GlobalTitle | undefined;
   830→    let ssn: SubsystemNumber | undefined;
   831→
   832→    // Point Code (if present)
   833→    if (pointCodeIndicator && offset + 2 <= addrData.length) {
   834→      const pcRaw = addrData.slice(offset, offset + 2).readUInt16BE(0);
   835→      offset += 2;
   836→    }
   837→
   838→    // Subsystem Number (if present)
   839→    if (ssnIndicator && offset < addrData.length) {
   840→      ssn = addrData[offset++] as SubsystemNumber;
   841→    }
   842→
   843→    // Global Title (if present)
   844→    if (gtIndicator && offset < addrData.length) {
   845→      const gtEncodingScheme = (addrData[offset] >> 4) & 0x07;
   846→      const numberingPlan = addrData[offset] & 0x0F;
   847→      offset++;
   848→
   849→      if (offset < addrData.length) {
   850→        const natureOfAddr = (addrData[offset] >> 4) & 0x07;
   851→        const digitsLength = addrData[offset] & 0x0F;
   852→        offset++;
   853→
   854→        if (offset + digitsLength <= addrData.length) {
   855→          const digits = decodeTBCDString(addrData.slice(offset, offset + digitsLength));
   856→          
   857→          globalTitle = {
   858→            type: numberingPlan === 1 ? GlobalTitleType.E164 :
   859→                   numberingPlan === 6 ? GlobalTitleType.E212 :
   860→                   numberingPlan === 7 ? GlobalTitleType.E214 :
   861→                   GlobalTitleType.E164,
   862→            translationType: gtTranslationType,
   863→            numberingPlan,
   864→            encodingScheme: gtEncodingScheme,
   865→            natureOfAddressIndicator: natureOfAddr,
   866→            digits,
   867→            masked: maskMSISDN(digits.startsWith('213') ? '+' + digits : digits),
   868→          };
   869→        }
   870→      }
   871→    }
   872→
   873→    if (globalTitle) {
   874→      globalTitle.destinationSSN = ssn;
   875→    }
   876→
   877→    return globalTitle;
   878→  }

   880→  private detectInnerProtocol(data: Buffer): string {
   881→    if (data.length < 2) return 'unknown';
   882→    
   883→    // TCAP tags are typically 0x60-0x68
   884→    if (data[0] >= 0x60 && data[0] <= 0x68) return 'TCAP';
   885→    
   886→    // Could be other protocols
   887→    return 'unknown';
   888→  }

   891→  // TCAP DECODING

   894→  decodeTCAP(buffer: Buffer, options?: any): DecodeResult<DecodedTCAPMessage> {
   895→    const result: DecodedTCAPMessage = {
   896→      id: generateMessageId(),
   897→      timestamp: new Date(),
   898→      protocol: SS7ProtocolLayer.TCAP,
   899→      direction: options?.direction || 'inbound',
   900→      opc: options?.opc ? parsePointCode(options.opc) : parsePointCode(0),
   901→      dpc: options?.dpc ? parsePointCode(options.dpc) : parsePointCode(0),
   902→      rawData: buffer,
   903→      hexDump: hexDump(buffer),
   904→      packetLength: buffer.length,
   905→      components: [],
   906→    };
   907→
   908→    try {
   909→      const decoder = new BERDecoder(buffer);
   910→      
   911→      // TCAP outer tag
   912→      const outerTag = decoder.decodeTag();
   913→      result.messageType = outerTag.tagNumber as TCAPMessageType;
   914→      result.messageName = TCAP_MESSAGE_NAMES[result.messageType] || `Unknown (${outerTag.tagNumber})`;
   915→
   916→      result.decodedFields = {
   917→        messageType: result.messageType,
   918→        messageName: result.messageName,
   919→      };
   920→
   921→      // Transaction portion
   922→      if (decoder.remaining > 0) {
   923→        const transTag = decoder.decodeTag();
   924→        const transData = decoder.readBytes(transTag.length);
   925→        
   926→        // Originating Transaction ID (OTID)
   927→        if (transData.length >= 4) {
   928→          result.otid = transData.slice(0, 4).toString('hex').toUpperCase();
   929→        }
   930→        
   931→        // Destination Transaction ID (DTID) - for some message types
   932→        if (transData.length >= 8) {
   933→          result.dtid = transData.slice(4, 8).toString('hex').toUpperCase();
   934→        }
   935→      }
   936→
   937→      result.decodedFields.otid = result.otid;
   938→      result.decodedFields.dtid = result.dtid;
   939→
   940→      // Dialogue portion (optional)
   941→      if (decoder.remaining > 0) {
   942→        const savedPos = decoder.position;
   943→        try {
   944→          const dlgTag = decoder.decodeTag();
   945→          if (dlgTag.tagClass === 1 && dlgTag.constructed) {
   946→            result.dialoguePortion = this.parseDialoguePortion(decoder, dlgTag.length);
   947→          } else {
   948→            decoder.position = savedPos;
   949→          }
   950→        } catch {
   951→          decoder.position = savedPos;
   952→        }
   953→      }
   954→
   955→      // Components portion
   956→      if (decoder.remaining > 0) {
   957→        const compTag = decoder.decodeTag();
   958→        if (compTag.constructed) {
   959→          result.components = this.parseComponents(decoder, compTag.length);
   960→        }
   961→      }
   962→
   963→      result.decodedFields.componentCount = result.components.length;
   964→
   965→      return { success: true, data: result };
   966→    } catch (error) {
   967→      return { success: false, error: `TCAP decode error: ${error}`, data: result };
   968→    }
   969→  }
   970→
   971→  private parseDialoguePortion(decoder: BERDecoder, length: number): DialoguePortion {
   972→    const result: DialoguePortion = {};
   973→    const endPos = decoder.position + length;
   974→
   975→    while (decoder.position < endPos) {
   976→      const tag = decoder.decodeTag();
   977→      
   978→      if (tag.tagNumber === 0x06) { // OID
   979→        const oid = decoder.readBytes(tag.length);
   980→        result.applicationContextName = this.formatOID(oid);
   981→      } else if (tag.constructed && tag.length > 0) {
   982→        // User information or other complex field
   983→        decoder.skip(tag.length);
   984→      } else {
   985→        decoder.skip(tag.length);
   986→      }
   987→    }
   988→
   989→    return result;
   990→  }
   991→
   992→  private formatOID(oid: Buffer): string {
   993→    const components: number[] = [];
   994→    let value = 0;
   995→    
   996→    for (let i = 0; i < oid.length; i++) {
   997→      const byte = oid[i];
   998→      value = (value << 7) | (byte & 0x7F);
   999→      if (!(byte & 0x80)) {
  1000→        components.push(value);
  1001→        value = 0;
  1002→      }
  1003→    }
  1004→
  1005→    // Well-known OIDs
  1006→    const oidStr = components.join('.');
  1007→    const knownOIDs: Record<string, string> = {
  1008→      '0.4.0.0.1.0.1.1': 'map-ac (MAP v1)',
  1009→      '0.4.0.0.1.0.1.2': 'map-ac (MAP v2)',
  1010→      '0.4.0.0.1.0.1.3': 'map-ac (MAP v3)',
  1011→      '0.4.0.0.1.0.19.2': 'cap-ac (CAP v2)',
  1012→      '0.4.0.0.1.0.24.3': 'cap-ac (CAP v3)',
  1013→      '0.4.0.0.1.0.20.2': 'IN-CSE-ac',
  1014→      '0.17.134.1': 'Remote Operations',
  1015→    };
  1016→
  1017→    return knownOIDs[oidStr] || oidStr;
  1018→  }

  1020→  private parseComponents(decoder: BERDecoder, length: number): TCAPComponent[] {
  1021→    const components: TCAPComponent[] = [];
  1022→    const endPos = decoder.position + length;
  1023→
  1024→    while (decoder.position < endPos) {
  1025→      try {
  1026→        const compTag = decoder.decodeTag();
  1027→        const component = this.parseComponent(decoder, compTag);
  1028→        components.push(component);
  1029→      } catch {
  1030→        // Skip malformed component
  1031→        break;
  1032→      }
  1033→    }
  1034→
  1035→    return components;
  1036→  }
  1037→
  1038→  private parseComponent(decoder: BERDecoder, tag: any): TCAPComponent {
  1039→    const component: TCAPComponent = {
  1040→      type: tag.tagNumber as ComponentType,
  1041→    };
  1042→
  1043→    switch (component.type) {
  1044→      case ComponentType.INVOKE:
  1045→        component.invokeId = this.decodeInvokeId(decoder);
  1046→        component.operationCode = this.decodeOperationCode(decoder);
  1047→        component.operationName = this.getOperationName(component.operationCode);
  1048→        component.parameters = this.decodeParameters(decoder);
  1049→        break;
  1050→
  1051→      case ComponentType.RETURN_RESULT_LAST:
  1052→      case ComponentType.RETURN_RESULT_NOT_LAST:
  1053→        component.invokeId = this.decodeInvokeId(decoder);
  1054→        // Skip result marker
  1055→        if (decoder.peek() === 0x02) {
  1056→          decoder.decodeTag(); // Result tag
  1057→          decoder.readBytes(decoder.decodeTag().length);
  1058→        }
  1059→        component.parameters = this.decodeParameters(decoder);
  1060→        break;
  1061→
  1062→      case ComponentType.RETURN_ERROR:
  1063→        component.invokeId = this.decodeInvokeId(decoder);
  1064→        component.error = this.decodeError(decoder);
  1065→        break;
  1066→
  1067→      case ComponentType.REJECT:
  1068→        component.invokeId = this.decodeInvokeId(decoder);
  1069→        component.problem = this.decodeProblem(decoder);
  1070→        component.rejectReason = this.getRejectReason(component.problem);
  1071→        break;
  1072→    }
  1073→
  1074→    return component;
  1075→  }
  1076→
  1077→  private decodeInvokeId(decoder: BERDecoder): number {
  1078→    const tag = decoder.decodeTag();
  1079→    if (tag.tagNumber === 0x02) { // INTEGER
  1080→      return decoder.readInteger();
  1081→    }
  1082→    return 0;
  1083→  }
  1084→
  1085→  private decodeOperationCode(decoder: BERDecoder): number {
  1086→    const tag = decoder.decodeTag();
  1087→    if (tag.tagNumber === 0x02) { // Local integer form
  1088→      return decoder.readInteger();
  1089→    } else if (tag.tagNumber === 0x06) { // OID form
  1090→      const oid = decoder.readBytes(tag.length);
  1091→      // Extract last component as operation code
  1092→      const oidStr = oid.toString('.');
  1093→      const parts = oidStr.split('.').map(Number);
  1094→      return parts[parts.length - 1] || 0;
  1095→    }
  1096→    return 0;
  1097→  }
  1098→
  1099→  private getOperationName(code: number): string {
  1100→    // Check both MAP and CAP operations
  1101→    return MAP_OPERATION_NAMES[code as MAPOperationCode] ||
  1102→           CAP_OPERATION_NAMES[code as CAPOperationCode] ||
  1103→           `Operation(${code})`;
  1104→  }
  1105→
  1106→  private decodeParameters(decoder: BERDecoder): any {
  1107→    if (decoder.remaining === 0) return null;
  1108→    
  1109→    try {
  1110→      const tag = decoder.decodeTag();
  1111→      if (!tag.constructed) {
  1112→        return decoder.readBytes(tag.length).toString('hex');
  1113→      }
  1114→      
  1115→      // For SET/SEQUENCE, recursively decode
  1116→      const params: Record<string, any> = {};
  1117→      const endPos = decoder.position + tag.length;
  1118→      
  1119→      while (decoder.position < endPos) {
  1120→        try {
  1121→          const paramTag = decoder.decodeTag();
  1122→          const paramName = `param_${paramTag.tagNumber}`;
  1123→          
  1124→          if (paramTag.constructed) {
  1125→            params[paramName] = this.decodeParameters(decoder);
  1126→          } else {
  1127→            const data = decoder.readBytes(paramTag.length);
  1128→            
  1129→            // Try to interpret common types
  1130→            if (paramTag.tagNumber === 0x04) { // OCTET STRING
  1131→              params[paramName] = decodeTBCDString(data);
  1132→            } else if (paramTag.tagNumber === 0x12) { // NumericString
  1133→              params[paramName] = data.toString('ascii');
  1134→            } else if (paramTag.tagNumber === 0x13) { // PrintableString
  1135→              params[paramName] = data.toString('ascii');
  1136→            } else {
  1137→              params[paramName] = data.toString('hex').toUpperCase();
  1138→            }
  1139→          }
  1140→        } catch {
  1141→          break;
  1142→        }
  1143→      }
  1144→      
  1145→      return Object.keys(params).length > 0 ? params : null;
  1146→    } catch {
  1147→      return null;
  1148→    }
  1149→  }

  1280→  decodeMAP(buffer: Buffer, options?: any): DecodeResult<DecodedMAPMessage> {
  1281→    // First decode as TCAP, then extract MAP-specific info
  1282→    const tcapResult = this.decodeTCAP(buffer, options);
  1283→    
  1284→    if (!tcapResult.success || !tcapResult.data) {
  1285→      return tcapResult as DecodeResult<DecodedMAPMessage>;
  1286→    }
  1287→
  1288→    const tcap = tcapResult.data;
  1289→    const result: DecodedMAPMessage = {
  1290→      ...tcap,
  1291→      protocol: SS7ProtocolLayer.MAP,
  1292→      operation: MAPOperationCode.UPDATE_LOCATION, // Will be updated
  1293→      operationName: '',
  1294→      parameters: {},
  1295→    };
  1296→
  1297→    // Extract MAP operation from TCAP components
  1298→    if (tcap.components.length > 0) {
  1299→      const invokeComp = tcap.components.find(c => c.type === ComponentType.INVOKE);
  1300→      if (invokeComp) {
  1301→        result.operation = (invokeComp.operationCode || 0) as MAPOperationCode;
  1302→        result.operationName = MAP_OPERATION_NAMES[result.operation] || `Unknown(${result.operation})`;
  1303→        result.invokeId = invokeComp.invokeId;
  1304→        result.parameters = invokeComp.parameters || {};
  1305→
  1306→        // Extract subscriber info from parameters
  1307→        this.extractMAPSubscribers(result, invokeComp.parameters);
  1308→      }
  1309→    }
  1310→
  1311→    return { success: true, data: result };
  1312→  }

  1314→  private extractMAPSubscribers(result: DecodedMAPMessage, params: any): void {
  1315→    if (!params || typeof params !== 'object') return;
  1316→
  1317→    // Look for IMSI/MSISDN in various parameter names
  1318→    for (const [key, value] of Object.entries(params)) {
  1319→      if (typeof value === 'string') {
  1320→        // IMSI is 15 digits starting with MCC
  1321→        if (/^\d{15}$/.test(value) && value.startsWith('603')) {
  1322→          result.sourceSubscriber = {
  1323→            imsi: value,
  1324→            msisdn: '', // Would need separate lookup
  1325→            maskedIMSI: maskIMSI(value),
  1326→            maskedMSISDN: '',
  1327→          };
  1328→        }
  1329→        // MSISDN detection (Algerian numbers)
  1330→        if (/^213\d{9}$/.test(value) || /^\+?213\d{9}$/.test(value)) {
  1331→          const msisdn = value.startsWith('+') ? value : '+213' + value;
  1332→          if (result.sourceSubscriber) {
  1333→            result.sourceSubscriber.msisdn = msisdn;
  1334→            result.sourceSubscriber.maskedMSISDN = maskMSISDN(msisdn);
  1335→          } else {
  1336→            result.sourceSubscriber = {
  1337→              imsi: '',
  1338→              msisdn,
  1339→              maskedIMSI: '',
  1340→              maskedMSISDN: maskMSISDN(msisdn),
  1341→            };
  1342→          }
  1343→        }
  1344→      }
  1345→    }
  1346→  }

  1405→  // ============================================================
  1406→
  1407→  decodeISUP(buffer: Buffer, options?: any): DecodeResult<DecodedISUPMessage> {
  1408→    const result: DecodedISUPMessage = {
  1409→      id: generateMessageId(),
  1410→      timestamp: new Date(),
  1411→      protocol: SS7ProtocolLayer.ISUP,
  1412→      direction: options?.direction || 'inbound',
  1413→      opc: options?.opc ? parsePointCode(options.opc) : parsePointCode(0),
  1414→      dpc: options?.dpc ? parsePointCode(options.dpc) : parsePointCode(0),
  1415→      rawData: buffer,
  1416→      hexDump: hexDump(buffer),
  1417→      packetLength: buffer.length,
  1418→      message: ISUPMessageType.IAM,
  1419→      messageName: '',
  1420→      cic: 0,
  1421→    };
  1422→
  1423→    try {
  1424→      // ISUP message structure:
  1425→      // Byte 0: Circuit Identification Code (CIC) - low bits
  1426→      // Byte 1: CIC - high bits + Message Type
  1427→      
  1428→      if (buffer.length < 4) {
  1429→        return { success: false, error: 'Buffer too short for ISUP', data: result };
  1430→      }
  1431→
  1432→      const cicLow = buffer[0];
  1433→      const cicHighAndMsgType = buffer[1];
  1434→      const cic = cicLow | ((cicHighAndMsgType & 0x0F) << 8);
  1435→      const msgType = (cicHighAndMsgType >> 4) & 0x0F;
  1436→      
  1437→      // Some implementations use different byte ordering
  1438→      // Let's also check if byte 2 is the message type
  1439→      const altMsgType = buffer[2];
  1440→      
  1441→      result.cic = cic;
  1442→      result.message = (msgType > 0 && msgType <= 0x30) ? msgType as ISUPMessageType : 
  1443→                       altMsgType as ISUPMessageType;
  1444→      result.messageName = ISUP_MESSAGE_NAMES[result.message] || `Unknown ISUP(${result.message})`;
  1445→
  1446→      result.decodedFields = {
  1447→        circuitIdentificationCode: cic,
  1448→        messageType: result.message,
  1449→        messageName: result.messageName,
  1450→      };
  1451→
  1452→      // Parse mandatory fixed parameters based on message type
  1453→      this.parseISUPMandatoryFixed(result, buffer.slice(3));
  1454→
  1455→      // Parse optional parameters
  1456→      this.parseISUPOptional(result, buffer);
  1457→
  1458→      return { success: true, data: result };
  1459→    } catch (error) {
  1460→      return { success: false, error: `ISUP decode error: ${error}`, data: result };
  1461→    }
  1462→  }
  1463→
  1464→  private parseISUPMandatoryFixed(result: DecodedISUPMessage, buffer: Buffer): void {
  1465→    let offset = 0;
  1466→
  1467→    switch (result.message) {
  1468→      case ISUPMessageType.IAM: // Initial Address Message
  1469→        if (buffer.length >= offset + 4) {
  1470→          result.iams = {
  1471→            natureOfConnection: buffer[offset++],
  1472→            forwardCallIndicators: buffer[offset++],
  1473→            callingPartyCategory: buffer[offset++],
  1474→            transmissionMediumRequirement: buffer[offset++],
  1475→          };
  1476→        }
  1477→        break;
  1478→
  1479→      case ISUPMessageType.ACM: // Address Complete
  1480→        if (buffer.length >= offset + 1) {
  1481→          result.decodedFields!.backwardCallIndicators = buffer[offset];
  1482→        }
  1483→        break;
  1484→
  1485→      case ISUPMessageType.ANM: // Answer
  1486→        // No mandatory fixed parameters
  1487→        break;
  1488→
  1489→      case ISUPMessageType.REL: // Release
  1490→        if (buffer.length >= offset + 2) {
  1491→          result.causeValue = buffer[offset + 1];
  1492→          result.causeDescription = this.getCauseDescription(result.causeValue);
  1493→          result.decodedFields!.causeIndicator = buffer[offset];
  1494→        }
  1495→        break;
  1496→
  1497→      case ISUPMessageType.RLC: // Release Complete
  1498→        // No mandatory fixed parameters
  1499→        break;
  1500→    }
  1501→  }
  1502→
  1503→  private parseISUPOptional(result: DecodedISUPMessage, buffer: Buffer): void {
  1504→    // Find optional parameters pointer
  1505→    // This varies by message type, but usually after mandatory fixed
  1506→    let optPtr = 0;
  1507→    
  1508→    switch (result.message) {
  1509→      case ISUPMessageType.IAM:
  1510→        optPtr = buffer[7] || 0; // Optional pointer after fixed fields
  1511→        break;
  1512→      case ISUPMessageType.REL:
  1513→        optPtr = buffer[3] || 0;
  1514→        break;
  1515→      default:
  1516→        optPtr = buffer[buffer.length - 1] || 0;
  1517→    }
  1518→
  1519→    if (optPtr >= buffer.length) return;
  1520→
  1521→    const optStart = optPtr;
  1522→    const optLength = buffer[optStart];
  1523→
  1524→    if (optStart + optLength > buffer.length) return;
  1525→
  1526→    let offset = optStart + 1;
  1527→    const endOpt = optStart + 1 + optLength;
  1528→
  1529→    while (offset < endOpt && offset < buffer.length) {
  1530→      const paramType = buffer[offset++];
  1531→      const paramLength = buffer[offset++];
  1532→
  1533→      if (offset + paramLength > buffer.length) break;
  1534→
  1535→      const paramData = buffer.slice(offset, offset + paramLength);
  1536→      offset += paramLength;
  1537→
  1538→      this.parseISUPParameter(result, paramType, paramData);
  1539→    }
  1540→  }
  1541→
  1542→  private parseISUPParameter(result: DecodedISUPMessage, paramType: number, paramData: Buffer): void {
  1543→    switch (paramType) {
  1544→      case 0x01: // Called Party Number
  1545→        result.calledPartyNumber = this.decodeISUPNumber(paramData);
  1546→        result.decodedFields!.calledPartyNumber = result.calledPartyNumber;
  1547→        break;
  1548→
  1549→      case 0x0A: // Calling Party Number
  1550→        result.callingPartyNumber = this.decodeISUPNumber(paramData);
  1551→        result.decodedFields!.callingPartyNumber = result.callingPartyNumber;
  1552→        break;
  1553→
  1554→      case 0x1E: // Cause Indicators
  1555→        if (paramData.length >= 2) {
  1556→          result.causeValue = paramData[1] & 0x7F;
  1557→          result.causeDescription = this.getCauseDescription(result.causeValue);
  1558→        }
  1559→        break;
  1560→
  1561→      default:
  1562→        // Store unknown parameters
  1563→        if (!result.decodedFields!.optionalParameters) {
  1564→          result.decodedFields!.optionalParameters = {};
  1565→        }
  1566→        result.decodedFields!.optionalParameters![`param_0x${paramType.toString(16)}`] = bufferToHex(paramData);
  1567→    }
  1568→  }

  1570→  private decodeISUPNumber(data: Buffer): string {
  1571→    if (data.length < 1) return '';
  1572→
  1573→    const oddEvenIndicator = data[0] & 0x80;
  1574→    const natureOfAddress = data[0] & 0x7F;
  1575→    const digits = decodeTBCDString(data.slice(1));
  1576→    
  1577→    // If odd number of digits, last nibble is filler
  1578→    let number = oddEvenIndicator ? digits.slice(0, -1) : digits;
  1579→    
  1580→    // Add + prefix for international numbers
  1581→    if (natureOfAddress === 0x04 || natureOfAddress === 0x05) {
  1582→      number = '+' + number;
  1583→    }
  1584→
  1585→    return number;
  1586→  }
