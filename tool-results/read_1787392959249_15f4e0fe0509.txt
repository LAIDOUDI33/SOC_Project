   447→function QueryBuilder({
   448→  conditions,
   449→  setConditions,
   450→  onExecute
   451→}: {
   452→  conditions: QueryCondition[]
   453→  setConditions: React.Dispatch<React.SetStateAction<QueryCondition[]>>
   454→  onExecute: () => void
   455→}) {
   456→  const addCondition = () => {
   457→    setConditions([
   458→      ...conditions,
   459→      {
   460→        id: `cond-${Date.now()}`,
   461→        field: '',
   462→        operator: 'equals',
   463→        value: '',
   464→        logic: conditions.length > 0 ? 'AND' : 'AND'
   465→      }
   466→    ])
   467→  }
   468→
   469→  const removeCondition = (id: string) => {
   470→    setConditions(conditions.filter(c => c.id !== id))
   471→  }
   472→
   473→  const updateCondition = (id: string, updates: Partial<QueryCondition>) => {
   474→    setConditions(conditions.map(c => c.id === id ? { ...c, ...updates } : c))
   475→  }
   476→
   477→  return (
   478→    <Card className="bg-slate-900 border-slate-700">
   479→      <CardHeader className="pb-3">
   480→        <CardTitle className="text-white text-base flex items-center gap-2">
   481→          <Terminal className="h-4 w-4 text-green-400" />
   482→          Query Builder
   483→        </CardTitle>
   484→      </CardHeader>
   485→      <CardContent className="space-y-3">
   486→        {conditions.map((condition, index) => (
