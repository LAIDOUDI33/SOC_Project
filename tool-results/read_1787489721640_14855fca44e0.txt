    33→// TYPES & INTERFACES FOR INCIDENT RESPONSE
    34→// ============================================================
    35→
    36→type IncidentSeverity = 'LOW' | 'MEDIUM' | 'HIGH' | 'CRITICAL'
    37→type IncidentStatus = 'NEW' | 'TRIAGE' | 'IN_PROGRESS' | 'CONTAINED' | 'ERADICATED' | 'RECOVERY' | 'CLOSED' | 'ESCALATED'
    38→type IncidentCategory = 'MALWARE' | 'PHISHING' | 'DDOS' | 'INTRUSION' | 'DATA_BREACH' | 'INSIDER_THREAT' | 'TELECOM_FRAUD' | 'POLICY_VIOLATION' | 'OTHER'
    39→
    40→interface IOCItem {
    41→  id: string
    42→  type: 'ip' | 'domain' | 'hash' | 'url' | 'email' | 'phone' | 'imsi' | 'imei'
    43→  value: string
    44→  isIndicatorsOfCompromise: boolean
    45→  confidence: number
    46→  source: string
    47→}
    48→
    49→interface Task {
    50→  id: string
    51→  title: string
    52→  description: string
    53→  assignee: string
    54→  status: 'TODO' | 'IN_PROGRESS' | 'BLOCKED' | 'COMPLETED'
    55→  priority: 'LOW' | 'MEDIUM' | 'HIGH' | 'CRITICAL'
    56→  dueDate: string
    57→  completedAt?: string
    58→}
    59→
    60→interface TimelineEvent {
    61→  id: string
    62→  timestamp: string
    63→  type: 'DETECTION' | 'TRIAGE' | 'CONTAINMENT' | 'ERADICATION' | 'RECOVERY' | 'ESCALATION' | 'NOTIFICATION' | 'EVIDENCE' | 'NOTE'
    64→  title: string
    65→  description: string
    66→  author: string
    67→  attachments?: string[]
    68→}
    69→
    70→interface EvidenceItem {
    71→  id: string
    72→  name: string
    73→  type: 'LOG' | 'MEMORY_DUMP' | 'DISK_IMAGE' | 'PCAP' | 'SCREENSHOT' | 'REPORT' | 'OTHER'
    74→  size: string
    75→  hash: string
    76→  collectedBy: string
    77→  collectedAt: string
    78→  description: string
    79→}
    80→
    81→interface CommunicationLog {
    82→  id: string
    83→  timestamp: string
    84→  channel: 'INTERNAL' | 'EMAIL' | 'PHONE' | 'TICKET' | 'SLACK' | 'EXTERNAL'
    85→  from: string
    86→  to: string
    87→  subject: string
    88→  content: string
    89→  isEncrypted: boolean
    90→}
    91→
    92→interface PlaybookStep {
    93→  id: number
    94→  phase: 'PREPARATION' | 'DETECTION_ANALYSIS' | 'CONTAINMENT' | 'ERADICATION' | 'RECOVERY' | 'LESSONS_LEARNED'
    95→  name: string
    96→  description: string
    97→  status: 'PENDING' | 'IN_PROGRESS' | 'COMPLETED' | 'SKIPPED'
    98→  assignee?: string
    99→  completedAt?: string
   100→  notes?: string
   101→}

   103→interface Incident {
   104→  id: string
   105→  ticketNumber: string
   106→  title: string
   107→  description: string
   108→  severity: IncidentSeverity
   109→  status: IncidentStatus
   110→  category: IncidentCategory
   111→  reporter: string
   112→  assignee: string
   113→  team: string
   114→  iocs: IOCItem[]
   115→  tasks: Task[]
   116→  timeline: TimelineEvent[]
   117→  evidence: EvidenceItem[]
   118→  communications: CommunicationLog[]
   119→  playbookSteps: PlaybookStep[]
   120→  createdAt: string
   121→  updatedAt: string
   122→  detectedAt: string
   123→  containmentTarget?: string
   124→  resolutionTarget?: string
   125→  slaBreachRisk: boolean
   126→  tags: string[]
   127→  affectedAssets: string[]
   128→  affectedUsers: number
   129→  businessImpact: 'LOW' | 'MEDIUM' | 'HIGH' | 'CRITICAL'

   351→export const IncidentResponseCenter: React.FC = () => {
   352→  // State Management
   353→  const [incidents, setIncidents] = useState<Incident[]>(mockIncidents)
   354→  const [selectedIncident, setSelectedIncident] = useState<Incident | null>(null)
   355→  const [activeTab, setActiveTab] = useState('overview')
   356→  const [searchQuery, setSearchQuery] = useState('')
   357→  const [filterSeverity, setFilterSeverity] = useState<string>('all')
   358→  const [filterStatus, setFilterStatus] = useState<string>('all')
   359→  const [showNewIncidentDialog, setShowNewIncidentDialog] = useState(false)
   360→  const [viewMode, setViewMode] = useState<'list' | 'kanban'>('list')
   361→
   362→  // Computed Values
   363→  const filteredIncidents = useMemo(() => {
   364→    return incidents.filter(inc => {
   365→      const matchesSearch = searchQuery === '' || 
   366→        inc.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
   367→        inc.ticketNumber.toLowerCase().includes(searchQuery.toLowerCase()) ||
   368→        inc.description.toLowerCase().includes(searchQuery.toLowerCase())
   369→      
   370→      const matchesSeverity = filterSeverity === 'all' || inc.severity === filterSeverity
   371→      const matchesStatus = filterStatus === 'all' || inc.status === filterStatus
   372→      
   373→      return matchesSearch && matchesSeverity && matchesStatus
   374→    })
   375→  }, [incidents, searchQuery, filterSeverity, filterStatus])
   376→
   377→  const stats = useMemo(() => ({
   378→    total: incidents.length,
   379→    newCount: incidents.filter(i => i.status === 'NEW').length,
   380→    activeCount: incidents.filter(i => ['IN_PROGRESS', 'TRIAGE'].includes(i.status)).length,
   381→    criticalCount: incidents.filter(i => i.severity === 'CRITICAL').length,
   382→    containedCount: incidents.filter(i => i.status === 'CONTAINED').length,
   383→    avgResolutionTime: '4h 23m' // Mock value
   384→  }), [incidents])
   385→
   386→  // Handlers
   387→  const handleUpdateIncidentStatus = useCallback((incidentId: string, newStatus: IncidentStatus) => {
   388→    setIncidents(prev => prev.map(inc => 
   389→      inc.id === incidentId ? { ...inc, status: newStatus, updatedAt: new Date().toISOString() } : inc
   390→    ))
