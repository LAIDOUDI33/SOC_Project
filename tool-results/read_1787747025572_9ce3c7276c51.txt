   156→-- Partition by month for retention management (7 years per ANRT)
   157→CREATE TABLE audit_log_partitioned (
   158→    LIKE audit_log INCLUDING ALL
   159→) PARTITION BY RANGE (timestamp);
   160→
   161→-- Create initial partitions (current year + buffer)
   162→DO $$
   163→DECLARE
   164→    start_date DATE := date_trunc('year', CURRENT_DATE);
   165→BEGIN
   166→    FOR i IN 0..11 LOOP
   167→        EXECUTE format('
   168→            CREATE TABLE audit_log_%s PARTITION OF audit_log_partitioned
   169→            FOR VALUES FROM (%s) TO (%s)',
   170→            to_char(start_date + interval '1 month' * i, 'YYYY_MM'),
   171→            start_date + interval '1 month' * i,
   172→            start_date + interval '1 month' * (i + 1)
   173→        );
   174→    END LOOP;
   175→END $$;

   895→-- Active Incidents Summary View
   896→CREATE OR REPLACE VIEW v_active_incidents AS
   897→SELECT 
   898→    i.id,
   899→    i.incident_number,
   900→    i.title,
   901→    i.severity,
   902→    i.status,
   903→    i.assigned_to,
   904→    u.display_name as assigned_to_name,
   905→    i.detected_at,
   906→    i.mttd_minutes,
   907→    i.mttr_minutes,
   908→    EXTRACT(EPOCH FROM (NOW() - i.detected_at))/3600 as age_hours,
   909→    COUNT(a.id) as alert_count
   910→FROM incidents i
   911→LEFT JOIN users u ON i.assigned_to = u.id
   912→LEFT JOIN alerts a ON a.correlated_incident = i.id AND a.status = 'active'
   913→WHERE i.status NOT IN ('closed', 'recovered')
   914→GROUP BY i.id, u.display_name;

   979→RETURNS TRIGGER AS $$
   980→DECLARE
   981→    prefix TEXT;
   982→    seq_num INTEGER;
   983→BEGIN
   984→    prefix := 'INC-' || TO_CHAR(NOW(), 'YYYY');
   985→

   987→    INTO seq_num
   988→    FROM incidents
   989→    WHERE incendid_number LIKE prefix || '%';
   990→

   993→END;

  1003→-- Get SOC Dashboard Summary
  1004→CREATE OR REPLACE PROCEDURE get_soc_dashboard_summary(
  1005→    IN p_hours INTEGER DEFAULT 24,
  1006→    OUT active_incidents INTEGER,
  1007→    OUT critical_alerts INTEGER,
  1008→    OUT unresolved_alerts INTEGER,
  1009→    OUT mttd_avg NUMERIC,
  1010→    OUT top_attack_vectors JSON
  1011→)
  1012→LANGUAGE plpgsql AS $$
  1013→BEGIN
  1014→    SELECT COUNT(*) INTO active_incidents
  1015→    FROM incidents WHERE status NOT IN ('closed', 'recovered');
  1016→    
  1017→    SELECT COUNT(*) INTO critical_alerts
  1018→    FROM alerts WHERE severity = 'critical' AND status = 'active';
  1019→    
  1020→    SELECT COUNT(*) INTO unresolved_alerts
  1021→    FROM alerts WHERE status = 'active';
  1022→    
  1023→    SELECT COALESCE(AVG(mttd_minutes), 0) INTO mttd_avg
  1024→    FROM incidents WHERE detected_at > NOW() - (p_hours || ' hours')::interval;
  1025→    
  1026→    SELECT json_agg(json_build_object(
  1027→        'vector', kill_chain_phase,
  1028→        'count', cnt
  1029→    )) INTO top_attack_vectors
  1030→    FROM (
  1031→        SELECT kill_chain_phase, COUNT(*) as cnt
  1032→        FROM alerts
  1033→        WHERE triggered_at > NOW() - (p_hours || ' hours')::interval
  1034→          AND kill_chain_phase IS NOT NULL
  1035→        GROUP BY kill_chain_phase
  1036→        ORDER BY cnt DESC
  1037→        LIMIT 5
  1038→    ) sub;
  1039→END;

  1041→-- Bulk acknowledge alerts procedure
  1042→CREATE OR REPLACE PROCEDURE bulk_acknowledge_alerts(
  1043→    p_alert_ids UUID[],
  1044→    p_user_id UUID
  1045→)
  1046→LANGUAGE plpgsql AS $$
  1047→DECLARE
  1048→    alert_uuid UUID;
  1049→BEGIN
  1050→    FOREACH alert_uuid IN ARRAY p_alert_ids
  1051→    LOOP
  1052→        UPDATE alerts SET
  1053→            status = 'acknowledged',
  1054→            acknowledged_at = NOW(),
  1055→            acknowledged_by = p_user_id,
  1056→            updated_at = NOW()
  1057→        WHERE id = alert_uuid AND status = 'active';
  1058→        
  1059→        -- Log the acknowledgment
  1060→        INSERT INTO audit_log (actor_id, actor_username, action, resource_type, resource_id, details)
  1061→        SELECT p_user_id, u.username, 'ALERT_ACKNOWLEDGE', 'alert', alert_id::TEXT,
  1062→               jsonb_build_object('alert_id', alert_id, 'bulk_operation', true)
  1063→        FROM users u WHERE u.id = p_user_id;
  1064→    END LOOP;
  1065→END;
