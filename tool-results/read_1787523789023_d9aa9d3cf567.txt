   303→
   304→      // Test Osquery Fleet connection
   305→      if (this.config.osquery.fleetApiUrl) {
   306→        console.log('[EDR] Testing Osquery Fleet connection...');
   307→        const fleetTest = await this.fetchFleet('/api/v1/version');
   308→        if (fleetTest.ok) {

   315→      this.emit('connected', { 
   316→        grr: true, 
   317→        osquery: !!this.config.osquery.fleetApiUrl,
   318→        timestamp: new Date()
   319→      });
   320→      
   321→    } catch (error) {

   350→  /**
   351→   * Search for GRR clients with filtering options
   352→   * Telco-specific: search by department, location, OS, labels
   353→   */
   354→  async searchClients(params: {
   355→    query?: string;       // Search in hostname, IP, username
   356→    limit?: number;       // Max results (default: 100)
   357→    offset?: number;      // Pagination offset
   358→    os?: string;          // Filter by OS (Linux, Windows, Darwin)
   359→    labels?: string[];    // Filter by labels
   360→    onlineThreshold?: number; // Seconds since last ping (default: 3600)
   361→    sortBy?: 'last_seen' | 'first_seen' | 'hostname';
   362→    sortOrder?: 'asc' | 'desc';
   363→  }): Promise<{ clients: GrrClient[]; total: number }> {
   364→    const {
   365→      query,
   366→      limit = 100,
   367→      offset = 0,
   368→      os,
   369→      labels,
   370→      onlineThreshold = 3600,
   371→      sortBy = 'last_seen',
   372→      sortOrder = 'desc'
   373→    } = params;
   374→
   375→    try {
   376→      const searchParams = new URLSearchParams({
   377→        limit: limit.toString(),
   378→        offset: offset.toString(),
   379→        sort_by: sortBy,
   380→        sort_order: sortOrder,
   381→        ...(query && { query }),
   382→        ...(os && { os }),
   383→        ...(onlineThreshold && { online_threshold: onlineThreshold.toString() }),
   384→      });
   385→
   386→      const response = await this.fetchGrr(`/clients?${searchParams}`);
   387→      
   388→      if (!response.ok) {
   389→        throw new Error(`GRR client search failed: ${response.status}`);
   390→      }
   391→
   392→      const data = await response.json();
   393→      const clients: GrrClient[] = (data.clients || []).map(this.transformGrrClient);
   394→      
   395→      // Apply label filtering locally if needed
   396→      let filteredClients = clients;
   397→      if (labels && labels.length > 0) {
   398→        filteredClients = clients.filter(client =>
   399→          labels.some(label => client.labels.includes(label))
   400→        );
   401→      }
   402→
   403→      return {
   404→        clients: filteredClients,
   405→        total: data.total_count || filteredClients.length
   406→      };
   407→
   408→    } catch (error) {
   409→      this.emit('error', error);
   410→      throw error;
   411→    }
   412→  }

   473→  /**
   474→   * Create a new GRR hunt for forensic artifact collection
   475→   * Telco-specific: supports department/location targeting
   476→   */
   477→  async createHunt(params: {
   478→    name: string;
   479→    description: string;
   480→    artifactNames: string[];     // Artifacts to collect
   481→    huntRunnerArgs?: Record<string, any>; // Additional runner arguments
   482→    clientLimit?: number;       // Max clients to target
   483→    clientRate?: number;        // Clients per second
   484→    priority?: 'critical' | 'high' | 'medium' | 'low';
   485→    incidentId?: string;        // Link to SOC incident
   486→    targetLabels?: string[];    // Target specific client labels
   487→    excludeLabels?: string[];   // Exclude certain clients
   488→  }): Promise<GrrHunt> {
   489→    const {
   490→      name,
   491→      description,
   492→      artifactNames,
   493→      huntRunnerArgs = {},
   494→      clientLimit,
   495→      clientRate = 50,
   496→      priority = 'medium',
   497→      incidentId,
   498→      targetLabels,
   499→      excludeLabels,
   500→    } = params;
   501→
   502→    try {
   503→      const huntPayload = {
   504→        hunt_name: name,
   505→        hunt_description: description,
   506→        runner_args: {
   507→          ...huntRunnerArgs,
   508→          ...(clientLimit && { client_limit: { count: clientLimit } }),
   509→          client_rate: clientRate,
   510→        },
   511→        // Use FlowArgs for artifact collection
   512→        flow_args: {
   513→          artifact_list: artifactNames,
   514→        },
   515→        // Telco-specific metadata
   516→        start_hunt: true,
   517→      };
   518→
   519→      const response = await this.fetchGrr('/hunts', {
   520→        method: 'POST',
   521→        body: JSON.stringify(huntPayload),
   522→      });
   523→
   524→      if (!response.ok) {
   525→        const errorText = await response.text();
   526→        throw new Error(`Failed to create hunt: ${response.status} - ${errorText}`);
   527→      }
   528→
   529→      const huntData = await response.json();
   530→      const hunt: GrrHunt = {
   531→        huntId: huntData.hunt_id || huntData.hunt?.huntId,
   532→        name,
   533→        description,
   534→        status: 'STARTED',
   535→        createdAt: new Date().toISOString(),
   536→        updatedAt: new Date().toISOString(),
   537→        createdBy: 'soc-platform',
   538→        clientLimit: clientLimit ? { count: clientLimit } : undefined,
   539→        clientRate,
   540→        artifactNames,
   541→        priority,
   542→        incidentId,
   543→      };
   544→
   545→      this.hunts.set(hunt.huntId, hunt);
   546→      this.stats.activeHunts++;
   547→
   548→      this.emit('huntCreated', hunt);
   549→      console.log(`[EDR] Hunt created: ${name} (${hunt.huntId})`);
   550→
   551→      return hunt;
   552→
   553→    } catch (error) {
   554→      this.emit('error', error);
   555→      throw error;
   556→    }
   557→  }

  1383→  private calculateEndpointRisk(client: any): number {
  1384→    let riskScore = 0;
  1385→    
  1386→    // Factor 1: Days since last seen (stale = risky)
  1387→    const lastSeen = new Date(client.last_seen_at || client.lastSeenAt);
  1388→    const daysSinceLastSeen = (Date.now() - lastSeen.getTime()) / (1000 * 60 * 60 * 24);
  1389→    if (daysSinceLastSeen > 30) riskScore += 30;
  1390→    else if (daysSinceLastSeen > 7) riskScore += 15;
  1391→
  1392→    // Factor 2: OS version (outdated = risky)
  1393→    const osVersion = client.os_version || '';
  1394→    if (osVersion.includes('Windows 7') || osVersion.includes('Server 2008')) riskScore += 25;
  1395→    else if (osVersion.includes('Windows 8') || osVersion.includes('Server 2012')) riskScore += 15;
  1396→
  1397→    // Factor 3: Labels indicating sensitive environments
  1398→    const labels = client.labels || [];
  1399→    if (labels.some(l => l.includes('DMZ'))) riskScore += 20;
  1400→    if (labels.some(l => l.includes('Production'))) riskScore += 10;
  1401→
  1402→    // Factor 4: Ping latency (high = potential issues)
  1403→    const ping = client.last_ping_ms || client.ping || 0;
  1404→    if (ping > 5000) riskScore += 15;
  1405→    else if (ping > 2000) riskScore += 5;
  1406→
  1407→    return Math.min(100, riskScore);
  1408→  }

  1410→  private analyzeHuntFindings(payload: any): TelcoHuntResult['findings'] {
  1411→    const findings: TelcoHuntResult['findings'] = [];
  1412→
  1413→    // Analyze different payload types

  1425→    if (payload.processes) {

  1428→          type: 'process',
  1429→          data: proc,

  1436→    if (payload.network_connections) {

  1439→          type: 'network',
  1440→          data: conn,

  1450→  private assessFileRisk(file: any): 'critical' | 'high' | 'medium' | 'low' | 'info' {
  1451→    const filename = (file.name || '').toLowerCase();
  1452→    const path = (file.path || '').toLowerCase();
  1453→    
  1454→    // Critical indicators
  1455→    if (filename.match(/\.(exe|dll|bat|ps1|vbs|js|jar)$/i) && 
  1456→        path.includes('temp')) return 'critical';
  1457→    if (filename.match(/(mimikatz|cobalt|covenant|sliver|metasploit)/i)) return 'critical';
  1458→    
  1459→    // High indicators
  1460→    if (path.includes('startup') || path.includes('run keys')) return 'high';
  1461→    if (file.size && file.size > 100 * 1024 * 1024 && filename.endsWith('.zip')) return 'high'; // Large zip
  1462→    
  1463→    // Medium indicators
  1464→    if (path.includes('downloads') && filename.match(/\.(exe|msi|scr)/i)) return 'medium';
  1465→    if (file.hash?.md5 && KNOWN_MALWARE_HASHES.includes(file.hash.md5)) return 'critical';
  1466→
  1467→    return 'info';
  1468→  }

  1470→  private assessProcessRisk(process: any): 'critical' | 'high' | 'medium' | 'low' | 'info' {
  1471→    const cmdline = (process.cmdline || process.command_line || '').toLowerCase();
  1472→    const name = (process.name || process.process_name || '').toLowerCase();
  1473→    
  1474→    // Critical process names
  1475→    if (name.match(/(mimikatz|psexec|ncat|netcat|plink|putty)/i)) return 'critical';
  1476→    
  1477→    // Suspicious command lines
  1478→    if (cmdline.includes('encodecommand') || cmdline.includes('-encodedcommand')) return 'high';
  1479→    if (cmdline.includes('downloadstring') || cmdline.includes('downloadfile')) return 'high';
  1480→    if (cmdline.includes('bypass') && cmdline.includes('executionpolicy')) return 'high';
  1481→    
  1482→    // Unusual parent processes
  1483→    if (name === 'cmd.exe' && process.parent?.match(/(word|excel|outlook|winword)/i)) return 'medium';
  1484→    if (name === 'powershell.exe' && process.parent?.match(/(chrome|firefox|edge)/i)) return 'high';
  1485→
  1486→    return 'info';
  1487→  }

  1489→  private assessConnectionRisk(connection: any): 'critical' | 'high' | 'medium' | 'low' | 'info' {
  1490→    const remoteIp = connection.remote_address || connection.remote_ip || '';
  1491→    const remotePort = connection.remote_port || connection.remoteport;
  1492→    const protocol = connection.protocol || connection.proto || '';
  1493→
  1494→    // Known bad ports
  1495→    const suspiciousPorts = [4444, 5555, 6667, 8888, 4443, 1337, 31337];
  1496→    if (suspiciousPorts.includes(remotePort)) return 'critical';
  1497→
  1498→    // Unusual outbound protocols
  1499→    if (protocol === 'irc' || protocol === 'torrent') return 'high';
  1500→    
  1501→    // Connections to known bad countries (simplified check)
  1502→    if (this.isSuspiciousGeolocation(remoteIp)) return 'high';
  1503→
  1504→    return 'info';
  1505→  }
