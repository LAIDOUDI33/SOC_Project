   388→export class OpenctiClient extends EventEmitter {
   389→  private config: OpenctiIntegrationConfig;
   390→  private isConnected: boolean = false;
   391→  
   392→  // Statistics cache
   393→  public stats: OpenctiStatistics = this.initializeStats();
   394→
   395→  constructor(config: OpenctiIntegrationConfig) {
   396→    super();
   397→    this.config = config;
   398→
   399→    // Set up internal event handlers
   400→    this.on('error', (err) => {
   401→      console.error('[OPENCTI] Error:', err.message);
   402→    });
   403→
   404→    this.on('indicatorCreated', (indicator: StixIndicator) => {
   405→      console.log(`[OPENCTI] New indicator: ${indicator.name || indicator.indicator_pattern}`);
   406→      
   407→      if (this.config.enableAutoCorrelation && indicator.x_opencti_score && indicator.x_opencti_score > 70) {
   408→        this.emit('highConfidenceIndicator', indicator);
   409→      }
   410→    });
   411→
   412→    this.on('threatActorUpdate', (actor: StixIntrusionSet) => {
   413→      console.log(`[OPENCTI] Actor update: ${actor.name}`);
   414→    });
   415→  }
   416→
   417→  // ============================================================
   418→  // Connection Management
   419→  // ============================================================
   420→
   421→  /**
   422→   * Initialize connection to OpenCTI platform
   423→   */
   424→  async connect(): Promise<void> {
   425→    try {
   426→      console.log('[OPENCTI] Connecting to', this.config.opencti.apiUrl);
   427→
   428→      // Test authentication with a simple query
   429→      const testResponse = await this.fetchApi('/query', {
   430→        method: 'POST',
   431→        body: JSON.stringify({
   432→          query: `
   433→            query {
   434→              info {
   435→                version
   436→                platform_version
   437→                auth_module
   438→              }
   439→            }
   440→          `,
   441→        }),
   442→      });
   443→
   444→      if (!testResponse.ok) {
   445→        throw new Error(`OpenCTI authentication failed: ${testResponse.status}`);
   446→      }
   447→
   448→      const testData = await testResponse.json();
   449→      console.log(`[OPENCTI] Connected to OpenCTI v${testData.data?.info?.version}`);
   450→
   451→      this.isConnected = true;
   452→
   453→      // Refresh initial statistics
   454→      await this.refreshStats();
   455→
   456→      this.emit('connected', {
   457→        version: testData.data?.info?.version,
   458→        timestamp: new Date(),
   459→      });
   460→
   461→    } catch (error) {
   462→      this.isConnected = false;
   463→      this.emit('error', error);
   464→      throw error;
   465→    }
   466→  }
   467→
   468→  /**
   469→   * Disconnect from OpenCTI
   470→   */
   471→  async disconnect(): Promise<void> {
   472→    this.isConnected = false;
   473→    this.emit('disconnected', { timestamp: new Date() });
   474→    console.log('[OPENCTI] Disconnected');
   475→  }
   476→
   477→  get connectionStatus(): boolean {
   478→    return this.isConnected;
   479→  }

   485→  /**
   486→   * Create a new STIX indicator
   487→   */
   488→  async createIndicator(params: {
   489→    pattern: string;
   490→    patternType: string;
   491→    name?: string;
   492→    description?: string;
   493→    validFrom?: string;
   494→    validUntil?: string;
   495→    confidence?: number;
   496→    markingDefinitions?: string[]; // Marking definition IDs
   497→    labels?: string[];
   498→    killChainPhases?: Array<{ killChainName: string; phaseName: string }>;
   499→    xOpenctiScore?: number;
   500→  }): Promise<StixIndicator> {
   501→    try {
   502→      const mutation = `
   503→        mutation ($input: IndicatorAddInput!) {
   504→          indicatorAdd(input: $input) {
   505→            id
   506→            entity_type
   507→            stix_id
   508→            standard_id
   509→            created_at
   510→            updated_at
   511→            name
   512→            description
   513→            indicator_pattern
   514→            pattern_type
   515→            pattern_version
   516→            valid_from
   517→            valid_until
   518→            confidence
   519→            x_opencti_score
   520→            x_opencti_main_observable_type
   521→            objectLabel {
   522→              edges {
   523→                node {
   524→                  id
   525→                  value
   526→                  color
   527→                }
   528→              }
   529→            }
   530→            objectMarking {
   531→              edges {
   532→                node {
   533→                  id
   534→                  definition_type
   535→                  color
   536→                }
   537→              }
   538→            }
   539→          }
   540→        }
   541→      `;
   542→
   543→      const variables = {
   544→        input: {
   545→          name: params.name || `Indicator-${Date.now()}`,
   546→          description: params.description,
   547→          indicator_pattern: params.pattern,
   548→          pattern_type: params.patternType,
   549→          pattern_version: params.patternVersion || '2.1',
   550→          valid_from: params.validFrom || new Date().toISOString(),
   551→          ...(params.validUntil && { valid_until: params.validUntil }),
   552→          ...(params.confidence !== undefined && { confidence: params.confidence }),
   553→          ...(params.xOpenctiScore !== undefined && { x_opencti_score: params.xOpenctiScore }),
   554→          ...(params.markingDefinitions?.length && {
   555→            objectMarking: params.markingDefinitions.map(id => ({ id })),
   556→          }),
   557→          ...(params.labels?.length && {
   558→            objectLabel: params.labels.map(value => ({ value })),
   559→          }),
   560→          ...(params.killChainPhases?.length && {
   561→            killChainPhases: params.killChainPhases,
   562→          }),
   563→          update: true, // Update if exists
   564→        },
   565→      };
   566→
   567→      const response = await this.fetchApi('/query', {
   568→        method: 'POST',
   569→        body: JSON.stringify({ query: mutation, variables }),
   570→      });
   571→
   572→      if (!response.ok) {
   573→        throw new Error(`Failed to create indicator: ${response.status}`);
   574→      }
   575→
   576→      const result = await response.json();
   577→      const indicator = result.data?.indicatorAdd;
   578→
   579→      if (!indicator) {
   580→        throw new Error('No indicator returned from mutation');
   581→      }
   582→
   583→      this.emit('indicatorCreated', indicator);
   584→      return indicator;
   585→
   586→    } catch (error) {
   587→      this.emit('error', error);
   588→      throw error;
   589→    }
   590→  }

   592→  /**
   593→   * Search indicators by value, pattern, or attributes
   594→   */
   595→  async searchIndicators(params: {
   596→    searchTerm?: string;
   597→    types?: string[];          // Observable types to search
   598→    minScore?: number;
   599→    maxScore?: number;
   600→    validOnly?: boolean;       // Only return currently valid indicators
   601→    limit?: number;
   602→    offset?: number;
   603→  }): Promise<{ results: StixIndicator[]; total: number }> {
   604→    const {
   605→      searchTerm,
   606→      types,
   607→      minScore,
   608→      maxScore,
   609→      validOnly = true,
   610→      limit = 50,
   611→      offset = 0,
   612→    } = params;
   613→
   614→    try {
   615→      const query = `
   616→        query (
   617→          $search: String
   618→          $types: [String]
   619→          $minScore: Int
   620→          $maxScore: Int
   621→          $validOnly: Boolean
   622→          $first: Int
   623→          $offset: Int
   624→        ) {
   625→          indicators(
   626→            search: $search
   627→            filterTypes: $types
   628→            minScore: $minScore
   629→            maxScore: $maxScore
   630→            validOnly: $validOnly
   631→            first: $first
   632→            offset: $offset
   633→            orderBy: x_opencti_score
   634→            orderMode: desc
   635→          ) {
   636→            edges {
   637→              node {
   638→                id
   639→                entity_type
   640→                stix_id
   641→                standard_id
   642→                created_at
   643→                updated_at
   644→                name
   645→                description
   646→                indicator_pattern
   647→                pattern_type
   648→                valid_from
   649→                valid_until
   650→                confidence
   651→                x_opencti_score
   652→                x_opencti_main_observable_type
   653→                objectLabel {
   654→                  edges {
   655→                    node {
   656→                      id
   657→                      value
   658→                      color
   659→                    }
   660→                  }
   661→                }
   662→              }
   663→            }
   664→            pageInfo {
   665→              startCursor
   666→              endCursor
   667→              hasNextPage
   668→              hasPreviousPage
   669→              globalCount
   670→            }
   671→          }
   672→        }
   673→      `;
   674→
   675→      const variables = {
   676→        search: searchTerm,
   677→        types,
   678→        minScore,
   679→        maxScore,
   680→        validOnly,
   681→        first: limit,
   682→        offset,
   683→      };
   684→
   685→      const response = await this.fetchApi('/query', {
   686→        method: 'POST',
   687→        body: JSON.stringify({ query, variables }),
   688→      });
   689→
   690→      if (!response.ok) {
   691→        throw new Error(`Indicator search failed: ${response.status}`);
   692→      }
   693→
   694→      const result = await response.json();
   695→      const edges = result.data?.indicators?.edges || [];
   696→
   697→      return {
   698→        results: edges.map((edge: any) => edge.node),
   699→        total: result.data?.indicators?.pageInfo?.globalCount || edges.length,
   700→      };
   701→
   702→    } catch (error) {
   703→      this.emit('error', error);
   704→      return { results: [], total: 0 };
   705→    }
   706→  }

   712→  async matchObservable(params: {

   809→          x_opencti_main_observable_type
   810→          objectLabel {
   811→            edges {
   812→              node {
   813→                id
   814→                value
   815→                color
   816→              }
   817→            }
   818→          }
   819→        }
   820→      `;
   821→
   822→      const variables = {
   823→        observableValue,
   824→        observableType,
   825→        includeContext,
   826→        includeMitre,

   843→      let context;

   871→        context,

   881→   * Bulk check multiple observables against known IOCs
   882→   * Optimized for high-throughput scenarios
   883→   */
   884→  async bulkMatchObservables(
   885→    observables: Array<{ value: string; type: string }>
   886→  ): Promise<Array<{
   887→    observable: { value: string; type: string };
   888→    matched: boolean;
   889→    topMatch?: StixIndicator;
   890→    score: number;
   891→  }>> {
   892→    try {
   893→      // Process in parallel batches for performance
   894→      const batchSize = 20;
   895→      const results = [];
   896→
   897→      for (let i = 0; i < observables.length; i += batchSize) {
   898→        const batch = observables.slice(i, i + batchSize);
   899→        const batchResults = await Promise.all(
   900→          batch.map(async (obs) => {
   901→            try {
   902→              const matchResult = await this.matchObservable({
   903→                observableValue: obs.value,
   904→                observableType: obs.type,
   905→                includeContext: false,
   906→                includeMitre: false,
   907→              });
   908→
   909→              return {
   910→                observable: obs,
   911→                matched: matchResult.matched,
   912→                topMatch: matchResult.indicators[0],
   913→                score: matchResult.highestScore,
   914→              };
   915→            } catch (error) {
   916→              return {
   917→                observable: obs,
   918→                matched: false,
   919→                score: 0,
   920→              };
   921→            }
   922→          })
   923→        );
   924→
   925→        results.push(...batchResults);
   926→
   927→        // Small delay between batches to avoid rate limiting
   928→        if (i + batchSize < observables.length) {
   929→          await new Promise(resolve => setTimeout(resolve, 100));
   930→        }
   931→      }
   932→
   933→      return results;
   934→
   935→    } catch (error) {
   936→      this.emit('error', error);
   937→      return observables.map(obs => ({
   938→        observable: obs,
   939→        matched: false,
   940→        score: 0,
   941→      }));

  1390→        categories.forEach(c => labels.push(`telco:${c.toLowerCase()}`));
  1391→      }
  1392→      if (severity) {
  1393→        severity.forEach(s => labels.push(`severity:${s}`));
  1394→      }
  1395→      if (tlp) {
  1396→        tlp.forEach(t => labels.push(`tlp:${t.toLowerCase()}`));
  1397→      }
  1398→
  1399→      // Search for indicators with these labels
  1400→      const { results: indicators } = await this.searchIndicators({
  1401→        limit,
  1402→        validOnly: true,
  1403→      });
  1404→
  1405→      // Transform into TelcoThreatIntel format
  1406→      // In production, this would be more sophisticated with proper report extraction
  1407→      const telcoIntel: TelcoThreatIntel[] = indicators
  1408→        .filter(indicator => 
  1409→          indicator.objectLabel?.edges?.some((l: any) => 
  1410→            l.node.value.startsWith('telco:')
  1411→          )
  1412→        )
  1413→        .map(indicator => {
  1414→          const categoryLabel = indicator.objectLabel?.edges?.find((l: any) => 
  1415→            l.node.value.startsWith('telco:')
  1416→          )?.node.value.replace('telco:', '') as TelcoThreatIntel['category'];
  1417→
  1418→          return {
  1419→            category: categoryLabel || 'sim_swap',
  1420→            title: indicator.name || '',
  1421→            description: indicator.description || '',
  1422→            severity: (indicator.objectLabel?.edges?.find((l: any) => 
  1423→              l.node.value.startsWith('severity:')
  1424→            )?.node.value.replace('severity:', '') || 'medium') as TelcoThreatIntel['severity'],
  1425→            iocs: [{
  1426→              type: 'ip' as const,
  1427→              value: this.extractObservableFromPattern(indicator.indicator_pattern),
  1428→              context: indicator.description || '',
  1429→            }],
  1430→            attack_vector: '',
  1431→            affected_network_segments: [],
  1432→            mitigation_steps: [],
  1433→            first_observed: indicator.created_at,
  1434→            last_observed: indicator.updated_at,
  1435→            tlp: (indicator.objectLabel?.edges?.find((l: any) => 
  1436→              l.node.value.startsWith('tlp:')
  1437→            )?.node.value.replace('tlp:', '').toUpperCase() || 'AMBER') as TelcoThreatIntel['tlp'],
  1438→          };
  1439→        });
  1440→
  1441→      return telcoIntel;
  1442→
  1443→    } catch (error) {
  1444→      this.emit('error', error);
  1445→      return [];
  1446→    }
  1447→  }

  1657→  private telcoIocToStix(type: string, value: string): {
  1658→    pattern: string;
  1659→    patternType: string;
  1660→    observableType: string;
  1661→  } {
  1662→    switch (type) {
  1663→      case 'msisdn':
  1664→        return {
  1665→          pattern: `[phone-number:value = '+213${value}']`,
  1666→          patternType: 'stix2',
  1667→          observableType: 'Phone-Number',
  1668→        };
  1669→      case 'imsi':
  1670→        return {
  1671→          pattern: `( [x-opencti-imsi:value = '${value}' )`,
  1672→          patternType: 'stix2',
  1673→          observableType: 'CustomAttribute',
  1674→        };
  1675→      case 'imei':
  1676→        return {
  1677→          pattern: `( [x-opencti-imei:value = '${value}' )`,
  1678→          patternType: 'stix2',
  1679→          observableType: 'CustomAttribute',
  1680→        };
  1681→      case 'ip':
  1682→        return {
  1683→          pattern: `[ipv4-addr:value = '${value}']`,
  1684→          patternType: 'stix2',
  1685→          observableType: 'IPv4-Addr',
  1686→        };
  1687→      case 'domain':
  1688→        return {
  1689→          pattern: `[domain-name:value = '${value}']`,
  1690→          patternType: 'stix2',
  1691→          observableType: 'Domain-Name',
  1692→        };
  1693→      case 'email':
  1694→        return {
  1695→          pattern: `[email-addr:value = '${value}']`,
  1696→          patternType: 'stix2',
  1697→          observableType: 'Email-Addr',
  1698→        };
  1699→      case 'range':
  1700→        return {
  1701→          pattern: `[ipv4-addr:value STARTS WITH '${value.split('.')[0]}']`,
  1702→          patternType: 'stix2',
  1703→          observableType: 'IPv4-Addr',
  1704→        };
  1705→      default:
  1706→        return {
  1707→          pattern: `[artifact:payload_bin = '${Buffer.from(value).toString('base64')}']`,
  1708→          patternType: 'stix2',
  1709→          observableType: 'Artifact',
  1710→        };
  1711→    }
  1712→  }

  1714→  private extractObservableFromPattern(pattern: string): string {
  1715→    // Simple extraction - would need proper STIX parser in production
  1716→    const match = pattern.match(/value\s*=\s*'?([^'\]]+)/);
  1717→    return match ? match[1] : pattern;
  1718→  }
