    86→/**
    87→ * Generate a unique request ID for tracing
    88→ */
    89→function generateRequestId(): string {
    90→  return `req_${Date.now()}_${crypto.randomUUID().replace(/-/g, '').substring(0, 8)}`;
    91→}

    93→/**
    94→ * Create standardized error response
    95→ */
    96→function createErrorResponse(
    97→  statusCode: number,
    98→  errorCode: string,
    99→  message: string,
   100→  requestId: string,
   101→  details?: Record<string, unknown>
   102→): NextResponse<ApiErrorResponse> {
   103→  return NextResponse.json(
   104→    {
   105→      success: false,
   106→      error: message,
   107→      errorCode,
   108→      ...(details && { details }),
   109→      requestId,
   110→      timestamp: new Date().toISOString()
   111→    },
   112→    { status: statusCode }
   113→  );
   114→}

   116→/**
   117→ * Extract client info for audit logging
   118→ */
   119→function extractClientInfo(request: Request): { ipAddress?: string; userAgent?: string } {
   120→  return {
   121→    ipAddress: request.headers.get('x-forwarded-for')?.split(',')[0] || 
   122→               request.headers.get('x-real-ip') || 
   123→               undefined,
   124→    userAgent: request.headers.get('user-agent')?.substring(0, 500) || undefined
   125→  };
   126→}

   155→/**
   156→ * Generate TATC code with proper uniqueness guarantees
   157→ */
   158→function generateTATCCode(): string {
   159→  const year = new Date().getFullYear();
   160→  const timestamp = Date.now().toString(36).toUpperCase();
   161→  const random = crypto.randomUUID().replace(/-/g, '').substring(0, 6).toUpperCase();
   162→  return `TATC-${year}-${timestamp}${random}`;
   163→}
   164→
   165→/**
   166→ * Validate and sanitize offset parameter
   167→ */
   168→function sanitizeOffset(offset: number): number {
   169→  if (offset < 0) return 0;
   170→  if (offset > MAX_OFFSET) return MAX_OFFSET;
   171→  return offset;
   172→}

   174→// ============================================================
   175→// GET /api/incidents - Fetch incidents with caching
   176→// ============================================================
   177→
   178→export async function GET(request: Request) {
   179→  const requestId = generateRequestId();
   180→  const startTime = Date.now();
   181→
   182→  // Authenticate request
   183→  const authResult = await authenticateRequest(request as NextRequest);
   184→  
   185→  if (!authResult.success || !authResult.user) {
   186→    return createErrorResponse(
   187→      401,
   188→      'UNAUTHORIZED',
   189→      authResult.error || 'Authentication required',
   190→      requestId
   191→    );
   192→  }
   193→
   194→  try {
   195→    const { searchParams } = new URL(request.url);
   196→    
   197→    // Validate query parameters using Zod schema
   198→    const queryValidation = IncidentQuerySchema.safeParse(Object.fromEntries(searchParams.entries()));
   199→    
   200→    if (!queryValidation.success) {
   201→      return createErrorResponse(
   202→        400,
   203→        'INVALID_QUERY',
   204→        'Invalid query parameters',
   205→        requestId,
   206→        formatZodError(queryValidation.error)
   207→      );
   208→    }
   209→
   210→    const query = queryValidation.data;
   211→    const limit = query.limit;
   212→    const offset = sanitizeOffset(query.offset);
   213→
   214→    // Build where clause with validated parameters
   215→    const where: any = {};
   216→    
   217→    if (query.severity) where.severity = query.severity;
   218→    if (query.status) where.status = query.status;
   219→    if (query.phase) where.phase = query.phase;
   220→    if (query.type) where.incidentType = query.type;
   221→    
   222→    if (query.search) {
   223→      where.OR = [
   224→        { title: { contains: query.search, mode: 'insensitive' } },
   225→        { description: { contains: query.search, mode: 'insensitive' } },
   226→        { tatcCode: { contains: query.search, mode: 'insensitive' } }
   227→      ];
   228→    }
   229→
   230→    if (query.assigneeId) where.assignedToId = query.assigneeId;
   231→    if (query.commanderId) where.commanderId = query.commanderId;
   232→    
   233→    if (query.dateFrom || query.dateTo) {
   234→      where.detectedAt = {};
   235→      if (query.dateFrom) where.detectedAt.gte = new Date(query.dateFrom);
   236→      if (query.dateTo) where.detectedAt.lte = new Date(query.dateTo);
   237→    }
   238→
   239→    if (query.slaBreach === 'true') where.slaBreach = true;
   240→    if (query.subscribersAffected === 'true') where.subscribersAffected = { gt: 0 };
   241→
   242→    if (query.tags) {
   243→      const tagArray = query.tags.split(',').map(t => t.trim()).filter(Boolean);
   244→      if (tagArray.length > 0) {
   245→        where.tags = { hasSome: tagArray };
   246→      }
   247→    }
   248→
   249→    const includeDetails = query.details === 'true';
   250→    const orderBy: any = {};
   251→    orderBy[query.sortBy] = query.sortOrder;
   252→
   253→    // Execute queries in parallel with caching wrapper
   254→    const [incidents, total, stats] = await Promise.all([
   255→      db.incident.findMany({
   256→        where,
   257→        orderBy,
   258→        take: limit,
   259→        skip: offset,
   260→        include: {
   261→          alerts: includeDetails ? {
   262→            take: 5,
   263→            orderBy: { firstSeen: 'desc' },
   264→            select: { id: true, title: true, severity: true, status: true }
   265→          } : false,
   266→          updates: includeDetails ? {
   267→            take: 3,
   268→            orderBy: { createdAt: 'desc' },
   269→            include: {
   270→              author: { select: { id: true, name: true } }
   271→            }
   272→          } : false,
   273→          tasks: includeDetails ? {
   274→            where: { status: { not: TaskStatus.COMPLETED } },
   275→            take: 5,
   276→            orderBy: { dueDate: 'asc' }
   277→          } : false,
   278→          evidence: includeDetails ? {
   279→            take: 3,
   280→            orderBy: { collectedAt: 'desc' }
   281→          } : false,
   282→          _count: {
   283→            select: {
   284→              alerts: true,
   285→              updates: true,
   286→              tasks: true,
   287→              evidence: true
   288→            }
   289→          }
   290→        },
   291→      }),
   292→      db.incident.count({ where }),
   293→      // Aggregate statistics
   294→      db.incident.groupBy({
   295→        by: ['status', 'severity'],
   296→        _count: { id: true },
   297→        where: {
   298→          status: { notIn: [IncidentStatus.RESOLVED, IncidentStatus.CLOSED] }
   299→        }
   300→      })
   301→    ]);
   302→
   303→    // Calculate processing time for monitoring
   304→    const processingTimeMs = Date.now() - startTime;
   305→
   306→    const response = NextResponse.json({
   307→      success: true,
   308→      data: incidents.map(incident => ({
   309→        id: incident.id,
   310→        tatcCode: incident.tatcCode,
   311→        title: incident.title,
   312→        description: incident.description,
   313→        incidentType: incident.incidentType.toLowerCase(),
   314→        severity: incident.severity.toLowerCase(),
   315→        status: incident.status.toLowerCase(),
   316→        phase: incident.phase.toLowerCase(),
   317→        priority: incident.priority,
   318→        impactScore: incident.impactScore,
   319→        confidenceScore: incident.confidenceScore,
   320→        detectedAt: incident.detectedAt,
   321→        resolvedAt: incident.resolvedAt,
   322→        targetResolution: incident.targetResolution,
   323→        slaBreach: incident.slaBreach,
   324→        assignedToId: incident.assignedToId,
   325→        affectedAssets: incident.affectedAssets ? JSON.parse(incident.affectedAssets) : [],
   326→        affectedServices: incident.affectedServices ? JSON.parse(incident.affectedServices) : [],
   327→        _count: incident._count,
   328→        alerts: incident.alerts,
   329→        updates: incident.updates?.map(update => ({
   330→          id: update.id,
   331→          message: update.message,
   332→          status: update.status?.toLowerCase(),
   333→          phase: update.phase?.toLowerCase(),
   334→          isInternal: update.isInternal,
   335→          createdAt: update.createdAt,
   336→          author: 'author' in update ? update.author : null
   337→        })),
   338→        tasks: incident.tasks,
   339→        evidence: incident.evidence,
   340→      })),
   341→      pagination: {
   342→        total,
   343→        limit,
   344→        offset,
   345→        hasMore: offset + limit < total,
   346→        totalPages: Math.ceil(total / limit),
   347→      },
   348→      statistics: {
   349→        byStatus: stats.reduce((acc, s) => ({
   350→          ...acc,
   351→          [s.status]: {
   352→            ...(acc[s.status] || {}),
   353→            [s.severity.toLowerCase()]: s._count.id
   354→          }
   355→        }), {} as Record<string, any>),
   356→        totalActive: incidents.filter(i => 
   357→          !([IncidentStatus.RESOLVED, IncidentStatus.CLOSED] as IncidentStatus[]).includes(i.status)
   358→        ).length,
   359→      },
   360→      meta: {
   361→        requestId,
   362→        processingTimeMs,
   363→        cached: false // Will be updated by cache middleware if applicable
   364→      },
   365→      timestamp: new Date().toISOString(),
   366→    });
   367→
   368→    // Add performance headers
   369→    response.headers.set('X-Request-ID', requestId);
   370→    response.headers.set('X-Processing-Time', String(processingTimeMs));
   371→    response.headers.set('X-Total-Count', String(total));
   372→
   373→    return response;
   374→
   375→  } catch (error) {
   376→    console.error(`[INCIDENTS][GET] Error (${requestId}):`, error);
   377→    
   378→    return createErrorResponse(
   379→      500,
   380→      'INTERNAL_ERROR',
   381→      'Failed to fetch incidents',
   382→      requestId,
   383→      {
   384→        error: error instanceof Error ? error.message : 'Unknown error',
   385→        stack: process.env.NODE_ENV === 'development' ? 
   386→          (error instanceof Error ? error.stack : undefined) : undefined
   387→      }
   388→    );
   389→  }
   390→}

   392→// ============================================================
   393→// POST /api/incidents - Create or update incidents
   394→// ============================================================
   395→
   396→export async function POST(request: Request) {
   397→  const requestId = generateRequestId();
   398→  const startTime = Date.now();
   399→  const clientInfo = extractClientInfo(request);
   400→
   401→  // Authenticate request
   402→  const authResult = await authenticateRequest(request as NextRequest);
   403→  
   404→  if (!authResult.success || !authResult.user) {
   405→    return createErrorResponse(
   406→      401,
   407→      'UNAUTHORIZED',
   408→      authResult.error || 'Authentication required',
   409→      requestId
   410→    );
   411→  }
   412→  
   413→  const user = authResult.user;
   414→  
   415→  try {
   416→    const body = await request.json();
   417→    const { action, id, ...incidentData } = body;
   418→    
   419→    // Validate action is present
   420→    if (!action || typeof action !== 'string') {
   421→      return createErrorResponse(
   422→        400,
   423→        'MISSING_ACTION',
   424→        'Action is required. Supported actions: create, update, addUpdate, linkAlert, bulkUpdate, bulkClose',
   425→        requestId,
   426→        { supportedActions: ['create', 'update', 'addUpdate', 'linkAlert', 'bulkUpdate', 'bulkClose'] }
   427→      );
   428→    }
   429→
   430→    // Log who is creating/modifying incidents
   431→    console.log(`[INCIDENTS][POST] User ${user.userId} (${user.roleName}) performing action: ${action}`, { requestId });
   432→
   433→    // ========================================
   434→    // CREATE NEW INCIDENT
   435→    // ========================================
   436→    if (action === "create") {
   437→      // Validate input using Zod schema
   438→      const validation = CreateIncidentSchema.safeParse(incidentData);
   439→      
   440→      if (!validation.success) {
   441→        return createErrorResponse(
   442→          400,
   443→          'VALIDATION_ERROR',
   444→          'Invalid incident data',
   445→          requestId,
   446→          formatZodError(validation.error)
   447→        );
   448→      }
   449→
   450→      const data = validation.data;
   451→
   452→      const incident = await db.incident.create({
   453→        data: {
   454→          title: data.title,
   455→          description: data.description,
   456→          incidentType: data.type,
   457→          severity: data.severity,
   458→          status: IncidentStatus.OPEN,
   459→          phase: IncidentPhase.DETECTION,
   460→          priority: data.priority,
   461→          tatcCode: data.tatcCode || generateTATCCode(),
   462→          reportedBy: data.reportedBy,
   463→          assignedToId: data.assigneeId,
   464→          affectedAssets: data.affectedAssets.length > 0 ? JSON.stringify(data.affectedAssets) : null,
   465→          affectedServices: data.affectedServices.length > 0 ? JSON.stringify(data.affectedServices) : null,
   466→          confidenceScore: data.confidenceScore,
   467→          impactScore: data.impactScore,
   468→          subscribersAffected: data.subscribersAffected,
   469→        },
   470→        include: {
   471→          alerts: { take: 5, orderBy: { firstSeen: 'desc' } }
   472→        }
   473→      });
   474→
   475→      // Create initial timeline entry
   476→      await db.incidentUpdate.create({
   477→        data: {
   478→          incidentId: incident.id,
   479→          authorId: data.reporterId || user.userId,
   480→          message: `Incident created: ${incident.title}`,
   481→          status: IncidentStatus.OPEN,
   482→          phase: IncidentPhase.DETECTION
   483→        }
   484→      });
   485→
   486→      // Write audit log (async)
   487→      await writeAuditLog({
   488→        action: 'CREATE',
   489→        entity: 'Incident',
   490→        entityId: incident.id,
   491→        userId: user.userId,
   492→        userName: user.name || user.email,
   493→        newValue: { tatcCode: incident.tatcCode, severity: incident.severity },
   494→        ...clientInfo,
   495→        timestamp: new Date()
   496→      });
   497→
   498→      // Invalidate related caches
   499→      await invalidateByTag(CacheTags.INCIDENTS[0]);
   500→
   501→      const processingTimeMs = Date.now() - startTime;
   502→
   503→      return NextResponse.json({
   504→        success: true,
   505→        message: "Incident created successfully",
   506→        data: incident,
   507→        meta: { requestId, processingTimeMs },
   508→        timestamp: new Date().toISOString(),
   509→      }, { status: 201 });
   510→    }
   511→
   512→    // ========================================
   513→    // UPDATE INCIDENT
   514→    // ========================================
   515→    if (action === "update" && id) {
   516→      // Validate input
   517→      const validation = UpdateIncidentSchema.safeParse(incidentData);
   518→      
   519→      if (!validation.success) {
   520→        return createErrorResponse(
   521→          400,
   522→          'VALIDATION_ERROR',
   523→          'Invalid update data',
   524→          requestId,
   525→          formatZodError(validation.error)
   526→        );
   527→      }
   528→
   529→      const updates = validation.data;
   530→
   531→      // Get current state for audit log
   532→      const existingIncident = await db.incident.findUnique({
   533→        where: { id },
   534→        select: { status: true, phase: true, severity: true, assignedToId: true }
   535→      });
   536→
   537→      if (!existingIncident) {
   538→        return createErrorResponse(
   539→          404,
   540→          'NOT_FOUND',
   541→          `Incident ${id} not found`,
   542→          requestId
   543→        );
   544→      }
   545→
   546→      const updateData: any = { ...updates };
   547→
   548→      // Auto-set resolvedAt when resolving
   549→      if (updates.status === IncidentStatus.RESOLVED) {
   550→        updateData.resolvedAt = new Date();
   551→      }
   552→
   553→      // Handle status transition validation
   554→      if (updates.status) {
   555→        const validTransitions: Record<string, string[]> = {
   556→          [IncidentStatus.OPEN]: ['IN_PROGRESS'],
   557→          [IncidentStatus.IN_PROGRESS]: ['CONTAINED', 'RESOLVED'],
   558→          [IncidentStatus.CONTAINED]: ['ERADICATED', 'RECOVERING'],
   559→          [IncidentStatus.ERADICATED]: ['RECOVERING'],
   560→          [IncidentStatus.RECOVERING]: ['RESOLVED'],
   561→          [IncidentStatus.RESOLVED]: ['CLOSED', 'POST_MORTEM']
   562→        };
   563→
   564→        const allowedTransitions = validTransitions[existingIncident.status] || [];
   565→        if (allowedTransitions.length > 0 && !allowedTransitions.includes(updates.status)) {
   566→          return createErrorResponse(
   567→            400,
   568→            'INVALID_STATUS_TRANSITION',
   569→            `Cannot transition from ${existingIncident.status} to ${updates.status}. Allowed: ${allowedTransitions.join(', ')}`,
   570→            requestId
   571→          );
   572→        }
   573→      }
   574→
   575→      const updatedIncident = await db.incident.update({
   576→        where: { id },
   577→        data: updateData,
   578→      });
   579→
   580→      // Add update log entry
   581→      if (updates.status || updates.phase) {
   582→        await db.incidentUpdate.create({
   583→          data: {
   584→            incidentId: id,
   585→            authorId: user.userId,
   586→            message: `Incident ${updates.status ? `status changed to ${updates.status}` : ''}${updates.phase ? `, phase updated to ${updates.phase}` : ''}`,
   587→            status: updates.status?.toUpperCase(),
   588→            phase: updates.phase?.toUpperCase()
   589→          }
   590→        });
   591→      }
   592→
   593→      // Write audit log
   594→      await writeAuditLog({
   595→        action: 'UPDATE',
   596→        entity: 'Incident',
   597→        entityId: id,
   598→        userId: user.userId,
   599→        userName: user.name || user.email,
   600→        previousValue: existingIncident,
   601→        newValue: updates,
   602→        ...clientInfo,
   603→        timestamp: new Date()
   604→      });
   605→
   606→      // Invalidate caches
   607→      await invalidateByTag(CacheTags.INCIDENTS[0]);
   608→      await invalidateByTag(CacheTags.INCIDENT_DETAIL(id)[0]);
   609→
   610→      const processingTimeMs = Date.now() - startTime;
   611→
   612→      return NextResponse.json({
   613→        success: true,
   614→        message: `Incident ${id} updated`,
   615→        data: updatedIncident,
   616→        meta: { requestId, processingTimeMs },
   617→        timestamp: new Date().toISOString(),
   618→      });
   619→    }
   620→
   621→    // ========================================
   622→    // ADD UPDATE/COMMENT TO INCIDENT
   623→    // ========================================
   624→    if (action === "addUpdate" && id) {
   625→      // Validate input
   626→      const validation = AddUpdateSchema.safeParse(incidentData);
   627→      
   628→      if (!validation.success) {
   629→        return createErrorResponse(
   630→          400,
   631→          'VALIDATION_ERROR',
   632→          'Invalid update data',
   633→          requestId,
   634→          formatZodError(validation.error)
   635→        );
   636→      }
   637→
   638→      const data = validation.data;
   639→
   640→      // Verify incident exists
   641→      const incident = await db.incident.findUnique({
   642→        where: { id },
   643→        select: { id: true, status: true }
   644→      });
   645→
   646→      if (!incident) {
   647→        return createErrorResponse(
   648→          404,
   649→          'NOT_FOUND',
   650→          `Incident ${id} not found`,
   651→          requestId
   652→        );
   653→      }
   654→
   655→      const update = await db.incidentUpdate.create({
   656→        data: {
   657→          incidentId: id,
   658→          authorId: data.authorId,
   659→          message: data.message,
   660→          isInternal: data.isInternal,
   661→          status: data.status?.toUpperCase(),
   662→          phase: data.phase?.toUpperCase()
   663→        },
   664→        include: {
   665→          author: { select: { id: true, name: true } }
   666→        }
   667→      });
   668→
   669→      // If status/phase change included, also update the incident
   670→      if (data.status || data.phase) {
   671→        const incidentUpdate: any = {};
   672→        if (data.status) incidentUpdate.status = data.status.toUpperCase();
   673→        if (data.phase) incidentUpdate.phase = data.phase.toUpperCase();
   674→        
   675→        if (data.status === IncidentStatus.RESOLVED) {
   676→          incidentUpdate.resolvedAt = new Date();
   677→        }
   678→
   679→        await db.incident.update({
   680→          where: { id },
   681→          data: incidentUpdate
   682→        });
   683→      }
   684→
   685→      // Audit log
   686→      await writeAuditLog({
   687→        action: 'ADD_UPDATE',
   688→        entity: 'Incident',
   689→        entityId: id,
   690→        userId: user.userId,
   691→        userName: user.name || user.email,
   692→        newValue: { message: data.message.substring(0, 100), isInternal: data.isInternal },
   693→        ...clientInfo,
   694→        timestamp: new Date()
   695→      });
   696→
   697→      // Invalidate caches
   698→      await invalidateByTag(CacheTags.INCIDENT_DETAIL(id)[0]);
   699→
   700→      const processingTimeMs = Date.now() - startTime;
   701→
   702→      return NextResponse.json({
   703→        success: true,
   704→        message: "Update added to incident",
   705→        data: update,
   706→        meta: { requestId, processingTimeMs },
   707→        timestamp: new Date().toISOString(),
   708→      });
   709→    }
   710→
   711→    // ========================================
   712→    // LINK ALERT TO INCIDENT
   713→    // ========================================
   714→    if (action === "linkAlert" && id) {
   715→      // Validate input
   716→      const validation = LinkAlertSchema.safeParse(incidentData);
   717→      
   718→      if (!validation.success) {
   719→        return createErrorResponse(
   720→          400,
   721→          'VALIDATION_ERROR',
   722→          'Invalid alert linking data',
   723→          requestId,
   724→          formatZodError(validation.error)
   725→        );
   726→      }
   727→
   728→      const data = validation.data;
   729→
   730→      // Verify both exist
   731→      const [incident, alert] = await Promise.all([
   732→        db.incident.findUnique({ where: { id }, select: { id: true, tatcCode: true } }),
   733→        db.alert.findUnique({ where: { id: data.alertId }, select: { id: true, incidentId: true } })
   734→      ]);
   735→
   736→      if (!incident) {
   737→        return createErrorResponse(404, 'NOT_FOUND', `Incident ${id} not found`, requestId);
   738→      }
   739→
   740→      if (!alert) {
   741→        return createErrorResponse(404, 'NOT_FOUND', `Alert ${data.alertId} not found`, requestId);
   742→      }
   743→
   744→      if (alert.incidentId) {
   745→        return createErrorResponse(
   746→          409,
   747→          'CONFLICT',
   748→          `Alert ${data.alertId} is already linked to another incident`,
   749→          requestId
   750→        );
   751→      }
   752→
   753→      await db.alert.update({
   754→        where: { id: data.alertId },
   755→        data: {
   756→          incidentId: id,
   757→          status: 'ESCALATED'
   758→        }
   759→      });
   760→
   761→      // Audit log
   762→      await writeAuditLog({
   763→        action: 'LINK_ALERT',
   764→        entity: 'Incident',
   765→        entityId: id,
   766→        userId: user.userId,
   767→        userName: user.name || user.email,
   768→        newValue: { alertId: data.alertId, reason: data.linkReason },
   769→        ...clientInfo,
   770→        timestamp: new Date()
   771→      });
   772→
   773→      // Invalidate caches
   774→      await invalidateByTag(CacheTags.INCIDENT_DETAIL(id)[0]);
   775→
   776→      const processingTimeMs = Date.now() - startTime;
   777→
   778→      return NextResponse.json({
   779→        success: true,
   780→        message: `Alert ${data.alertId} linked to incident ${incident.tatcCode || id}`,
   781→        meta: { requestId, processingTimeMs },
   782→        timestamp: new Date().toISOString(),
   783→      });
   784→    }
   785→
   786→    // ========================================
   787→    // UNKNOWN ACTION
   788→    // ========================================
   789→    return createErrorResponse(
   790→      400,
   791→      'INVALID_ACTION',
   792→      `Unsupported action: ${action}. Supported actions: create, update, addUpdate, linkAlert`,
   793→      requestId,
   794→      { supportedActions: ['create', 'update', 'addUpdate', 'linkAlert'] }
   795→    );
   796→
   797→  } catch (error) {
   798→    console.error(`[INCIDENTS][POST] Error (${requestId}):`, error);
   799→    
   800→    // Handle specific Prisma errors
   801→    let errorMessage = 'Failed to process request';
   802→    let errorCode = 'INTERNAL_ERROR';
   803→    
   804→    if (error && typeof error === 'object' && 'code' in error) {
   805→      const prismaError = error as { code: string; meta?: Record<string, unknown> };
   806→      
   807→      if (prismaError.code === 'P2025') {
   808→        errorMessage = 'Record not found';
   809→        errorCode = 'NOT_FOUND';
   810→      } else if (prismaError.code === 'P2002') {
   811→        errorMessage = 'Unique constraint violation';
   812→        errorCode = 'DUPLICATE';
   813→      } else if (prismaError.code === 'P2003') {
   814→        errorMessage = 'Foreign key constraint failed';
   815→        errorCode = 'INVALID_REFERENCE';
   816→      }
   817→    }
   818→    
   819→    return createErrorResponse(
   820→      500,
   821→      errorCode,
   822→      errorMessage,
   823→      requestId,
   824→      {
   825→        error: error instanceof Error ? error.message : 'Unknown error',
   826→        action: body.action
   827→      }
   828→    );
   829→  }
   830→}

   832→// ============================================================
   833→// DELETE /api/incidents - Soft delete incident (admin only)
   834→// ============================================================
   835→
   836→export async function DELETE(request: Request) {
   837→  const requestId = generateRequestId();
   838→  
   839→  const authResult = await authenticateRequest(request as NextRequest);
   840→  
   841→  if (!authResult.success || !authResult.user) {
   842→    return createErrorResponse(401, 'UNAUTHORIZED', 'Authentication required', requestId);
   843→  }
   844→
   845→  // Check admin permissions
   846→  if (user.role !== 'ADMIN' && user.role !== 'SUPERADMIN') {
   847→    return createErrorResponse(403, 'FORBIDDEN', 'Admin privileges required', requestId);
   848→  }
   849→
   850→  try {
   851→    const { searchParams } = new URL(request.url);
   852→    const id = searchParams.get('id');
   853→
   854→    if (!id) {
   855→      return createErrorResponse(400, 'MISSING_ID', 'Incident ID is required', requestId);
   856→    }
   857→
   858→    // Soft delete - mark as deleted instead of removing
   859→    const deletedIncident = await db.incident.update({
   860→      where: { id },
   861→      data: {
   862→        status: 'DELETED',
   863→        deletedAt: new Date(),
   864→        deletedBy: user.userId
   865→      }
   866→    });
   867→
   868→    // Audit log
   869→    await writeAuditLog({
   870→      action: 'DELETE',
   871→      entity: 'Incident',
   872→      entityId: id,
   873→      userId: user.userId,
   874→      userName: user.name || user.email,
   875→      ...extractClientInfo(request),
   876→      timestamp: new Date()
   877→    });
   878→
   879→    // Invalidate caches
   880→    await invalidateByTag(CacheTags.INCIDENTS[0]);
   881→    await invalidateByTag(CacheTags.INCIDENT_DETAIL(id)[0]);
   882→
   883→    return NextResponse.json({
   884→      success: true,
   885→      message: `Incident ${id} soft-deleted`,
   886→      meta: { requestId },
   887→      timestamp: new Date().toISOString()
   888→    });
   889→
   890→  } catch (error) {
   891→    console.error(`[INCIDENTS][DELETE] Error (${requestId}):`, error);
   892→    return createErrorResponse(500, 'INTERNAL_ERROR', 'Failed to delete incident', requestId);
   893→  }
   894→}
