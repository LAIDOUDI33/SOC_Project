   100→// Sample data generators
   101→function generateTimeSeriesData(points: number = 60): TrafficData[] {
   102→  const data: TrafficData[] = [];
   103→  const now = new Date();
   104→  
   105→  for (let i = points; i >= 0; i--) {
   106→    const time = new Date(now.getTime() - i * 60000);
   107→    const baseMPS = 2500 + Math.sin(i / 10) * 500;
   108→    const noise = (Math.random() - 0.5) * 400;
   109→    
   110→    data.push({
   111→      timestamp: time.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' }),
   112→      mps: Math.round(Math.max(100, baseMPS + noise)),
   113→      bytes: Math.round((baseMPS + noise) * 150),
   114→      mapCount: Math.round((baseMPS + noise) * 0.45),
   115→      capCount: Math.round((baseMPS + noise) * 0.15),
   116→      isupCount: Math.round((baseMPS + noise) * 0.25),
   117→      sccpCount: Math.round((baseMPS + noise) * 0.10),
   118→    });
   119→  }
   120→  
   121→  return data;
   122→}
