   102→/**
   103→ * Fetch data from API with error handling
   104→ */
   105→async function fetchFromAPI<T>(endpoint: string, params?: Record<string, string>): Promise<T> {
   106→  const searchParams = new URLSearchParams(params);
   107→  const url = `/api/grafana${endpoint}${params ? `?${searchParams.toString()}` : ''}`;
   108→
   109→  const response = await fetch(url);
   110→  
   111→  if (!response.ok) {
   112→    const errorData = await response.json().catch(() => ({}));
   113→    throw new Error(errorData.message || `API error: ${response.status}`);
   114→  }
   115→
   116→  const result = await response.json();
   117→  return result.data as T;
   118→}
   119→
   120→/**
   121→ * Post data to API with error handling
   122→ */
   123→async function postToAPI<T>(endpoint: string, body: Record<string, unknown>): Promise<T> {
   124→  const response = await fetch(`/api/grafana${endpoint}`, {
   125→    method: 'POST',
   126→    headers: { 'Content-Type': 'application/json' },
   127→    body: JSON.stringify(body),
   128→  });
   129→
   130→  if (!response.ok) {
   131→    const errorData = await response.json().catch(() => ({}));
   132→    throw new Error(errorData.message || `API error: ${response.status}`);
   133→  }
   134→
   135→  const result = await response.json();
   136→  return result.data as T;
   137→}

   143→/**
   144→ * Hook for fetching and managing Grafana dashboards
   145→ * @param initialLimit Initial page size
   146→ * @returns Dashboard list with filtering capabilities
   147→ * 
   148→ * @example
   149→ * ```tsx
   150→ * const { data, loading, filters, setFilters, refetch } = useDashboards(20);
   151→ * ```
   152→ */
   153→export function useDashboards(initialLimit: number = 50): UseDashboardsReturn {
   154→  const [data, setData] = useState<DashboardSearchResult[] | null>(null);
   155→  const [loading, setLoading] = useState<boolean>(true);
   156→  const [error, setError] = useState<string | null>(null);
   157→  const [total, setTotal] = useState<number>(0);
   158→  const [filters, setFiltersState] = useState<UseDashboardsReturn['filters']>({});
   159→
   160→  const fetchData = useCallback(async () => {
   161→    setLoading(true);
   162→    setError(null);
   163→
   164→    try {
   165→      const params: Record<string, string> = {
   166→        limit: initialLimit.toString(),
   167→      };
   168→
   169→      if (filters.query) params.query = filters.query;
   170→      if (filters.tag?.length) params.tag = filters.tag.join(',');
   171→      if (filters.starred !== undefined) params.starred = filters.starred.toString();
   172→      if (filters.type) params.type = filters.type;
   173→
   174→      const result = await fetchFromAPI<{
   175→        items: DashboardSearchResult[];
   176→        total: number;
   177→        page: number;
   178→        pageSize: number;
   179→        totalPages: number;
   180→      }>('/dashboards', params);
   181→
   182→      setData(result.items);
   183→      setTotal(result.total);
   184→    } catch (err) {
   185→      setError(err instanceof Error ? err.message : 'Failed to fetch dashboards');
   186→      setData(null);
   187→    } finally {
   188→      setLoading(false);
   189→    }
   190→  }, [initialLimit, filters]);
   191→
   192→  useEffect(() => {
   193→    fetchData();
   194→  }, [fetchData]);
   195→
   196→  const setFilters = useCallback((newFilters: Partial<typeof filters>) => {
   197→    setFiltersState(prev => ({ ...prev, ...newFilters }));
   198→  }, []);
   199→
   200→  return {
   201→    data,
   202→    loading,
   203→    error,
   204→    total,
   205→    filters,
   206→    setFilters,
   207→    refetch: fetchData,
   208→  };
   209→}

   211→/**
   212→ * Hook for getting a single dashboard by UID
   213→ * @param uid Dashboard UID to fetch
   214→ * @returns Full dashboard object
   215→ */
   216→export function useDashboard(uid: string | null): UseDataReturn<GrafanaDashboard> {
   217→  const [data, setData] = useState<GrafanaDashboard | null>(null);
   218→  const [loading, setLoading] = useState<boolean>(!!uid);
   219→  const [error, setError] = useState<string | null>(null);
   220→
   221→  const fetchData = useCallback(async () => {
   222→    if (!uid) {
   223→      setData(null);
   224→      setLoading(false);
   225→      return;
   226→    }
   227→
   228→    setLoading(true);
   229→    setError(null);
   230→
   231→    try {
   232→      const dashboard = await fetchFromAPI<GrafanaDashboard>(`/dashboards/${uid}`);
   233→      setData(dashboard);
   234→    } catch (err) {
   235→      setError(err instanceof Error ? err.message : 'Failed to fetch dashboard');
   236→      setData(null);
   237→    } finally {
   238→      setLoading(false);
   239→    }
   240→  }, [uid]);
   241→
   242→  useEffect(() => {
   243→    fetchData();
   244→  }, [fetchData]);
   245→
   246→  return {
   247→    data,
   248→    loading,
   249→    error,
   250→    refetch: fetchData,
   251→  };
   252→}

   258→/**
   259→ * Hook for monitoring alert rules
   260→ * @returns Alert rules with management functions
   261→ * 
   262→ * @example
   263→ * ```tsx
   264→ * const { data, summary, pauseAlert, resumeAlert, togglePause } = useAlertRules();
   265→ * ```
   266→ */
   267→export function useAlertRules(): UseAlertRulesReturn {
   268→  const [data, setData] = useState<AlertRule[] | null>(null);
   269→  const [loading, setLoading] = useState<boolean>(true);
   270→  const [error, setError] = useState<string | null>(null);
   271→  const [total, setTotal] = useState<number>(0);
   272→
   273→  const fetchData = useCallback(async () => {
   274→    setLoading(true);
   275→    setError(null);
   276→
   277→    try {
   278→      const result = await fetchFromAPI<{
   279→        items: AlertRule[];
   280→        total: number;
   281→        summary: {
   282→          byState: Record<string, number>;
   283→          bySeverity: Record<string, number>;
   284→        };
   285→      }>('/alerts');
   286→
   287→      setData(result.items);
   288→      setTotal(result.total);
   289→    } catch (err) {
   290→      setError(err instanceof Error ? err.message : 'Failed to fetch alerts');
   291→      setData(null);
   292→    } finally {
   293→      setLoading(false);
   294→    }
   295→  }, []);
   296→
   297→  useEffect(() => {
   298→    fetchData();
   299→
   300→    // Auto-refresh every 30 seconds for real-time alert status
   301→    const interval = setInterval(fetchData, 30000);
   302→    return () => clearInterval(interval);
   303→  }, [fetchData]);
   304→
   305→  // Calculate summary from data
   306→  const summary = useMemo(() => {
   307→    if (!data) {
   308→      return { ok: 0, alerting: 0, paused: 0, critical: 0, high: 0, warning: 0 };
   309→    }
   310→
   311→    return {
   312→      ok: data.filter(a => a.currentState === 'ok').length,
   313→      alerting: data.filter(a => a.currentState === 'alerting').length,
   314→      paused: data.filter(a => a.isPaused).length,
   315→      critical: data.filter(a => a.labels.severity === 'critical').length,
   316→      high: data.filter(a => a.labels.severity === 'high').length,
   317→      warning: data.filter(a => a.labels.severity === 'warning').length,
   318→    };
   319→  }, [data]);
   320→
   321→  /**
   322→   * Pause an alert rule
   323→   */
   324→  const pauseAlert = useCallback(async (uid: string): Promise<void> => {
   325→    try {
   326→      await postToAPI('/alerts', { action: 'pause', uid, paused: true });
   327→      await fetchData(); // Refresh data
   328→    } catch (err) {
   329→      console.error('Failed to pause alert:', err);
   330→      throw err;
   331→    }
   332→  }, [fetchData]);
   333→
   334→  /**
   335→   * Resume a paused alert rule
   336→   */
   337→  const resumeAlert = useCallback(async (uid: string): Promise<void> => {
   338→    try {
   339→      await postToAPI('/alerts', { action: 'pause', uid, paused: false });
   340→      await fetchData(); // Refresh data
   341→    } catch (err) {
   342→      console.error('Failed to resume alert:', err);
   343→      throw err;
   344→    }
   345→  }, [fetchData]);
   346→
   347→  /**
   348→   * Toggle pause state of an alert rule
   349→   */
   350→  const togglePause = useCallback(async (uid: string): Promise<void> => {
   351→    const alert = data?.find(a => a.uid === uid);
   352→    if (alert?.isPaused) {
   353→      await resumeAlert(uid);
   354→    } else {
   355→      await pauseAlert(uid);
   356→    }
   357→  }, [data, pauseAlert, resumeAlert]);
   358→
   359→  return {
   360→    data,
   361→    loading,
   362→    error,
   363→    total,
   364→    summary,
   365→    pauseAlert,
   366→    resumeAlert,
   367→    togglePause,
   368→    refetch: fetchData,
   369→  };
   370→}

   529→        message: result.message || (result.success ? 'Connection successful' : 'Connection failed'),

   532→      return {
   533→        success: false,
   534→        message: err instanceof Error ? err.message : 'Connection test failed',
   535→      };
   536→    }
   537→  }, [fetchData]);
   538→
   539→  return {
   540→    data,
   541→    loading,
   542→    error,
   543→    healthyCount,
   544→    unhealthyCount,
   545→    testConnection,
   546→    refetch: fetchData,
   547→  };
   548→}

   816→/**
   817→ * Hook for rendering embedded Grafana dashboards
   818→ * @param config Initial embedding configuration
   819→ * @returns Embedding utilities
   820→ * 
   821→ * @example
   822→ * ```tsx
   823→ * const { embedUrl, iframeHtml, updateConfig } = useDashboardView({
   824→ *   dashboardUid: 'soc-overview',
   825→ *   theme: 'dark',
   826→ * });
   827→ * 
   828→ * <div dangerouslySetInnerHTML={{ __html: iframeHtml }} />
   829→ * ```
   830→ */
   831→export function useDashboardView(config: EmbeddedDashboardConfig): UseDashboardViewReturn {
   832→  const [currentConfig, setCurrentConfig] = useState<EmbeddedDashboardConfig>(config);
   833→  const [isLoading, setIsLoading] = useState<boolean>(true);
   834→  const [error, setError] = useState<string | null>(null);
   835→
   836→  // Get Grafana base URL from environment or default
   837→  const grafanaUrl = process.env.NEXT_PUBLIC_GRAFANA_URL || 'http://localhost:3000';
   838→
   839→  /**
   840→   * Generate embedded URL based on configuration
   841→   */
   842→  const generateEmbedUrl = useCallback((cfg: EmbeddedDashboardConfig): string => {
   843→    const params = new URLSearchParams();
   844→
   845→    // Theme
   846→    if (cfg.theme && cfg.theme !== 'current') {
   847→      params.set('theme', cfg.theme);
   848→    }
   849→
   850→    // Panel selection
   851→    if (cfg.panelId) {
   852→      params.set('viewPanel', cfg.panelId.toString());
   853→    }
   854→
   855→    // Time range
   856→    if (cfg.initialTime) {
   857→      params.set('from', cfg.initialTime.from);
   858→      params.set('to', cfg.initialTime.to);
   859→    }
   860→
   861→    // Variables
   862→    if (cfg.initialVariables) {
   863→      Object.entries(cfg.initialVariables).forEach(([key, value]) => {
   864→        params.set(`var-${key}`, value);
   865→      });
   866→    }
   867→
   868→    // Display options
   869→    if (cfg.autoRefresh) {
   870→      params.set('refresh', cfg.autoRefresh);
   871→    }
   872→
   873→    // Kiosk mode (hide unnecessary UI)
   874→    if (!cfg.showToolbar) {
   875→      params.set('kiosk', 'tv');
   876→    }
   877→    if (cfg.showTimePicker !== undefined) {
   878→      params.set('showTimePicker', String(cfg.showTimePicker));
   879→    }
   880→    if (cfg.showHeader !== undefined) {
   881→      params.set('showHeader', String(cfg.showHeader));
   882→    }
   883→
   884→    const queryString = params.toString();
   885→    return `${grafanaUrl}/d/${cfg.dashboardUid}${queryString ? `?${queryString}` : ''}`;
   886→  }, [grafanaUrl]);
   887→
   888→  /**
   889→   * Generate iframe HTML
   890→   */
   891→  const generateIframeHtml = useCallback((url: string, width?: number, height?: number): string => {
   892→    const style = [
   893→      'border: none',
   894→      'border-radius: 8px',
   895→      width: width ? `${width}px` : '100%',
   896→      height: height ? `${height}px` : '700px',
   897→      'box-shadow: 0 4px 6px rgba(0, 0, 0, 0.1)',
   898→    ].join('; ');
   899→
   900→    return `<iframe src="${url}" style="${style}" allow="fullscreen" loading="lazy"></iframe>`;
   901→  }, []);
   902→
   903→  const embedUrl = useMemo(() => generateEmbedUrl(currentConfig), [currentConfig, generateEmbedUrl]);
   904→  
   905→  const iframeHtml = useMemo(
   906→    () => generateIframeHtml(embedUrl, currentConfig.width, currentConfig.height),
   907→    [embedUrl, currentConfig.width, currentConfig.height, generateIframeHtml]
   908→  );
   909→
   910→  /**
   911→   * Update embedding configuration
   912→   */
   913→  const updateConfig = useCallback((updates: Partial<EmbeddedDashboardConfig>) => {
   914→    setCurrentConfig(prev => ({ ...prev, ...updates }));
   915→    setIsLoading(true); // Trigger reload animation
   916→    
   917→    // Simulate loading
   918→    setTimeout(() => setIsLoading(false), 500);
   919→  }, []);
   920→
   921→  /**
   922→   * Force refresh of embedded content
   923→   */
   924→  const refresh = useCallback(() => {
   925→    setIsLoading(true);
   926→    setTimeout(() => setIsLoading(false), 500);
   927→  }, []);
   928→
   929→  // Initialize
   930→  useEffect(() => {
   931→    setIsLoading(true);
   932→    
   933→    // Simulate iframe load check
   934→    const timer = setTimeout(() => {
   935→      setIsLoading(false);
   936→    }, 1000);
   937→
   938→    return () => clearTimeout(timer);
   939→  }, [currentConfig.dashboardUid]);
   940→
   941→  return {
   942→    config: currentConfig,
   943→    embedUrl,
   944→    iframeHtml,
   945→    isLoading,
   946→    error,
   947→    updateConfig,
   948→    refresh,
   949→  };
   950→}

   952→// ============================================================
   953→// Combined Monitoring Hook
   954→// ============================================================
   955→
   956→/**
   957→ * Comprehensive hook combining all Grafana monitoring features
   958→ * Ideal for main SOC dashboard view
   959→ * @returns Complete monitoring state
   960→ * 
   961→ * @example
   962→ * ```tsx
   963→ * const { stats, dashboards, alerts, datasources, isReady, refreshAll } = useGrafanaDashboard();
   964→ * 
   965→ * if (!isReady) return <LoadingSpinner />;
   966→ * 
   967→ * return (
   968→ *   <Dashboard>
   969→ *     <KPICards kpis={stats.kpis} />
   970→ *     <AlertSummary {...alerts} />
   971→ *     <DashboardGrid dashboards={dashboards} />
   972→ *   </Dashboard>
   973→ * );
   974→ * ```
   975→ */
   976→export function useGrafanaDashboard() {
   977→  const stats = useGrafanaStats();
   978→  const dashboards = useDashboards(10);
   979→  const alerts = useAlertRules();
   980→  const datasources = useDataSources();
   981→  const incidents = useAlertIncidents('alerting');
   982→
   983→  /** Check if all data is loaded */
   984→  const isReady = useMemo(
   985→    () => !stats.loading && !dashboards.loading && !alerts.loading && !datasources.loading,
   986→    [stats.loading, dashboards.loading, alerts.loading, datasources.loading]
   987→  );
   988→
   989→  /** Has any errors */
   990→  const hasErrors = useMemo(
   991→    () => !!(stats.error || dashboards.error || alerts.error || datasources.error),
   992→    [stats.error, dashboards.error, alerts.error, datasources.error]
   993→  );
   994→
   995→  /** Refresh all data sources */
   996→  const refreshAll = useCallback(() => {
   997→    stats.refetch();
   998→    dashboards.refetch();
   999→    alerts.refetch();
  1000→    datasources.refetch();
  1001→    incidents.refetch();
  1002→  }, [stats, dashboards, alerts, datasources, incidents]);
  1003→
  1004→  /** Overall system health status */
  1005→  const systemStatus = useMemo<'healthy' | 'degraded' | 'critical'>(() => {
  1006→    if (alerts.summary.alerting > 5 || alerts.summary.critical > 2) return 'critical';
  1007→    if (alerts.summary.alerting > 0 || datasources.unhealthyCount > 0) return 'degraded';
  1008→    return 'healthy';
  1009→  }, [alerts.summary, datasources.unhealthyCount]);
  1010→
  1011→  return {
  1012→    // Individual hooks
  1013→    stats,
  1014→    dashboards,
  1015→    alerts,
  1016→    datasources,
  1017→    incidents,
  1018→
  1019→    // Computed values
  1020→    isReady,
  1021→    hasErrors,
  1022→    systemStatus,
  1023→    refreshAll,
  1024→  };
  1025→}
