   487→/** Create standardized API response */
   488→function createApiResponse<T>(
   489→  data: T,
   490→  status: number = 200,
   491→  message?: string
   492→): NextResponse<GrafanaAPIResponse<T>> {
   493→  return NextResponse.json(
   494→    {
   495→      success: status >= 200 && status < 300,
   496→      message: message || (status >= 200 && status < 300 ? 'Success' : 'Error'),
   497→      data,
   498→    },
   499→    { status }
   500→  );
   501→}
   502→
   503→/** Parse search parameters */
   504→function parseSearchParams(request: NextRequest): {
   505→  state?: AlertState;
   506→  limit?: number;
   507→  page?: number;
   508→  query?: string;
   509→} {
   510→  const { searchParams } = new URL(request.url);
   511→  
   512→  return {
   513→    state: searchParams.get('state') as AlertState | undefined,
   514→    limit: parseInt(searchParams.get('limit') || '50', 10),
   515→    page: parseInt(searchParams.get('page') || '1', 10),
   516→    query: searchParams.get('query') || undefined,
   517→  };
   518→}
   519→
   520→/** Filter alerts based on parameters */
   521→function filterAlerts(params: ReturnType<typeof parseSearchParams>): {
   522→  alerts: AlertRule[];
   523→  total: number;
   524→} {
   525→  let filtered = [...mockAlertRules];
   526→  
   527→  if (params.state) {
   528→    filtered = filtered.filter(a => 
   529→      a.currentState === params.state ||
   530→      (params.state === 'paused' && a.isPaused)
   531→    );
   532→  }
   533→  
   534→  if (params.query) {
   535→    const query = params.query.toLowerCase();
   536→    filtered = filtered.filter(a =>
   537→      a.title.toLowerCase().includes(query) ||
   538→      a.ruleGroup.toLowerCase().includes(query) ||
   539→      Object.values(a.labels).some(l => String(l).toLowerCase().includes(query))
   540→    );
   541→  }
   542→  
   543→  const total = filtered.length;
   544→  const limit = params.limit || 50;
   545→  const page = params.page || 1;
   546→  const start = (page - 1) * limit;
   547→  
   548→  return {
   549→    alerts: filtered.slice(start, start + limit),
   550→    total,
   551→  };
   552→}

   558→/**
   559→ * GET /api/grafana/alerts
   560→ * List all alert rules with optional filtering
   561→ */
   562→export async function GET(request: NextRequest): Promise<NextResponse> {
   563→  try {
   564→    const { searchParams } = new URL(request.url);
   565→    const action = searchParams.get('action');
   566→
   567→    // Handle different actions
   568→    if (action === 'history') {
   569→      return getAlertHistory(request);
   570→    }
   571→
   572→    if (action === 'incidents') {
   573→      return getAlertIncidents(request);
   574→    }
   575→
   576→    if (action === 'summary') {
   577→      return getAlertSummary();
   578→    }
   579→
   580→    // Default: list alert rules
   581→    const params = parseSearchParams(request);
   582→    const { alerts, total } = filterAlerts(params);
   583→
   584→    // Calculate summary stats
   585→    const summary = {
   586→      total: mockAlertRules.length,
   587→      byState: {
   588→        ok: mockAlertRules.filter(a => a.currentState === 'ok').length,
   589→        alerting: mockAlertRules.filter(a => a.currentState === 'alerting').length,
   590→        paused: mockAlertRules.filter(a => a.isPaused).length,
   591→        nodata: mockAlertRules.filter(a => a.health === 'nodata').length,
   592→        error: mockAlertRules.filter(a => a.health === 'error').length,
   593→      },
   594→      bySeverity: {
   595→        critical: mockAlertRules.filter(a => a.labels.severity === 'critical').length,
   596→        high: mockAlertRules.filter(a => a.labels.severity === 'high').length,
   597→        warning: mockAlertRules.filter(a => a.labels.severity === 'warning').length,
   598→        info: mockAlertRules.filter(a => a.labels.severity === 'info').length,
   599→      },
   600→    };
   601→
   602→    return createApiResponse({
   603→      items: alerts,
   604→      total,
   605→      page: params.page || 1,
   606→      pageSize: params.limit || 50,
   607→      totalPages: Math.ceil(total / (params.limit || 50)),
   608→      summary,
   609→    });
   610→  } catch (error) {
   611→    console.error('Error fetching alerts:', error);
   612→    return createApiResponse(
   613→      { error: 'Failed to fetch alerts' } as unknown as never,
   614→      500,
   615→      'Internal server error'
   616→    );
   617→  }
   618→}

   620→/**
   621→ * POST /api/grafana/alerts
   622→ * Create new alert rule or batch operations
   623→ */
   624→export async function POST(request: NextRequest): Promise<NextResponse> {
   625→  try {
   626→    const body = await request.json();
   627→    const { action, ...alertData } = body;
   628→
   629→    if (action === 'pause') {
   630→      return handlePauseResume(alertData);
   631→    }
   632→
   633→    if (action === 'batch-pause') {
   634→      return handleBatchPauseResume(alertData);
   635→    }
   636→
   637→    // Create new alert rule
   638→    const newAlert: AlertRule = {
   639→      id: mockAlertRules.length + 1,
   640→      uid: `alert-${Date.now()}`,
   641→      orgId: 1,
   642→      folderId: body.folderId || 0,
   643→      folderUid: body.folderUid || 'default',
   644→      title: body.title || 'Untitled Alert',
   645→      condition: body.condition || 'B',
   646→      data: body.data || [],
   647→      ruleGroup: body.ruleGroup || 'Default',
   648→      noDataState: body.noDataState || 'NoData',
   649→      execErrState: body.execErrState || 'Alerting',
   650→      for: body.for || '5m',
   651→      annotations: body.annotations || {},
   652→      labels: body.labels || {},
   653→      isPaused: body.isPaused ?? false,
   654→      currentState: 'normal',
   655→      stateSince: new Date().toISOString(),
   656→      stateChangesCount: 0,
   657→      intervalSeconds: body.intervalSeconds || 60,
   658→      notifications: body.notifications || [],
   659→      created: new Date().toISOString(),
   660→      updated: new Date().toISOString(),
   661→      version: 1,
   662→      health: 'ok',
   663→    };
   664→
   665→    mockAlertRules.push(newAlert);
   666→
   667→    return createApiResponse(newAlert, 201, 'Alert rule created successfully');
   668→  } catch (error) {
   669→    console.error('Error creating alert:', error);
   670→    return createApiResponse(
   671→      { error: 'Failed to create alert rule' } as unknown as never,
   672→      400,
   673→      'Invalid request body'
   674→    );
   675→  }
   676→}

   754→/**
   755→ * Handle pause/resume single alert
   756→ */
   757→function handlePauseResume(data: Record<string, unknown>): NextResponse {
   758→  const { uid, paused } = data;
   759→
   760→  if (!uid || typeof paused !== 'boolean') {
   761→    return createApiResponse(null, 400, 'uid and paused fields are required');
   762→  }
   763→
   764→  const alertIndex = mockAlertRules.findIndex(a => a.uid === uid);
   765→  
   766→  if (alertIndex === -1) {
   767→    return createApiResponse(null, 404, 'Alert rule not found');
   768→  }
   769→
   770→  mockAlertRules[alertIndex].isPaused = paused;
   771→  mockAlertRules[alertIndex].currentState = paused ? 'paused' : 'ok';
   772→  mockAlertRules[alertIndex].updated = new Date().toISOString();
   773→
   774→  return createApiResponse({
   775→    ...mockAlertRules[alertIndex],
   776→    message: `Alert ${paused ? 'paused' : 'resumed'} successfully`,
   777→  });
   778→}
   779→
   780→/**
   781→ * Handle batch pause/resume
   782→ */
   783→function handleBatchPauseResume(data: Record<string, unknown>): NextResponse {
   784→  const { uids, paused } = data;
   785→
   786→  if (!Array.isArray(uids) || typeof paused !== 'boolean') {
   787→    return createApiResponse(null, 400, 'uids array and paused boolean are required');
   788→  }
   789→
   790→  let successCount = 0;
   791→  let failedCount = 0;
   792→  const errors: Array<{ item: string; reason: string }> = [];
   793→
   794→  uids.forEach((uid: unknown) => {
   795→    const alertIndex = mockAlertRules.findIndex(a => a.uid === uid);
   796→    
   797→    if (alertIndex === -1) {
   798→      failedCount++;
   799→      errors.push({ item: String(uid), reason: 'Alert not found' });
   800→      return;
   801→    }
   802→
   803→    mockAlertRules[alertIndex].isPaused = paused;
   804→    mockAlertRules[alertIndex].currentState = paused ? 'paused' : 'ok';
   805→    mockAlertRules[alertIndex].updated = new Date().toISOString();
   806→    successCount++;
   807→  });
   808→
   809→  return createApiResponse({
   810→    successCount,
   811→    failedCount,
   812→    errors,
   813→    message: `${successCount} alerts ${paused ? 'paused' : 'resumed'}, ${failedCount} failed`,
   814→  });
   815→}

   851→/**
   852→ * Update alert by ID
   853→ */
   854→async function updateAlert(id: string, request: NextRequest): Promise<NextResponse> {
   855→  const alertIndex = mockAlertRules.findIndex(a => a.id === parseInt(id, 10));
   856→
   857→  if (alertIndex === -1) {
   858→    return createApiResponse(null, 404, 'Alert rule not found');
   859→  }
   860→
   861→  try {
   862→    const updates = await request.json();
   863→
   864→    mockAlertRules[alertIndex] = {
   865→      ...mockAlertRules[alertIndex],
   866→      ...(updates.title && { title: updates.title }),
   867→      ...(updates.annotations && { annotations: updates.annotations }),
   868→      ...(updates.labels && { labels: updates.labels }),
   869→      ...(updates.isPaused !== undefined && { isPaused: updates.isPaused }),
   870→      ...(updates.condition && { condition: updates.condition }),
   871→      ...(updates.data && { data: updates.data }),
   872→      ...(updates.for && { for: updates.for }),
   873→      ...(updates.noDataState && { noDataState: updates.noDataState }),
   874→      ...(updates.execErrState && { execErrState: updates.execErrState }),
   875→      ...(updates.notifications && { notifications: updates.notifications }),
   876→      updated: new Date().toISOString(),
   877→      version: mockAlertRules[alertIndex].version + 1,
   878→    };
   879→
   880→    return createApiResponse(mockAlertRules[alertIndex]);
   881→  } catch (error) {
   882→    return createApiResponse(
   883→      { error: 'Invalid request body' } as unknown as never,
   884→      400,
   885→      'Invalid request body'
   886→    );
   887→  }
   888→}

   893→async function deleteAlert(id: string): Promise<NextResponse> {
   894→  const alertIndex = mockAlertRules.findIndex(a => a.id === parseInt(id, 10));

   896→  if (alertIndex === -1) {
