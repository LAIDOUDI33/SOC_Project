    51→function ClockDisplay() {
    52→  const [time, setTime] = useState<string>('')
    53→  const [mounted, setMounted] = useState(false)
    54→
    55→  useEffect(() => {
    56→    setMounted(true)
    57→    setTime(new Date().toLocaleTimeString())
    58→    
    59→    const timer = setInterval(() => {
    60→      setTime(new Date().toLocaleTimeString())
    61→    }, 1000)
    62→
    63→    return () => clearInterval(timer)
    64→  }, [])
    65→
    66→  if (!mounted) {
    67→    return <div className="text-cyan-400 font-mono w-20">&nbsp;</div>
    68→  }
    69→
    70→  return <div className="text-cyan-400 font-mono">{time}</div>
    71→}
