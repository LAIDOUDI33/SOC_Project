   150→/**
   151→ * Initialize ANSSI alignments in database
   152→ */
   153→export async function initializeAnssiAlignments(frameworkId: string): Promise<void> {
   154→  const existingCount = await db.ansiAlignment.count();
   155→  if (existingCount > 0) {
   156→    console.log(`[Compliance Engine] ${existingCount} ANSSI alignments already exist`);
   157→    return;
   158→  }
   159→
   160→  // Get all controls mapped by controlId for lookup
   161→  const controls = await db.complianceControl.findMany({
   162→    where: { frameworkId }
   163→  });
   164→  const controlMap = new Map(controls.map(c => [c.controlId, c.id]));
   165→
   166→  let created = 0;
   167→  for (const alignment of ANSSI_ALIGNMENTS) {
   168→    // Find related ARTP control IDs
   169→    const artpControlIds = alignment.artpMapping
   170→      .map(ref => controlMap.get(ref))
   171→      .filter(Boolean) as string[];
   172→
   173→    // Use first mapped control as primary, or create without specific control
   174→    const localControlId = artpControlIds[0];
   175→
   176→    await db.ansiAlignment.create({
   177→      data: {
   178→        localControlId: localControlId || '',
   179→        anssiReference: alignment.ansiReference,
   180→        anssiDomain: alignment.ansiDomain,
   181→        anssiCategory: alignment.ansiCategory,
   182→        mappingStrength: (alignment.id.includes('FULL') ? MappingStrength.FULL :
   183→                          alignment.id.includes('SUBSTANTIAL') ? MappingStrength.SUBSTANTIAL :
   184→                          alignment.id.includes('PARTIAL') ? MappingStrength.PARTIAL :
   185→                          MappingStrength.MINIMAL) as MappingStrength,
   186→        implementationStatus: AnssiImplementationStatus.NOT_IMPLEMENTED,
   187→        certRelevant: alignment.certRelevant,
   188→        certLevel: alignment.certLevel
   189→      }
   190→    });
   191→    created++;
   192→  }
   193→
   194→  console.log(`[Compliance Engine] Initialized ${created} ANSSI alignments`);
   195→}

   288→/**
   289→ * Recalculate overall assessment scores from control assessments
   290→ */
   291→async function recalculateAssessmentScores(assessmentId: string): Promise<void> {
   292→  const controlAssessments = await db.controlAssessment.findMany({
   293→    where: { assessmentId },
   294→    include: { control: true }
   295→  });
   296→
   297→  let compliantCount = 0;
   298→  let partialCount = 0;
   299→  let nonCompliantCount = 0;
   300→  let naCount = 0;
   301→  let notAssessedCount = 0;
   302→  let weightedSum = 0;
   303→  let totalWeight = 0;
   304→
   305→  const criticalityWeights = {
   306→    [ControlCriticality.CRITICAL]: 5,
   307→    [ControlCriticality.HIGH]: 4,
   308→    [ControlCriticality.MEDIUM]: 3,
   309→    [ControlCriticality.LOW]: 2
   310→  };
   311→
   312→  for (const ca of controlAssessments) {
   313→    const weight = criticalityWeights[ca.control.criticality] || 3;
   314→    totalWeight += weight;
   315→
   316→    switch (ca.result) {
   317→      case AssessmentResult.COMPLIANT:
   318→        compliantCount++;
   319→        weightedSum += weight * 100;
   320→        break;
   321→      case AssessmentResult.PARTIALLY_COMPLIANT:
   322→        partialCount++;
   323→        weightedSum += weight * (ca.score || 50);
   324→        break;
   325→      case AssessmentResult.NON_COMPLIANT:
   326→        nonCompliantCount++;
   327→        weightedSum += weight * 0;
   328→        break;
   329→      case AssessmentResult.NOT_APPLICABLE:
   330→        naCount++;
   331→        totalWeight -= weight; // Exclude from weighting
   332→        break;
   333→      default:
   334→        notAssessedCount++;
   335→        break;
   336→    }
   337→  }
   338→
   339→  const overallScore = totalWeight > 0 ? Math.round((weightedSum / totalWeight) * 100) / 100 : 0;
   340→
   341→  await db.complianceAssessment.update({
   342→    where: { id: assessmentId },
   343→    data: {
   344→      overallScore,
   345→      compliantCount,
   346→      partialCount,
   347→      nonCompliantCount,
   348→      notApplicableCount: naCount,
   349→      notAssessedCount: notAssessedCount
   350→    }
   351→  });
   352→}

   354→/**
   355→ * Complete an assessment with final calculations
   356→ */
   357→export async function completeAssessment(
   358→  assessmentId: string,
   359→  executiveSummary?: string,
   360→  recommendations?: string[]
   361→): Promise<ComplianceAssessment> {
   362→  // Final score calculation
   363→  await recalculateAssessmentScores(assessmentId);
   364→
   365→  // Generate gap findings from non-compliant controls
   366→  await generateGapFindings(assessmentId);
   367→
   368→  const assessment = await db.complianceAssessment.update({
   369→    where: { id: assessmentId },
   370→    data: {
   371→      status: AssessmentStatus.COMPLETED,
   372→      completedAt: new Date(),
   373→      executiveSummary,
   374→      recommendations: JSON.stringify(recommendations || [])
   375→    }
   376→  });
   377→
   378→  console.log(`[Compliance Engine] Completed assessment ${assessmentId} with score ${assessment.overallScore}%`);
   379→  return assessment;
   380→}

   421→/**
   422→ * Calculate score for a specific assessment
   423→ */
   424→export async function calculateAssessmentScore(
   425→  assessmentId: string
   426→): Promise<ComplianceScore> {
   427→  const controlAssessments = await db.controlAssessment.findMany({
   428→    where: { assessmentId },
   429→    include: { control: true }
   430→  });
   431→
   432→  let compliant = 0, partial = 0, nonCompliant = 0, na = 0, notAssessed = 0;
   433→  let criticalGaps = 0;
   434→  let weightedSum = 0, totalWeight = 0;
   435→
   436→  const weights = {
   437→    [ControlCriticality.CRITICAL]: 5,
   438→    [ControlCriticality.HIGH]: 4,
   439→    [ControlCriticality.MEDIUM]: 3,
   440→    [ControlCriticality.LOW]: 2
   441→  };
   442→
   443→  for (const ca of controlAssessments) {
   444→    const w = weights[ca.control.criticality] || 3;
   445→    totalWeight += w;
   446→
   447→    switch (ca.result) {
   448→      case AssessmentResult.COMPLIANT:
   449→        compliant++;
   450→        weightedSum += w * 100;
   451→        break;
   452→      case AssessmentResult.PARTIALLY_COMPLIANT:
   453→        partial++;
   454→        weightedSum += w * (ca.score || 50);
   455→        break;
   456→      case AssessmentResult.NON_COMPLIANT:
   457→        nonCompliant++;
   458→        if (ca.control.criticality === ControlCriticality.CRITICAL) {
   459→          criticalGaps++;
   460→        }
   461→        break;
   462→      case AssessmentResult.NOT_APPLICABLE:
   463→        na++;
   464→        totalWeight -= w;
   465→        break;
   466→      default:
   467→        notAssessed++;
   468→        break;
   469→    }
   470→  }
   471→
   472→  return {
   473→    overall: totalWeight > 0 ? Math.round((weightedSum / totalWeight) * 100) / 100 : 0,
   474→    compliant,
   475→    partiallyCompliant: partial,
   476→    nonCompliant,
   477→    notAssessed,
   478→    notApplicable: na,
   479→    criticalGaps,
   480→    weightedScore: totalWeight > 0 ? Math.round((weightedSum / totalWeight) * 100) / 100 : 0
   481→  };
   482→}

   488→/**
   489→ * Generate gap findings from assessment results
   490→ */
   491→async function generateGapFindings(assessmentId: string): Promise<void> {
   492→  const nonCompliantControls = await db.controlAssessment.findMany({
   493→    where: {
   494→      assessmentId,
   495→      result: { in: [AssessmentResult.NON_COMPLIANT, AssessmentResult.PARTIALLY_COMPLIANT] }
   496→    },
   497→    include: { control: true }
   498→  });
   499→
   500→  for (const ca of nonCompliantControls) {
   501→    // Check if gap finding already exists for this control
   502→    const existing = await db.gapFinding.findFirst({
   503→      where: {
   504→        assessmentId,
   505→        controlId: ca.controlId
   506→      }
   507→    });
   508→
   509→    if (!existing) {
   510→      await db.gapFinding.create({
   511→        data: {
   512→          assessmentId,
   513→          controlId: ca.controlId,
   514→          title: `Gap: ${ca.control.name}`,
   515→          description: ca.findings || `Control ${ca.control.controlId} is assessed as ${ca.result}`,
   516→          gapType: 'CONTROL_GAP',
   517→          severity: mapCriticalityToGapSeverity(ca.control.criticality),
   518→          remediationStatus: RemediationStatus.OPEN,
   519→          ownerId: ca.owner || undefined,
   520→          dueDate: ca.targetDate
   521→        }
   522→      });
   523→    }
   524→  }
   525→}
   526→
   527→function mapCriticalityToGapSeverity(criticality: ControlCriticality): any {
   528→  switch (criticality) {
   529→    case ControlCriticality.CRITICAL: return 'CRITICAL';
   530→    case ControlCriticality.HIGH: return 'HIGH';
   531→    case ControlCriticality.MEDIUM: return 'MEDIUM';
   532→    case ControlCriticality.LOW: return 'LOW';
   533→    default: return 'INFORMATIONAL';
   534→  }
   535→}

   537→/**
   538→ * Get comprehensive gap analysis for an organization
   539→ */
   540→export async function getGapAnalysis(frameworkId?: string): Promise<GapAnalysisResult> {
   541→  const where: any = {};
   542→  if (frameworkId) {
   543→    where.assessment = { frameworkId };
   544→  }
   545→
   546→  const findings = await db.gapFinding.findMany({
   547→    where,
   548→    include: {
   549→      assessment: true,
   550→      control: true
   551→    },
   552→    orderBy: { severity: 'asc' }
   553→  });
   554→
   555→  const bySeverity: Record<string, number> = {};
   556→  const byCategory: Record<string, number> = {};
   557→  const byType: Record<string, number> = {};
   558→
   559→  let openRemediations = 0;
   560→  let overdueRemediations = 0;
   561→  let totalRemediationDays = 0;
   562→  let resolvedWithDate = 0;
   563→
   564→  for (const finding of findings) {
   565→    // By severity
   566→    bySeverity[finding.severity] = (bySeverity[finding.severity] || 0) + 1;
   567→
   568→    // By category
   569→    if (finding.control) {
   570→      const cat = finding.control.category;
   571→      byCategory[cat] = (byCategory[cat] || 0) + 1;
   572→    }
   573→
   574→    // By type
   575→    byType[finding.gapType] = (byType[finding.gapType] || 0) + 1;
   576→
   577→    // Remediation status
   578→    if (finding.remediationStatus !== RemediationStatus.CLOSED &&
   579→        finding.remediationStatus !== RemediationStatus.ACCEPTED_RISK) {
   580→      openRemediations++;
   581→
   582→      if (finding.dueDate && new Date(finding.dueDate) < new Date()) {
   583→        overdueRemediations++;
   584→      }
   585→    }
   586→
   587→    // Average remediation time
   588→    if (finding.resolvedAt && finding.createdAt) {
   589→      const days = Math.abs(new Date(finding.resolvedAt).getTime() - new Date(finding.createdAt).getTime()) / (1000 * 60 * 60 * 24);
   590→      totalRemediationDays += days;
   591→      resolvedWithDate++;
   592→    }
   593→  }
   594→
   595→  return {
   596→    totalGaps: findings.length,
   597→    bySeverity,
   598→    byCategory,
   599→    byType,
   600→    openRemediations,
   601→    overdueRemediations,
   602→    averageRemediationTime: resolvedWithDate > 0 ? Math.round(totalRemediationDays / resolvedWithDate) : 0,
   603→    findings
   604→  };
   605→}

   671→    fraudDetails: [],
   672→    complianceStatus: {},
   673→    recommendations: []
   674→  };
   675→
   676→  // Gather incident data if linked
   677→  if (submission.incidentId) {
   678→    const incident = await db.incident.findUnique({
   679→      where: { id: submission.incidentId },
   680→      include: { alerts: true, updates: true }
   681→    });
   682→    
   683→    if (incident) {
   684→      reportData.summary.totalIncidents = 1;
   685→      reportData.incidentDetails.push({
   686→        id: incident.id,
   687→        title: incident.title,
   688→        severity: incident.severity,
   689→        status: incident.status,
   690→        detectedAt: incident.detectedAt,
   691→        description: incident.description
   692→      });
   693→    }
   694→  }
   695→
   696→  // Gather fraud data if linked
   697→  if (submission.fraudCaseId) {
   698→    const fraudCase = await db.fraudCase.findUnique({
   699→      where: { id: submission.fraudCaseId }
   700→    });
   701→    
   702→    if (fraudCase) {
   703→      reportData.summary.fraudCases = 1;
   704→      reportData.summary.fraudLossDZD = fraudCase.estimatedLossDZD || 0;
   705→      reportData.fraudDetails.push({
   706→        caseNumber: fraudCase.caseNumber,
   707→        type: fraudCase.type,
   708→        severity: fraudCase.severity,
   709→        status: fraudCase.status,
   710→        estimatedLossDZD: fraudCase.estimatedLossDZD
   711→      });
   712→    }
   713→  }
   714→
   715→  // Gather compliance data if linked or for quarterly reports
   716→  if (submission.assessmentId || submission.submissionType === 'QUARTERLY_REPORT') {
   717→    const latestAssessment = submission.assessmentId
   718→      ? await db.complianceAssessment.findUnique({ where: { id: submission.assessmentId } })
   719→      : await db.complianceAssessment.findFirst({
   720→          where: { status: 'COMPLETED' },
   721→          orderBy: { completedAt: 'desc' }
   722→        });
   723→
   724→    if (latestAssessment) {
   725→      reportData.summary.complianceScore = latestAssessment.overallScore || 0;
   726→      
   727→      const gaps = await db.gapFinding.count({
   728→        where: {
   729→          assessmentId: latestAssessment.id,
   730→          remediationStatus: { notIn: ['CLOSED', 'ACCEPTED_RISK'] }
   731→        }
   732→      });
   733→      
   734→      const criticalGaps = await db.gapFinding.count({
   735→        where: {
   736→          assessmentId: latestAssessment.id,
   737→          severity: 'CRITICAL',
   738→          remediationStatus: { notIn: ['CLOSED', 'ACCEPTED_RISK'] }
   739→        }
   740→      });
   741→
   742→      reportData.summary.openFindings = gaps;
   743→      reportData.summary.criticalFindings = criticalGaps;
   744→    }
   745→  }
   746→
   747→  // Generate recommendations based on gaps
   748→  reportData.recommendations = await generateRecommendations();
   749→
   750→  // Update submission with report data
   751→  const updated = await db.artpSubmission.update({
   752→    where: { id: submissionId },
   753→    data: {
   754→      detailedReport: JSON.stringify(reportData),
   755→      summary: generateExecutiveSummary(reportData)
   756→    }
   757→  });
   758→
   759→  return updated;
   760→}

   845→async function generateRecommendations(): Promise<string[]> {
   846→  const openGaps = await db.gapFinding.findMany({
   847→    where: {
   848→      remediationStatus: { in: ['OPEN', 'IN_PROGRESS'] },
   849→      severity: { in: ['CRITICAL', 'HIGH'] }
   850→    },
   851→    take: 10,
   852→    orderBy: { severity: 'asc' }
   853→  });
   854→
   855→  const recommendations: string[] = [];
   856→
   857→  if (openGaps.some(g => g.severity === 'CRITICAL')) {
   858→    recommendations.push('IMMEDIATE: Address critical security gaps - allocate emergency resources');
   859→  }
   860→
   861→  if (openGaps.some(g => g.severity === 'HIGH')) {
   862→    recommendations.push('SHORT-TERM: Develop remediation plan for high-severity findings within 30 days');
   863→  }
   864→
   865→  const categoryCounts: Record<string, number> = {};
   866→  for (const gap of openGaps) {
   867→    if (gap.control) {
   868→      const cat = gap.control.category;
   869→      categoryCounts[cat] = (categoryCounts[cat] || 0) + 1;
   870→    }
   871→  }
   872→
   873→  const topCategory = Object.entries(categoryCounts).sort((a, b) => b[1] - a[1])[0];
   874→  if (topCategory) {
   875→    recommendations.push(`FOCUS AREA: ${topCategory[0]} has highest concentration of gaps (${topCategory[1]} items)`);
   876→  }
   877→
   878→  recommendations.push('Continue regular security awareness training for all personnel');
   879→  recommendations.push('Maintain 24/7 SOC monitoring and incident response readiness');
   880→
   881→  return recommendations;
   882→}

   955→export async function updateComplianceMetrics(frameworkId?: string): Promise<void> {

   959→    orderBy: { completedAt: 'desc' }

   969→    previousValue: undefined, // Will be set by upsert logic
   970→    targetValue: 80, // Target 80% compliance
   971→    unit: '%',
   972→    trend: currentScore >= 80 ? MetricTrend.IMPROVING : 
   973→           currentScore >= 60 ? MetricTrend.STABLE : MetricTrend.DECLINING,
   974→    calculationMethod: 'Weighted average of control assessments',
   975→    dataSource: 'Compliance Assessment Engine'

   980→    where: {
   981→      status: { in: ['DRAFT', 'IN_REVIEW', 'PENDING_APPROVAL'] },

   985→
   986→  await upsertMetric({
   987→    metricName: 'pending_artp_submissions',

   994→    calculationMethod: 'Count of submissions due within 7 days',
   995→    dataSource: 'ARTP Submissions table'
   996→  });
   997→
   998→  // Open gap findings
   999→  const openGaps = await db.gapFinding.count({
  1000→    where: {
  1001→      remediationStatus: { notIn: ['CLOSED', 'ACCEPTED_RISK'] }

  1003→  });
  1004→
  1005→  await upsertMetric({
  1006→    metricName: 'open_gap_findings',
  1007→    metricCategory: 'REMEDIATION_PROGRESS',

  1010→    targetValue: 10,
  1011→    unit: 'count',
  1012→    trend: openGaps > 20 ? MetricTrend.DECLINING : MetricTrend.STABLE,

  1077→/**
  1078→ * Calculate ANSSI alignment maturity score
  1079→ */
  1080→export async function calculateAnssiAlignmentScore(): Promise<{
  1081→  overallAlignment: number;
  1082→  byDomain: Record<string, { total: number; implemented: number; partial: number; score: number }>;
  1083→  certificationReadiness: any;
  1084→}> {
  1085→  const alignments = await db.ansiAlignment.findMany();
  1086→
  1087→  const byDomain: Record<string, any> = {};
  1088→  let totalImplemented = 0;
  1089→  let totalFullAlignment = 0;
  1090→
  1091→  for (const alignment of alignments) {
  1092→    const domain = alignment.ansiDomain;
  1093→    if (!byDomain[domain]) {
  1094→      byDomain[domain] = { total: 0, implemented: 0, partial: 0, score: 0 };
  1095→    }
  1096→
  1097→    byDomain[domain].total++;
  1098→
  1099→    switch (alignment.implementationStatus) {
  1100→      case AnssiImplementationStatus.IMPLEMENTED:
  1101→        byDomain[domain].implemented++;
  1102→        totalImplemented++;
  1103→        if (alignment.mappingStrength === MappingStrength.FULL ||
  1104→            alignment.mappingStrength === MappingStrength.SUBSTANTIAL) {
  1105→          totalFullAlignment++;
  1106→        }
  1107→        break;
  1108→      case AnssiImplementationStatus.PARTIALLY_IMPLEMENTED:
  1109→        byDomain[domain].partial++;
  1110→        totalImplemented += 0.5;
  1111→        break;
  1112→    }
  1113→  }
  1114→
  1115→  // Calculate per-domain scores
  1116→  for (const domain of Object.keys(byDomain)) {
  1117→    const d = byDomain[domain];
  1118→    d.score = d.total > 0 ? Math.round(((d.implemented + d.partial * 0.5) / d.total) * 100) : 0;
  1119→  }
  1120→
  1121→  const overallAlignment = alignments.length > 0 
  1122→    ? Math.round((totalImplemented / alignments.length) * 100) 
  1123→    : 0;
  1124→
  1125→  // Determine certification readiness
  1126→  const certRelevant = alignments.filter(a => a.certRelevant);
  1127→  const certImplemented = certRelevant.filter(a => 
  1128→    a.implementationStatus === AnssiImplementationStatus.IMPLEMENTED
  1129→  );
  1130→  const certificationReadiness = {
  1131→    basic: certImplemented.filter(a => a.certLevel === 'BASIC').length >= 
  1132→           certRelevant.filter(a => a.certLevel === 'BASIC').length,
  1133→    standard: certImplemented.filter(a => a.certLevel === 'STANDARD').length >= 
  1134→               certRelevant.filter(a => a.certLevel === 'STANDARD').length,
  1135→    advanced: certImplemented.filter(a => a.certLevel === 'ADVANCED').length >= 
  1136→               certRelevant.filter(a => a.certLevel === 'ADVANCED').length
  1137→  };
  1138→
  1139→  return {
  1140→    overallAlignment,
  1141→    byDomain,
  1142→    certificationReadiness
  1143→  };
  1144→}
