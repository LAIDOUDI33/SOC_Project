   230→# ============== Kafka Producer ==============
   231→
   232→class SS7KafkaProducer:
   233→    """Kafka producer for SS7 events with delivery guarantees."""
   234→    
   235→    def __init__(self):
   236→        config = {
   237→            'bootstrap.servers': settings.kafka_bootstrap_servers,
   238→            'client.id': 'ss7-collector',
   239→            'linger.ms': settings.kafka_linger_ms,
   240→            'batch.size': settings.kafka_batch_size,
   241→            'compression.type': settings.kafka_compression_type,
   242→            'acks': 'all',
   243→            'retries': 5,
   244→            'max.in.flight.requests.per.connection': 5,
   245→            'enable.idempotence': True,
   246→            'queue.buffering.max.messages': settings.buffer_size,
   247→            'socket.keepalive.enable': True,
   248→        }
   249→        
   250→        self.producer = Producer(config)
   251→        self.messages_sent = 0
   252→        self.messages_failed = 0
   253→        
   254→        logger.info("kafka_initialized", bootstrap_servers=settings.kafka_bootstrap_servers)
   255→    
   256→    def delivery_callback(self, err, msg):
   257→        """Callback for message delivery reports."""
   258→        if err:
   259→            self.messages_failed += 1
   260→            logger.error("kafka_delivery_error", error=str(err), topic=msg.topic(), partition=msg.partition())
   261→        else:
   262→            pass  # Successful delivery
   263→    
   264→    async def send_message(self, message: SS7Message):
   265→        """Send a normalized SS7 message to Kafka."""
   266→        try:
   267→            key = message.imsi or message.msisdn or message.message_id
   268→            value = orjson.dumps(message.to_dict())
   269→            
   270→            self.producer.produce(
   271→                topic=settings.output_topic,
   272→                key=key.encode('utf-8'),
   273→                value=value,
   274→                callback=self.delivery_callback
   275→            )
   276→            
   277→            self.messages_sent += 1
   278→            
   279→            # Poll for delivery callbacks periodically
   280→            if self.messages_sent % 100 == 0:
   281→                self.producer.poll(0)
   282→                
   283→        except Exception as e:
   284→            logger.exception("kafka_send_error", message_id=message.message_id)
   285→            raise
   286→    
   287→    def flush(self):
   288→        """Flush any pending messages."""
   289→        remaining = self.producer.flush(timeout=30)
   290→        if remaining > 0:
   291→            logger.warning("kafka_flush_incomplete", remaining=remaining)
   292→        return remaining
   293→    
   294→    def close(self):
   295→        """Close the producer gracefully."""
   296→        self.flush()
   297→        self.producer.flush(infinity=10)
   298→        logger.info("kafka_producer_closed", total_sent=self.messages_sent)

   303→class M3UAParser:
   304→    """Parse M3UA (MTP3 User Adaptation) messages."""
   305→    
   306→    MESSAGE_CLASS_MGMT = 0
   307→    MESSAGE_CLASS_TRANSFER = 1
   308→    MESSAGE_CLASS_SSNM = 2
   309→    MESSAGE_CLASS_ASPSM = 3
   310→    MESSAGE_CLASS_ASPTM = 4
   311→    
   312→    @classmethod
   313→    def parse_header(cls, data: bytes) -> Dict:
   314→        """Parse M3UA header (8 bytes)."""
   315→        if len(data) < 8:
   316→            raise ValueError("M3UA header too short")
   317→        
   318→        version, reserved, message_class, message_type = struct.unpack('!BBHH', data[:8])
   319→        
   320→        return {
   321→            "version": version,
   322→            "reserved": reserved,
   323→            "message_class": message_class,
   324→            "message_type": message_type,
   325→            "length": struct.unpack('!I', data[4:8])[0] & 0x00FFFFFF,
   326→        }
   327→    
   328→    @classmethod
   329→    def parse_payload_data(cls, data: bytes) -> Dict:
   330→        """Parse DATA message payload (MTP3 payload)."""
   331→        if len(data) < 16:
   332→            raise ValueError("M3UA DATA payload too short")
   333→        
   334→        # Parse MTP3 routing label
   335→        opc, dpc, si, ni, mp, sls = struct.unpack('!IIIBBBI', data[:16])
   336→        
   337→        return {
   338→            "opc": opc & 0x3FFFFFFF,  # Originating Point Code (14-bit or 24-bit)
   339→            "dpc": dpc & 0x3FFFFFFF,  # Destination Point Code
   340→            "service_indicator": si,
   341→            "network_indicator": ni,
   342→            "message_priority": mp,
   343→            "signaling_link_selection": sls,
   344→            "mtp3_payload": data[16:] if len(data) > 16 else b'',
   345→        }

   348→class SCCPParser:
   349→    """Parse SCCP (Signaling Connection Control Part) messages."""
   350→    
   351→    MESSAGE_TYPE_CR = 0x01  # Connection Request
   352→    MESSAGE_TYPE_CC = 0x02  # Connection Confirm
   353→    MESSAGE_TYPE_DT1 = 0x05  # Data Form 1
   354→    MESSAGE_TYPE_UDT = 0x09  # Unitdata (connectionless)
   355→    MESSAGE_TYPE_XUDT = 0x11  # Extended Unitdata
   356→    
   357→    GT_INDICATOR_NO_GT = 0x00
   358→    GT_INDICATOR_GT_ONLY = 0x01
   359→    GT_INDICATOR_GT_SSN = 0x02
   360→    
   361→    @classmethod
   362→    def parse_called_party_address(cls, data: bytes) -> Dict:
   363→        """Parse SCCP called party address."""
   364→        if not data:
   365→            return {}
   366→        
   367→        result = {}
   368→        byte0 = data[0]
   369→        
   370→        result["point_code_indicator"] = bool(byte0 & 0x80)
   371→        result["ssn_indicator"] = bool(byte0 & 0x40)
   372→        result["global_title_indicator"] = (byte0 >> 2) & 0x03
   373→        result["routing_indicator"] = byte0 & 0x03
   374→        
   375→        offset = 1
   376→        
   377→        if result["global_title_indicator"] != cls.GT_INDICATOR_NO_GT:
   378→            if offset < len(data):
   379→                result["nature_of_address"] = data[offset] & 0x07
   380→                result["numbering_plan"] = (data[offset] >> 4) & 0x0F
   381→                offset += 1
   382→                if offset < len(data):
   383→                    odd_flag = data[offset] & 0x80
   384→                    digits = b''
   385→                    while offset < len(data):
   386→                        b = data[offset]
   387→                        high = (b >> 4) & 0x0F
   388→                        low = b & 0x0F
   389→                        if high <= 9:
   390→                            digits += bytes([high + 0x30])
   391→                        if low <= 9:
   392→                            digits += bytes([low + 0x30])
   393→                        offset += 1
   394→                    result["address_digits"] = digits.decode('ascii')
   395→        
   396→        if result["ssn_indicator"] and offset < len(data):
   397→            result["subsystem_number"] = data[offset]
   398→        
   399→        return result

   402→class TCAPParser:
   403→    """Parse TCAP (Transaction Capabilities Application Part)."""
   404→    
   405→    @classmethod
   406→    def parse_transaction_id(cls, data: bytes) -> tuple:
   407→        """Extract OTID (Originating Transaction ID) and DTID (Destination Transaction ID)."""
   408→        # Simplified OTID/DTID extraction from TCAP
   409→        otid = dtid = None
   410→        try:
   411→            if len(data) >= 12:
   412→                # OTID is typically first 4 bytes of transaction portion
   413→                otid = struct.unpack('!I', data[4:8])[0]
   414→                dtid = struct.unpack('!I', data[8:12])[0]
   415→        except Exception:
   416→            pass
   417→        return otid, dtid

   420→class MAPParser:
   421→    """Parse MAP (Mobile Application Part) operations."""
   422→    
   423→    OPERATION_CODES = {
   424→        2: "sendRoutingInfoForSM",
   425→        18: "provideRoamingNumber",
   426→        2: "updateLocation",
   427→        56: "cancelLocation",
   428→        52: "insertSubscriberData",
   429→        53: "deleteSubscriberData",
   430→        46: "forwardShortMessage",
   431→        59: "processUnstructuredSSRequest",
   432→        66: "authenticationFailureReport",
   433→        68: "noteSubscriberPresent",
   434→        67: "readyForSM",
   435→        88: "anyTimeInterrogation",
   436→    }
   437→    
   438→    @classmethod
   439→    def get_operation_name(cls, opcode: int) -> str:
   440→        return cls.OPERATION_CODES.get(opcode, f"unknown_operation_{opcode}")
   441→    
   442→    @classmethod
   443→    def extract_imsi(cls, data: bytes) -> Optional[str]:
   444→        """Attempt to extract IMSI from MAP message."""
   445→        # IMSI is typically encoded as BCD string in specific parameters
   446→        # This is a simplified extraction - real implementation needs full ASN.1 parsing
   447→        imsi_patterns = [
   448→            b'\x08\x29\x80\x00',  # Common IMSI tag prefix
   449→            b'\x80\x91\xA3',       # Alternative encoding
   450→        ]
   451→        for pattern in imsi_patterns:
   452→            idx = data.find(pattern)
   453→            if idx != -1:
   454→                start = idx + len(pattern)
   455→                # Extract up to 15 digits (IMSI length)
   456→                end = min(start + 8, len(data))
   457→                try:
   458→                    digits = ''
   459→                    for b in data[start:end]:
   460→                        high = (b >> 4) & 0x0F
   461→                        low = b & 0x0F
   462→                        if high <= 9:
   463→                            digits += chr(high + 0x30)
   464→                        if low <= 9:
   465→                            digits += chr(low + 0x30)
   466→                    if len(digits) >= 8 and digits.isdigit():
   467→                        return digits
   468→                except Exception:
   469→                    continue
   470→        return None

   672→# ============== PCAP Writer ==============
   673→
   674→class PCAPWriter:
   675→    """Rotating PCAP file writer for SS7 traffic capture."""
   676→    
   677→    def __init__(self, base_dir: Path, max_size_mb: int = 512, rotation_hours: int = 1):
   678→        self.base_dir = base_dir
   679→        self.max_size_bytes = max_size_mb * 1024 * 1024
   680→        self.rotation_hours = rotation_hours
   681→        self.current_file = None
   682→        self.current_size = 0
   683→        self.rotation_time = time.time()
   684→        self.base_dir.mkdir(parents=True, exist_ok=True)
   685→    
   686→    def write_packet(self, data: bytes, timestamp: float = None):
   687→        """Write a packet to current PCAP file."""
   688→        import os
   689→        
   690→        if timestamp is None:
   691→            timestamp = time.time()
   692→        
   693→        # Check rotation conditions
   694→        need_rotation = (
   695→            self.current_file is None or
   696→            self.current_size > self.max_size_bytes or
   697→            (time.time() - self.rotation_time) > (self.rotation_hours * 3600)
   698→        )
   699→        
   700→        if need_rotation:
   701→            self._rotate_file(timestamp)
   702→        
   703→        # Write packet (simplified - use scapy for proper PCAP format)
   704→        try:
   705→            self.current_file.write(data)
   706→            self.current_size += len(data)
   707→        except Exception as e:
   708→            logger.error("pcap_write_error", error=str(e))
   709→    
   710→    def _rotate_file(self, timestamp: float):
   711→        """Rotate to new PCAP file."""
   712→        if self.current_file:
   713→            self.current_file.close()
   714→        
   715→        filename = f"ss7_capture_{datetime.fromtimestamp(timestamp).strftime('%Y%m%d_%H%M%S')}.pcap"
   716→        filepath = self.base_dir / filename
   717→        
   718→        self.current_file = open(filepath, 'ab')
   719→        self.current_size = 0
   720→        self.rotation_time = timestamp
   721→        
   722→        logger.info("pcap_rotated", file=str(filepath))
   723→
