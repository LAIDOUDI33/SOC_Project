   298→function generateTimelineData(): Array<{ time: string; alerts: number; blocked: number; resolved: number }> {
   299→  const data = [];
   300→  for (let i = 24; i >= 0; i--) {
   301→    data.push({
   302→      time: `${i}h ago`,
   303→      alerts: Math.floor(Math.random() * 15) + (i < 4 ? 10 : 0),
   304→      blocked: Math.floor(Math.random() * 5) + (i < 2 ? 3 : 0),
   305→      resolved: Math.floor(Math.random() * 8),
   306→    });
   307→  }
   308→  return data.reverse();
   309→}

   639→function AlertCard({ 
   640→  alert, 
   641→  isExpanded, 
   642→  onToggle, 
   643→  onBlock 
   644→}: { 
   645→  alert: FraudAlert; 
   646→  isExpanded: boolean; 
   647→  onToggle: () => void; 
   648→  onBlock: () => void;
   649→}) {
   650→  const severityCfg = SEVERITY_CONFIG[alert.severity];
   651→  const statusCfg = STATUS_CONFIG[alert.status];
   652→
   653→  return (
   654→    <div 
   655→      className={`rounded-lg border p-4 cursor-pointer transition-all ${
   656→        severityCfg.bgClass
   657→      } ${isExpanded ? 'ring-1 ring-white/20' : ''}`}
   658→      onClick={onToggle}
   659→    >
   660→      <div className="flex items-start justify-between">
   661→        <div className="flex items-start gap-3">
   662→          {severityCfg.icon}
   663→          <div>
   664→            <div className="flex items-center gap-2 mb-1">
   665→              <span className="text-white font-medium">{alert.ruleName}</span>
   666→              <Badge variant="outline" className={`text-xs ${severityCfg.bgClass}`}>
   667→                {alert.severity.toUpperCase()}
   668→              </Badge>
   669→            </div>
   670→            
   671→            <div className="flex items-center gap-3 text-xs text-gray-400">
   672→              <span className="flex items-center gap-1">
   673→                {statusCfg.icon}
   674→                {statusCfg.label}
   675→              </span>
   676→              <span>{alert.timestamp.toLocaleTimeString()}</span>
   677→              {alert.subscriber && (
   678→                <code className="bg-slate-800 px-1 rounded">
   679→                  {alert.subscriber.maskedMSISDN}
   680→                </code>
   681→              )}
   682→            </div>
   683→          </div>
   684→        </div>
   685→
   686→        <div className="flex items-center gap-2">
   687→          <span className="text-sm font-mono font-bold">{alert.confidence}%</span>
   688→          <ChevronDown className={`w-4 h-4 transition-transform ${isExpanded ? 'rotate-180' : ''}`} />
   689→        </div>
   690→      </div>
   691→
   692→      {/* Expanded Content */}
   693→      {isExpanded && (
   694→        <div className="mt-4 pt-4 border-t border-white/10 space-y-3">
   695→          {/* Indicators */}
   696→          <div>
   697→            <span className="text-xs text-gray-400 uppercase tracking-wider">Indicators</span>
   698→            <div className="flex flex-wrap gap-1 mt-1">
   699→              {alert.indicators.map(ind => (
   700→                <Badge key={ind} variant="outline" className="text-xs bg-slate-800/50 border-slate-600">
   701→                  {ind.replace(/_/g, ' ')}
   702→                </Badge>
   703→              ))}
   704→            </div>
   705→          </div>
   706→
   707→          {/* Financial Impact */}
   708→          {alert.financialImpact && (
   709→            <div className="flex items-center justify-between p-2 bg-slate-800/50 rounded">
   710→              <span className="text-gray-400 text-sm">Est. Financial Impact:</span>
   711→              <span className="text-red-400 font-bold font-mono">
   712→                {alert.financialImpact.estimatedLossDZD.toLocaleString()} DZD
   713→              </span>
   714→            </div>
   715→          )}
   716→
   717→          {/* Actions */}
   718→          <div className="flex gap-2">
   719→            {alert.status !== AlertStatus.BLOCKED && (
   720→              <Button 
   721→                size="sm" 
   722→                variant="destructive"
   723→                onClick={(e) => { e.stopPropagation(); onBlock(); }}
   724→              >
   725→                <Ban className="w-3 h-3 mr-1" />
   726→                Block Subscriber
   727→              </Button>
   728→            )}
   729→            <Button 
   730→              size="sm" 
   731→              variant="outline"
   732→              onClick={(e) => e.stopPropagation()}
   733→            >
   734→              <Eye className="w-3 h-3 mr-1" />
   735→              View Details
   736→            </Button>
   737→          </div>
   738→        </div>
   739→      )}
   740→    </div>
