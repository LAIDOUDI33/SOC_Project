   574→-- Active Incidents View
   575→CREATE OR REPLACE VIEW v_active_incidents AS
   576→SELECT 
   577→    i.*,
   578→    u.name as assignee_name,

   580→    COUNT(DISTINCT t.id) as task_count

   584→LEFT JOIN tasks t ON t.incident_id = i.id

   644→-- Function to calculate subscriber risk score
   645→CREATE OR REPLACE FUNCTION calculate_subscriber_risk(p_subscriber_id TEXT)
   646→RETURNS INTEGER AS $$
   647→DECLARE
   648→    v_risk_score INTEGER := 0;
   649→    v_alert_count INTEGER;
   650→    v_anomaly_avg NUMERIC;
   651→BEGIN
   652→    -- Count recent high-severity alerts for this subscriber
   653→    SELECT COUNT(*) INTO v_alert_count
   654→    FROM alerts a
   655→    JOIN ss7_messages s ON s.imsi = (SELECT imsi FROM subscribers WHERE id = p_subscriber_id)
   656→    WHERE a.severity IN ('CRITICAL', 'HIGH')
   657→    AND a.first_seen > NOW() - INTERVAL '7 days';
   658→    
   659→    -- Get average anomaly score from recent sessions
   660→    SELECT COALESCE(AVG(anomaly_score), 0) INTO v_anomaly_avg
   661→    FROM gtp_sessions
   662→    WHERE imsi = (SELECT imsi FROM subscribers WHERE id = p_subscriber_id)
   663→    AND started_at > NOW() - INTERVAL '24 hours';
   664→    
   665→    -- Calculate composite score
   666→    v_risk_score := LEAST(100, (v_alert_count * 10) + ROUND(v_anomaly_avg::INTEGER));
   667→    
   668→    UPDATE subscribers SET risk_score = v_risk_score WHERE id = p_subscriber_id;
   669→    
   670→    RETURN v_risk_score;
   671→END;
   672→$$ LANGUAGE plpgsql;
