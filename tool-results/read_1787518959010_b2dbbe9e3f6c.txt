   250→class FraudDetectionEngine {
   251→  private rules: Map<string, FraudRule> = new Map();
   252→  private recentAlerts: Map<string, Date> = new Map(); // subscriberId -> last alert time
   253→
   254→  constructor() {
   255→    // Load default rules
   256→    this.loadRules(DEFAULT_FRAUD_RULES);
   257→  }
   258→
   259→  /**
   260→   * Load fraud detection rules
   261→   */
   262→  loadRules(rules: FraudRule[]): void {
   263→    for (const rule of rules) {
   264→      this.rules.set(rule.id, rule);
   265→    }
   266→    console.log(`Loaded ${rules.length} fraud detection rules`);
   267→  }
   268→
   269→  /**
   270→   * Analyze event for fraud patterns
   271→   */
   272→  async analyzeEvent(
   273→    eventType: 'ss7' | 'gtp' | 'sip' | 'diameter',
   274→    eventData: Record<string, any>
   275→  ): Promise<FraudDetectionResult[]> {
   276→    const results: FraudDetectionResult[] = [];
   277→    
   278→    // Get subscriber identifier
   279→    const subscriberId = eventData.imsi || eventData.msisdn || eventData.callingParty;
   280→
   281→    if (!subscriberId) return results;
   282→
   283→    // Check each rule
   284→    for (const [ruleId, rule] of this.rules) {
   285→      if (!rule.enabled) continue;
   286→
   287→      try {
   288→        const matchResult = await this.evaluateRule(rule, eventType, eventData);
   289→        
   290→        if (matchResult.matched) {
   291→          // Check cooldown
   292→          const lastAlert = this.recentAlerts.get(`${subscriberId}:${ruleId}`);
   293→          if (lastAlert && Date.now() - lastAlert.getTime() < rule.cooldownMinutes * 60000) {
   294→            continue; // Still in cooldown period
   295→          }
   296→
   297→          // Create fraud case or alert
   298→          const result = await this.createFraudAlert(rule, matchResult, eventData);
   299→          results.push(result);
   300→
   301→          // Update last alert time
   302→          this.recentAlerts.set(`${subscriberId}:${ruleId}`, new Date());
   303→        }
   304→      } catch (error) {
   305→        console.error(`Error evaluating rule ${ruleId}:`, error);
   306→      }
   307→    }
   308→
   309→    return results;
   310→  }
   311→
   312→  /**
   313→   * Evaluate a single rule against event data
   314→   */
   315→  private async evaluateRule(
   316→    rule: FraudRule,
   317→    eventType: string,
   318→    eventData: Record<string, any>
   319→  ): Promise<{ matched: boolean; confidence: number; matchedConditions: RuleCondition[] }> {
   320→    const matchedConditions: RuleCondition[] = [];
   321→    let totalWeight = 0;
   322→    let matchedWeight = 0;
   323→
   324→    for (const condition of rule.conditions) {
   325→      totalWeight += condition.weight;
   326→      
   327→      const fieldValue = await this.getFieldValue(condition.field, eventType, eventData);
   328→      const matches = this.evaluateCondition(fieldValue, condition);
   329→      
   330→      if (matches) {
   331→        matchedConditions.push(condition);
   332→        matchedWeight += condition.weight;
   333→      }
   334→    }
   335→
   336→    // All conditions must match (AND logic)
   337→    const allMatched = matchedConditions.length === rule.conditions.length;
   338→    const confidence = allMatched ? Math.round((matchedWeight / totalWeight) * 100) : 0;
   339→
   340→    return {
   341→      matched: allMatched && confidence >= 50, // Minimum 50% confidence to trigger
   342→      confidence,
   343→      matchedConditions
   344→    };
   345→  }

   347→  /**
   348→   * Get field value from event data or database query
   349→   */
   350→  private async getFieldValue(
   351→    field: string,
   352→    eventType: string,
   353→    eventData: Record<string, any>
   354→  ): Promise<any> {
   355→    switch (field) {
   356→      // Direct fields from event
   357→      case 'callingParty':
   358→        return eventData.callingParty || eventData.msisdn;
   359→      case 'calledParty':
   360→        return eventData.calledParty || eventData.calledParty;
   361→      case 'imsi':
   362→        return eventData.imsi;
   363→      case 'msisdn':
   364→        return eventData.msisdn;
   365→      case 'callDuration':
   366→        return eventData.duration || eventData.callDuration || 0;
   367→      case 'isRoaming':
   368→        return eventData.isRoaming === true || eventData.visitedPLMN?.startsWith('6');
   369→      case 'dataVolumeGB':
   370→        return ((eventData.bytesUp || 0) + (eventData.bytesDown || 0)) / 1073741824;
   371→      case 'calledCountryCode':
   372→        const called = eventData.calledParty || '';
   373→        return called.substring(0, called.indexOf(' ') || 3);
   374→      
   375→      // Computed fields requiring database queries
   376→      case 'updateLocationCount':
   377→        return await this.countRecentOperations(eventData.imsi, 'UpdateLocation', 3600000);
   378→      case 'differentMSCs':
   379→        return await this.countDifferentMSCs(eventData.imsi, 3600000);
   380→      case 'shortCallCount':
   381→        return await this.countShortCalls(eventData.callingParty, 3600000);
   382→      case 'uniqueCalledNumbers':
   383→        return await this.countUniqueCalledNumbers(eventData.callingParty, 3600000);
   384→      case 'avgCallDuration':
   385→        return await this.getAverageCallDuration(eventData.callingParty, 3600000);
   386→      case 'activeSessionsWithSameIMEI':
   387→        return await this.countActiveSessionsByIMEI(eventData.imei);
   388→      case 'differentIMSIs':
   389→        return await this.countDifferentIMSIByIMEI(eventData.imei);
   390→      case 'outgoingCallCount':
   391→        return await this.countOutgoingCalls(eventData.callingParty, 300000);
   392→      case 'uniqueCalledNumbers': // For call spam
   393→        return await this.countUniqueCalledNumbers(eventData.callingParty, 300000);
   394→      case 'averageCallInterval':
   395→        return await this.getAverageCallInterval(eventData.callingParty, 300000);
   396→      case 'authAttempts':
   397→        return await this.countAuthAttempts(eventData.imsi, 60000);
   398→      case 'authFailureRate':
   399→        return await this.getAuthFailureRate(eventData.imsi, 60000);
   400→      case 'callerRiskScore':
   401→        const sub = await this.getSubscriber(eventData.imsi || eventData.msisdn);
   402→        return sub?.riskScore || 0;
   403→      case 'homeDataAvgGB':
   404→        return await this.getAverageHomeDataUsage(eventData.imsi, 86400000 * 7); // 7 days
   405→      
   406→      default:
   407→        return eventData[field];
   408→    }
   409→  }

   411→  /**
   412→   * Evaluate single condition
   413→   */
   414→  private evaluateCondition(value: any, condition: RuleCondition): boolean {
   415→    if (value === undefined || value === null) return false;
   416→
   417→    switch (condition.operator) {
   418→      case 'eq':
   419→        return value === condition.value;
   420→      case 'neq':
   421→        return value !== condition.value;
   422→      case 'gt':
   423→        return Number(value) > Number(condition.value);
   424→      case 'gte':
   425→        return Number(value) >= Number(condition.value);
   426→      case 'lt':
   427→        return Number(value) < Number(condition.value);
   428→      case 'lte':
   429→        return Number(value) <= Number(condition.value);
   430→      case 'contains':
   431→        return String(value).includes(String(condition.value));
   432→      case 'in':
   433→        return Array.isArray(condition.value) && condition.value.includes(value);
   434→      case 'regex':
   435→        return new RegExp(condition.value).test(String(value));
   436→      default:
   437→        return false;
   438→    }
   439→  }

   441→  /**
   442→   * Create fraud alert from matched rule
   443→   */
   444→  private async createFraudAlert(
   445→    rule: FraudRule,
   446→    evaluation: { matched: boolean; confidence: number; matchedConditions: RuleCondition[] },
   447→    eventData: Record<string, any>
   448→  ): Promise<FraudDetectionResult> {
   449→    
   450→    const caseNumber = `FRD-${Date.now().toString(36).toUpperCase()}`;
   451→    const subscriberId = eventData.imsi || eventData.msisdn || eventData.callingParty;
   452→
   453→    // Calculate estimated financial impact (if applicable)
   454→    const financialImpact = await this.estimateFinancialImpact(rule.type, eventData);
   455→
   456→    // Determine ARTP deadline (24 hours from detection)
   457→    const artpDeadline = rule.artpReportable 
   458→      ? new Date(Date.now() + 24 * 60 * 60 * 1000)
   459→      : undefined;
   460→
   461→    // Execute rule actions
   462→    await this.executeActions(rule.actions, subscriberId, eventData);
   463→
   464→    // Create database record
   465→    const fraudCase = await db.fraudCase.create({
   466→      data: {
   467→        caseNumber,
   468→        type: rule.type,
   469→        severity: rule.severity.toUpperCase(),
   470→        status: 'OPEN',
   471→        confidence: evaluation.confidence,
   472→        description: this.generateDescription(rule, eventData),
   473→        subscriberId,
   474→        evidence: JSON.stringify(evaluation.matchedConditions.map(c => ({
   475→          type: 'detection_rule',
   476→          timestamp: new Date(),
   477→          description: `${c.field} ${c.operator} ${c.value}`,
   478→          data: { field: c.field, actualValue: eventData[c.field] },
   479→          riskContribution: c.weight
   480→        }))),
   481→        estimatedLossDZD: financialImpact,
   482→        artpReportable: rule.artpReportable,
   483→        artpDeadline,
   484→        metadata: JSON.stringify({
   485→          ruleId: rule.id,
   486→          ruleName: rule.name,
   487→          eventType: 'telecom',
   488→          sourceEvent: eventData,
   489→          detectedAt: new Date()
   490→        })
   491→      }
   492→    });
   493→
   494→    // Also create security alert
   495→    await db.alert.create({
   496→      data: {
   497→        title: `[FRAUD] ${rule.name}`,
   498→        description: this.generateDescription(rule, eventData),
   499→        severity: rule.severity.toUpperCase() as any,
   500→        status: 'NEW',
   501→        source: 'FRAUD_DETECTION_ENGINE',
   502→        category: 'FRAUD_DETECTION',
   503→        rawEvent: {
   504→          fraudCaseId: fraudCase.id,
   505→          caseNumber,
   506→          ruleName: rule.name,
   507→          fraudType: rule.type,
   508→          confidence: evaluation.confidence,
   509→          eventData
   510→        },
   511→        tags: ['fraud', rule.type.toLowerCase(), rule.severity],
   512→        context: {
   513→          artpReportable: rule.artpReportable,
   514→          artpDeadline: artpDeadline?.toISOString(),
   515→          financialImpactDZD: financialImpact,
   516→          recommendedActions: rule.actions.map(a => a.type)
   517→        }
   518→      }
   519→    });
   520→
   521→    return {
   522→      caseId: fraudCase.id,
   523→      caseNumber,
   524→      type: rule.type,
   525→      severity: rule.severity,
   526→      confidence: evaluation.confidence,
   527→      actionsTaken: rule.actions.map(a => a.type),
   528→      artpReportable: rule.artpReportable,
   529→      artpDeadline
   530→    };
   531→  }

   563→      IRSF: 200000,         // International fraud can be costly
   564→      WANGIRI: 5000,        // Lower per-incident but high volume
   565→      PBX_HIJACKING: 500000,// Can be very expensive
   566→      TOLL_FRAUD: 100000,
   567→      IMEI_CLONING: 75000,
   568→      DATA_TUNNELING: 15000,
   569→      ROAMING_ABUSE: 25000,
   570→      CALL_SPAM: 8000,
   571→      PREMIUM_RATE_ABUSE: 35000,
   572→      LOCATION_SPOOFING: 40000,
   573→      AUTHENTICATION_STORM: 10000,
   574→      SUBSCRIBER_TAKEOVER: 60000
   575→    };
   576→
   577→    let impact = baseImpacts[fraudType] || 10000;
   578→
   579→    // Adjust based on duration/volume
   580→    if (eventData.duration) {
   581→      impact *= Math.min(eventData.duration / 3600, 24); // Scale by hours, max 24x
   582→    }
   583→    if (eventData.bytesUp || eventData.bytesDown) {
   584→      const gb = ((eventData.bytesUp || 0) + (eventData.bytesDown || 0)) / 1073741824;
   585→      impact *= Math.min(gb / 10, 50); // Scale by GB, max 50x
   586→    }
   587→
   588→    return Math.round(impact);
   589→  }

   620→        
   621→        case 'report_artp':
   622→          console.log(`[ARTP] Report required for fraud case involving ${subscriberId}`);

   625→    }

   788→  /**
   789→   * Get fraud statistics dashboard
   790→   */
   791→  async getFraudStatistics(timeframe: string = '24h'): Promise<any> {
   792→    const now = new Date();
   793→    const timeRangeMap: Record<string, Date> = {
   794→      '1h': new Date(now.getTime() - 3600000),
   795→      '6h': new Date(now.getTime() - 21600000),
   796→      '24h': new Date(now.getTime() - 86400000),
   797→      '7d': new Date(now.getTime() - 604800000),
   798→      '30d': new Date(now.getTime() - 2592000000)
   799→    };
   800→    const since = timeRangeMap[timeframe] || timeRangeMap['24h'];
   801→
   802→    const [
   803→      totalCases,
   804→      casesBySeverity,
   805→      casesByType,
   806→      casesByStatus,
   807→      totalEstimatedLoss,
   808→      artpReportableCases,
   809→      resolvedCases
   810→    ] = await Promise.all([
   811→      db.fraudCase.count({ where: { createdAt: { gte: since } } }),
   812→      db.fraudCase.groupBy({ by: ['severity'], _count: { id: true }, where: { createdAt: { gte: since } } }),
   813→      db.fraudCase.groupBy({ by: ['type'], _count: { id: true }, where: { createdAt: { gte: since } } }),
   814→      db.fraudCase.groupBy({ by: ['status'], _count: { id: true }, where: { createdAt: { gte: since } } }),
   815→      db.fraudCase.aggregate({ _sum: { estimatedLossDZD: true }, where: { createdAt: { gte: since } } }),
   816→      db.fraudCase.count({ where: { artpReportable: true, status: { notIn: ['RESOLVED', 'CLOSED'] }, createdAt: { gte: since } } }),
   817→      db.fraudCase.count({ where: { status: 'RESOLVED', resolvedAt: { gte: since } } })
   818→    ]);
   819→
   820→    return {
   821→      period: timeframe,
   822→      since: since.toISOString(),
   823→      summary: {
   824→        totalCases,
   825→        totalEstimatedLossDZD: Math.round((totalEstimatedLoss._sum.estimatedLossDZD || 0)),
   826→        artpReportableCases,
   827→        resolutionRate: totalCases > 0 ? Math.round((resolvedCases / totalCases) * 100) : 0
   828→      },
   829→      breakdown: {
   830→        bySeverity: casesBySeverity.reduce((acc, s) => ({ ...acc, [s.severity.toLowerCase()]: s._count.id }), {}),
   831→        byType: casesByType.reduce((acc, t) => ({ ...acc, [t.type]: t._count.id }), {}),
   832→        byStatus: casesByStatus.reduce((acc, s) => ({ ...acc, [s.status.toLowerCase()]: s._count.id }), {})
   833→      }
   834→    };
   835→  }
