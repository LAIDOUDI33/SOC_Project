   375→function ComplianceScoreCard({
   376→  score,
   377→  trend,
   378→  label,
   379→  subtitle
   380→}: {
   381→  score: number
   382→  trend?: number
   383→  label: string
   384→  subtitle?: string
   385→}) {
   386→  const getScoreColor = (s: number) => {
   387→    if (s >= 90) return 'text-green-400'
   388→    if (s >= 70) return 'text-yellow-400'
   389→    if (s >= 50) return 'text-orange-400'
   390→    return 'text-red-400'
   391→  }
   392→
   393→  const getProgressColor = (s: number) => {
   394→    if (s >= 90) return '[&>div]:bg-green-500'
   395→    if (s >= 70) return '[&>div]:bg-yellow-500'
   396→    if (s >= 50) return '[&>div]:bg-orange-500'
   397→    return '[&>div]:bg-red-500'
   398→  }
   399→
   400→  return (
   401→    <Card className="bg-slate-900 border-slate-700">
   402→      <CardContent className="pt-6">
   403→        <p className="text-sm text-slate-400 mb-1">{label}</p>
   404→        <div className={`text-3xl font-bold ${getScoreColor(score)} mb-2`}>
   405→          {score}%
   406→        </div>
   407→        <Progress value={score} className={`h-2 ${getProgressColor(score)}`} />
   408→        <div className="flex items-center justify-between mt-2">
   409→          {trend !== undefined && (
   410→            <MetricTrend value={trend} showArrow size="sm" />
   411→          )}
   412→          {subtitle && (
   413→            <span className="text-xs text-slate-500">{subtitle}</span>
   414→          )}
   415→        </div>
   416→      </CardContent>
   417→    </Card>
   418→  )
   419→}

   421→function ANRTRequirementsMatrix({ requirements }: { requirements: ANRTRequirement[] }) {
   422→  const getStatusBadge = (status: ANRTRequirement['status']) => {
   423→    switch (status) {
   424→      case 'compliant':
   425→        return <Badge variant="outline" className="border-green-500/50 text-green-400">Compliant</Badge>
   426→      case 'partial':
   427→        return <Badge variant="outline" className="border-yellow-500/50 text-yellow-400">Partial</Badge>
   428→      case 'non-compliant':
   429→        return <Badge variant="outline" className="border-red-500/50 text-red-400">Non-Compliant</Badge>
   430→      case 'not-assessed':
   431→        return <Badge variant="outline" className="border-slate-500/50 text-slate-400">Not Assessed</Badge>
   432→    }
   433→  }
   434→
   435→  const getRiskBadge = (risk: ANRTRequirement['riskLevel']) => {
   436→    switch (risk) {
   437→      case 'critical': return <StatusIndicator status="critical" size="sm" />

   439→      case 'medium': return <StatusIndicator status="good" size="sm" />

   603→function DeadlineCountdown({ deadlines }: { deadlines: RegulatoryDeadline[] }) {
   604→  const getDeadlineStatus = (status: RegulatoryDeadline['status']) => {
   605→    switch (status) {
   606→      case 'on-track': return { color: 'bg-green-500/20 border-green-500/30', badge: 'bg-green-500/20 text-green-400' }
   607→      case 'at-risk': return { color: 'bg-yellow-500/20 border-yellow-500/30', badge: 'bg-yellow-500/20 text-yellow-400' }
   608→      case 'overdue': return { color: 'bg-red-500/20 border-red-500/30', badge: 'bg-red-500/20 text-red-400' }
   609→      case 'completed': return { color: 'bg-slate-500/20 border-slate-500/30', badge: 'bg-slate-500/20 text-slate-400' }
   610→    }
   611→  }
   612→
   613→  const daysRemaining = (date: Date) => Math.ceil((date.getTime() - Date.now()) / 86400000)
   614→
   615→  return (
   616→    <Card className="bg-slate-900 border-slate-700">
   617→      <CardHeader className="pb-3">
   618→        <CardTitle className="text-white text-base flex items-center gap-2">
   619→          <Timer className="h-4 w-4 text-red-400" />
   620→          Regulatory Deadline Countdown
   621→        </CardTitle>
   622→      </CardHeader>
   623→      <CardContent>
   624→        <div className="space-y-3">
   625→          {deadlines.map(deadline => {
   626→            const style = getDeadlineStatus(deadline.status)
   627→            const remaining = daysRemaining(deadline.deadline)
   628→            
   629→            return (
   630→              <div key={deadline.id} className={`p-3 rounded-lg border ${style.color}`}>
   631→                <div className="flex items-start justify-between mb-2">
   632→                  <div>
   633→                    <p className="font-medium text-white text-sm">{deadline.requirement}</p>
   634→                    <p className="text-xs text-slate-400">{deadline.regulation}</p>
   635→                  </div>
   636→                  <Badge variant="secondary" className={style.badge}>
   637→                    {deadline.status.replace('-', ' ').toUpperCase()}
   638→                  </Badge>
   639→                </div>
   640→                
   641→                <div className="flex items-center gap-4 mt-3">
   642→                  <div className="flex-1">
   643→                    <div className="flex items-center justify-between text-xs mb-1">
   644→                      <span className="text-slate-400">Progress</span>
   645→                      <span className="text-white font-medium">{deadline.progress}%</span>
   646→                    </div>
   647→                    <Progress value={deadline.progress} className="h-1.5" />
   648→                  </div>
   649→                  
   650→                  <div className="text-right min-w-[80px]">
   651→                    <p className={`text-lg font-bold ${
   652→                      remaining < 0 ? 'text-red-400' :
   653→                      remaining <= 7 ? 'text-orange-400' :
   654→                      remaining <= 30 ? 'text-yellow-400' :
   655→                      'text-green-400'
   656→                    }`}>
   657→                      {remaining < 0 ? `${Math.abs(remaining)}d overdue` : `${remaining}d left`}
   658→                    </p>
   659→                    <p className="text-[10px] text-slate-500">{deadline.owner}</p>
   660→                  </div>
   661→                </div>
   662→              </div>
   663→            )
   664→          })}
   665→        </div>
   666→      </CardContent>
   667→    </Card>
   668→  )
   669→}
