    61→    estimatedLossDZD: number;
    62→    currency: 'DZD';

    64→  };
    65→  
    66→  // Detection metadata
    67→  ruleId: string;
    68→  ruleName: string;
    69→  source: string;           // e.g., 'ss7-analyzer', 'ml-model'
    70→  
    71→  // Actions taken/recommended

    73→  
    74→  // ARTP reporting
    75→  artpReportable: boolean;
    76→  artpDeadline?: Date;
    77→  artpCaseNumber?: string;

   409→// ============================================================
   410→// FRAUD DETECTOR CLASS
   411→// ============================================================
   412→
   413→export class FraudDetector {
   414→  private rules: Map<string, FraudRule>;
   415→  private alertHistory: Map<string, FraudAlert[]>;
   416→  private subscriberProfiles: Map<string, SubscriberProfile>;
   417→  private activeAlerts: Set<string>;
   418→  private onAlertCallback?: (alert: FraudAlert) => void;
   419→
   420→  constructor() {
   421→    this.rules = new Map();
   422→    this.alertHistory = new Map();
   423→    this.subscriberProfiles = new Map();
   424→    this.activeAlerts = new Set();
   425→
   426→    // Load default rules
   427→    DEFAULT_RULES.forEach(rule => this.rules.set(rule.id, rule));
   428→  }
   429→
   430→  /**
   431→   * Set callback for real-time alerts
   432→   */
   433→  onAlert(callback: (alert: FraudAlert) => void): void {
   434→    this.onAlertCallback = callback;
   435→  }
   436→
   437→  /**
   438→   * Add or update a fraud detection rule
   439→   */
   440→  updateRule(rule: FraudRule): void {
   441→    this.rules.set(rule.id, rule);
   442→  }
   443→
   444→  /**
   445→   * Get all configured rules
   446→   */
   447→  getRules(): FraudRule[] {
   448→    return Array.from(this.rules.values());
   449→  }
   450→
   451→  /**
   452→   * Enable/disable a rule
   453→   */
   454→  setRuleEnabled(ruleId: string, enabled: boolean): boolean {
   455→    const rule = this.rules.get(ruleId);
   456→    if (!rule) return false;
   457→    rule.enabled = enabled;
   458→    return true;
   459→  }
   460→
   461→  /**
   462→   * Main analysis entry point - analyze an SS7 message for fraud indicators
   463→   */
   464→  analyze(message: SS7Message): FraudIndicator[] {
   465→    const indicators: FraudIndicator[] = [];
   466→
   467→    switch (message.protocol) {
   468→      case SS7ProtocolLayer.MAP:
   469→        indicators.push(...this.analyzeMAPMessage(message as DecodedMAPMessage));
   470→        break;
   471→      case SS7ProtocolLayer.ISUP:
   472→        indicators.push(...this.analyzeISUPMessage(message as DecodedISUPMessage));
   473→        break;
   474→      case SS7ProtocolLayer.CAP:
   475→        indicators.push(...this.analyzeCAPMessage(message));
   476→        break;
   477→      default:
   478→        // Generic analysis for other protocols
   479→        break;
   480→    }
   481→
   482→    return indicators;
   483→  }
   484→
   485→  /**
   486→   * Analyze batch of messages and generate alerts
   487→   */
   488→  analyzeBatch(messages: SS7Message[]): FraudAlert[] {
   489→    const alerts: FraudAlert[] = [];
   490→    
   491→    // Update subscriber profiles first
   492→    messages.forEach(msg => this.updateSubscriberProfile(msg));
   493→
   494→    // Check each message against rules
   495→    for (const message of messages) {
   496→      const indicators = this.analyze(message);
   497→      
   498→      if (indicators.length > 0) {
   499→        // Generate alerts based on indicators
   500→        const generatedAlerts = this.generateAlerts(message, indicators);
   501→        alerts.push(...generatedAlerts);
   502→      }
   503→    }
   504→
   505→    return alerts;
   506→  }
   507→
   508→  // ============================================================
   509→  // MAP MESSAGE ANALYSIS
   510→  // ============================================================
   511→
   512→  private analyzeMAPMessage(message: DecodedMAPMessage): FraudIndicator[] {
   513→    const indicators: FraudIndicator[] = [];
   514→
   515→    switch (message.operation) {
   516→      case MAPOperationCode.UPDATE_LOCATION:
   517→        indicators.push(...this.checkLocationUpdateAnomalies(message));
   518→        break;
   519→
   520→      case MAPOperationCode.SEND_AUTHENTICATION_INFO:
   521→        indicators.push(...this.checkAuthPatterns(message));
   522→        break;
   523→
   524→      case MAPOperationCode.INSERT_SUBSCRIBER_DATA:
   525→        indicators.push(...this.checkProvisioningPatterns(message));
   526→        break;
   527→
   528→      case MAPOperationCode.ROUTING_INFO_FOR_SM:
   529→      case MAPOperationCode.MO_FORWARD_SHORT_MESSAGE:
   530→      case MAPOperationCode.MT_FORWARD_SHORT_MESSAGE:
   531→        indicators.push(...this.checkSMSAbusePatterns(message));
   532→        break;
   533→
   534→      case MAPOperationCode.PROVIDE_ROAMING_NUMBER:
   535→        indicators.push(...this.checkRoamingPatterns(message));
   536→        break;
   537→
   538→      default:
   539→        break;
   540→    }
   541→
   542→    return indicators;
   543→  }
   544→
   545→  private checkLocationUpdateAnomalies(message: DecodedMAPMessage): FraudIndicator[] {
   546→    const indicators: FraudIndicator[] = [];
   547→    const profile = this.getOrCreateProfile(message);
   548→
   549→    // Check for location updates during active call
   550→    if (profile.activeCalls > 0) {
   551→      indicators.push(FraudIndicator.LOCATION_UPDATE_WHILE_ACTIVE_CALL);
   552→      
   553→      // Add evidence
   554→      profile.addEvidence({
   555→        id: `ev_${Date.now()}`,
   556→        timestamp: new Date(),
   557→        type: 'location_update',
   558→        description: `Location update received while ${profile.activeCalls} call(s) active`,
   559→        data: { operation: MAP_OPERATION_NAMES[message.operation] },
   560→        riskContribution: 75,
   561→      });
   562→    }
   563→
   564→    // Check for impossible roaming speed
   565→    if (profile.lastLocation && message.sccp?.destinationGlobalTitle?.digits) {
   566→      const distance = this.calculateDistance(
   567→        profile.lastLocation,
   568→        message.sccp.destinationGlobalTitle.digits
   569→      );
   570→      const timeDiff = Date.now() - profile.lastLocationTime.getTime();
   571→      const hoursDiff = timeDiff / (1000 * 60 * 60);
   572→
   573→      if (hoursDiff > 0 && distance / hoursDiff > 800) { // Faster than plane
   574→        indicators.push(FraudIndicator.ROAMING_ANOMALY_FAST_TRAVEL);
   575→        
   576→        profile.addEvidence({
   577→          id: `ev_${Date.now()}`,
   578→          timestamp: new Date(),
   579→          type: 'location_update',
   580→          description: `Impossible travel: ${Math.round(distance)}km in ${hoursDiff.toFixed(1)}h`,
   581→          data: { distance, timeHours: hoursDiff, speed: distance / hoursDiff },
   582→          riskContribution: 90,
   583→        });
   584→      }
   585→    }
   586→
   587→    // Update last location
   588→    if (message.sccp?.sourceGlobalTitle?.digits) {
   589→      profile.lastLocation = message.sccp.sourceGlobalTitle.digits;
   590→      profile.lastLocationTime = new Date();
   591→    }
   592→    profile.locationsSeen.add(message.sccp?.destinationGlobalTitle?.digits || 'unknown');
   593→
   594→    return indicators;
   595→  }
   596→
   597→  private checkAuthPatterns(message: DecodedMAPMessage): FraudIndicator[] {
   598→    const indicators: FraudIndicator[] = [];
   599→    const profile = this.getOrCreateProfile(message);
   600→
   601→    // Track authentication attempts
   602→    profile.authAttempts.push(Date.now());
   603→
   604→    // Clean old auth attempts (keep only last hour)
   605→    const oneHourAgo = Date.now() - 60 * 60 * 1000;
   606→    profile.authAttempts = profile.authAttempts.filter(t => t > oneHourAgo);
   607→
   608→    // Check for auth failure burst
   609→    if (profile.authAttempts.length >= 5) {
   610→      indicators.push(FraudIndicator.SIM_SWAP_AUTH_FAILURE_BURST);
   611→      
   612→      profile.addEvidence({
   613→        id: `ev_${Date.now()}`,
   614→        timestamp: new Date(),
   615→        type: 'authentication',
   616→        description: `${profile.authAttempts.length} authentication attempts in last hour`,
   617→        data: { attemptCount: profile.authAttempts.length },
   618→        riskContribution: 70,
   619→      });
   620→    }
   621→
   622→    return indicators;
   623→  }
   624→
   625→  private checkProvisioningPatterns(message: DecodedMAPMessage): FraudIndicator[] {
   626→    const indicators: FraudIndicator[] = [];
   627→    const profile = this.getOrCreateProfile(message);
   628→
   629→    // Track provisioning attempts
   630→    profile.provisioningAttempts.push(Date.now());
   631→
   632→    // Clean old attempts (keep last hour)
   633→    const oneHourAgo = Date.now() - 60 * 60 * 1000;

   793→  private generateAlerts(message: SS7Message, indicators: FraudIndicator[]): FraudAlert[] {
   794→    const alerts: FraudAlert[] = [];
   795→    const subscriberKey = this.getSubscriberKey(message);
   796→
   797→    // Group indicators by fraud type
   798→    const indicatorGroups = this.groupIndicatorsByType(indicators);
   799→

   801→      // Find matching rules
   802→      for (const [ruleId, rule] of this.rules.entries()) {

   804→
   805→        // Check cooldown
   806→        if (this.isInCooldown(subscriberKey, ruleId)) continue;
   807→
   808→        // Evaluate rule conditions
   809→        const evaluation = this.evaluateRule(rule, message, groupIndicators);
   810→        
   811→        if (evaluation.triggered) {
   812→          const alert = this.createAlert(rule, message, groupIndicators, evaluation);
   813→          alerts.push(alert);
   814→          
   815→          // Store alert
   816→          this.storeAlert(alert, subscriberKey);
   817→          
   818→          // Trigger callback
   819→          if (this.onAlertCallback) {
   820→            this.onAlertCallback(alert);
   821→          }
   822→        }
   823→      }
   824→    }
   825→
   826→    return alerts;

   829→  private evaluateRule(
   830→    rule: FraudRule,
   831→    message: SS7Message,
   832→    indicators: FraudIndicator[]
   833→  ): { triggered: boolean; score: number; reason: string } {
   834→    let score = 0;
   835→    const reasons: string[] = [];
   836→    const profile = this.getOrCreateProfile(message);
   837→    const threshold = rule.threshold;
   838→
   839→    switch (rule.type) {
   840→      case FraudType.IRSF:
   841→        if (threshold.internationalCallsPerHour && 
   842→            profile.internationalCallCount >= threshold.internationalCallsPerHour) {
   843→          score += 40;
   844→          reasons.push(`${profile.internationalCallCount} international calls/hour`);
   845→        }
   846→        if (threshold.uniqueInternationalDestinations &&
   847→            profile.internationalDestinations.size >= threshold.uniqueInternationalDestinations) {
   848→          score += 35;
   849→          reasons.push(`${profile.internationalDestinations.size} unique destinations`);
   850→        }
   851→        if (indicators.includes(FraudIndicator.IRSF_PREMIUM_RATE_TARGET)) {
   852→          score += 25;
   853→          reasons.push('Premium rate targets detected');
   854→        }
   855→        break;
   856→
   857→      case FraudType.SIM_SWAP:
   858→        if (threshold.provisioningAttemptsPerHour &&
   859→            profile.provisioningAttempts.length >= threshold.provisioningAttemptsPerHour) {
   860→          score += 45;
   861→          reasons.push(`${profile.provisioningAttempts.length} provisioning attempts`);
   862→        }
   863→        if (threshold.authFailuresPerHour &&
   864→            profile.authAttempts.length >= threshold.authFailuresPerHour) {
   865→          score += 35;
   866→          reasons.push(`${profile.authAttempts.length} auth failures`);
   867→        }
   868→        if (indicators.includes(FraudIndicator.LOCATION_UPDATE_WHILE_ACTIVE_CALL)) {
   869→          score += 20;
   870→          reasons.push('Location update during active call');
   871→        }
   872→        break;
   873→
   874→      case FraudType.WANGIRI:
   875→        if (threshold.shortDurationCallsPerHour) {
   876→          const shortCalls = profile.callsToday.filter(
   877→            c => c.duration <= (threshold.maxWangiriDurationSeconds || 5)
   878→          ).length;
   879→          if (shortCalls >= threshold.shortDurationCallsPerHour) {
   880→            score += 50;
   881→            reasons.push(`${shortCalls} short duration calls`);
   882→          }
   883→        }
   884→        if (threshold.uniqueCalledNumbersPerHour) {
   885→          const uniqueNumbers = new Set(profile.callsToday.map(c => c.calledNumber)).size;
   886→          if (uniqueNumbers >= threshold.uniqueCalledNumbersPerHour) {
   887→            score += 30;
   888→            reasons.push(`${uniqueNumbers} unique numbers called`);
   889→          }
   890→        }
   891→        break;
   892→
   893→      case FraudType.BYPASS_FRAUD:
   894→        // Simbox detection based on patterns
   895→        if (indicators.includes(FraudIndicator.BYPASS_FRAUD_GSM_GATEWAY)) {
   896→          score += 50;
   897→          reasons.push('GSM gateway pattern detected');
   898→        }
   899→        if (indicators.includes(FraudIndicator.BYPASS_FRAUD_SIMBOX)) {
   900→          score += 50;
   901→          reasons.push('Simbox pattern detected');
   902→        }
   903→        break;
   904→
   905→      case FraudType.ROAMING_ANOMALY:
   906→        if (indicators.includes(FraudIndicator.ROAMING_ANOMALY_FAST_TRAVEL)) {
   907→          score += 60;
   908→          reasons.push('Impossible travel speed detected');
   909→        }
   910→        if (indicators.includes(FraudIndicator.ROAMING_ANOMALY_IMPOSSIBLE)) {
   911→          score += 40;
   912→          reasons.push('Impossible roaming detected');
   913→        }
   914→        if (threshold.countriesVisitedPerDay &&
   915→            profile.locationsSeen.size >= threshold.countriesVisitedPerDay) {
   916→          score += 30;
   917→          reasons.push(`${profile.locationsSeen.size} locations visited`);
   918→        }
   919→        break;
   920→    }
   921→
   922→    const triggered = score >= (threshold.riskScoreThreshold || 50);
   923→    
   924→    return {
   925→      triggered,
   926→      score: Math.min(score, 100),
   927→      reason: reasons.join('; ') || 'Threshold met',
   928→    };
   929→  }

  1102→  private calculateDistance(location1: string, location2: string): number {
  1103→    // Simplified distance calculation
  1104→    // In production, this would use actual coordinates or MCC/MNC lookup
  1105→    if (location1 === location2) return 0;
  1106→    
  1107→    // Return a random-ish but deterministic value for demo
  1108→    // Real implementation would use cell tower coordinates
  1109→    const hash = (loc: string) => {
  1110→      let h = 0;
  1111→      for (let i = 0; i < loc.length; i++) {
  1112→        h = ((h << 5) - h + loc.charCodeAt(i)) | 0;
  1113→      }
  1114→      return Math.abs(h);
  1115→    };
  1116→    
  1117→    return Math.abs(hash(location1) - hash(location2)) % 5000; // Max ~5000km
  1118→  }

  1145→  getStatistics(): FraudStatistics {
  1146→    const stats: FraudStatistics = {
  1147→      totalAlerts: 0,
  1148→      alertsByStatus: {},

  1150→      alertsBySeverity: {},

  1155→    };
  1156→
  1157→    const now = Date.now();

  1159→
  1160→    for (const alerts of this.alertHistory.values()) {
  1161→      for (const alert of alerts) {
  1162→        if (alert.timestamp.getTime() < twentyFourHoursAgo) continue;
  1163→        
  1164→        stats.totalAlerts++;
  1165→        stats.alertsByStatus[alert.status] = (stats.alertsByStatus[alert.status] || 0) + 1;

  1171→        }
  1172→        
  1173→        if (alert.status === AlertStatus.BLOCKED) {
  1174→          stats.blockedSubscribers++;
  1175→        }
  1176→      }
  1177→    }
  1178→
  1179→    // Top fraud types
  1180→    stats.topFraudTypes = Object.entries(stats.alertsByType)
  1181→      .sort(([,a], [,b]) => b - a)
  1182→      .slice(0, 5)
  1183→      .map(([type, count]) => ({ type, count }));
  1184→
  1185→    // Recent alerts (last 24h)
  1186→    stats.recentAlerts = this.getActiveAlerts().slice(0, 10);
  1187→
  1188→    return stats;
  1189→  }

  1191→  /**
  1192→   * Block a suspicious subscriber
  1193→   */
  1194→  async blockSubscriber(alertId: string, reason: string): Promise<boolean> {
  1195→    const alert = Array.from(this.alertHistory.values())
  1196→      .flat()
  1197→      .find(a => a.id === alertId);
  1198→    
  1199→    if (!alert) return false;
  1200→
  1201→    const action: FraudAction = {
  1202→      id: `action_${Date.now()}`,
  1203→      type: FraudActionType.BLOCK_SUBSCRIBER,
  1204→      status: ActionStatus.EXECUTED,
  1205→      timestamp: new Date(),
  1206→      performedBy: 'system',
  1207→      details: reason,
  1208→      result: 'Subscriber blocked successfully',
  1209→    };
  1210→
  1211→    if (!alert.actions) alert.actions = [];
  1212→    alert.actions.push(action);
  1213→    alert.status = AlertStatus.BLOCKED;
  1214→
  1215→    return true;
  1216→  }

  1218→  /**
  1219→   * Clear old data (memory management)
  1220→   */
  1221→  cleanup(maxAgeHours: number = 24): void {
  1222→    const cutoff = Date.now() - maxAgeHours * 60 * 60 * 1000;
  1223→
  1224→    // Clean up alert history
  1225→    for (const [key, alerts] of this.alertHistory.entries()) {
  1226→      const filtered = alerts.filter(a => a.timestamp.getTime() > cutoff);
  1227→      if (filtered.length === 0) {
  1228→        this.alertHistory.delete(key);
  1229→      } else {
  1230→        this.alertHistory.set(key, filtered);
  1231→      }
  1232→    }
  1233→
  1234→    // Clean up subscriber profiles
  1235→    for (const [key, profile] of this.subscriberProfiles.entries()) {
  1236→      profile.cleanup(cutoff);
  1237→      if (profile.isEmpty()) {
  1238→        this.subscriberProfiles.delete(key);
  1239→      }
  1240→    }
  1241→  }
