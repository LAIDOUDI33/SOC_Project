  1089→/**
  1090→ * Parse a point code based on format
  1091→ */
  1092→export function parsePointCode(raw: number, format: PointCodeFormat = PointCodeFormat.ITU_14BIT): PointCode {
  1093→  switch (format) {
  1094→    case PointCodeFormat.ITU_14BIT:
  1095→      // 3-8-3 format
  1096→      return {
  1097→        raw,
  1098→        format,
  1099→        network: (raw >> 11) & 0x07,
  1100→        cluster: (raw >> 3) & 0xFF,
  1101→        member: raw & 0x07,
  1102→        display: `${(raw >> 11) & 0x07}-${((raw >> 3) & 0xFF).toString().padStart(3, '0')}-${(raw & 0x07).toString()}`,
  1103→      };
  1104→    
  1105→    case PointCodeFormat.ANSI_24BIT:
  1106→      // 8-8-8 format
  1107→      return {
  1108→        raw,
  1109→        format,
  1110→        network: (raw >> 16) & 0xFF,
  1111→        cluster: (raw >> 8) & 0xFF,
  1112→        member: raw & 0xFF,
  1113→        display: `${((raw >> 16) & 0xFF).toString().padStart(3, '0')}-${((raw >> 8) & 0xFF).toString().padStart(3, '0')}-${(raw & 0xFF).toString().padStart(3, '0')}`,
  1114→      };
  1115→    
  1116→    case PointCodeFormat.JAPAN_16BIT:
  1117→      // 4-4-8 format
  1118→      return {
  1119→        raw,
  1120→        format,
  1121→        network: (raw >> 12) & 0x0F,
  1122→        cluster: (raw >> 8) & 0x0F,
  1123→        member: raw & 0xFF,
  1124→        display: `${((raw >> 12) & 0x0F).toString()}-${((raw >> 8) & 0x0F).toString()}-${(raw & 0xFF).toString().padStart(3, '0')}`,
  1125→      };
  1126→    
  1127→    default:
  1128→      return {
  1129→        raw,
  1130→        format,
  1131→        network: 0,
  1132→        cluster: 0,
  1133→        member: raw,
  1134→        display: raw.toString(),
  1135→      };
  1136→  }
  1137→}

  1175→/**
  1176→ * Mask MSISDN for privacy compliance (GDPR/ARTP)
  1177→ * Format: +213XXXXXXXXX -> +213XXX***XX
  1178→ */
  1179→export function maskMSISDN(msisdn: string): string {
  1180→  if (!msisdn || msisdn.length < 6) return msisdn.replace(/\d/g, '*');
  1181→  
  1182→  // Keep country code and first 2 digits, mask the rest except last 2
  1183→  const cleaned = msisdn.replace(/[^+\d]/g, '');
  1184→  if (cleaned.startsWith('+213')) {
  1185→    return '+213' + cleaned.slice(4, 6) + '****' + cleaned.slice(-2);
  1186→  }
  1187→  
  1188→  // Generic masking for non-Algerian numbers
  1189→  if (cleaned.startsWith('+')) {
  1190→    return cleaned.substring(0, 4) + '****' + cleaned.slice(-2);
  1191→  }
  1192→  
  1193→  return cleaned.substring(0, 3) + '****' + cleaned.slice(-2);
  1194→}

  1197→ * Mask IMSI for privacy compliance
  1198→ * Format: 60301XXXXXXXXX -> 60301********
  1199→ */
  1200→export function maskIMSI(imsi: string): string {
  1201→  if (!imsi || imsi.length < 5) return imsi.replace(/\d/g, '*');
  1202→  // Keep MCC+MNC (5 digits), mask MSIN
  1203→  return imsi.substring(0, 5) + '********';
  1204→}

  1225→/**
  1226→ * Format bytes as hex dump
  1227→ */
  1228→export function hexDump(buffer: Buffer, bytesPerLine: number = 16): string {
  1229→  const lines: string[] = [];
  1230→  for (let offset = 0; offset < buffer.length; offset += bytesPerLine) {
  1231→    const chunk = buffer.slice(offset, Math.min(offset + bytesPerLine, buffer.length));
  1232→    const hex = Array.from(chunk).map(b => b.toString(16).padStart(2, '0')).join(' ');
  1233→    const ascii = Array.from(chunk).map(b => b >= 32 && b <= 126 ? String.fromCharCode(b) : '.').join('');
  1234→    lines.push(`${offset.toString(16).padStart(8, '0')}  ${hex.padEnd(bytesPerLine * 3)} |${ascii}|`);
  1235→  }
  1236→  return lines.join('\n');
  1237→}
