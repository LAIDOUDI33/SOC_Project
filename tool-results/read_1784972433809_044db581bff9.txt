   190→  /**
   191→   * Build URL with query parameters
   192→   */
   193→  private buildUrl(path: string, params?: Record<string, unknown>): string {
   194→    const url = new URL(`${this.config.baseUrl}${path}`);
   195→    
   196→    if (params) {
   197→      Object.entries(params).forEach(([key, value]) => {
   198→        if (value !== undefined && value !== null && value !== '') {
   199→          if (Array.isArray(value)) {
   200→            value.forEach(v => url.searchParams.append(key, String(v)));
   201→          } else {
   202→            url.searchParams.append(key, String(value));
   203→          }
   204→        }
   205→      });
   206→    }
   207→    
   208→    return url.toString();
   209→  }
   210→
   211→  /**
   212→   * Get cached data or fetch fresh
   213→   */
   214→  private async getCached<T>(cacheKey: string, fetcher: () => Promise<T>): Promise<T> {
   215→    if (!this.config.enableCache) {
   216→      return fetcher();
   217→    }
   218→
   219→    const cached = this.cache.get(cacheKey) as CacheEntry<T> | undefined;
   220→    const now = Date.now();
   221→
   222→    if (cached && now < cached.expiresAt) {
   223→      if (this.config.debug) {
   224→        console.log(`[SuricataClient] Cache hit: ${cacheKey}`);
   225→      }
   226→      return cached.data;
   227→    }
   228→
   229→    const data = await fetcher();
   230→    this.cache.set(cacheKey, {
   231→      data,
   232→      timestamp: now,
   233→      expiresAt: now + this.config.cacheTtl
   234→    });
   235→
   236→    // Clean up old entries periodically
   237→    if (this.requestCount % 100 === 0) {
   238→      this.cleanCache();
   239→    }
   240→
   241→    return data;
   242→  }
   243→
   244→  /**
   245→   * Remove expired cache entries
   246→   */
   247→  private cleanCache(): void {
   248→    const now = Date.now();
   249→    for (const [key, entry] of this.cache.entries()) {
   250→      if (now >= entry.expiresAt) {
   251→        this.cache.delete(key);
   252→      }
   253→    }
   254→  }
   255→
   256→  /**
   257→   * Clear all cached data
   258→   */
   259→  clearCache(): void {
   260→    this.cache.clear();
   261→    if (this.config.debug) {
   262→      console.log('[SuricataClient] Cache cleared');
   263→    }
   264→  }
   265→
   266→  /**
   267→   * Execute HTTP request with error handling and retries
   268→   */
   269→  private async request<T>(
   270→    method: string,
   271→    path: string,
   272→    body?: unknown,
   273→    options?: { skipCache?: boolean; headers?: Record<string, string> }
   274→  ): Promise<T> {
   275→    const url = this.buildUrl(path);
   276→    const cacheKey = `${method}:${url}:${JSON.stringify(body)}`;
   277→
   278→    // Check rate limiting
   279→    this.enforceRateLimit();
   280→
   281→    const controller = new AbortController();
   282→    const timeoutId = setTimeout(() => controller.abort(), this.config.timeout);
   283→
   284→    try {
   285→      const headers: Record<string, string> = {
   286→        'Content-Type': 'application/json',
   287→        'Accept': 'application/json',
   288→        ...(this.config.apiKey ? { 'X-API-Key': this.config.apiKey } : {}),
   289→        ...(options?.headers || {})
   290→      };
   291→
   292→      if (this.config.debug) {
   293→        console.log(`[SuricataClient] ${method} ${url}`, body ? { body } : '');
   294→      }
   295→
   296→      const response = await fetch(url, {
   297→        method,
   298→        headers,
   299→        body: body ? JSON.stringify(body) : undefined,
   300→        signal: controller.signal
   301→      });
   302→
   303→      clearTimeout(timeoutId);
   304→      this.requestCount++;
   305→      this.lastRequestTime = Date.now();
   306→
   307→      // Handle specific status codes
   308→      if (response.status === 401) {
   309→        throw new SuricataAuthError('Invalid or missing API key');
   310→      }
   311→
   312→      if (response.status === 429) {
   313→        const retryAfter = parseInt(response.headers.get('Retry-After') || '60', 10);
   314→        throw new SuricataRateLimitError(retryAfter);
   315→      }
   316→
   317→      if (!response.ok) {
   318→        const errorBody = await response.json().catch(() => ({}));
   319→        throw new SuricataError(
   320→          errorBody.message || `HTTP ${response.status}: ${response.statusText}`,
   321→          `HTTP_${response.status}`,
   322→          response.status,
   323→          errorBody.details
   324→        );
   325→      }
   326→
   327→      const data = await response.json();
   328→      
   329→      if (options?.skipCache) {
   330→        return data;
   331→      }
   332→      
   333→      return this.getCached(cacheKey, () => Promise.resolve(data));
   334→
   335→    } catch (error) {
   336→      clearTimeout(timeoutId);
   337→
   338→      if (error instanceof SuricataError) {
   339→        throw error;
   340→      }
   341→
   342→      if (error instanceof DOMException && error.name === 'AbortError') {
   343→        throw new SuricataConnectionError(`Request timed out after ${this.config.timeout}ms`);
   344→      }
   345→
   346→      throw new SuricataConnectionError(
   347→        error instanceof Error ? error.message : 'Unknown connection error'
   348→      );
   349→    }
   350→  }
   351→
   352→  /**
   353→   * Enforce rate limiting between requests
   354→   */
   355→  private enforceRateLimit(): void {
   356→    const minInterval = 100; // Minimum 100ms between requests
   357→    const timeSinceLastRequest = Date.now() - this.lastRequestTime;
   358→
   359→    if (timeSinceLastRequest < minInterval && this.lastRequestTime > 0) {
   360→      const waitTime = minInterval - timeSinceLastRequest;
   361→      if (this.config.debug) {
   362→        console.log(`[SuricataClient] Rate limit: waiting ${waitTime}ms`);
   363→      }
   364→    }
   365→  }

   572→  /**
   573→   * Create custom rule
   574→   */
   575→  async createRule(ruleData: Omit<SuricataRule, 'id' | 'created_at' | 'updated_at' | 'hit_count' | 'false_positive_count'>): Promise<SuricataRule> {
   576→    // Validate rule syntax before creation
   577→    const validation = await this.validateRule(ruleData.raw);
   578→    if (validation.status !== RuleValidationStatus.VALID) {
   579→      throw new RuleCompilationError('new-rule', validation.errors || ['Validation failed']);
   580→    }
   581→
   582→    return this.request('POST', '/rules', {
   583→      ...ruleData,
   584→      source: SignatureSource.CUSTOM,
   585→      created_at: new Date().toISOString()
   586→    });
   587→  }

   590→   * Update existing rule
   591→   */
   592→  async updateRule(sid: number, updates: Partial<SuricataRule>): Promise<SuricataRule> {
   593→    // If raw rule changed, re-validate
   594→    if (updates.raw) {
   595→      const validation = await this.validateRule(updates.raw);
   596→      if (validation.status !== RuleValidationStatus.VALID) {
   597→        throw new RuleCompilationError(String(sid), validation.errors || ['Validation failed']);
   598→      }
   599→    }
   600→
   601→    return this.request('PUT', `/rules/${sid}`, {
   602→      ...updates,
   603→      updated_at: new Date().toISOString()
   604→    });
   605→  }

   905→  /**
   906→   * Format alert for MISP event creation
   907→   */
   908→  formatAlertForMISP(alert: EveEvent): {
   909→    title: string;
   910→    description: string;
   911→    attributes: Array<{ type: string; value: string; category: string }>;
   912→    tags: string[];
   913→  } {
   914→    if (!alert.alert) {
   915→      throw new Error('Event is not an alert type');
   916→    }
   917→
   918→    const { alert: alertData } = alert;
   919→    
   920→    return {

   924→        `Severity: ${alertData.severity}\n` +

   927→        `Timestamp: ${alert.timestamp}`,

   930→        { type: 'ip-dst', value: alert.dest_ip, category: 'Network activity' },
   931→        { type: 'domain', value: alert.http?.hostname || '', category: 'Network activity' },
   932→        { type: 'text', value: alertData.signature, category: 'Payload delivery' },
   933→        { type: 'text', value: alertData.signature_id.toString(), category: 'External analysis' },
   934→        { type: 'port', value: alert.src_port.toString(), category: 'Network activity' },
   935→        { type: 'port', value: alert.dest_port.toString(), category: 'Network activity' }
   936→      ].filter(attr => attr.value),
   937→      tags: [
   938→        `suricata:${alertData.signature_id}`,
   939→        `severity:${alertData.severity}`,
   940→        `category:${alertData.category.toLowerCase().replace(/\s+/g, '_')}`,
   941→        ...Object.keys(alertData.metadata || {}).map(k => `metadata:${k}`)

   943→    };

   947→   * Format alert for TheHive case creation
   948→   */
   949→  formatAlertForTheHive(alert: EveEvent): {
   950→    title: string;
   951→    description: string;
   952→    severity: number;
   953→    tags: string[];
   954→    observables: Array<{ dataType: string; data: string; message: string }>;
   955→    tlp: number;

   957→    if (!alert.alert) {
   958→      throw new Error('Event is not an alert type');

   960→

   992→  /**
   993→   * Generate summary report for dashboard
   994→   */
   995→  async generateSummaryReport(): Promise<{
   996→    total_alerts: number;
   997→    critical_alerts: number;
   998→    unique_signatures: number;
   999→    unique_source_ips: number;
  1000→    unique_dest_ips: number;
  1001→    top_attack_countries: Array<{ country: string; count: number }>;
  1002→    packet_stats: Awaited<ReturnType<SuricataClient['getPacketStats']>>;
  1003→    sensor_status: Array<{ id: string; name: string; status: SensorStatus }>;
  1004→    last_rule_update: string;
  1005→  }> {
  1006→    const [stats, sensors, packetStats] = await Promise.all([
  1007→      this.getStatistics(),
  1008→      this.getSensors(),
  1009→      this.getPacketStats()
  1010→    ]);
  1011→
  1012→    // Aggregate countries from top source IPs
  1013→    const countryCounts: Record<string, number> = {};
  1014→    stats.top_source_ips.forEach(ip => {
  1015→      if (ip.country) {
  1016→        countryCounts[ip.country] = (countryCounts[ip.country] || 0) + ip.alert_count;
  1017→      }
  1018→    });
  1019→
  1020→    return {
  1021→      total_alerts: Object.values(stats.alerts_by_severity).reduce((a, b) => a + b, 0),
  1022→      critical_alerts: stats.alerts_by_severity[SeverityLevel.CRITICAL] || 0,
  1023→      unique_signatures: stats.top_signatures.length,
  1024→      unique_source_ips: stats.top_source_ips.length,
  1025→      unique_dest_ips: stats.top_destination_ips.length,
  1026→      top_attack_countries: Object.entries(countryCounts)
  1027→        .sort(([, a], [, b]) => b - a)
  1028→        .slice(0, 10)
  1029→        .map(([country, count]) => ({ country, count })),
  1030→      packet_stats: packetStats,
  1031→      sensor_status: sensors.map(s => ({
  1032→        id: s.id,
  1033→        name: s.name,
  1034→        status: s.status
  1035→      })),
  1036→      last_rule_update: sensors[0]?.last_rule_update || 'Unknown'
  1037→    };
  1038→  }

  1134→  static parseEVEJson(line: string): EveEvent | null {
