   118→ * Handle GET requests - List models and get performance

   121→ */
   122→export async function GET(request: NextRequest) {
   123→  const searchParams = request.nextUrl.searchParams;
   124→  const action = searchParams.get('action');
   125→
   126→  try {
   127→    switch (action) {
   128→      case 'models':
   129→        return handleListModels();
   130→      
   131→      case 'performance':
   132→        return handleGetPerformance(searchParams);
   133→      
   134→      case 'status':
   135→        return handleGetStatus();
   136→      
   137→      default:
   138→        return NextResponse.json(
   139→          { error: 'Invalid action. Use: models, performance, or status' },
   140→          { status: 400 }
   141→        );
   142→    }
   143→  } catch (error) {
   144→    console.error('ML API GET error:', error);
   145→    return NextResponse.json(
   146→      { error: 'Internal server error', message: (error as Error).message },
   147→      { status: 500 }
   148→    );
   149→  }
   150→}

   280→/**
   281→ * Get model performance metrics
   282→ */
   283→async function handleGetPerformance(searchParams: URLSearchParams): Promise<NextResponse> {
   284→  const modelId = searchParams.get('modelId');
   285→  const timeRange = searchParams.get('timeRange') || '24h';
   286→
   287→  if (modelId) {
   288→    // Get specific model performance
   289→    const perf = performanceTracker.getLatestPerformance(modelId);
   290→    const trend = performanceTracker.getPerformanceTrend(modelId);
   291→
   292→    return NextResponse.json({
   293→      success: true,
   294→      data: {
   295→        modelId,
   296→        performance: perf,
   297→        trend,
   298→        timeRange,
   299→        timestamp: new Date().toISOString(),
   300→      },
   301→    });
   302→  }
   303→
   304→  // Get aggregate performance across all models
   305→  const allMetrics = Array.from(models.keys()).map(id => ({
   306→    modelId: id,
   307→    ...performanceTracker.getLatestPerformance(id),
   308→  }));
   309→
   310→  // Calculate aggregates
   311→  const avgAccuracy = allMetrics.reduce((sum, m) => sum + (m.accuracy || 0), 0) / (allMetrics.length || 1);
   312→  const avgF1 = allMetrics.reduce((sum, m) => sum + (m.f1Score || 0), 0) / (allMetrics.length || 1);
   313→
   314→  return NextResponse.json({
   315→    success: true,
   316→    data: {
   317→      summary: {
   318→        totalModelsTracked: allMetrics.length,
   319→        averageAccuracy: avgAccuracy,
   320→        averageF1Score: avgF1,
   321→      },
   322→      models: allMetrics,
   323→      timeRange,
   324→      timestamp: new Date().toISOString(),
   325→    },
   326→  });
   327→}

   364→/**
   365→ * Run prediction with a model
   366→ */
   367→async function handlePredict(body: any): Promise<NextResponse> {
   368→  const { modelId, features, options } = body;
   369→
   370→  if (!features || !Array.isArray(features)) {
   371→    return NextResponse.json(
   372→      { error: 'features array is required' },
   373→      { status: 400 }
   374→    );
   375→  }
   376→
   377→  let modelEntry = modelId ? models.get(modelId) : undefined;
   378→
   379→  // Use default model if none specified or found
   380→  if (!modelEntry && models.size > 0) {
   381→    modelEntry = models.values().next().value;
   382→  }
   383→
   384→  // Create default model if needed
   385→  if (!modelEntry) {
   386→    const nn = new SimplifiedNeuralNetwork(DEFAULT_THREAT_NN_CONFIG);
   387→    nn.initialize(features.length);
   388→    
   389→    modelEntry = {
   390→      instance: nn,
   391→      metadata: {
   392→        id: 'auto-created-' + generateId(),
   393→        name: 'Auto-created Model',
   394→        type: 'neural_network',
   395→        version: '1.0.0',
   396→        status: 'active',
   397→        performance: {
   398→          accuracy: 0,
   399→          precision: 0,
   400→          recall: 0,
   401→          f1Score: 0,
   402→          aucROC: 0,
   403→          aucPR: 0,
   404→          falsePositiveRate: 0,
   405→          falseNegativeRate: 0,
   406→          calibrationError: 0,
   407→          inferenceTimeAvgMs: 0,
   408→          inferenceTimeP95Ms: 0,
   409→          throughputPerSecond: 0,
   410→        },
   411→        trainingInfo: {
   412→          totalEpochs: 0,
   413→          bestEpoch: 0,
   414→          finalLoss: 0,
   415→          bestLoss: 0,
   416→          trainingSamples: 0,
   417→          validationSamples: 0,
   418→          convergenceEpoch: 0,
   419→          earlyStopped: false,
   420→          trainingHistory: [],
   421→        },
   422→        config: DEFAULT_THREAT_NN_CONFIG,
   423→        createdAt: new Date(),
   424→        lastUpdated: new Date(),
   425→        description: 'Auto-created for prediction',
   426→        tags: ['auto-created'],
   427→      },
   428→    };
   429→    models.set(modelEntry.metadata.id, modelEntry);
   430→  }
   431→
   432→  try {
   433→    const startTime = Date.now();
   434→    
   435→    let result: PredictionResult;
   436→    
   437→    if (modelEntry.instance instanceof SimplifiedNeuralNetwork) {
   438→      result = modelEntry.instance.predict(features);
   439→    } else if (modelEntry.instance instanceof IsolationForestDetector) {
   440→      result = modelEntry.instance.predict(features);
   441→    } else if (modelEntry.instance instanceof EnsembleDetector) {
   442→      result = modelEntry.instance.predict(features);
   443→    } else {
   444→      throw new Error('Unknown model type');
   445→    }
   446→
   447→    const processingTime = Date.now() - startTime;
   448→
   449→    // Record performance
   450→    performanceTracker.recordPerformance(modelEntry.metadata.id, {
   451→      ...(modelEntry.metadata.performance),
   452→      inferenceTimeAvgMs: processingTime,
   453→      throughputPerSecond: Math.floor(1000 / processingTime),
   454→    });
   455→
   456→    return NextResponse.json({
   457→      success: true,
   458→      data: {
   459→        prediction: result,
   460→        modelId: modelEntry.metadata.id,
   461→        modelName: modelEntry.metadata.name,
   462→        processingTimeMs: processingTime,
   463→        timestamp: new Date().toISOString(),
   464→      },
   465→    });
   466→
   467→  } catch (error) {
   468→    return NextResponse.json(
   469→      { 
   470→        error: 'Prediction failed',
   471→        message: (error as Error).message 
   472→      },
   473→      { status: 500 }
   474→    );
   475→  }
   476→}

   478→/**
   479→ * Train a model
   480→ */
   481→async function handleTrain(body: any): Promise<NextResponse> {
   482→  const { 
   483→    modelId, 
   484→    config, 
   485→    trainingData, 
   486→    validationData,
   487→    modelType = 'neural_network'
   488→  } = body;
   489→
   490→  if (!trainingData || !Array.isArray(trainingData) || trainingData.length === 0) {
   491→    return NextResponse.json(
   492→      { error: 'trainingData array is required and must not be empty' },
   493→      { status: 400 }
   494→    );
   495→  }
   496→
   497→  // Convert to TrainingSample format
   498→  const samples: TrainingSample[] = trainingData.map((d: any) => ({
   499→    input: d.input || d.features || [],
   500→    output: d.output || d.labels || [d.label ?? 0],
   501→    weight: d.weight,
   502→  }));
   503→
   504→  try {
   505→    const startTime = Date.now();
   506→    let trainedModel: { instance: any; metadata: ModelMetadata };
   507→
   508→    switch (modelType) {
   509→      case 'neural_network': {
   510→        const nn = new SimplifiedNeuralNetwork(config || DEFAULT_THREAT_NN_CONFIG);
   511→        
   512→        // Validate input size from first sample
   513→        if (samples[0].input.length > 0) {
   514→          nn.initialize(samples[0].input.length);
   515→        }
   516→
   517→        const valSamples = validationData?.map((d: any) => ({
   518→          input: d.input || d.features || [],
   519→          output: d.output || d.labels || [d.label ?? 0],
   520→        }));
   521→
   522→        const trainingInfo = nn.train(samples, valSamples);

   525→        // Estimate performance (would use actual test set in production)

   553→            lastUpdated: new Date(),
   554→            lastTrainedAt: new Date(),
   555→            description: 'Trained neural network model',
   556→            tags: ['trained', 'custom'],

   560→      }

   611→
   612→      default:

   638→
   639→  } catch (error) {
   640→    return NextResponse.json(
   641→      { 
   642→        error: 'Training failed',
   643→        message: (error as Error).message 
   644→      },
   645→      { status: 500 }
   646→    );
   647→  }
   648→}

   650→/**
   651→ * Analyze network flow through NIDS
   652→ */
   653→async function handleNIDSAnalyze(body: any): Promise<NextResponse> {
   654→  initializeEngines();
   655→
   656→  const { flow, flows } = body;
   657→
   658→  if (!flow && !flows) {
   659→    return NextResponse.json(
   660→      { error: 'flow or flows array is required' },
   661→      { status: 400 }
   662→    );
   663→  }
   664→
   665→  try {
   666→    let detections: NIDSDetectionResult[];
   667→
   668→    if (Array.isArray(flows)) {
   669→      detections = nidsEngine!.processBatch(flows as NetworkFlow[]);
   670→    } else {
   671→      detections = nidsEngine!.processFlow(flow as NetworkFlow);
   672→    }
   673→
   674→    const stats = nidsEngine!.getStatistics();
   675→
   676→    return NextResponse.json({
   677→      success: true,

   698→ * Check for fraud patterns
   699→ */
   700→async function handleFraudCheck(body: any): Promise<NextResponse> {
   701→  initializeEngines();
   702→
   703→  const { event, events, subscriber } = body;
   704→
   705→  if (!event && !events) {
   706→    return NextResponse.json(
   707→      { error: 'event or events array is required' },
   708→      { status: 400 }
   709→    );
   710→  }
   711→
   712→  try {
   713→    // Register subscriber if provided
   714→    if (subscriber) {
   715→      fraudEngine!.registerSubscriber(subscriber as SubscriberProfile);
   716→    }
   717→
   718→    const results: FraudDetectionResult[] = [];
   719→
   720→    if (Array.isArray(events)) {
   721→      for (const e of events) {
   722→        const detections = fraudEngine!.processEvent(e as FraudEvent);
   723→        results.push(...detections);
   724→      }
   725→    } else {
   726→      const detections = fraudEngine!.processEvent(event as FraudEvent);
   727→      results.push(...detections);
   728→    }
   729→
   730→    const stats = fraudEngine!.getStatistics();
   731→
   732→    return NextResponse.json({
   733→      success: true,
   734→      data: {
   735→        detections: results,
   736→        statistics: stats,
   737→        analyzedCount: Array.isArray(events) ? events.length : 1,
   738→        hasFraudAlerts: results.some(r => r.fraudScore >= 70),
   739→        timestamp: new Date().toISOString(),
   740→      },
   741→    });
   742→
   743→  } catch (error) {
   744→    return NextResponse.json(
   745→      { 
   746→        error: 'Fraud check failed',
   747→        message: (error as Error).message 
   748→      },
   749→      { status: 500 }
   750→    );
   751→  }

   847→/**
   848→ * Create a new model with specified configuration
   849→ */
   850→async function handleCreateModel(body: any): Promise<NextResponse> {
   851→  const { modelType, config, name, description } = body;
   852→
   853→  if (!modelType) {
   854→    return NextResponse.json(
   855→      { error: 'modelType is required' },
   856→      { status: 400 }
   857→    );
   858→  }
   859→
   860→  try {
   861→    let instance: SimplifiedNeuralNetwork | IsolationForestDetector | EnsembleDetector;
   862→    let actualConfig: any;
   863→    let type: string;
   864→
   865→    switch (modelType) {
   866→      case 'neural_network':
   867→      case 'nn':
   868→        instance = new SimplifiedNeuralNetwork(config || DEFAULT_THREAT_NN_CONFIG);
   869→        actualConfig = config || DEFAULT_THREAT_NN_CONFIG;
   870→        type = 'neural_network';
   871→        break;
   872→
   873→      case 'isolation_forest':
   874→      case 'if':
   875→        instance = new IsolationForestDetector(config || DEFAULT_STATISTICAL_CONFIG);
   876→        actualConfig = config || DEFAULT_STATISTICAL_CONFIG;
   877→        type = 'isolation_forest';
   878→        break;
   879→
   880→      case 'ensemble':
   881→        instance = new EnsembleDetector(config || DEFAULT_ENSEMBLE_CONFIG);
   882→        actualConfig = config || DEFAULT_ENSEMBLE_CONFIG;
   883→        type = 'ensemble';
   884→        break;
   885→
   886→      default:
   887→        return NextResponse.json(
   888→          { error: `Unsupported modelType: ${modelType}` },
   889→          { status: 400 }
   890→        );
   891→    }
   892→
   893→    const modelId = generateId();
   894→    const metadata: ModelMetadata = {
   895→      id: modelId,
   896→      name: name || `${type}-${modelId}`,
   897→      type: type as any,

   911→        inferenceTimeP95Ms: 0,
   912→        throughputPerSecond: 0,
   913→      },
   914→      trainingInfo: {
   915→        totalEpochs: 0,
   916→        bestEpoch: 0,
   917→        finalLoss: 0,
   918→        bestLoss: 0,
   919→        trainingSamples: 0,
   920→        validationSamples: 0,
   921→        convergenceEpoch: 0,
   922→        earlyStopped: false,
   923→        trainingHistory: [],
   924→      },
   925→      config: actualConfig,
   926→      createdAt: new Date(),
   927→      lastUpdated: new Date(),
   928→      description: description || `Created ${type} model`,
   929→      tags: ['custom'],
   930→    };
   931→
   932→    models.set(modelId, { instance, metadata });
   933→
   934→    return NextResponse.json({
   935→      success: true,

   941→        nextSteps: [
   942→          'Use POST /api/analytics/ml/train with this modelId to train the model',
   943→          'Or use POST /api/analytics/ml/predict with this modelId after training',
   944→        ],
   945→        timestamp: new Date().toISOString(),
   946→      },
   947→    });
   948→
   949→  } catch (error) {
   950→    return NextResponse.json(
   951→      { 
   952→        error: 'Model creation failed',
   953→        message: (error as Error).message 
   954→      },
   955→      { status: 500 }
   956→    );
   957→  }
