   362→
   363→/**
   364→ * Security Event Audit Logger
   365→ * 
   366→ * Provides comprehensive audit logging with:
   367→ * - Structured event capture
   368→ * - Integrity verification via hash chains
   369→ * - PII masking for privacy
   370→ * - Multiple output backends
   371→ * - SIEM forwarding
   372→ * 
   373→ * @example
   374→ * ```typescript
   375→ * const auditLogger = new AuditLogger(DEFAULT_AUDIT_CONFIG);
   376→ * 
   377→ * // Log authentication failure
   378→ * await auditLogger.log({
   379→ *   eventType: 'AUTH_FAILURE',
   380→ *   sourceIp: clientIp,
   381→ *   resource: '/api/auth/login',
   382→ *   action: 'LOGIN_ATTEMPT',
   383→ *   outcome: 'FAILURE',
   384→ *   message: 'Invalid credentials for user john.doe',
   385→ *   userId: 'user-123',
   386→ *   username: 'john.doe',
   387→ *   metadata: { reason: 'invalid_password' },
   388→ * });
   389→ * ```
   390→ */
   391→export class AuditLogger {
   392→  private config: AuditLoggerConfig;
   393→  private previousEventHash: string = '';
   394→  private eventBuffer: AuditEvent[] = [];
   395→  private flushTimer: ReturnType<typeof setInterval> | null = null;
   396→
   397→  constructor(config: Partial<AuditLoggerConfig> = {}) {
   398→    this.config = { ...DEFAULT_AUDIT_CONFIG, ...config };
   399→    
   400→    // Start auto-flush timer if SIEM forwarding enabled
   401→    if (this.config.siemForwarding?.enabled) {
   402→      this.flushTimer = setInterval(
   403→        () => this.flush(),
   404→        this.config.siemForwarding.flushIntervalMs
   405→      );
   406→    }
   407→  }
   408→
   409→  /**
   410→   * Main logging method - creates and persists an audit event
   411→   */
   412→  async log(eventInput: Omit<AuditEvent, keyof AuditEventBase>): Promise<AuditEvent> {
   413→    const event = await this.createEvent(eventInput);
   414→    
   415→    // Write to configured outputs
   416→    await this.writeEvent(event);
   417→    
   418→    // Buffer for batch forwarding
   419→    if (this.config.siemForwarding?.enabled) {
   420→      this.eventBuffer.push(event);
   421→      if (this.eventBuffer.length >= this.config.siemForwarding.batchSize) {
   422→        await this.flush();
   423→      }
   424→    }
   425→    
   426→    return event;
   427→  }

   525→    resource: string;
   526→    action: 'READ' | 'EXPORT' | 'QUERY' | 'DOWNLOAD';
   527→    dataType: string;
   528→    recordCount?: number;
   529→    dataClassification: DataClassification;
   530→    maskingApplied?: boolean;
   531→    metadata?: Record<string, unknown>;
   532→  }): Promise<AuditEvent> {
   533→    const eventType = params.action === 'EXPORT' ? 'DATA_EXPORT' :
   534→                      params.action === 'QUERY' ? 'DATA_QUERY' : 'DATA_ACCESS';
   535→    
   536→    return this.log({

   544→      outcome: 'SUCCESS',
   545→      message: `Data ${params.action.toLowerCase()} on ${params.dataType}${params.recordCount ? ` (${params.recordCount} records)` : ''} by ${params.username}`,
   546→      metadata: {
   547→        ...params.metadata,
   548→        dataType: params.dataType,
   549→        recordCount: params.recordCount,
   550→        maskingApplied: params.maskingApplied,
   551→      },
   552→      dataClassification: params.dataClassification,

   557→  /** Log administrative action */
   558→  async logAdminAction(params: {
   559→    userId: string;
   560→    username: string;
   561→    userRole: string;
   562→    sourceIp: string;
   563→    action: string;
   564→    targetResource: string;
   565→    targetType: 'user' | 'role' | 'config' | 'system';
   566→    outcome: 'SUCCESS' | 'FAILURE';
   567→    details: string;
   568→    metadata?: Record<string, unknown>;
   569→  }): Promise<AuditEvent> {
   570→    const eventTypeMap: Record<string, AuditEventType> = {
   571→      user: params.outcome === 'SUCCESS' ? 'ADMIN_USER_CREATE' : 'ADMIN_USER_UPDATE',
   572→      role: 'ADMIN_ROLE_ASSIGN',
   573→      config: 'ADMIN_CONFIG_CHANGE',
   574→      system: params.action.toUpperCase().includes('SHUTDOWN') ? 'ADMIN_SYSTEM_SHUTDOWN' : 'ADMIN_SYSTEM_STARTUP',
   575→    };
   576→
   577→    return this.log({
   578→      eventType: eventTypeMap[params.targetType] || 'ADMIN_CONFIG_CHANGE',
   579→      userId: params.userId,
   580→      username: params.username,
   581→      userRole: params.userRole,
   582→      sourceIp: params.sourceIp,
   583→      resource: params.targetResource,
   584→      action: params.action,
   585→      outcome: params.outcome,
   586→      message: `Admin action: ${params.details}`,
   587→      metadata: { ...params.metadata, targetType: params.targetType },
   588→      dataClassification: 'CONFIDENTIAL',
   589→      requiresReview: params.targetType === 'user' && params.action.includes('DELETE'),
   590→    });
   591→  }
   592→
   593→  /** Log security incident/violation */
   594→  async logSecurityEvent(params: {
   595→    eventType: 'SECURITY_POLICY_VIOLATION' | 'SECURITY_RATE_LIMIT_EXCEEDED' | 'SECURITY_WAF_BLOCK' | 'SECURITY_INTRUSION_ATTEMPT';
   596→    sourceIp: string;
   597→    resource: string;
   598→    severity?: AuditEventSeverity;
   599→    details: string;
   600→    metadata?: Record<string, unknown>;
   601→  }): Promise<AuditEvent> {
   602→    return this.log({
   603→      eventType: params.eventType,
   604→      sourceIp: params.sourceIp,
   605→      resource: params.resource,
   606→      action: 'SECURITY_EVENT',
   607→      outcome: 'BLOCKED',
   608→      message: `Security event: ${params.details}`,
   609→      metadata: params.metadata,
   610→      dataClassification: 'INTERNAL',
   611→      requiresReview: true,
   612→      severity: params.severity || EVENT_SEVERITY_MAP[params.eventType],
   613→    });
   614→  }
   615→
   616→  /**
   617→   * Flush buffered events to SIEM
   618→   */
   619→  async flush(): Promise<void> {
   620→    if (!this.config.siemForwarding?.enabled || this.eventBuffer.length === 0) {
   621→      return;
   622→    }
   623→
   624→    const eventsToSend = [...this.eventBuffer];
   625→    this.eventBuffer = [];
   626→
   627→    try {
   628→      const response = await fetch(this.config.siemForwarding.endpoint, {
   629→        method: 'POST',
   630→        headers: {
   631→          'Content-Type': 'application/json',
   632→          'X-Audit-Source': this.config.applicationId,
   633→          'X-Event-Count': eventsToSend.length.toString(),
   634→        },
   635→        body: JSON.stringify({
   636→          source: this.config.applicationId,
   637→          environment: this.config.environment,
   638→          timestamp: new Date().toISOString(),
   639→          eventCount: eventsToSend.length,
   640→          events: eventsToSend.map(this.sanitizeForSIEM),
   641→        }),
   642→      });
   643→
   644→      if (!response.ok) {
   645→        console.error(`Failed to send audit events to SIEM: ${response.status}`);
   646→        // Re-add failed events to buffer
   647→        this.eventBuffer.unshift(...eventsToSend);
   648→      }
   649→    } catch (error) {
   650→      console.error('Error flushing audit events to SIEM:', error);
   651→      this.eventBuffer.unshift(...eventsToSend);
   652→    }
   653→  }
   654→
   655→  /**
   656→   * Shutdown logger gracefully
   657→   */
   658→  async shutdown(): Promise<void> {
   659→    if (this.flushTimer) {
   660→      clearInterval(this.flushTimer);
   661→    }
   662→    await this.flush();
   663→  }

   669→  /**
   670→   * Create a complete audit event from input
   671→   */
   672→  private async createEvent(eventInput: Omit<AuditEvent, keyof AuditEventBase>): Promise<AuditEvent> {
   673→    const now = new Date();
   674→    const eventType = eventInput.eventType;
   675→    
   676→    // Determine retention period based on data classification
   677→    const classification = eventInput.dataClassification || 'INTERNAL';
   678→    const retentionDays = this.config.retentionPolicies[classification];
   679→
   680→    // Calculate integrity hash
   681→    const eventData = {
   682→      timestamp: now.toISOString(),
   683→      eventType,
   684→      ...eventInput,
   685→    };
   686→    
   687→    const integrityHash = this.calculateIntegrityHash(eventData);
   688→
   689→    const event: AuditEvent = {
   690→      eventId: randomUUID(),
   691→      timestamp: this.formatTimestamp(now),
   692→      eventType,
   693→      severity: eventInput.severity || EVENT_SEVERITY_MAP[eventType],
   694→      category: EVENT_CATEGORY_MAP[eventType],
   695→      sourceIp: this.maskPII('ip', eventInput.sourceIp),
   696→      riskScore: EVENT_RISK_SCORE_MAP[eventType],
   697→      retentionDays,
   698→      requiresReview: eventInput.requiresReview ?? (EVENT_RISK_SCORE_MAP[eventType] >= 50),
   699→      integrityHash,
   700→      previousEventHash: this.previousEventHash || undefined,
   701→      ...eventInput,
   702→    };
   703→
   704→    // Update hash chain
   705→    this.previousEventHash = integrityHash;
   706→
   707→    return event;
   708→  }

   782→  /**
   783→   * Calculate HMAC-SHA256 integrity hash for event
   784→   */
   785→  private calculateIntegrityHash(eventData: Record<string, unknown>): string {
   786→    const eventString = JSON.stringify({
   787→      ...eventData,
   788→      previousHash: this.previousEventHash,
   789→    });
   790→    
   791→    return createHmac('sha256', this.config.integrityKey)
   792→      .update(eventString)
   793→      .digest('hex');
   794→  }

   796→  /**
   797→   * Mask PII in event data
   798→   */
   799→  private maskPII(type: 'ip' | 'email' | 'imsi' | 'msisdn', value: string): string {
   800→    if (!this.config.piiMasking.enabled) {
   801→      return value;
   802→    }
   803→
   804→    switch (type) {
   805→      case 'ip':
   806→        if (this.config.piiMasking.maskIP) {
   807→          // Mask last octet
   808→          return value.replace(/\.\d+$/, '.***');
   809→        }
   810→        return value;
   811→      case 'email':
   812→        if (this.config.piiMasking.maskEmail) {
   813→          // Mask local part
   814→          const [local, domain] = value.split('@');
   815→          if (local && domain) {
   816→            return `${local[0]}***@${domain}`;
   817→          }
   818→        }
   819→        return value;
   820→      case 'imsi':
   821→      case 'msisdn':
   822→        if (this.config.piiMasking.maskIMSIMSISDN) {
   823→          return `***${value.slice(-4)}`;
   824→        }
   825→        return value;
   826→      default:
   827→        return value;
   828→    }
   829→  }
