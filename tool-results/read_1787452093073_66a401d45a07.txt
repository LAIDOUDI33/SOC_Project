   579→/**
   580→ * Generate complete cache control headers based on route configuration
   581→ */
   582→export function getCacheHeaders(route: string): Record<string, string> {
   583→  const routeConfig = RouteCacheConfig[route as keyof typeof RouteCacheConfig];
   584→  
   585→  if (!routeConfig) {
   586→    return { 'Cache-Control': CacheConfig.HEADERS.API_PRIVATE };
   587→  }
   588→  
   589→  // Explicitly set headers
   590→  if ('headers' in routeConfig && routeConfig.headers) {
   591→    const headers: Record<string, string> = { 'Cache-Control': routeConfig.headers };
   592→    
   593→    // Add CDN headers for cacheable routes
   594→    if (routeConfig.cache === 'ISR' || routeConfig.cache === 'EDGE') {
   595→      headers['CDN-Cache-Control'] = getCDNCacheControl(routeConfig);
   596→      headers['Vercel-CDN-Cache-Control'] = getCDNCacheControl(routeConfig);
   597→    }
   598→    
   599→    return headers;
   600→  }
   601→  
   602→  // ISR routes
   603→  if (routeConfig.cache === 'ISR' && 'revalidate' in routeConfig) {
   604→    const revalidate = (routeConfig as any).revalidate || CacheConfig.TTL.ISR_REVALIDATION.DASHBOARD;
   605→    const swr = 'swr' in routeConfig ? (routeConfig as any).swr : CacheConfig.TTL.SWR.DEFAULT;
   606→    
   607→    return {
   608→      'Cache-Control': `s-maxage=${revalidate}, stale-while-revalidate=${swr}`,
   609→      'CDN-Cache-Control': `public, s-maxage=${revalidate}, stale-while-revalidate=${swr}`,
   610→      'Vercel-CDN-Cache-Control': `public, s-maxage=${revalidate}, stale-while-revalidate=${swr}`,
   611→    };
   612→  }
   613→  
   614→  // EDGE cache routes
   615→  if (routeConfig.cache === 'EDGE' && 'ttl' in routeConfig) {
   616→    const ttl = (routeConfig as any).ttl || 60;
   617→    return {
   618→      'Cache-Control': `public, s-maxage=${ttl}, stale-while-revalidate=${ttl * 2}`,
   619→      'CDN-Cache-Control': `public, s-maxage=${ttl}, stale-while-revalidate=${ttl * 2}`,
   620→    };
   621→  }
   622→  
   623→  // Default: no caching
   624→  return { 'Cache-Control': CacheConfig.HEADERS.API_PRIVATE };
   625→}
   626→
   627→/**
   628→ * Get CDN-specific cache control header
   629→ */
   630→function getCDNCacheControl(routeConfig: any): string {
   631→  const ttl = routeConfig.revalidate || routeConfig.ttl || 60;
   632→  const swr = routeConfig.swr || CacheConfig.TTL.SWR.DEFAULT;
   633→  return `public, s-maxage=${ttl}, stale-while-revalidate=${swr}`;
   634→}

   672→/**
   673→ * Service Worker cache strategy constants
   674→ */
   675→export const SW_CACHE_STRATEGIES = {
   676→  CACHE_NAMES: {
   677→    STATIC: 'djezzy-soc-static-v2',
   678→    DYNAMIC: 'djezzy-soc-dynamic-v2',
   679→    API: 'djezzy-soc-api-v2',
   680→    IMAGES: 'djezzy-soc-images-v2',
   681→    FONTS: 'djezzy-soc-fonts-v2',
   682→  },
   683→  
   684→  STRATEGIES: {
   685→    // Cache first, network fallback for static assets
   686→    CACHE_FIRST: 'cache-first',
   687→    
   688→    // Network first, cache fallback for dynamic data
   689→    NETWORK_FIRST: 'network-first',
   690→    
   691→    // Stale while revalidate for near-real-time data
   692→    STALE_WHILE_REVALIDATE: 'swr',
   693→    
   694→    // Network only for sensitive operations
   695→    NETWORK_ONLY: 'network-only',
   696→    
   697→    // Cache only for offline-first content
   698→    CACHE_ONLY: 'cache-only',
   699→  },
   700→};
   701→
   702→/**
   703→ * Determine cache strategy for a request URL
   704→ * Used by Service Worker for client-side caching decisions
   705→ */
   706→export function getCacheStrategy(url: string): string {
   707→  // Static assets - cache first
   708→  if (url.match(/\.(js|css|woff2?|ttf|otf|eot)$/)) {
   709→    return SW_CACHE_STRATEGIES.STRATEGIES.CACHE_FIRST;
   710→  }
   711→  
   712→  // Images - cache first
   713→  if (url.match(/\.(png|jpg|jpeg|gif|svg|webp|avif|ico)$/)) {
   714→    return SW_CACHE_STRATEGIES.STRATEGIES.CACHE_FIRST;
   715→  }
   716→  
   717→  // Metrics and dashboard data - stale while revalidate
   718→  if (
   719→    url.startsWith('/api/metrics') ||
   720→    url.startsWith('/api/dashboard') ||
   721→    url.startsWith('/api/kpi')
   722→  ) {
   723→    return SW_CACHE_STRATEGIES.STRATEGIES.STALE_WHILE_REVALIDATE;
   724→  }
   725→  
   726→  // Alerts - short SWR for near-realtime
   727→  if (url.startsWith('/api/alerts')) {
   728→    return SW_CACHE_STRATEGIES.STRATEGIES.STALE_WHILE_REVALIDATE;
   729→  }
   730→  
   731→  // Authentication - never cache
   732→  if (url.startsWith('/api/auth') || url.includes('/login')) {
   733→    return SW_CACHE_STRATEGIES.STRATEGIES.NETWORK_ONLY;
   734→  }
   735→  
   736→  // Event ingestion - network only
   737→  if (url.includes('/events') && !url.includes('stream')) {
   738→    return SW_CACHE_STRATEGIES.STRATEGIES.NETWORK_ONLY;
   739→  }
   740→  
   741→  // Other API endpoints - network first with cache fallback
   742→  if (url.startsWith('/api/')) {
   743→    return SW_CACHE_STRATEGIES.STRATEGIES.NETWORK_FIRST;
   744→  }
   745→  
   746→  // Pages - network first
   747→  return SW_CACHE_STRATEGIES.STRATEGIES.NETWORK_FIRST;
   748→}
   749→
   750→/**
   751→ * Get appropriate cache name for a URL
   752→ */
   753→export function getCacheName(url: string): string {
   754→  if (url.match(/\.(js|css|woff2?|ttf|otf|eot)$/)) {
   755→    return SW_CACHE_STRATEGIES.CACHE_NAMES.STATIC;
   756→  }
   757→  if (url.match(/\.(png|jpg|jpeg|gif|svg|webp|avif|ico)$/)) {
   758→    return SW_CACHE_STRATEGIES.CACHE_NAMES.IMAGES;
   759→  }
   760→  if (url.match(/fonts/i)) {
   761→    return SW_CACHE_STRATEGIES.CACHE_NAMES.FONTS;
   762→  }
   763→  if (url.startsWith('/api/')) {
   764→    return SW_CACHE_STRATEGIES.CACHE_NAMES.API;
   765→  }
   766→  return SW_CACHE_STRATEGIES.CACHE_NAMES.DYNAMIC;
   767→}
