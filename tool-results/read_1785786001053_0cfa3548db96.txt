    45→// Only renders time on client-side after mount
    46→// ============================================================
    47→function ClockDisplay() {
    48→  const [time, setTime] = useState<string>('')
    49→  const [mounted, setMounted] = useState(false)
    50→
    51→  useEffect(() => {
    52→    setMounted(true)
    53→    setTime(new Date().toLocaleTimeString())
    54→    
    55→    const timer = setInterval(() => {
    56→      setTime(new Date().toLocaleTimeString())
    57→    }, 1000)
    58→
    59→    return () => clearInterval(timer)
    60→  }, [])
    61→
    62→  if (!mounted) {
    63→    return <div className="text-cyan-400 font-mono w-20">&nbsp;</div>
    64→  }
    65→
    66→  return <div className="text-cyan-400 font-mono">{time}</div>
    67→}
