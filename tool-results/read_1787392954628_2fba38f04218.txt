   350→function FraudDetectionPanel({ alerts }: { alerts: FraudAlert[] }) {
   351→  const getFraudTypeIcon = (type: FraudAlert['type']) => {
   352→    switch (type) {
   353→      case 'irsf': return <PhoneCall className="h-4 w-4" />
   354→      case 'sim_swap': return <UserX className="h-4 w-4" />
   355→      case 'premium_rate': return <CreditCard className="h-4 w-4" />
   356→      case 'bypass_fraud': return <Router className="h-4 w-4" />
   357→      case 'wangiri': return <Phone className="h-4 w-4" />
   358→    }
   359→  }
   360→
   361→  const getSeverityColor = (severity: FraudAlert['severity']) => {
   362→    switch (severity) {
   363→      case 'critical': return 'border-red-500/50 bg-red-500/10'
   364→      case 'high': return 'border-orange-500/50 bg-orange-500/10'
   365→      case 'medium': return 'border-yellow-500/50 bg-yellow-500/10'
   366→      case 'low': return 'border-blue-500/50 bg-blue-500/10'
   367→    }
   368→  }
   369→
   370→  return (
   371→    <Card className="bg-slate-900 border-slate-700">
   372→      <CardHeader className="pb-3">
   373→        <div className="flex items-center justify-between">
   374→          <CardTitle className="text-white text-base flex items-center gap-2">
   375→            <AlertTriangle className="h-4 w-4 text-orange-400" />
   376→            Fraud Detection Alerts
   377→          </CardTitle>
   378→          <Badge variant="secondary" className="bg-orange-500/20 text-orange-400">
   379→            {alerts.filter(a => a.status !== 'false_positive').length} Active
   380→          </Badge>
   381→        </div>
   382→      </CardHeader>
   383→      <CardContent>
   384→        <div className="space-y-3">
   385→          {alerts.map(alert => (
   386→            <div key={alert.id} className={`p-3 rounded-lg border ${getSeverityColor(alert.severity)}`}>
   387→              <div className="flex items-start justify-between mb-2">

   395→                    alert.severity === 'high' ? 'border-orange-500/50 text-orange-400' :
