   240→
   241→  private async apiRequest<T>(
   242→    path: string,
   243→    options: RequestInit = {}
   244→  ): Promise<T> {
   245→    const url = `${this.config.apiUrl}${path}`;
   246→    
   247→    try {
   248→      const response = await fetch(url, {
   249→        ...options,
   250→        headers: {
   251→          'Authorization': `Bearer ${this.config.apiKey}`,
   252→          'Content-Type': 'application/json',
   253→          ...(this.config.organisation ? { 'X-Organisation': this.config.organisation } : {}),
   254→          ...(options.headers as Record<string, string>),
   255→        },
   256→        signal: AbortSignal.timeout(this.config.timeout!),
   257→      });
   258→
   259→      if (!response.ok) {
   260→        const errorBody = await response.json().catch(() => ({}));
   261→        throw new TheHiveApiError(
   262→          `TheHive API error (${response.status}): ${errorBody.message || response.statusText}`,
   263→          response.status,
   264→          undefined,
   265→          errorBody
   266→        );
   267→      }
   268→
   269→      return response.json() as Promise<T>;
   270→    } catch (error) {
   271→      if (error instanceof TheHiveApiError || error instanceof SoarIntegrationError) throw error;
   272→      throw new SoarIntegrationError(
   273→        `TheHive request failed: ${error instanceof Error ? error.message : String(error)}`,
   274→        'REQUEST_ERROR',
   275→        error as Error
   276→      );
   277→    }
   278→  }

   636→  /**
   637→   * Run multiple analyzers on an observable (common use case)
   638→   */
   639→  async runFullAnalysis(observable: {
   640→    dataType: string;
   641→    data: string;
   642→  }, category?: 'ip' | 'domain' | 'hash' | 'url' | 'mail' | 'telco'): Promise<CortexJob[]> {
   643→    // Select appropriate analyzers based on data type and category
   644→    let analyzerIds: string[];
   645→
   646→    switch (category || observable.dataType) {
   647→      case 'ip':
   648→        analyzerIds = [
   649→          'AbuseIPDB_2_0',
   650→          'VirusTotal_GetIPReport_3_1',
   651→          'Shodan_Search',
   652→          'Censys_Search',
   653→          'IPInfo_1_0',
   654→          'GreyNoise_Community_3_1',
   655→          'ThreatFox_1_0',
   656→        ];
   657→        break;
   658→      
   659→      case 'domain':
   660→        analyzerIds = [
   661→          'VirusTotal_GetDomainReport_3_1',
   662→          'DNSdb_Tools_3_1',
   663→          'SecurityTrails_2_1',
   664→          'Urlhaus_Query_1_0',
   665→          'PhishTank_Submit_2_1',
   666→          'AlienVault_OTXQuery_2_1',
   667→        ];
   668→        break;
   669→      
   670→      case 'hash':
   671→        analyzerIds = [
   672→          'VirusTotal_GetReport_3_1',
   673→          'HybridAnalysis_GetReport_2_1',
   674→          'MalwareBazaar_GetFile_2_1',
   675→          'XForce_Malware_2_0',
   676→          'MISP_Search_2_1',
   677→          'Unpac_ME_EXE_Unpacker_1_0',
   678→        ];
   679→        break;
   680→      
   681→      case 'url':
   682→        analyzerIds = [
   683→          'VirusTotal_ScanURL_3_1',
   684→          'UrlScan_IO_1_0',
   685→          'Phishtank_2_0',
   686→          'URLhaus_Query_1_0',
   687→          'GoogleSafeBrowsing_1_0',
   688→        ];
   689→        break;
   690→      
   691→      case 'mail':
   692→        analyzerIds = [
   693→          'HaveIBeenPwned_Breaches_2_1',
   694→          'Hunter_io_2_0',
   695→          'EmailRep_2_0',
   696→          'MISP_Search_2_1',
   697→        ];
   698→        break;
   699→
   700→      case 'telco':
   701→        // Djezzy-specific telco analyzers
   702→        analyzerIds = [
   703→          'Djezzy_Subscriber_Lookup_1_0',
   704→          'Djezzy_Fraud_Check_1_0',
   705→          'SIMSwap_Database_1_0',
   706→          'Roaming_Analysis_1_0',
   707→        ];
   708→        break;
   709→      
   710→      default:
   711→        analyzerIds = ['VirusTotal_GetReport_3_1'];
   712→    }
   713→
   714→    // Run each analyzer (or could batch them)
   715→    const jobPromises = analyzerIds.map(analyzerId =>
   716→      this.runAnalysis(observable, [analyzerId]).catch(error => {
   717→        console.error(`[Cortex] Analyzer ${analyzerId} failed:`, error);
   718→        return null;
   719→      })
   720→    );
   721→
   722→    const jobs = await Promise.all(jobPromises);
   723→    return jobs.filter((job): job is CortexJob => job !== null);
   724→  }

   726→  /**
   727→   * Get job status and results
   728→   */
   729→  async getJob(jobId: string): Promise<CortexJob> {
   730→    return this.apiRequest<CortexJob>(`/api/job/${jobId}`);
   731→  }

   733→  /**
   734→   * Wait for job completion (polling)
   735→   */
   736→  async waitForJob(
   737→    jobId: string,
   738→    options?: {
   739→      pollIntervalMs?: number;
   740→      maxWaitMs?: number;
   741→    }
   742→  ): Promise<CortexJob> {
   743→    const pollInterval = options?.pollIntervalMs || 2000;
   744→    const maxWait = options?.maxWaitMs || 120000; // Default 2 minutes
   745→    const startTime = Date.now();
   746→
   747→    while (Date.now() - startTime < maxWait) {
   748→      const job = await this.getJob(jobId);
   749→
   750→      if (job.status === 'Success' || job.status === 'Failure' || job.status === 'Timeout') {
   751→        return job;
   752→      }
   753→
   754→      // Wait before polling again
   755→      await new Promise(resolve => setTimeout(resolve, pollInterval));
   756→    }
   757→
   758→    throw new SoarIntegrationError('Job wait timeout exceeded', 'TIMEOUT');
   759→  }

   761→  /**
   762→   * Analyze observable and wait for all results
   763→   */
   764→  async analyzeAndWait(
   765→    observable: {
   766→      dataType: string;
   767→      data: string;
   768→    },
   769→    category?: string
   770→  ): Promise<CortexJob[]> {
   771→    // Start all analyses
   772→    const jobs = await this.runFullAnalysis(observable, category);
   773→
   774→    // Wait for all to complete
   775→    const completedJobs = await Promise.allSettled(
   776→      jobs.map(job => this.waitForJob(job.jobId))
   777→    );
   778→
   779→    return completedJobs
   780→      .filter((r): r is PromiseFulfilledResult<CortexJob> => r.status === 'fulfilled')
   781→      .map(r => r.value);
   782→  }
   783→
   784→  /**
   785→   * Get report for a completed job
   786→   */
   787→  async getReport(jobId: string, format: 'full' | 'summary' = 'summary'): Promise<CortexReport> {
   788→    const job = await this.getJob(jobId);
   789→    
   790→    if (!job.report) {
   791→      throw new SoarIntegrationError('No report available for this job', 'NO_REPORT');
   792→    }
   793→
   794→    return format === 'full' ? job.report.fullReport : job.report;
   795→  }

   797→  /**
   798→   * Search past jobs
   799→   */
   800→  async searchJobs(filters?: {
   801→    dataType?: string;
   802→    data?: string;
   803→    status?: string;
   804→    analyzerId?: string;
   805→    startDate?: string;
   806→    endDate?: string;
   807→    range?: string;
   808→    count?: number;
   809→    from?: number;
   810→  }): Promise<{ jobs: CortexJob[]; total: number }> {
   811→    const params = new URLSearchParams();
   812→    if (filters?.dataType) params.append('dataType', filters.dataType);
   813→    if (filters?.data) params.append('data', filters.data);
   814→    if (filters?.status) params.append('status', filters.status);
   815→    if (filters?.analyzerId) params.append('analyzerId', filters.analyzerId);
   816→    if (filters?.startDate) params.append('startDate', filters.startDate);
   817→    if (filters?.endDate) params.append('endDate', filters.endDate);
   818→    if (filters?.range) params.append('range', filters.range);
   819→    if (filters?.count) params.append('count', String(filters.count));
   820→    if (filters?.from) params.append('from', String(filters.from));
   821→
   822→    return this.apiRequest(`/api/job/search?${params}`);
   823→  }

   825→  /**
   826→   * Test connectivity to Cortex
   827→   */
   828→  async healthCheck(): Promise<{
   829→    status: 'healthy' | 'unhealthy';
   830→    version: string;
   831→    analyzerCount: number;
   832→    latency: number;
   833→  }> {
   834→    const startTime = Date.now();
   835→
   836→    try {
   837→      const analyzers = await this.getAnalyzers();
   838→      const latency = Date.now() - startTime;
   839→
   840→      return {
   841→        status: 'healthy',
   842→        version: 'connected',
   843→        analyzerCount: analyzers.length,
   844→        latency,
   845→      };
   846→    } catch (error) {
   847→      return {
   848→        status: 'unhealthy',
   849→        version: 'unknown',
   850→        analyzerCount: 0,
   851→        latency: Date.now() - startTime,
   852→      };
   853→    }
   854→  }

   910→  /**
   911→   * Create incident case from security alert
   912→   */
   913→  async createIncidentCase(alert: {
   914→    title: string;
   915→    description: string;
   916→    severity: 'critical' | 'high' | 'medium' | 'low';
   917→    eventType: string;
   918→    sourceIp?: string;
   919→    destinationIp?: string;
   920→    subscriberId?: string;
   921→    rawLog?: string;
   922→  }): Promise<{
   923→    case: HiveCase;
   924→    initialTask: HiveTask;
   925→    observablesAdded: HiveObservable[];
   926→  }> {
   927→    if (!this.isRunning) {
   928→      throw new SoarIntegrationError('SOAR service is not running', 'SERVICE_NOT_RUNNING');
   929→    }
   930→
   931→    // Map severity
   932→    const severityMap: Record<string, 1 | 2 | 3 | 4> = {
   933→      critical: 1,
   934→      high: 2,
   935→      medium: 3,
   936→      low: 4,
   937→    };
   938→
   939→    // Create the case
   940→    const hiveCase = await this.thehiveClient.createCase({
   941→      title: alert.title,
   942→      description: alert.description,
   943→      severity: severityMap[alert.severity] || 3,
   944→      tags: ['soc-platform', alert.eventType, 'auto-created'],
   945→      tlp: 2, // Amber by default for security incidents
   946→      pap: 2,
   947→      taskTitle: 'Initial Triage and Investigation',
   948→    });
   949→
   950→    // Add observables based on available data
   951→    const observables: Array<{ dataType: string; data: string; message?: string; tags?: string[] }> = [];
   952→
   953→    if (alert.sourceIp) {
   954→      observables.push({
   955→        dataType: 'ip',
   956→        data: alert.sourceIp,
   957→        message: 'Source IP address from alert',
   958→        tags: ['source-ip'],
   959→      });
   960→    }
   961→
   962→    if (alert.destinationIp) {
   963→      observables.push({
   964→        dataType: 'ip',
   965→        data: alert.destinationIp,
   966→        message: 'Destination IP address from alert',
   967→        tags: ['destination-ip'],
   968→      });
   969→    }
   970→
   971→    if (alert.subscriberId) {
   972→      observables.push({
   973→        dataType: 'other',
   974→        data: alert.subscriberId,
   975→        message: 'Subscriber MSISDN (masked)',
   976→        tags: ['subscriber', 'telco'],
   977→      });
   978→    }
   979→
   980→    // Add observables to case
   981→    const addedObservables = observables.length > 0
   982→      ? await this.thehiveClient.addObservablesBatch(hiveCase.id, observables)
   983→      : [];
   984→
   985→    // Get the auto-created initial task
   986→    const tasks = await this.thehiveClient.getTasks(hiveCase.id);
   987→    const initialTask = tasks[0];
   988→
   989→    this.emit('incident_created', { case: hiveCase, alert });
   990→
   991→    return {
   992→      case: hiveCase,
   993→      initialTask,
   994→      observablesAdded: addedObservables,
   995→    };
   996→  }

   998→  /**
   999→   * Enrich observables with automatic Cortex analysis
  1000→   */
  1001→  async enrichObservables(
  1002→    caseId: string,
  1003→    observables: Array<{ dataType: string; data: string }>
  1004→  ): Promise<Array<{
  1005→    observable: { dataType: string; data: string };
  1006→    analyses: CortexJob[];
  1007→  }>> {
  1008→    if (!this.isRunning) {
  1009→      throw new SoarIntegrationError('SOAR service is not running', 'SERVICE_NOT_RUNNING');
  1010→    }
  1011→
  1012→    const results = [];
  1013→
  1014→    for (const observable of observables) {
  1015→      try {
  1016→        console.log(`[SOAR] Enriching ${observable.dataType}: ${observable.data}`);
  1017→
  1018→        // Run full analysis and wait for results
  1019→        const analyses = await this.cortexClient.analyzeAndWait(observable);
  1020→
  1021→        // Update observable in TheHive with analysis results
  1022→        // (In production, you'd call thehiveClient.updateObservable)
  1023→
  1024→        results.push({
  1025→          observable,
  1026→          analyses,
  1027→        });
  1028→
  1029→        this.emit('observable_enriched', { observable, analyses });
  1030→      } catch (error) {
  1031→        console.error(`[SOAR] Failed to enrich ${observable.dataType}:${observable.data}:`, error);
  1032→        
  1033→        results.push({
  1034→          observable,
  1035→          analyses: [],
  1036→        });
  1037→
  1038→        this.emit('enrichment_error', { observable, error });
  1039→      }
  1040→    }
  1041→
  1042→    return results;
  1043→  }

  1072→          await this.executeAction('Submit URLs to URLhaus', () => /* submit */ ({})),

  1078→      case 'malware_investigation':

  1137→  // Private helper for executing actions with error handling
  1138→  private async executeAction<T>(
  1139→    actionName: string,
  1140→    fn: () => Promise<T>
  1141→  ): Promise<{
  1142→    action: string;
  1143→    status: 'success' | 'failed' | 'skipped';
  1144→    result?: T;
  1145→    error?: string;
  1146→  }> {
  1147→    try {
  1148→      const result = await fn();
  1149→      return { action: actionName, status: 'success', result };
  1150→    } catch (error) {
  1151→      return {
  1152→        action: actionName,
  1153→        status: 'failed',
  1154→        error: error instanceof Error ? error.message : String(error),
  1155→      };
  1156→    }
  1157→  }
