    71→/** Status badge component */
    72→const StatusBadge: React.FC<{ status: string; variant?: 'default' | 'dot' }> = ({ status, variant = 'default' }) => {
    73→  const getStatusColor = (s: string) => {
    74→    switch (s.toLowerCase()) {
    75→      case 'ok':
    76→      case 'healthy':
    77→      case 'good':
    78→        return 'bg-green-100 text-green-800';
    79→      case 'alerting':
    80→      case 'critical':
    81→      case 'error':
    82→        return 'bg-red-100 text-red-800';
    83→      case 'warning':
    84→      case 'degraded':
    85→        return 'bg-yellow-100 text-yellow-800';
    86→      case 'paused':
    87→      case 'unknown':
    88→        return 'bg-gray-100 text-gray-800';
    89→      default:
    90→        return 'bg-blue-100 text-blue-800';
    91→    }
    92→  };
    93→
    94→  const getDotColor = (s: string) => {
    95→    switch (s.toLowerCase()) {
    96→      case 'ok':
    97→      case 'healthy':
    98→      case 'good':
    99→        return 'bg-green-500';
   100→      case 'alerting':
   101→      case 'critical':
   102→      case 'error':
   103→        return 'bg-red-500';
   104→      case 'warning':
   105→      case 'degraded':
   106→        return 'bg-yellow-500';
   107→      case 'paused':
   108→      case 'unknown':
   109→        return 'bg-gray-400';
   110→      default:
   111→        return 'bg-blue-500';
   112→    }
   113→  };
   114→
   115→  if (variant === 'dot') {
   116→    return (
   117→      <span className={`inline-flex items-center gap-1.5`}>
   118→        <span className={`w-2 h-2 rounded-full ${getDotColor(status)}`}></span>
   119→        <span className={`px-2 py-0.5 text-xs font-medium rounded-full ${getStatusColor(status)}`}>
   120→          {status}
   121→        </span>
   122→      </span>
   123→    );
   124→  }
   125→
   126→  return (
   127→    <span className={`inline-flex px-2 py-0.5 text-xs font-medium rounded-full ${getStatusColor(status)}`}>
   128→      {status}
   129→    </span>
   130→  );
   131→};

   150→/** KPI Card Component */
   151→const KPICard: React.FC<{
   152→  metric: SOCKpiMetric;
   153→  onClick?: () => void;
   154→}> = ({ metric, onClick }) => {
   155→  const getTrendIcon = () => {
   156→    if (metric.trend === 'up') {
   157→      return metric.status === 'good' ? (
   158→        <svg className="w-4 h-4 text-green-500" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   159→          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M5 10l7-7m0 0l7 7m-7-7v18" />
   160→        </svg>
   161→      ) : (
   162→        <svg className="w-4 h-4 text-red-500" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   163→          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M5 10l7-7m0 0l7 7m-7-7v18" />
   164→        </svg>
   165→      );
   166→    }
   167→    if (metric.trend === 'down') {
   168→      return metric.status === 'good' ? (
   169→        <svg className="w-4 h-4 text-green-500" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   170→          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 14l-7 7m0 0l-7-7m7 7V3" />
   171→        </svg>
   172→      ) : (
   173→        <svg className="w-4 h-4 text-red-500" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   174→          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 14l-7 7m0 0l-7-7m7 7V3" />
   175→        </svg>
   176→      );
   177→    }
   178→    return (
   179→      <svg className="w-4 h-4 text-gray-400" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   180→        <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M20 12H4" />
   181→      </svg>
   182→    );
   183→  };
   184→
   185→  const getCardBorder = () => {
   186→    switch (metric.status) {
   187→      case 'critical': return 'border-l-4 border-l-red-500';
   188→      case 'warning': return 'border-l-4 border-l-yellow-500';
   189→      case 'good': return 'border-l-4 border-l-green-500';
   190→      default: return 'border-l-4 border-l-blue-500';
   191→    }
   192→  };
   193→
   194→  return (
   195→    <div
   196→      onClick={onClick}
   197→      className={`bg-white rounded-lg shadow p-4 hover:shadow-md transition-shadow cursor-pointer ${getCardBorder()}`}
   198→    >
   199→      <div className="flex items-start justify-between mb-2">
   200→        <p className="text-sm font-medium text-gray-600">{metric.name}</p>
   201→        {getTrendIcon()}
   202→      </div>
   203→      
   204→      <div className="flex items-baseline gap-2">
   205→        <span className="text-2xl font-bold text-gray-900">{metric.currentValue.toLocaleString()}</span>
   206→        <span className="text-sm text-gray-500">{metric.unit}</span>
   207→      </div>
   208→
   209→      {metric.trendPercent !== undefined && (
   210→        <p className={`text-xs mt-1 ${
   211→          (metric.trend === 'up' && metric.status !== 'good') || (metric.trend === 'down' && metric.status === 'good')
   212→            ? 'text-green-600'
   213→            : 'text-red-600'
   214→        }`}>
   215→          {metric.trend === 'stable' ? 'No change' : `${metric.trend > 0 ? '+' : ''}${metric.trendPercent}%`}
   216→        </p>
   217→      )}
   218→
   219→      <p className="text-xs text-gray-400 mt-2">Source: {metric.sourceSystem}</p>
   220→    </div>
   221→  );
   222→};

   357→/** Dashboards Tab Content */
   358→const DashboardsTabContent: React.FC<{
   359→  dashboards: DashboardSearchResult[] | null;
   360→  loading: boolean;
   361→  error: string | null;
   362→  total: number;
   363→  onOpenDashboard?: (uid: string) => void;
   364→}> = ({ dashboards, loading, error, total, onOpenDashboard }) => {
   365→  const [searchQuery, setSearchQuery] = useState('');
   366→  const [viewMode, setViewMode] = useState<'grid' | 'list'>('grid');
   367→  const [selectedTag, setSelectedTag] = useState<string | null>(null);
   368→
   369→  // Get all unique tags
   370→  const allTags = useMemo(() => {
   371→    if (!dashboards) return [];
   372→    const tags = new Set<string>();
   373→    dashboards.forEach(d => d.tags.forEach(t => tags.add(t)));
   374→    return Array.from(tags).sort();
   375→  }, [dashboards]);
   376→
   377→  // Filter dashboards
   378→  const filteredDashboards = useMemo(() => {
   379→    if (!dashboards) return [];
   380→    
   381→    return dashboards.filter(d => {
   382→      const matchesSearch = !searchQuery ||
   383→        d.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
   384→        d.tags.some(t => t.toLowerCase().includes(searchQuery.toLowerCase()));
   385→      
   386→      const matchesTag = !selectedTag || d.tags.includes(selectedTag);
   387→      
   388→      return matchesSearch && matchesTag;
   389→    });
   390→  }, [dashboards, searchQuery, selectedTag]);
   391→
   392→  if (loading) return <LoadingSpinner message="Loading dashboards..." />;
   393→  if (error) return <ErrorState message={error} />;
   394→
   395→  return (
   396→    <div className="space-y-4">
   397→      {/* Controls */}
   398→      <div className="flex flex-col sm:flex-row gap-4 items-start sm:items-center justify-between">
   399→        <div className="relative flex-1 max-w-md">
   400→          <input
   401→            type="text"
   402→            placeholder="Search dashboards..."
   403→            value={searchQuery}
   404→            onChange={(e) => setSearchQuery(e.target.value)}
   405→            className="w-full pl-10 pr-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
   406→          />
   407→          <svg className="absolute left-3 top-2.5 w-5 h-5 text-gray-400" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   408→            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z" />
   409→          </svg>
   410→        </div>
   411→
   412→        <div className="flex items-center gap-2">
   413→          {/* View Mode Toggle */}
   414→          <div className="flex bg-gray-100 rounded-lg p-1">
   415→            <button
   416→              onClick={() => setViewMode('grid')}
   417→              className={`p-2 rounded ${viewMode === 'grid' ? 'bg-white shadow' : ''}`}
   418→              title="Grid view"
   419→            >
   420→              <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   421→                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 6a2 2 0 012-2h2a2 2 0 012 2v2a2 2 0 01-2 2H6a2 2 0 01-2-2V6zM14 6a2 2 0 012-2h2a2 2 0 012 2v2a2 2 0 01-2 2h-2a2 2 0 01-2-2V6zM4 16a2 2 0 012-2h2a2 2 0 012 2v2a2 2 0 01-2 2H6a2 2 0 01-2-2v-2zM14 16a2 2 0 012-2h2a2 2 0 012 2v2a2 2 0 01-2 2h-2a2 2 0 01-2-2v-2z" />
   422→              </svg>
   423→            </button>
   424→            <button
   425→              onClick={() => setViewMode('list')}
   426→              className={`p-2 rounded ${viewMode === 'list' ? 'bg-white shadow' : ''}`}
   427→              title="List view"
   428→            >
   429→              <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   430→                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 6h16M4 10h16M4 14h16M4 18h16" />
   431→              </svg>
   432→            </button>
   433→          </div>
   434→
   435→          <span className="text-sm text-gray-500">{filteredDashboards.length} of {total} dashboards</span>
   436→        </div>
   437→      </div>
   438→
   439→      {/* Tag Filters */}
   440→      {allTags.length > 0 && (
   441→        <div className="flex flex-wrap gap-2">
   442→          <button
   443→            onClick={() => setSelectedTag(null)}
   444→            className={`px-3 py-1 text-sm rounded-full transition-colors ${
   445→              !selectedTag ? 'bg-blue-600 text-white' : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
   446→            }`}
   447→          >
   448→            All Tags
   449→          </button>
   450→          {allTags.map(tag => (
   451→            <button
   452→              key={tag}
   453→              onClick={() => setSelectedTag(selectedTag === tag ? null : tag)}
   454→              className={`px-3 py-1 text-sm rounded-full transition-colors ${
   455→                selectedTag === tag ? 'bg-blue-600 text-white' : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
   456→              }`}
   457→            >
   458→              {tag}
   459→            </button>
   460→          ))}
   461→        </div>
   462→      )}
   463→
   464→      {/* Dashboard Grid/List */}
   465→      {filteredDashboards.length === 0 ? (
   466→        <div className="text-center py-12 text-gray-500">
   467→          <svg className="w-16 h-16 mx-auto mb-4 text-gray-300" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   468→            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={1.5} d="M9 17v-2m3 2v-4m3 4v-6m2 10H7a2 2 0 01-2-2V5a2 2 0 012-2h5.586a1 1 0 01.707.293l5.414 5.414a1 1 0 01.293.707V19a2 2 0 01-2 2z" />
   469→          </svg>
   470→          <p>No dashboards found matching your criteria</p>
   471→        </div>
   472→      ) : viewMode === 'grid' ? (
   473→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   474→          {filteredDashboards.map(dashboard => (
   475→            <div
   476→              key={dashboard.uid}
   477→              onClick={() => onOpenDashboard?.(dashboard.uid)}
   478→              className="bg-white rounded-lg shadow hover:shadow-md transition-all cursor-pointer p-4 group"
