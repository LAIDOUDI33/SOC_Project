   275→enum LogLevel {
   276→  DEBUG = 'DEBUG',
   277→  INFO = 'INFO',
   278→  WARN = 'WARN',
   279→  ERROR = 'ERROR',
   280→}
   281→
   282→function log(level: LogLevel, message: string, ...args: unknown[]): void {
   283→  const timestamp = new Date().toISOString();
   284→  const prefix = `[RateLimit][${timestamp}][${level}]`;
   285→  
   286→  switch (level) {
   287→    case LogLevel.ERROR:
   288→      console.error(prefix, message, ...args);
   289→      break;
   290→    case LogLevel.WARN:
   291→      console.warn(prefix, message, ...args);
   292→      break;
   293→    case LogLevel.DEBUG:
   294→      if (DEFAULT_MIDDLEWARE_CONFIG.verboseLogging) {
   295→        console.debug(prefix, message, ...args);
   296→      }
   297→      break;
   298→    default:
   299→      console.log(prefix, message, ...args);
   300→  }
   301→}

   304→// In-Memory Fallback Store (for Redis unavailability)

   307→/**
   308→ * In-memory rate limit store for fallback when Redis is unavailable.
   309→ * Uses a simple sliding window implementation.
   310→ */
   311→class InMemoryRateLimitStore {
   312→  private static instance: InMemoryRateLimitStore | null = null;
   313→  private store: Map<string, { count: number; windowStart: number; resetTime: number }> = new Map();
   314→  private connectionStore: Map<string, Set<string>> = new Map(); // Track concurrent connections
   315→
   316→  private constructor() {}
   317→
   318→  public static getInstance(): InMemoryRateLimitStore {
   319→    if (!InMemoryRateLimitStore.instance) {
   320→      InMemoryRateLimitStore.instance = new InMemoryRateLimitStore();
   321→    }
   322→    return InMemoryRateLimitStore.instance;
   323→  }
   324→
   325→  /**
   326→   * Check and increment rate limit using in-memory sliding window
   327→   */
   328→  public check(
   329→    key: string,
   330→    options: RateLimitOptions
   331→  ): RateLimitResult {
   332→    const now = Date.now();
   333→    const windowMs = options.duration * 1000;
   334→    const windowStart = now - windowMs;
   335→
   336→    // Get or create entry
   337→    let entry = this.store.get(key);
   338→    
   339→    // Clean up expired entries
   340→    if (entry && entry.windowStart < windowStart) {
   341→      entry = undefined;
   342→      this.store.delete(key);
   343→    }
   344→
   345→    if (!entry) {
   346→      entry = {
   347→        count: 0,
   348→        windowStart: now,
   349→        resetTime: now + windowMs,
   350→      };
   351→      this.store.set(key, entry);
   352→    }
   353→
   354→    // Check if allowed
   355→    const allowed = entry.count < options.points;
   356→
   357→    if (allowed) {
   358→      entry.count++;
   359→    }
   360→
   361→    return {
   362→      allowed,
   363→      remaining: Math.max(0, options.points - entry.count),
   364→      resetTime: entry.resetTime,
   365→      limit: options.points,
   366→      current: entry.count,
   367→      retryAfterMs: allowed ? undefined : entry.resetTime - now,
   368→      reason: allowed ? undefined : 'In-memory rate limit exceeded (Redis unavailable)',
   369→    };
   370→  }
   371→
   372→  /**
   373→   * Add a concurrent connection
   374→   */
   375→  public addConnection(userId: string, connectionId: string): boolean {
   376→    const connections = this.connectionStore.get(userId) || new Set();
   377→    
   378→    if (connections.size >= 10) {
   379→      return false; // Max connections reached
   380→    }
   381→    
   382→    connections.add(connectionId);
   383→    this.connectionStore.set(userId, connections);
   384→    return true;
   385→  }
   386→
   387→  /**
   388→   * Remove a concurrent connection
   389→   */
   390→  public removeConnection(userId: string, connectionId: string): void {
   391→    const connections = this.connectionStore.get(userId);
   392→    if (connections) {
   393→      connections.delete(connectionId);
   394→      if (connections.size === 0) {
   395→        this.connectionStore.delete(userId);
   396→      }
   397→    }
   398→  }
   399→
   400→  /**
   401→   * Get current connection count for user
   402→   */
   403→  public getConnectionCount(userId: string): number {
   404→    return this.connectionStore.get(userId)?.size || 0;
   405→  }
   406→
   407→  /**
   408→   * Clean up old entries (call periodically)
   409→   */
   410→  public cleanup(): void {
   411→    const now = Date.now();
   412→    for (const [key, entry] of this.store.entries()) {
   413→      if (entry.resetTime < now) {
   414→        this.store.delete(key);
   415→      }
   416→    }
   417→  }
   418→}

   421→// Singleton RateLimiter Instance

   423→
   424→/** Global RateLimiter instance (lazy initialized) */
   425→let rateLimiterInstance: RateLimiter | null = null;
   426→let inMemoryStore: InMemoryRateLimitStore | null = null;
   427→let isInitialized = false;
   428→let initializationPromise: Promise<void> | null = null;
   429→
   430→/**
   431→ * Initialize the singleton RateLimiter instance.
   432→ * Call once at application startup.
   433→ */
   434→async function initializeRateLimiter(
   435→  config?: Partial<RateLimitMiddlewareConfig>
   436→): Promise<void> {
   437→  if (isInitialized && rateLimiterInstance) {
   438→    return;
   439→  }
   440→
   441→  if (initializationPromise) {
   442→    return initializationPromise;
   443→  }
   444→
   445→  initializationPromise = doInitialize(config);
   446→  await initializationPromise;
   447→}
   448→
   449→async function doInitialize(
   450→  config?: Partial<RateLimitMiddlewareConfig>
   451→): Promise<void> {
   452→  try {
   453→    log(LogLevel.INFO, 'Initializing production rate limiter...');
   454→
   455→    // Initialize Redis connection
   456→    await initRedis();
   457→
   458→    // Check Redis availability
   459→    const redis = await getRedis();
   460→    
   461→    if (redis) {
   462→      log(LogLevel.INFO, '✅ Redis available, creating distributed rate limiter');
   463→      
   464→      // Merge custom endpoints with defaults
   465→      const mergedEndpoints = [
   466→        ...ENDPOINT_SPECIFIC_LIMITS,
   467→        ...(config?.customEndpoints || []),
   468→      ];
   469→
   470→      // Create RateLimiter configuration
   471→      const limiterConfig: RateLimiterConfig = {
   472→        redisClient: redis as Redis,
   473→        defaultLimits: DEFAULT_LIMITS.analyst,
   474→        endpointConfigs: mergedEndpoints,
   475→        roleBasedLimits: DEFAULT_LIMITS,
   476→        ipBasedLimits: {
   477→          enabled: true,
   478→          maxRequestsPerIP: parseInt(process.env.MAX_REQUESTS_PER_IP || '1000', 10),
   479→          windowSeconds: 60,
   480→        },
   481→        globalLimits: {
   482→          enabled: true,
   483→          maxRequestsTotal: parseInt(process.env.GLOBAL_MAX_REQUESTS || '10000', 10),
   484→          windowSeconds: 60,
   485→        },
   486→        verboseLogging: config?.verboseLogging ?? DEFAULT_MIDDLEWARE_CONFIG.verboseLogging,
   487→      };
   488→
   489→      rateLimiterInstance = new RateLimiter(limiterConfig);
   490→      log(LogLevel.INFO, '✅ Distributed rate limiter initialized successfully');
   491→    } else {
   492→      log(LogLevel.WARN, '⚠️ Redis unavailable, using in-memory fallback');
   493→      inMemoryStore = InMemoryRateLimitStore.getInstance();
   494→    }
   495→
   496→    isInitialized = true;
   497→  } catch (error) {
   498→    log(LogLevel.ERROR, '❌ Failed to initialize rate limiter:', error);
   499→    // Fall back to in-memory store
   500→    inMemoryStore = InMemoryRateLimitStore.getInstance();
   501→    isInitialized = true;
   502→  }
   503→}

   527→/**
   528→ * Extract client IP address from NextRequest
   529→ * Handles various proxy configurations
   530→ */
   531→function extractIP(request: NextRequest): string {
   532→  // Try the utility function first
   533→  const ipFromUtil = extractClientIP(request as Request);
   534→  if (ipFromUtil && ipFromUtil !== 'unknown') {
   535→    return ipFromUtil;
   536→  }
   537→
   538→  // Additional checks specific to NextRequest
   539→  const cfConnectingIp = request.headers.get('cf-connecting-ip');
   540→  if (cfConnectingIp) {
   541→    return cfConnectingIp.trim();
   542→  }
   543→
   544→  const flyClientIp = request.headers.get('fly-client-ip');
   545→  if (flyClientIp) {
   546→    return flyClientIp.trim();
   547→  }
   548→
   549→  const vercelIp = request.headers.get('x-vercel-forwarded-for');
   550→  if (vercelIp) {
   551→    return vercelIp.split(',')[0].trim();
   552→  }
   553→
   554→  // Return unknown with hash for tracking
   555→  return `unknown:${Math.random().toString(36).substring(2, 10)}`;
   556→}

   558→/**
   559→ * Determine user role from user object
   560→ */
   561→function resolveUserRole(user?: RateLimitUser | any): keyof RoleBasedLimits {
   562→  if (!user) {
   563→    return 'unauthenticated';
   564→  }
   565→
   566→  const role = user.role;
   567→  
   568→  // Validate role
   569→  const validRoles: (keyof RoleBasedLimits)[] = [
   570→    'analyst',
   571→    'engineer',
   572→    'admin',
   573→    'serviceAccount',
   574→    'unauthenticated',
   575→  ];
   576→
   577→  if (validRoles.includes(role)) {
   578→    return role;
   579→  }
   580→
   581→  // Default to analyst for authenticated users without explicit role
   582→  if (user.id) {
   583→    return 'analyst';
   584→  }
   585→
   586→  return 'unauthenticated';
   587→}

   590→ * Determine user identity for rate limiting

   592→function resolveUserIdentity(user?: RateLimitUser | any): string | null {
   593→  if (!user?.id) {
   594→    return null;
   595→  }
   596→  return String(user.id);

   599→/**
   600→ * Match request path to endpoint configuration
   601→ * Handles wildcard paths like /api/threat-hunting/*
   602→ */
   603→function matchEndpointPath(requestPath: string): string {
   604→  // Normalize path
   605→  const normalizedPath = requestPath.split('?')[0].toLowerCase();
   606→  
   607→  // Check for exact matches or prefix matches for wildcards
   608→  const pathSegments = normalizedPath.split('/');
   609→  
   610→  // Check threat-hunting sub-paths
   611→  if (normalizedPath.startsWith('/api/threat-hunting')) {
   612→    return '/api/threat-hunting';
   613→  }
   614→  
   615→  // Check stream sub-paths
   616→  if (normalizedPath.startsWith('/api/stream')) {
   617→    return '/api/stream';
   618→  }
   619→  
   620→  // Return exact path
   621→  return normalizedPath;
   622→}

   624→// ============================================================================
   625→// Main Rate Limit Check Function
   626→// ============================================================================
   627→
   628→/**
   629→ * Check rate limit for a request.
   630→ * 
   631→ * This is the main exported function that should be used in API routes
   632→ * and middleware to check rate limits.
   633→ * 
   634→ * @param request - The NextRequest object
   635→ * @param user - Optional user object with id and role
   636→ * @returns RateLimitCheckResult with allowed status, headers, and optional error response
   637→ * 
   638→ * @example
   639→ * ```typescript
   640→ * import { checkRateLimit } from '@/lib/production/rate-limit-middleware';
   641→ * import { NextRequest, NextResponse } from 'next/server';
   642→ * 
   643→ * export async function GET(request: NextRequest) {
   644→ *   const user = await getCurrentUser(request);
   645→ *   const rateLimitResult = await checkRateLimit(request, user);
   646→ *   
   647→ *   if (!rateLimitResult.allowed) {
   648→ *     return rateLimitResult.error!;
   649→ *   }
   650→ *   
   651→ *   // Process request...
   652→ *   return NextResponse.json(data, {
   653→ *     headers: rateLimitResult.headers,
   654→ *   });
   655→ * }
   656→ * ```
   657→ */
   658→export async function checkRateLimit(
   659→  request: NextRequest,
   660→  user?: RateLimitUser | any,
   661→  config?: Partial<RateLimitMiddlewareConfig>
   662→): Promise<RateLimitCheckResult> {
   663→  // Ensure limiter is initialized
   664→  if (!isInitialized) {
   665→    await initializeRateLimiter(config);
   666→  }
   667→
   668→  const mergedConfig = { ...DEFAULT_MIDDLEWARE_CONFIG, ...config };
   669→  const ipAddress = extractIP(request);
   670→  const userIdentity = resolveUserIdentity(user);
   671→  const role = resolveUserRole(user);
   672→  const path = matchEndpointPath(request.nextUrl.pathname);
   673→  const method = request.method;
   674→
   675→  log(LogLevel.DEBUG, `Checking rate limit`, {
   676→    path,
   677→    method,
   678→    ipAddress: ipAddress.replace(/\d+\.\d+\.(\d+)\.\d+/, '$1.*.*.*'), // Mask IP for privacy
   679→    userIdentity: userIdentity || '(anonymous)',
   680→    role,
   681→  });
   682→
   683→  try {
   684→    // Try distributed rate limiter first
   685→    const limiter = getRateLimiter();
   686→    
   687→    if (limiter) {
   688→      return await checkWithDistributedLimiter(
   689→        limiter,
   690→        userIdentity,
   691→        ipAddress,
   692→        path,
   693→        method,
   694→        role,
   695→        mergedConfig
   696→      );
   697→    }
   698→
   699→    // Fall back to in-memory store
   700→    return await checkWithInMemoryStore(
   701→      userIdentity,
   702→      ipAddress,
   703→      path,
   704→      role,
   705→      mergedConfig
   706→    );
   707→
   708→  } catch (error) {
   709→    log(LogLevel.ERROR, 'Rate limit check failed:', error);
   710→    
   711→    // Handle based on fail behavior
   712→    if (mergedConfig.failBehavior === 'fail-closed') {
   713→      // Deny all requests when rate limiting fails
   714→      return createDeniedResponse(
   715→        {
   716→          allowed: false,
   717→          remaining: 0,
   718→          resetTime: Date.now() + 60000,
   719→          limit: 0,
   720→          current: 0,
   721→          retryAfterMs: 60000,
   722→          reason: 'Rate limiting service unavailable',
   723→        },
   724→        mergedConfig.errorMessage
   725→      );
   726→    }
   727→
   728→    // Fail-open: allow request but log warning
   729→    log(LogLevel.WARN, '⚠️ Rate limiter failed, allowing request (fail-open mode)');
   730→    return {
   731→      allowed: true,
   732→      headers: {
   733→        'X-RateLimit-Limit': 'N/A',
   734→        'X-RateLimit-Remaining': 'N/A',
   735→        'X-RateLimit-Reset': 'N/A',
   736→        'X-RateLimit-Mode': 'fail-open',
   737→      },
   738→    };
   739→  }
   740→}
   741→
   742→/**
   743→ * Check rate limit using distributed Redis-backed limiter
   744→ */
   745→async function checkWithDistributedLimiter(
   746→  limiter: RateLimiter,
   747→  userIdentity: string | null,
   748→  ipAddress: string,
   749→  path: string,
   750→  method: string,
   751→  role: keyof RoleBasedLimits,
   752→  config: Required<RateLimitMiddlewareConfig>
   753→): Promise<RateLimitCheckResult> {
   754→  // Special handling for streaming endpoints (concurrent connections)
   755→  if (path === '/api/stream' && config.enableConnectionTracking) {
   756→    return await checkStreamingConnection(
   757→      limiter,
   758→      userIdentity,
   759→      ipAddress,
   760→      role,
   761→      config
   762→    );
   763→  }
   764→
   765→  // Standard rate limit check
   766→  const result = await limiter.check(userIdentity, ipAddress, path, method, role);
   767→
   768→  if (!result.allowed) {
   769→    log(LogLevel.WARN, `🚫 Rate limit exceeded`, {
   770→      path,
   771→      userIdentity: userIdentity || ipAddress,
   772→      reason: result.reason,
   773→      retryAfter: result.retryAfterMs,
   774→    });
   775→    return createDeniedResponse(result, config.errorMessage);
   776→  }
   777→
   778→  return {
   779→    allowed: true,
   780→    headers: buildRateLimitHeaders(result),
   781→    result,
   782→  };
   783→}

   785→/**
   786→ * Check streaming connection limits (concurrent connections)
   787→ */
   788→async function checkStreamingConnection(
   789→  limiter: RateLimiter,
   790→  userIdentity: string | null,
   791→  ipAddress: string,
   792→  role: keyof RoleBasedLimits,
   793→  config: Required<RateLimitMiddlewareConfig>
   794→): Promise<RateLimitCheckResult> {
   795→  const redis = await getRedis();
   796→  const connectionKey = `stream:conn:${userIdentity || ipAddress}`;
   797→  const maxConnections = role === 'admin' ? 20 : role === 'serviceAccount' ? 15 : 10;
   798→
   799→  if (redis) {
   800→    try {
   801→      // Use Redis to track concurrent connections
   802→      const currentConnections = await redis.scard(connectionKey);
   803→      
   804→      if (currentConnections >= maxConnections) {
   805→        const result: RateLimitResult = {
   806→          allowed: false,
   807→          remaining: 0,
   808→          resetTime: Date.now() + 60000,
   809→          limit: maxConnections,
   810→          current: currentConnections,
   811→          retryAfterMs: 60000,
   812→          reason: `Maximum concurrent streaming connections (${maxConnections}) reached`,
   813→        };
   814→        
   815→        return createDeniedResponse(
   816→          result,
   817→          'Maximum concurrent streaming connections reached. Please close existing connections.'
   818→        );
   819→      }
   820→
   821→      return {
   822→        allowed: true,
   823→        headers: {
   824→          'X-RateLimit-Limit': String(maxConnections),
   825→          'X-RateLimit-Remaining': String(maxConnections - currentConnections - 1),
   826→          'X-RateLimit-Reset': String(Date.now() + 60000),
   827→          'X-RateLimit-Connection-Current': String(currentConnections),
   828→          'X-RateLimit-Policy': `concurrent;max=${maxConnections}`,
   829→        },
   830→      };
   831→    } catch {
   832→      // Redis command failed, fall through to standard check
   833→    }
   834→  }
   835→
   836→  // Fallback to standard rate limit check
   837→  const result = await limiter.check(userIdentity, ipAddress, '/api/stream', 'GET', role);
   838→  
   839→  if (!result.allowed) {
   840→    return createDeniedResponse(result, config.errorMessage);
   841→  }
   842→
   843→  return {
   844→    allowed: true,
   845→    headers: buildRateLimitHeaders(result),
   846→    result,
   847→  };
   848→}

   850→/**
   851→ * Check rate limit using in-memory fallback store
   852→ */
   853→async function checkWithInMemoryStore(
   854→  userIdentity: string | null,
   855→  ipAddress: string,
   856→  path: string,
   857→  role: keyof RoleBasedLimits,
   858→  config: Required<RateLimitMiddlewareConfig>
   859→): Promise<RateLimitCheckResult> {
   860→  const store = getInMemoryStore();
   861→  const key = `${userIdentity || 'anon'}:${path}:${role}`;
   862→  
   863→  // Get appropriate limits based on path and role
   864→  const options = getOptionsForEndpoint(path, role);
   865→  const result = store.check(key, options);
   866→
   867→  if (!result.allowed) {
   868→    log(LogLevel.WARN, `🚫 Rate limit exceeded (in-memory)`, {
   869→      path,
   870→      userIdentity: userIdentity || ipAddress,
   871→    });
   872→    return createDeniedResponse(result, config.errorMessage);
   873→  }
   874→
   875→  return {
   876→    allowed: true,
   877→    headers: {
   878→      ...buildRateLimitHeaders(result),
   879→      'X-RateLimit-Mode': 'in-memory-fallback',
   880→    },
   881→    result,
   882→  };
   883→}

   885→/**
   886→ * Get rate limit options for an endpoint and role (for in-memory fallback)
   887→ */
   888→function getOptionsForEndpoint(path: string, role: keyof RoleBasedLimits): RateLimitOptions {
   889→  // Find matching endpoint configuration
   890→  const endpointConfig = ENDPOINT_SPECIFIC_LIMITS.find(ec => {
   891→    if (ec.path === path) return true;
   892→    // Handle prefix matching
   893→    if (path.startsWith(ec.path + '/') || path.startsWith(ec.path)) return true;
   894→    return false;
   895→  });
   896→
   897→  if (endpointConfig && 'analyst' in endpointConfig.limits) {
   898→    const roleLimits = endpointConfig.limits as RoleBasedLimits;
   899→    return roleLimits[role] || DEFAULT_LIMITS[role];
   900→  }
   901→
   902→  if (endpointConfig && !('analyst' in endpointConfig.limits)) {
   903→    return endpointConfig.limits as RateLimitOptions;
   904→  }
   905→
   906→  return DEFAULT_LIMITS[role] || DEFAULT_LIMITS.analyst;
   907→}

   913→/**
   914→ * Build rate limit headers from result

   916→function buildRateLimitHeaders(result: RateLimitResult): Record<string, string> {
   917→  return {
   918→    'X-RateLimit-Limit': String(result.limit),
   919→    'X-RateLimit-Remaining': String(Math.max(0, result.remaining)),
   920→    'X-RateLimit-Reset': String(result.resetTime),
   921→    ...(result.retryAfterMs ? {
   922→      'Retry-After': String(Math.ceil(result.retryAfterMs / 1000)),
   923→    } : {}),
   924→  };

   927→/**
   928→ * Create a denied (rate limited) response
   929→ */
   930→function createDeniedResponse(
   931→  result: RateLimitResult,
   932→  errorMessage: string
   933→): RateLimitCheckResult {
   934→  const headers: Record<string, string> = {
   935→    'X-RateLimit-Limit': String(result.limit),
   936→    'X-RateLimit-Remaining': '0',
   937→    'X-RateLimit-Reset': String(result.resetTime),
   938→    'Content-Type': 'application/json',
   939→  };
   940→
   941→  if (result.retryAfterMs) {
   942→    headers['Retry-After'] = String(Math.ceil(result.retryAfterMs / 1000));
   943→  }
   944→
   945→  const body = {
   946→    error: 'Too Many Requests',
   947→    message: errorMessage,
   948→    statusCode: 429,
   949→    retryAfter: result.retryAfterMs ? Math.ceil(result.retryAfterMs / 1000) : undefined,
   950→    reason: result.reason,
   951→    documentation: 'https://security.djezzy.dz/rate-limits',
   952→    timestamp: new Date().toISOString(),
   953→  };
   954→
   955→  const errorResponse = new Response(JSON.stringify(body), {
   956→    status: 429,
   957→    statusText: 'Too Many Requests',
   958→    headers,
   959→  });
   960→
   961→  return {
   962→    allowed: false,
   963→    headers,
   964→    error: errorResponse,
   965→    result,
   966→  };
   967→}

   973→/**
   974→ * Add rate limit headers to a NextResponse.
   975→ * 
   976→ * Use this function to add rate limit information to successful responses.
   977→ * 
   978→ * @param response - The NextResponse to modify
   979→ * @param headers - The rate limit headers from checkRateLimit
   980→ * @returns The modified NextResponse
   981→ * 
   982→ * @example
   983→ * ```typescript
   984→ * const rateLimitResult = await checkRateLimit(request, user);
   985→ * if (!rateLimitResult.allowed) {
   986→ *   return rateLimitResult.error!;
   987→ * }
   988→ * 
   989→ * return addRateLimitHeaders(
   990→ *   NextResponse.json(data),
   991→ *   rateLimitResult.headers
   992→ * );
   993→ * ```
   994→ */
   995→export function addRateLimitHeaders<T extends NextResponse>(
   996→  response: T,
   997→  headers: Record<string, string>
   998→): T {
   999→  for (const [key, value] of Object.entries(headers)) {
  1000→    // Skip content-type as it's already set on the response
  1001→    if (key.toLowerCase() !== 'content-type') {
  1002→      response.headers.set(key, value);
  1003→    }
  1004→  }
  1005→  return response;
  1006→}

  1056→/**
  1057→ * Register a new streaming connection for rate limiting.
  1058→ * Call this when a streaming connection is established.
  1059→ */
  1060→export async function registerStreamConnection(
  1061→  userId: string,
  1062→  connectionId: string
  1063→): Promise<boolean> {
  1064→  try {
  1065→    const redis = await getRedis();
  1066→    if (redis) {
  1067→      const key = `stream:conn:${userId}`;
  1068→      const current = await redis.scard(key);
  1069→      
  1070→      if (current >= 10) {
  1071→        return false;
  1072→      }
  1073→      
  1074→      await redis.sadd(key, connectionId);
  1075→      await redis.expire(key, 3600); // 1 hour expiry
  1076→      return true;
  1077→    }
  1078→
  1079→    // Fallback to in-memory
  1080→    return getInMemoryStore().addConnection(userId, connectionId);
  1081→  } catch (error) {
  1082→    log(LogLevel.ERROR, 'Failed to register stream connection:', error);
  1083→    return getInMemoryStore().addConnection(userId, connectionId);
  1084→  }
  1085→}

  1132→/**
  1133→ * Wrap a route handler with automatic rate limiting.
  1134→ * 
  1135→ * @example
  1136→ * ```typescript
  1137→ * export const GET = withRateLimit(async (request) => {
  1138→ *   return NextResponse.json(data);
  1139→ * });
  1140→ * 
  1141→ * // With custom config

  1143→ *   async (request) => {
  1144→ *     return NextResponse.json({ success: true });
  1145→ *   },
  1146→ *   { failBehavior: 'fail-closed' }
  1147→ * );
  1148→ * ```
  1149→ */
  1150→export function withRateLimit<T extends NextRequest>(
  1151→  handler: (request: T, ...args: any[]) => Promise<NextResponse>,
  1152→  config?: Partial<RateLimitMiddlewareConfig> & { getUser?: (req: T) => Promise<any> }
  1153→): (request: T, ...args: any[]) => Promise<NextResponse> {
  1154→  return async (request: T, ...args: any[]): Promise<NextResponse> => {
  1155→    // Get user if getUser function provided
  1156→    let user: any = undefined;
  1157→    if (config?.getUser) {
  1158→      try {
  1159→        user = await config.getUser(request);
  1160→      } catch {
  1161→        // Continue without user
  1162→      }
  1163→    }
  1164→
  1165→    // Check rate limit
  1166→    const result = await checkRateLimit(request, user, config);
  1167→
  1168→    if (!result.allowed && result.error) {
  1169→      return result.error as NextResponse;
  1170→    }
  1171→
  1172→    // Execute handler
  1173→    const response = await handler(request, ...args);
  1174→
  1175→    // Add rate limit headers to response
  1176→    return addRateLimitHeaders(response, result.headers);
  1177→  };
  1178→}
