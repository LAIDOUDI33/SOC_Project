    33→interface IOC {
    34→  id: string
    35→  type: 'ip' | 'domain' | 'hash' | 'url' | 'email' | 'pattern'
    36→  value: string
    37→  confidence: number
    38→  source: string
    39→  firstSeen: string
    40→  lastSeen: string
    41→  tags: string[]
    42→}
    43→
    44→interface HuntQuery {
    45→  id: string
    46→  name: string
    47→  queryType: 'SQL' | 'LUCENE' | 'SIGMA' | 'YARA' | 'CUSTOM'
    48→  queryString: string
    49→  dataSource: string
    50→  resultCount: number
    51→  executionTime: number
    52→  executedAt: string
    53→  status: 'pending' | 'running' | 'completed' | 'error'
    54→}
    55→
    56→interface HuntFinding {
    57→  id: string
    58→  queryId: string
    59→  title: string
    60→  description: string
    61→  severity: 'INFO' | 'LOW' | 'MEDIUM' | 'HIGH' | 'CRITICAL'
    62→  confidence: number
    63→  status: 'NEW' | 'INVESTIGATING' | 'CONFIRMED' | 'FALSE_POSITIVE' | 'ESCALATED'
    64→  rawEvidence: Record<string, unknown>
    65→  iocs: IOC[]
    66→  timestamp: string
    67→}
    68→
    69→interface HuntSession {
    70→  id: string
    71→  name: string
    72→  description: string
    73→  hypothesis: string
    74→  status: 'DRAFT' | 'ACTIVE' | 'PAUSED' | 'COMPLETED' | 'CANCELLED'
    75→  hunter: string
    76→  queries: HuntQuery[]
    77→  findings: HuntFinding[]
    78→  createdAt: string
    79→  updatedAt: string
    80→}
    81→
    82→interface HypothesisTemplate {
    83→  id: string
    84→  name: string
    85→  category: string
    86→  description: string
    87→  templateQuery: string
    88→}
