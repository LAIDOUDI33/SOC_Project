   453→// IOC Pivot Analysis
   454→function IOCPivotAnalysis({ iocs }: { iocs: IOCRecord[] }) {
   455→  const [selectedIOC, setSelectedIOC] = useState<IOCRecord | null>(null)
   456→
   457→  const getTypeIcon = (type: IOCRecord['type']) => {
   458→    switch (type) {
   459→      case 'ip': return <Network className="h-4 w-4 text-blue-400" />
   460→      case 'domain': return <Database className="h-4 w-4 text-green-400" />
   461→      case 'hash': return <Lock className="h-4 w-4 text-orange-400" />
   462→      case 'url': return <FileText className="h-4 w-4 text-purple-400" />
   463→      case 'email': return <User className="h-4 w-4 text-cyan-400" />
   464→    }
   465→  }
   466→
   467→  const getConfidenceColor = (confidence: number) => {
   468→    if (confidence >= 80) return 'text-green-400'
   469→    if (confidence >= 60) return 'text-yellow-400'
   470→    return 'text-red-400'
   471→  }
   472→
   473→  return (
   474→    <Card className="bg-slate-900 border-slate-700">
   475→      <CardHeader className="pb-3">
   476→        <div className="flex items-center justify-between">
   477→          <CardTitle className="text-white text-lg flex items-center gap-2">
   478→            <Shield className="h-5 w-5 text-red-400" />
   479→            IOC Pivot Analysis
   480→          </CardTitle>
   481→          <Button size="sm" variant="outline" className="border-slate-600 text-slate-300 h-8">
   482→            <Share2 className="h-4 w-4 mr-1" />
   483→            Push to MISP
   484→          </Button>
   485→        </div>
   486→      </CardHeader>
   487→      <CardContent>
   488→        <div className="space-y-2 max-h-[400px] overflow-y-auto">
   489→          {iocs.map((ioc, index) => (
   490→            <div 
   491→              key={index}
   492→              className={`p-3 rounded-lg border cursor-pointer transition-all ${
   493→                selectedIOC?.value === ioc.value 
   494→                  ? 'border-blue-500 bg-blue-500/10' 
   495→                  : 'border-slate-700 bg-slate-800 hover:border-slate-600'
   496→              }`}
   497→              onClick={() => setSelectedIOC(ioc)}
   498→            >
   499→              <div className="flex items-center justify-between">
   500→                <div className="flex items-center gap-3">

   625→// Query Builder Component
   626→function QueryBuilder() {

   628→    { field: 'source_ip', operator: '=', value: '' }

   636→    setQueryParts(queryParts.filter((_, i) => i !== index))
