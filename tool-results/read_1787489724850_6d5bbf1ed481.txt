   452→function QueryBuilder({
   453→  conditions,
   454→  setConditions,
   455→  onExecute
   456→}: {
   457→  conditions: QueryCondition[]
   458→  setConditions: React.Dispatch<React.SetStateAction<QueryCondition[]>>
   459→  onExecute: () => void
   460→}) {
   461→  const addCondition = () => {
   462→    setConditions([
   463→      ...conditions,
   464→      {
   465→        id: `cond-${Date.now()}`,
   466→        field: '',
   467→        operator: 'equals',
   468→        value: '',
   469→        logic: conditions.length > 0 ? 'AND' : 'AND'
   470→      }
   471→    ])
   472→  }
   473→
   474→  const removeCondition = (id: string) => {
   475→    setConditions(conditions.filter(c => c.id !== id))
   476→  }
   477→
   478→  const updateCondition = (id: string, updates: Partial<QueryCondition>) => {
   479→    setConditions(conditions.map(c => c.id === id ? { ...c, ...updates } : c))
   480→  }
   481→
   482→  return (
   483→    <Card className="bg-slate-900 border-slate-700">
   484→      <CardHeader className="pb-3">
   485→        <CardTitle className="text-white text-base flex items-center gap-2">
   486→          <Terminal className="h-4 w-4 text-green-400" />
   487→          Query Builder
   488→        </CardTitle>
   489→      </CardHeader>
