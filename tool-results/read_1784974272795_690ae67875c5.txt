   176→/** HTTP response wrapper */
   177→interface HttpResponse<T> {
   178→  data: T;
   179→  status: number;
   180→  headers: Headers;
   181→}
   182→
   183→/** Request options interface */
   184→interface RequestOptions {
   185→  method: string;
   186→  path: string;
   187→  body?: Record<string, unknown>;
   188→  params?: Record<string, string>;
   189→  headers?: Record<string, string>;
   190→  expectEmpty?: boolean;
   191→}
   192→
   193→/**
   194→ * Core HTTP client for Grafana API communication
   195→ */
   196→class GrafanaHttpClient {
   197→  private config: Required<GrafanaClientConfig> & { username?: string; password?: string; apiKey?: string };
   198→  
   199→  constructor(config: GrafanaClientConfig) {
   200→    this.config = { ...DEFAULT_CONFIG, ...config };
   201→    
   202→    // Validate required configuration
   203→    if (!this.config.url) {
   204→      throw new GrafanaValidationError('Grafana URL is required');
   205→    }
   206→    
   207→    // Ensure URL doesn't have trailing slash
   208→    this.config.url = this.config.url.replace(/\/+$/, '');
   209→  }
   210→
   211→  /**
   212→   * Build authorization headers
   213→   */
   214→  private getAuthHeaders(): Record<string, string> {
   215→    const headers: Record<string, string> = {};
   216→    
   217→    if (this.config.apiKey) {
   218→      headers['Authorization'] = `Bearer ${this.config.apiKey}`;
   219→    } else if (this.config.username && this.config.password) {
   220→      // Basic auth would be handled differently in browser environment
   221→      // For server-side or proxy scenarios
   222→      const credentials = btoa(`${this.config.username}:${this.config.password}`);
   223→      headers['Authorization'] = `Basic ${credentials}`;
   224→    }
   225→    
   226→    return headers;
   227→  }
   228→
   229→  /**
   230→   * Execute HTTP request
   231→   */
   232→  async request<T>(options: RequestOptions): Promise<HttpResponse<T>> {
   233→    const { method, path, body, params, headers: customHeaders, expectEmpty } = options;
   234→    
   235→    // Build URL
   236→    let url = `${this.config.url}${API_PREFIX}${path}`;
   237→    
   238→    // Add query parameters
   239→    if (params && Object.keys(params).length > 0) {
   240→      const searchParams = new URLSearchParams();
   241→      Object.entries(params).forEach(([key, value]) => {
   242→        if (value !== undefined && value !== null) {
   243→          searchParams.append(key, value);
   244→        }
   245→      });
   246→      url += `?${searchParams.toString()}`;
   247→    }
   248→
   249→    // Build headers
   250→    const headers: Record<string, string> = {
   251→      'Content-Type': 'application/json',
   252→      'Accept': 'application/json',
   253→      ...this.getAuthHeaders(),
   254→      ...customHeaders,
   255→    };
   256→
   257→    // Log request in debug mode
   258→    if (this.config.debug) {
   259→      console.log(`[Grafana] ${method} ${url}`, body ? JSON.stringify(body) : '');
   260→    }
   261→
   262→    try {
   263→      // Build fetch options
   264→      const fetchOptions: RequestInit = {
   265→        method,
   266→        headers,
   267→        signal: AbortSignal.timeout(this.config.timeout),
   268→      };
   269→
   270→      if (body && ['POST', 'PUT', 'PATCH'].includes(method)) {
   271→        fetchOptions.body = JSON.stringify(body);
   272→      }
   273→
   274→      const response = await fetch(url, fetchOptions);
   275→
   276→      // Handle empty responses
   277→      if (expectEmpty || response.status === 204) {
   278→        return { data: undefined as T, status: response.status, headers: response.headers };
   279→      }
   280→
   281→      // Parse response
   282→      const data = await response.json();
   283→
   284→      // Handle errors
   285→      if (!response.ok) {
   286→        return this.handleError(response, data);
   287→      }
   288→
   289→      return { data, status: response.status, headers: response.headers };
   290→    } catch (error) {
   291→      if (error instanceof DOMException && error.name === 'TimeoutError') {
   292→        throw new GrafanaError('Request timeout', undefined, 'TIMEOUT_ERROR');
   293→      }
   294→      
   295→      if (error instanceof TypeError) {
   296→        throw new GrafanaConnectionError(error);
   297→      }
   298→      
   299→      throw error;
   300→    }
   301→  }
   302→
   303→  /**
   304→   * Handle API error responses
   305→   */
   306→  private handleError<T>(response: Response, data: unknown): never {
   307→    switch (response.status) {
   308→      case 401:
   309→      case 403:
   310→        throw new GrafanaAuthError(
   311→          (data as { message?: string })?.message || 'Authentication failed',
   312→          response.status
   313→        );
   314→        
   315→      case 404:
   316→        throw new GrafanaNotFoundError('Resource');
   317→        
   318→      case 409:
   319→        throw new GrafanaVersionConflictError(
   320→          (data as { version?: number })?.version || 0,
   321→          0
   322→        );
   323→        
   324→      case 429:
   325→        const retryAfter = parseInt(response.headers.get('Retry-After') || '60', 10);
   326→        throw new GrafanaRateLimitError(retryAfter);
   327→        
   328→      case 400:
   329→        throw new GrafanaValidationError(
   330→          (data as { message?: string })?.message || 'Validation failed',
   331→          (data as { errors?: Array<{ field: string; message: string }> })?.errors
   332→        );
   333→        
   334→      default:
   335→        throw new GrafanaError(
   336→          (data as { message?: string })?.message || `Request failed with status ${response.status}`,
   337→          response.status,
   338→          undefined,
   339→          data
   340→        );
   341→    }
   342→  }
   343→

   345→   * GET request helper

   347→  async get<T>(path: string, params?: Record<string, string>): Promise<HttpResponse<T>> {

   349→  }

   365→  /**
   366→   * PATCH request helper
   367→   */

   372→  /**
   373→   * DELETE request helper
   374→   */

   377→  }

   447→  /**
   448→   * Search for dashboards
   449→   * @param params Search parameters
   450→   * @returns Array of matching dashboards
   451→   */
   452→  async searchDashboards(params?: DashboardSearchParams): Promise<DashboardSearchResult[]> {
   453→    const searchParams: Record<string, string> = {};
   454→
   455→    if (params?.query) searchParams.query = params.query;
   456→    if (params?.tag) searchParams.tag = params.tag.join(',');
   457→    if (params?.type) searchParams.type = params.type;
   458→    if (params?.folderIds) searchParams.folderIds = params.folderIds.join(',');
   459→    if (params?.limit) searchParams.limit = params.limit.toString();
   460→    if (params?.page) searchParams.page = params.page.toString();
   461→    if (params?.sort) searchParams.sort = params.sort;
   462→    if (params?.starred) searchParams.starred = 'true';
   463→
   464→    const response = await this.http.get<DashboardSearchResult[]>('/search', searchParams);
   465→    return response.data;
   466→  }

   536→    uid: string,
   537→    dashboard: Partial<GrafanaDashboard>,

   539→      folderUid?: string;
   540→      message?: string;

   556→      message: options?.message || 'Updated via API',

   898→  async batchSetAlertRulesPaused(ruleUids: string[], paused: boolean): Promise<BulkOperationResult> {
   899→    const results: BulkOperationResult = {
   900→      successCount: 0,
   901→      failedCount: 0,
   902→      errors: [],
   903→    };
   904→
   905→    for (const uid of ruleUids) {

   907→        await this.setAlertRulePaused(uid, paused);

   909→      } catch (error) {
   910→        results.failedCount++;
   911→        results.errors.push({
   912→          item: uid,
   913→          reason: error instanceof Error ? error.message : 'Unknown error',
   914→        });
   915→      }
   916→    }
   917→
   918→    return results;
   919→  }

  1579→  /**
  1580→   * Generate embedded dashboard URL
  1581→   * @param config Embedding configuration
  1582→   * @returns Embedded URL
  1583→   */
  1584→  generateEmbeddedUrl(config: {
  1585→    dashboardUid: string;
  1586→    panelId?: number;
  1587→    initialTime?: TimeRange;
  1588→    initialVariables?: Record<string, string>;
  1589→    theme?: 'light' | 'dark' | 'current';
  1590→    width?: number;
  1591→    height?: number;
  1592→    autoRefresh?: RefreshInterval;
  1593→    showToolbar?: boolean;
  1594→    showTimePicker?: boolean;
  1595→    showHeader?: boolean;
  1596→  }): string {
  1597→    const baseUrl = this.http.getBaseUrl();
  1598→    const params = new URLSearchParams();
  1599→
  1600→    // Set theme
  1601→    if (config.theme && config.theme !== 'current') {
  1602→      params.set('theme', config.theme);
  1603→    }
  1604→
  1605→    // Set panel if specified
  1606→    if (config.panelId) {
  1607→      params.set('viewPanel', config.panelId.toString());
  1608→    }
  1609→
  1610→    // Set time range
  1611→    if (config.initialTime) {
  1612→      params.set('from', config.initialTime.from);
  1613→      params.set('to', config.initialTime.to);
  1614→    }
  1615→
  1616→    // Set variables
  1617→    if (config.initialVariables) {
  1618→      Object.entries(config.initialVariables).forEach(([key, value]) => {
  1619→        params.set(`var-${key}`, value);
  1620→      });
  1621→    }
  1622→
  1623→    // Set refresh interval
  1624→    if (config.autoRefresh) {
  1625→      params.set('refresh', config.autoRefresh);
  1626→    }
  1627→
  1628→    // Set display options
  1629→    if (config.showToolbar !== undefined) {
  1630→      params.set('kiosk', config.showToolbar ? '' : 'tv');
  1631→    }
  1632→    if (config.showTimePicker !== undefined) {
  1633→      params.set('showTimePicker', String(config.showTimePicker));
  1634→    }
  1635→    if (config.showHeader !== undefined) {
  1636→      params.set('showHeader', String(config.showHeader));
  1637→    }
  1638→
  1639→    const queryString = params.toString();
  1640→    return `${baseUrl}/d/${config.dashboardUid}${queryString ? `?${queryString}` : ''}`;
  1641→  }

  1685→  /**
  1686→   * Validate dashboard JSON structure
  1687→   * @param json Dashboard JSON to validate
  1688→   * @returns Validation result with errors if any
  1689→   */
  1690→  validateDashboard(json: Record<string, unknown>): {
  1691→    valid: boolean;
  1692→    errors: string[];
  1693→    warnings: string[];
  1694→  } {
  1695→    const errors: string[] = [];
  1696→    const warnings: string[] = [];
  1697→
  1698→    // Check required fields
  1699→    if (!json.title) {
  1700→      errors.push('Dashboard must have a title');
  1701→    }
  1702→
  1703→    if (!json.panels || !Array.isArray(json.panels)) {
  1704→      errors.push('Dashboard must have panels array');
  1705→    } else if ((json.panels as unknown[]).length === 0) {
  1706→      warnings.push('Dashboard has no panels');
  1707→    }
  1708→
  1709→    // Check schema version
  1710→    if (json.schemaVersion && typeof json.schemaVersion === 'number' && json.schemaVersion < 30) {
  1711→      warnings.push(`Schema version ${json.schemaVersion} may be outdated. Consider updating.`);
  1712→    }
  1713→
  1714→    // Check panels structure
  1715→    if (Array.isArray(json.panels)) {
  1716→      json.panels.forEach((panel: unknown, index: number) => {
  1717→        const p = panel as Record<string, unknown>;
  1718→        if (!p.id) {
  1719→          errors.push(`Panel at index ${index} missing required "id" field`);
  1720→        }
  1721→        if (!p.type) {
  1722→          errors.push(`Panel at index ${index} missing required "type" field`);
  1723→        }
  1724→        if (!p.gridPos) {
  1725→          warnings.push(`Panel "${p.title || index}" has no grid position defined`);
  1726→        }
  1727→      });
  1728→    }
  1729→
  1730→    return {
  1731→      valid: errors.length === 0,
  1732→      errors,
  1733→      warnings,
  1734→    };
  1735→  }
  1736→}
