   349→function FraudDetectionPanel({ alerts }: { alerts: FraudAlert[] }) {
   350→  const getFraudTypeIcon = (type: FraudAlert['type']) => {
   351→    switch (type) {
   352→      case 'irsf': return <PhoneCall className="h-4 w-4" />
   353→      case 'sim_swap': return <UserX className="h-4 w-4" />
   354→      case 'premium_rate': return <CreditCard className="h-4 w-4" />
   355→      case 'bypass_fraud': return <Router className="h-4 w-4" />
   356→      case 'wangiri': return <Phone className="h-4 w-4" />
   357→    }
   358→  }
   359→
   360→  const getSeverityColor = (severity: FraudAlert['severity']) => {
   361→    switch (severity) {
   362→      case 'critical': return 'border-red-500/50 bg-red-500/10'
   363→      case 'high': return 'border-orange-500/50 bg-orange-500/10'
   364→      case 'medium': return 'border-yellow-500/50 bg-yellow-500/10'
   365→      case 'low': return 'border-blue-500/50 bg-blue-500/10'
   366→    }
   367→  }
   368→
   369→  return (
   370→    <Card className="bg-slate-900 border-slate-700">
   371→      <CardHeader className="pb-3">
   372→        <div className="flex items-center justify-between">
   373→          <CardTitle className="text-white text-base flex items-center gap-2">
   374→            <AlertTriangle className="h-4 w-4 text-orange-400" />
   375→            Fraud Detection Alerts
   376→          </CardTitle>
   377→          <Badge variant="secondary" className="bg-orange-500/20 text-orange-400">
   378→            {alerts.filter(a => a.status !== 'false_positive').length} Active
   379→          </Badge>
   380→        </div>
   381→      </CardHeader>
   382→      <CardContent>
   383→        <div className="space-y-3">
   384→          {alerts.map(alert => (
   385→            <div key={alert.id} className={`p-3 rounded-lg border ${getSeverityColor(alert.severity)}`}>
   386→              <div className="flex items-start justify-between mb-2">
   387→                <div className="flex items-center gap-2">
   388→                  {getFraudTypeIcon(alert.type)}
   389→                  <span className="font-medium text-white capitalize text-sm">
   390→                    {alert.type.replace('_', ' ')}
   391→                  </span>
   392→                  <Badge variant="outline" className={`text-xs ${
   393→                    alert.severity === 'critical' ? 'border-red-500/50 text-red-400' :
   394→                    alert.severity === 'high' ? 'border-orange-500/50 text-orange-400' :
   395→                    alert.severity === 'medium' ? 'border-yellow-500/50 text-yellow-400' :
   396→                    'border-blue-500/50 text-blue-400'
   397→                  }`}>
   398→                    {alert.severity}
   399→                  </Badge>
   400→                </div>
   401→                <StatusIndicator

   403→                    alert.status === 'blocked' ? 'excellent' :
   404→                    alert.status === 'monitoring' ? 'warning' :
   405→                    alert.status === 'investigating' ? 'good' : 'critical'
   406→                  } 
   407→                  size="sm"
   408→                  showLabel
   409→                />
   410→              </div>
   411→              <div className="flex items-center justify-between text-xs text-slate-400">
   412→                <div className="flex items-center gap-4">
   413→                  <span className="font-mono">{alert.msisdn}</span>
