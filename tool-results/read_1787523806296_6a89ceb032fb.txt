   184→function generateTrendData(days: number): TrendDataPoint[] {
   185→  const data: TrendDataPoint[] = []
   186→  const now = new Date()
   187→  
   188→  // Base values for realistic Djezzy SOC data
   189→  const baseAlerts = 85
   190→  const baseIncidents = 12
   191→  const baseResolved = 68
   192→  const baseRiskScore = 28
   193→  
   194→  for (let i = days; i >= 0; i--) {
   195→    const date = new Date(now)
   196→    date.setDate(date.getDate() - i)
   197→    
   198→    // Add realistic variation with slight trends
   199→    const dayVariation = Math.sin(i / 5) * 15
   200→    const randomFactor = (Math.random() - 0.5) * 20
   201→    
   202→    data.push({
   203→      date: date.toISOString().split('T')[0],
   204→      alerts: Math.floor(baseAlerts + dayVariation + randomFactor + (i < 10 ? -15 : 5)),
   205→      incidents: Math.floor(baseIncidents + (Math.random() - 0.3) * 8),
   206→      resolved: Math.floor(baseResolved + dayVariation * 0.8 + randomFactor * 0.5),
   207→      riskScore: Math.max(15, Math.min(45, Math.floor(baseRiskScore + (days - i) * 0.3 + (Math.random() - 0.5) * 5)))
   208→    })
   209→  }
   210→  
   211→  return data
   212→}

   512→function generateSS7TimeSeries(hours: number): TimeSeriesPoint[] {
   513→  const data: TimeSeriesPoint[] = []
   514→  const now = new Date()
   515→  
   516→  // Realistic traffic patterns for Algerian mobile network
   517→  for (let i = hours; i >= 0; i--) {
   518→    const timestamp = new Date(now)
   519→    timestamp.setHours(timestamp.getHours() - i)
   520→    
   521→    // Simulate daily patterns: low at night, peaks during day
   522→    const hour = timestamp.getHours()
   523→    let baseLoad = 800
   524→    
   525→    if (hour >= 7 && hour <= 9) baseLoad = 1800  // Morning rush
   526→    else if (hour >= 11 && hour <= 13) baseLoad = 2200  // Lunch period
   527→    else if (hour >= 17 && hour <= 20) baseLoad = 2500  // Evening peak
   528→    else if (hour >= 0 && hour <= 5) baseLoad = 400  // Night
   529→    
   530→    const variation = (Math.random() - 0.5) * baseLoad * 0.3
   531→    
   532→    data.push({
   533→      timestamp: timestamp.toISOString(),
   534→      value: Math.round(Math.max(200, baseLoad + variation))
   535→    })
   536→  }
   537→  
   538→  return data
   539→}
