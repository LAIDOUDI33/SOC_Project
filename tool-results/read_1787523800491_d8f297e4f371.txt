   371→  private bufferFlushInterval: NodeJS.Timeout | null = null;

   373→  // Statistics

   375→
   376→  constructor(config: NsmIntegrationConfig) {
   377→    super();
   378→    this.config = config;
   379→
   380→    // Set up internal event handlers
   381→    this.on('error', (err) => {
   382→      console.error('[NSM] Error:', err.message);
   383→    });
   384→
   385→    this.on('alert', (event: NsmEvent) => {
   386→      console.log(`[NSM] Alert (${event.severity}): ${event.title}`);
   387→      
   388→      // Auto-escalation logic
   389→      if (this.config.enableAutoEscalation && event.severity === 'critical') {
   390→        this.escalateEvent(event);
   391→      }
   392→    });
   393→
   394→    // Start buffer flush interval (every 5 seconds)
   395→    this.bufferFlushInterval = setInterval(() => this.flushEventBuffer(), 5000);
   396→  }

   402→  /**
   403→   * Initialize connections to all NSM components
   404→   */
   405→  async connect(): Promise<void> {
   406→    try {
   407→      console.log('[NSM] Initializing NSM integrations...');
   408→
   409→      // Test Suricata connection
   410→      console.log('[NSM] Connecting to Suricata at', this.config.suricata.apiUrl);
   411→      const suricataHealth = await this.fetchSuricata('/health');
   412→      if (suricataHealth.ok) {
   413→        console.log('[NSM] Suricata connected successfully');
   414→      } else {
   415→        console.warn('[NSM] Suricata health check returned:', suricataHealth.status);
   416→      }
   417→
   418→      // Test Zeek controller (if configured)
   419→      if (this.config.zeek.controllerUrl) {
   420→        console.log('[NSM] Testing Zeek controller...');
   421→        const zeekHealth = await this.fetchZeek('/health');
   422→        if (zeekHealth.ok) {
   423→          console.log('[NSM] Zeek controller connected');
   424→        } else {
   425→          console.warn('[NSM] Zeek controller unavailable, using log-based mode');
   426→        }
   427→      }
   428→
   429→      // Test Arkime connection
   430→      console.log('[NSM] Connecting to Arkime at', this.config.arkime.apiUrl);
   431→      const arkimeHealth = await this.fetchArkime('/api/session?startDate=2024-01-01&stopDate=2024-01-02&length=1');
   432→      if (arkimeHealth.ok || arkimeHealth.status === 401) { // 401 means auth required but service is up
   433→        console.log('[NSM] Arkime connected successfully');
   434→      } else {
   435→        console.warn('[NSM] Arkime health check returned:', arkimeHealth.status);
   436→      }
   437→
   438→      this.isConnected = true;
   439→
   440→      // Start real-time event streams if configured
   441→      if (this.config.suricata.eveLogPath) {
   442→        this.startEveJsonStream();
   443→      }
   444→
   445→      this.emit('connected', {
   446→        suricata: true,
   447→        zeek: !!this.config.zeek.controllerUrl,
   448→        arkime: true,
   449→        timestamp: new Date(),
   450→      });
   451→
   452→    } catch (error) {
   453→      this.isConnected = false;
   454→      this.emit('error', error);
   455→      throw error;
   456→    }
   457→  }

   459→  /**
   460→   * Disconnect from all services

   464→

   466→      clearInterval(this.bufferFlushInterval);
   467→      this.bufferFlushInterval = null;
   468→    }
   469→
   470→    // Flush remaining events
   471→    await this.flushEventBuffer();
   472→
   473→    this.emit('disconnected', { timestamp: new Date() });
   474→    console.log('[NSM] Disconnected from all NSM services');
   475→  }

   485→  /**
   486→   * Get Suricata alerts with filtering and pagination
   487→   */
   488→  async getAlerts(params: {
   489→    startTime?: string;
   490→    endTime?: string;
   491→    severity?: number[];       // 1-7 (1 is most severe)
   492→    signatureId?: number[];
   493→    sourceIp?: string;
   494→    destIp?: string;
   495→    destPort?: number[];
   496→    protocol?: string[];
   497→    action?: ('alerted' | 'blocked')[];
   498→    limit?: number;
   499→    offset?: number;
   500→    sortBy?: 'timestamp' | 'severity' | 'source_ip';
   501→    sortOrder?: 'asc' | 'desc';
   502→  } = {}): Promise<{ alerts: SuricataAlert[]; total: number }> {
   503→    const {
   504→      startTime,
   505→      endTime,
   506→      severity,
   507→      signatureId,
   508→      sourceIp,
   509→      destIp,
   510→      destPort,
   511→      protocol,
   512→      action,
   513→      limit = 100,
   514→      offset = 0,
   515→      sortBy = 'timestamp',
   516→      sortOrder = 'desc',
   517→    } = params;
   518→
   519→    try {
   520→      // Build query parameters for Elasticsearch/EVE JSON query
   521→      const queryParams: Record<string, any> = {
   522→        size: limit,
   523→        from: offset,
   524→        sort: [{ [sortBy]: { order: sortOrder } }],
   525→        _source: ['*'], // Return all fields
   526→      };
   527→
   528→      // Build bool query for filtering
   529→      const must: any[] = [];
   530→      const filter: any[] = [];
   531→
   532→      // Time range
   533→      if (startTime || endTime) {
   534→        filter.push({
   535→          range: {
   536→            timestamp: {
   537→              ...(startTime && { gte: startTime }),
   538→              ...(endTime && { lte: endTime }),
   539→            },
   540→          },
   541→        });
   542→      }
   543→
   544→      // Severity filter
   545→      if (severity && severity.length > 0) {
   546→        must.push({ terms: { 'alert.severity': severity } });
   547→      }
   548→
   549→      // Signature ID filter
   550→      if (signatureId && signatureId.length > 0) {
   551→        must.push({ terms: { 'alert.signature_id': signatureId } });
   552→      }
   553→
   554→      // IP filters
   555→      if (sourceIp) {
   556→        must.push({ term: { 'src_ip': sourceIp } });
   557→      }
   558→      if (destIp) {
   559→        must.push({ term: { 'dest_ip': destIp } });
   560→      }
   561→
   562→      // Port filter
   563→      if (destPort && destPort.length > 0) {
   564→        must.push({ terms: { 'dest_port': destPort } });
   565→      }
   566→
   567→      // Protocol filter
   568→      if (protocol && protocol.length > 0) {
   569→        must.push({ terms: { 'proto': protocol.map(p => p.toLowerCase()) } });
   570→      }
   571→
   572→      // Action filter
   573→      if (action && action.length > 0) {
   574→        must.push({ terms: { 'alert.action': action } });
   575→      }
   576→
   577→      // Combine into query
   578→      if (must.length > 0 || filter.length > 0) {
   579→        queryParams.query = { bool: { must, filter } };
   580→      }
   581→
   582→      const response = await this.fetchSuricata('/alerts/search', {
   583→        method: 'POST',
   584→        body: JSON.stringify(queryParams),
   585→      });
   586→
   587→      if (!response.ok) {
   588→        throw new Error(`Suricata alert search failed: ${response.status}`);
   589→      }
   590→
   591→      const data = await response.json();
   592→      const alerts: SuricataAlert[] = (data.hits?.hits || []).map((hit: any) =>
   593→        this.transformSuricataAlert(hit._source)
   594→      );
   595→
   596→      return {
   597→        alerts,
   598→        total: data.hits?.total?.value || alerts.length,
   599→      };
   600→
   601→    } catch (error) {
   602→      this.emit('error', error);
   603→      throw error;
   604→    }
   605→  }

   636→  /**
   637→   * Manage Suricata rules
   638→   */
   639→  async getRules(params: {
   640→    enabled?: boolean;
   641→    category?: string;
   642→    search?: string;
   643→    limit?: number;
   644→  } = {}): Promise<SuricataRule[]> {
   645→    const { enabled, category, search, limit = 500 } = params;
   646→
   647→    try {

   718→  /**
   719→   * Query Zeek logs with flexible filtering
   720→   */
   721→  async queryZeekLogs<T extends ZeekLogEntry>(params: {
   722→    logType: '_path';         // conn, dns, http, ssl, files, etc.
   723→    startTime?: string;
   724→    endTime?: string;
   725→    filters?: Array<{
   726→      field: keyof T;
   727→      operator: 'eq' | 'neq' | 'contains' | 'in' | 'gt' | 'lt' | 'exists';
   728→      value: any;
   729→    }>;
   730→    limit?: number;
   731→    offset?: number;
   732→  }): Promise<{ logs: T[]; total: number }> {
   733→    const {
   734→      logType,
   735→      startTime,
   736→      endTime,
   737→      filters = [],
   738→      limit = 100,
   739→      offset = 0,
   740→    } = params;
   741→
   742→    try {
   743→      // Build Zeek query (using Zeek's JSON API or direct log search)
   744→      const query: Record<string, any> = {
   745→        log_type: logType,
   746→        from: offset,
   747→        count: limit,
   748→        time_range: {
   749→          ...(startTime && { from: startTime }),
   750→          ...(endTime && { to: endTime }),
   751→        },
   752→        where: filters.map(f => ({
   753→          field: f.field,
   754→          op: f.operator,
   755→          value: f.value,
   756→        })),
   757→      };
   758→
   759→      const response = await this.fetchZeek('/logs/query', {
   760→        method: 'POST',
   761→        body: JSON.stringify(query),
   762→      });
   763→
   764→      if (!response.ok) {
   765→        throw new Error(`Zeek log query failed: ${response.status}`);
   766→      }
   767→
   768→      const data = await response.json();
   769→      const logs: T[] = (data.logs || []).map((log: any) => ({
   770→        ...log,
   771→        _time: new Date(log._time * 1000).toISOString(), // Convert Unix epoch to ISO
   772→      }));
   773→
   774→      return {
   775→        logs,
   776→        total: data.total || logs.length,
   777→      };
   778→
   779→    } catch (error) {
   780→      this.emit('error', error);
   781→      throw error;
   782→    }
   783→  }

   785→  /**
   786→   * Get Zeek connection summary with anomaly detection
   787→   */
   788→  async getConnectionSummary(params: {
   789→    sourceIp?: string;
   790→    destIp?: string;
   791→    timeRangeMinutes?: number;
   792→  } = {}): Promise<{
   793→    totalConnections: number;
   794→    uniqueDestinations: number;
   795→    protocols: Record<string, number>;
   796→    topTalkers: Array<{ ip: string; bytes: number; connections: number }>;
   797→    anomalies: Array<{
   798→      type: string;
   799→      description: string;
   800→      severity: string;
   801→      evidence: Record<string, any>;
   802→    }>;
   803→  }> {
   804→    const { sourceIp, destIp, timeRangeMinutes = 60 } = params;
   805→
   806→    try {
   807→      // Query connection logs
   808→      const { logs } = await this.queryZeekLogs<ZeekLogEntry>({
   809→        logType: 'conn',
   810→        startTime: new Date(Date.now() - timeRangeMinutes * 60000).toISOString(),
   811→        filters: [
   812→          ...(sourceIp ? [{ field: 'id_orig_h' as const, operator: 'eq' as const, value: sourceIp }] : []),
   813→          ...(destIp ? [{ field: 'id_resp_h' as const, operator: 'eq' as const, value: destIp }] : []),
   814→        ],
   815→        limit: 10000,
   816→      });
   817→
   818→      // Analyze connections
   819→      const protocols: Record<string, number> = {};
   820→      const talkerMap = new Map<string, { bytes: number; connections: number }>();
   821→      const anomalies: Array<{
   822→        type: string;
   823→        description: string;
   824→        severity: string;
   825→        evidence: Record<string, any>;
   826→      }> = [];
   827→
   828→      for (const conn of logs) {
   829→        // Protocol distribution
   830→        const proto = conn.proto || 'unknown';
   831→        protocols[proto] = (protocols[proto] || 0) + 1;
   832→
   833→        // Top talkers
   834→        const srcIp = conn.id_orig_h!;
   835→        const existing = talkerMap.get(srcIp) || { bytes: 0, connections: 0 };
   836→        existing.bytes += (conn.resp_bytes || 0) + (conn.orig_bytes || 0);
   837→        existing.connections += 1;
   838→        talkerMap.set(srcIp, existing);
   839→
   840→        // Anomaly detection
   841→        this.detectConnectionAnomalies(conn, anomalies);
   842→      }
   843→
   844→      // Sort talkers by bytes
   845→      const topTalkers = Array.from(talkerMap.entries())
   846→        .map(([ip, stats]) => ({ ip, ...stats }))
   847→        .sort((a, b) => b.bytes - a.bytes)
   848→        .slice(0, 20);
   849→
   850→      return {
   851→        totalConnections: logs.length,
   852→        uniqueDestinations: new Set(logs.map(l => l.id_resp_h)).size,
   853→        protocols,
   854→        topTalkers,
   855→        anomalies,
   856→      };
   857→
   858→    } catch (error) {
   859→      this.emit('error', error);
   860→      throw error;
   861→    }

   929→        .sort((a, b) => b.queries - a.queries)
   930→        .slice(0, 50);

   933→        totalQueries: logs.length,
   934→        uniqueDomains: domainMap.size,

  1277→    // Search Zeek logs for DNS/HTTP/SSL context

  1284→            startTime,
  1285→            filters: [{
  1286→              field: 'query' as const,
  1287→              operator: 'contains' as const,
  1288→              value: domain,
  1289→            }],
  1290→            limit: 50,
  1291→          });
  1292→
  1293→          for (const log of dnsLogs) {
  1294→            correlatedEvents.push({
  1295→              eventId: `dns-${log._uid}`,
  1296→              source: 'zeek',
  1297→              eventType: 'metadata',
  1298→              severity: 'info',
  1299→              title: `DNS Query: ${log.query}`,
  1300→              description: `DNS ${log.qtype_name} query for ${log.query}`,

  1304→              destinationIp: log.id_resp_h!,
  1305→              destinationPort: log.id_resp_p,
  1306→              protocol: log.proto || 'udp',
  1307→              zeekLog: log,
  1308→            });
  1309→          }
  1310→        }
  1311→
  1312→        // HTTP connections
  1313→        if (sourceIp || destIp) {
  1314→          const { logs: httpLogs } = await this.queryZeekLogs<ZeekLogEntry>({
  1315→            logType: 'http',
  1316→            startTime,
  1317→            filters: [
  1318→              ...(sourceIp ? [{ field: 'id_orig_h' as const, operator: 'eq' as const, value: sourceIp }] : []),
  1319→              ...(destIp ? [{ field: 'id_resp_h' as const, operator: 'eq' as const, value: destIp }] : []),
  1320→            ],
  1321→            limit: 50,
  1322→          });
  1323→
  1324→          for (const log of httpLogs) {
  1325→            correlatedEvents.push({
  1326→              eventId: `http-${log._uid}`,
  1327→              source: 'zeek',
  1328→              eventType: log.status_code && log.status_code >= 400 ? 'anomaly' : 'metadata',

  1345→    }
  1346→
  1347→    // Search Arkime for PCAP sessions
  1348→    if (sourceIp || destIp) {

  1352→          endTime: new Date().toISOString(),

  1388→  /**
  1389→   * Execute pre-built threat hunt queries

  1391→  async runThreatHunt(huntType: keyof typeof ThreatHuntPresets, targetIp?: string): Promise<{
  1392→    huntName: string;
  1393→    findings: NsmEvent[];
  1394→    riskScore: number;
  1395→    recommendations: string[];
  1396→  }> {
  1397→    const preset = ThreatHuntPresets[huntType];
  1398→    if (!preset) {
  1399→      throw new Error(`Unknown hunt type: ${huntType}`);
  1400→    }
  1401→
  1402→    const findings: NsmEvent[] = [];
  1403→
  1404→    // Execute each query in the preset
  1405→    for (const query of preset.queries) {
  1406→      switch (query.type) {
  1407→        case 'suricata':
  1408→          try {
  1409→            const { alerts } = await this.getAlerts({
  1410→              startTime: new Date(Date.now() - preset.timeRangeHours * 3600000).toISOString(),
  1411→              ...(targetIp && { sourceIp: targetIp }),
  1412→              signatureId: query.signatureIds,
  1413→              limit: 100,
  1414→            });
  1415→            
  1416→            findings.push(...alerts.map(a => ({
  1417→              eventId: `hunt-sur-${a.alertId}`,
  1418→              source: 'suricata' as const,
  1419→              eventType: 'alert' as const,
  1420→              severity: this.mapSeverity(a.alert.severity),

  1423→              timestamp: a.timestamp,
  1424→              sourceIp: a.sourceIp,
  1425→              sourcePort: a.sourcePort,
  1426→              destinationIp: a.destinationIp,
  1427→              destinationPort: a.destinationPort,
  1428→              protocol: a.protocol,
  1429→              suricataAlert: a,
  1430→            })));
  1431→          } catch (e) {
  1432→            console.error(`[NSM] Hunt query error (${query.type}):`, e.message);
  1433→          }
  1434→          break;
  1435→
  1436→        case 'zeek-dns':
  1437→          try {
  1438→            const { logs } = await this.queryZeekLogs<ZeekLogEntry>({
  1439→              logType: 'dns',
  1440→              startTime: new Date(Date.now() - preset.timeRangeHours * 3600000).toISOString(),
  1441→              filters: targetIp ? [{
  1442→                field: 'id_orig_h' as const,
  1443→                operator: 'eq' as const,
  1444→                value: targetIp,
  1445→              }] : [],
  1446→              limit: 200,
  1447→            });
  1448→
  1449→            findings.push(...logs.filter(l => 
  1450→              this.matchesDnsPattern(l, query.patterns || [])
  1451→            ).map(l => ({
  1452→              eventId: `hunt-dns-${l._uid}`,
  1453→              source: 'zeek' as const,
  1454→              eventType: 'reconnaissance' as const,
  1455→              severity: 'medium',
  1456→              title: `[Hunt] Suspicious DNS: ${l.query}`,
  1457→              description: preset.description,
  1458→              timestamp: l._time,
  1459→              sourceIp: l.id_orig_h!,
  1460→              sourcePort: l.id_orig_p,
  1461→              destinationIp: l.id_resp_h!,
  1462→              destinationPort: l.id_resp_p,
  1463→              protocol: 'udp',
  1464→              zeekLog: l,
  1465→            })));
  1466→          } catch (e) {
  1467→            console.error(`[NSM] Hunt query error (${query.type}):`, e.message);
  1468→          }
  1469→          break;
  1470→
  1471→        case 'zeek-files':
  1472→          try {
  1473→            const { logs } = await this.queryZeekLogs<ZeekLogEntry>({
  1474→              logType: 'files',
  1475→              startTime: new Date(Date.now() - preset.timeRangeHours * 3600000).toISOString(),
  1476→              limit: 100,
  1477→            });
  1478→
  1479→            findings.push(...logs.filter(l => 
  1480→              (l.size || 0) > (query.minSizeMb || 10) * 1024 * 1024 ||
  1481→              query.mimeTypes?.includes(l.mime_type || '')
  1482→            ).map(l => ({
  1483→              eventId: `hunt-file-${l.fuid}`,
  1484→              source: 'zeek' as const,
  1485→              eventType: 'exfiltration' as const,
  1486→              severity: 'high',
  1487→              title: `[Hunt] Large/Suspicious File: ${l.filename}`,
  1488→              description: `${l.filename} (${l.mime_type}, ${(l.size || 0) / 1024 / 1024} MB)`,
  1489→              timestamp: l._time,
  1490→              sourceIp: '', // Files don't always have IP context
  1491→              protocol: 'tcp',
  1492→              zeekLog: l,
  1493→            })));
  1494→          } catch (e) {
  1495→            console.error(`[NSM] Hunt query error (${query.type}):`, e.message);
  1496→          }
  1497→          break;
  1498→
  1499→        case 'arkime':
  1500→          if (targetIp) {
  1501→            try {
  1502→              const { sessions } = await this.searchSessions({
  1503→                startTime: new Date(Date.now() - preset.timeRangeHours * 3600000).toISOString(),
  1504→                endTime: new Date().toISOString(),
  1505→                sourceIp: targetIp,
  1506→                minBytes: query.minBytes || 1048576, // 1MB default
  1507→                length: 30,
  1508→              });
  1509→
  1510→              findings.push(...sessions.map(s => ({
  1511→                eventId: `hunt-ark-${s.sessionId}`,
  1512→                source: 'arkime' as const,
  1513→                eventType: 'exfiltration' as const,
  1514→                severity: s.bytes > 104857600 ? 'critical' : 'high', // >100MB = critical
  1515→                title: `[Hunt] High-volume Session: ${s.bytes} bytes`,
  1516→                description: `${s.sourceIp} → ${s.destinationIp}, ${s.packets} packets`,
  1517→                timestamp: s.firstPacket,
  1518→                sourceIp: s.sourceIp,
  1519→                sourcePort: s.sourcePort,
  1520→                destinationIp: s.destinationIp,
  1521→                destinationPort: s.destinationPort,
  1522→                protocol: s.protocol,
  1523→                arkimeSession: s,
  1524→              })));
  1525→            } catch (e) {
  1526→              console.error(`[NSM] Hunt query error (${query.type}):`, e.message);
  1527→            }
  1528→          }
  1529→          break;
  1530→      }
  1531→    }
  1532→
  1533→    // Calculate risk score based on findings
  1534→    const riskScore = Math.min(100, findings.reduce((score, f) => {
  1535→      switch (f.severity) {
  1536→        case 'critical': return score + 25;
  1537→        case 'high': return score + 15;
  1538→        case 'medium': return score + 8;
  1539→        case 'low': return score + 3;
  1540→        default: return score + 1;
  1541→      }
  1542→    }, 0));
  1543→
  1544→    return {
  1545→      huntName: preset.name,
  1546→      findings,
  1547→      riskScore,
  1548→      recommendations: preset.recommendations,
  1549→    };
  1550→  }

  1758→  private detectConnectionAnomalies(conn: ZeekLogEntry, anomalies: any[]): void {
  1759→    // Data exfiltration indicators
  1760→    if ((conn.resp_bytes || 0) > 104857600) { // >100MB upload
  1761→      anomalies.push({
  1762→        type: 'data_exfil',
  1763→        description: `Large data transfer detected: ${(conn.resp_bytes! / 1024 / 1024).toFixed(2)} MB to ${conn.id_resp_h}`,
  1764→        severity: 'high',
  1765→        evidence: conn,
  1766→      });
  1767→    }
  1768→
  1769→    // Port scanning indicators
  1770→    if (conn.conn_state === 'S0') { // SYN only - potential scan
  1771→      anomalies.push({
  1772→        type: 'port_scan',
  1773→        description: `Potential SYN scan from ${conn.id_orig_h} to ${conn.id_resp_h}:${conn.id_resp_p}`,
  1774→        severity: 'medium',
  1775→        evidence: conn,
  1776→      });
  1777→    }
  1778→
  1779→    // Long-lived connection (potential tunnel/C2)
  1780→    if ((conn.duration || 0) > 86400) { // >24 hours
  1781→      anomalies.push({
  1782→        type: 'persistent_connection',
  1783→        description: `Long-lived connection: ${conn.duration}s duration to ${conn.id_resp_h}`,
  1784→        severity: 'medium',
  1785→        evidence: conn,
  1786→      });
  1787→    }
  1788→
  1789→    // Unusual protocol combinations
  1790→    if (conn.service && !['http', 'dns', 'ssl', 'ssh', 'ftp', 'smtp'].includes(conn.service)) {
  1791→      anomalies.push({
