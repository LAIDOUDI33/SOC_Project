   251→export class MispClient extends EventEmitter {
   252→  private config: MispConfig;
   253→
   254→  constructor(config: MispConfig) {
   255→    super();
   256→    this.config = {
   257→      ...config,
   258→      timeout: config.timeout || 30000,
   259→      sslVerify: config.sslVerify ?? true,
   260→    };
   261→  }
   262→
   263→  private async apiRequest<T>(
   264→    path: string,
   265→    options: RequestInit = {}
   266→  ): Promise<T> {
   267→    const url = `${this.config.apiUrl}${path}`;
   268→
   269→    try {
   270→      const response = await fetch(url, {
   271→        ...options,
   272→        headers: {
   273→          'Authorization': this.config.apiKey,
   274→          'Content-Type': 'application/json',
   275→          'Accept': 'application/json',
   276→          ...(options.headers as Record<string, string>),
   277→        },
   278→        signal: AbortSignal.timeout(this.config.timeout!),
   279→      });
   280→
   281→      if (!response.ok) {
   282→        const errorBody = await response.json().catch(() => ({ message: response.statusText }));
   283→        throw new MispApiError(
   284→          `MISP API error (${response.status}): ${errorBody.message}`,
   285→          response.status,
   286→          undefined,
   287→          errorBody
   288→        );
   289→      }
   290→
   291→      return response.json() as Promise<T>;
   292→    } catch (error) {
   293→      if (error instanceof MispApiError || error instanceof ThreatIntelError) throw error;
   294→      throw new ThreatIntelError(
   295→        `MISP request failed: ${error instanceof Error ? error.message : String(error)}`,
   296→        'REQUEST_ERROR',
   297→        error as Error
   298→      );
   299→    }
   300→  }

   302→  /**
   303→   * Test connectivity to MISP instance
   304→   */
   305→  async healthCheck(): Promise<{
   306→    status: 'healthy' | 'unhealthy';
   307→    version: string;
   308→    user: { id: number; email: string; org_id: number; role: string };
   309→    latency: number;
   310→  }> {
   311→    const startTime = Date.now();
   312→
   313→    try {
   314→      const [user, serverVersion] = await Promise.all([
   315→        this.apiRequest<any>('/users/me/view.json'),
   316→        this.apiRequest<any>('/servers/getVersion.json'),
   317→      ]);
   318→
   319→      return {
   320→        status: 'healthy',
   321→        version: serverVersion.version?.version || 'connected',
   322→        user: {
   323→          id: user.User.id,
   324→          email: User.email,
   325→          org_id: User.org_id,
   326→          role: User.role_name,
   327→        },
   328→        latency: Date.now() - startTime,
   329→      };
   330→    } catch (error) {
   331→      return {
   332→        status: 'unhealthy',
   333→        version: 'unknown',
   334→        user: null as any,
   335→        latency: Date.now() - startTime,
   336→      };
   337→    }

   344→    info: string;

   347→    distribution?: number;
   348→    date?: string;

   383→  /**
   384→   * Search events with advanced filters
   385→   */
   386→  async searchEvents(filters: SearchFilters): Promise<{
   387→    events: MispEvent[];
   388→    total: number;
   389→  }> {
   390→    const params = new URLSearchParams();
   391→    
   392→    if (filters.value) params.append('value', filters.value);
   393→    if (filters.type) params.append('type', Array.isArray(filters.type) ? filters.type.join(',') : filters.type);
   394→    if (filters.category) params.append('category', Array.isArray(filters.category) ? filters.category.join(',') : filters.category);
   395→    if (filters.org) params.append('org', filters.org);
   396→    if (filters.tags) params.append('tags', filters.tags);
   397→    if (filters.from) params.append('from', filters.from);
   398→    if (filters.to) params.append('to', filters.to);
   399→    if (filters.last) params.append('last', filters.last);
   400→    if (filters.eventid) params.append('eventid', filters.eventid);
   401→    if (filters.limit) params.append('limit', String(filters.limit));
   402→    if (filters.page) params.append('page', String(filters.page));
   403→    if (filters.order) params.append('order', filters.order);
   404→    if (filters.sort) params.append('sort', filters.sort);
   405→
   406→    const result = await this.apiRequest<{
   407→      Event: MispEvent[];
   408→      response?: { totalCount: number };
   409→    }>(`/events/restSearch?${params}`);
   410→
   411→    return {
   412→      events: result.Event || [],
   413→      total: result.response?.totalCount || result.Event?.length || 0,
   414→    };
   415→  }

   417→  /**
   418→   * Search for specific IOCs across all events
   419→   */
   420→  async searchIocs(value: string, type?: string): Promise<MispAttribute[]> {
   421→    const filters: SearchFilters = { value, limit: 100 };
   422→    if (type) filters.type = type;
   423→
   424→    const result = await this.searchEvents(filters);
   425→
   426→    // Extract attributes from matching events
   427→    const attributes: MispAttribute[] = [];
   428→    
   429→    for (const event of result.events) {
   430→      if (event.Attribute) {
   431→        for (const attr of event.Attribute) {
   432→          if (!attr.deleted && attr.to_ids) {
   433→            attributes.push({
   434→              ...attr,
   435→              _eventId: event.id,
   436→              _eventInfo: event.info,
   437→              _eventDate: event.date,
   438→              _eventOrg: event.Org?.name,
   439→            });
   440→          }
   441→        }
   442→      }
   443→    }
   444→
   445→    return attributes;
   446→  }

   605→export class ThreatIntelService extends EventEmitter {
   606→  private mispClient: MispClient;
   607→  private config: MispConfig;
   608→  private isRunning = false;
   609→  private syncHistory: Map<string, SyncResult[]> = new Map();
   610→

   617→  /**
   618→   * Initialize the Threat Intel service
   619→   */
   620→  async initialize(): Promise<void> {
   621→    console.log('[TI] Initializing Threat Intelligence service...');
   622→
   623→    try {
   624→      const health = await this.mispClient.healthCheck();
   625→      console.log(`[TI] MISP connection: ${health.status} (${health.latency}ms)`);
   626→
   627→      this.isRunning = true;
   628→      this.emit('initialized', health);
   629→
   630→      console.log('[TI] Threat Intelligence service initialized successfully');
   631→    } catch (error) {
   632→      console.error('[TI] Failed to initialize:', error);
   633→      this.emit('error', error);
   634→      throw error;
   635→    }
   636→  }
   637→
   638→  /**
   639→   * Shutdown gracefully
   640→   */
   641→  async shutdown(): Promise<void> {
   642→    console.log('[TI] Shutting down Threat Intelligence service...');
   643→    this.isRunning = false;
   644→    this.emit('shutdown');
   645→    console.log('[TI] Threat Intelligence service shutdown complete');
   646→  }
   647→
   648→  /**
   649→   * Ingest IOCs from external source into local database
   650→   */
   651→  async ingestIocs(iocs: Array<Omit<IoCRecord, 'id' | 'createdAt' | 'updatedAt' | 'matchCount'>>): Promise<{
   652→    success: number;
   653→    failed: number;
   654→    duplicates: number;
   655→  }> {
   656→    if (!this.isRunning) {
   657→      throw new ThreatIntelError('TI service is not running', 'SERVICE_NOT_RUNNING');
   658→    }
   659→
   660→    let success = 0;
   661→    let failed = 0;
   662→    let duplicates = 0;
   663→
   664→    for (const ioc of iocs) {
   665→      try {
   666→        // Check for duplicates (by value + type)
   667→        const existing = await this.mispClient.searchIocs(ioc.value, ioc.iocType);
   668→        
   669→        if (existing.length > 0) {
   670→          duplicates++;
   671→          continue;
   672→        }
   673→
   674→        // Create MISP event or add to existing event based on feed/type
   675→        // For now, create individual events per IOC (in production, batch by feed)
   676→        await this.mispClient.createEvent({
   677→          info: `IOC: ${ioc.value} (${ioc.iocType})`,
   678→          threat_level_id: this.mapConfidenceToThreatLevel(ioc.confidence),
   679→          analysis: 2, // Completed
   680→          distribution: 3, // All communities
   681→          attributes: [{
   682→            type: this.mapIocTypeToMispType(ioc.iocType),
   683→            category: this.mapIocTypeToCategory(ioc.iocType),
   684→            value: ioc.value,
   685→            to_ids: true,
   686→            comment: ioc.description || `Source: ${ioc.feedName || 'manual'}`,
   687→          }],
   688→          tags: [
   689→            ...(ioc.threatTypes || []),
   690→            ...(ioc.malwareFamilies || []),
   691→            ...(ioc.tags || []),
   692→            `tlp:${ioc.tlp}`,
   693→            ...(ioc.feedName ? [`feed:${ioc.feedName}`] : []),
   694→          ],
   695→        });
   696→
   697→        success++;
   698→        this.emit('ioc_ingested', { ioc });
   699→      } catch (error) {
   700→        failed++;
   701→        console.error(`[TI] Failed to ingest IOC ${ioc.value}:`, error);
   702→        this.emit('ingest_error', { ioc, error });
   703→      }
   704→    }
   705→
   706→    return { success, failed, duplicates };
   707→  }

   709→  /**
   710→   * Lookup IOC against MISP and all configured feeds
   711→   */
   712→  async lookupIoc(value: string, type?: string): Promise<{
   713→    found: boolean;
   714→    matches: Array<{
   715→      attribute: MispAttribute;
   716→      event: { id: string; info: string; date: string; org: string };
   717→      riskScore: number; // 0-100
   718→    }>;
   719→    enrichment?: {
   720→      whois?: Record<string, any>;
   721→      geo?: Record<string, any>;
   722→      passiveDns?: Array<{ query: string; answer: string }>;
   723→    };
   724→    onWarninglist?: boolean;
   725→  }> {
   726→    if (!this.isRunning) {
   727→      throw new ThreatIntelError('TI service is not running', 'SERVICE_NOT_RUNNING');
   728→    }
   729→
   730→    try {
   731→      // Search MISP for matches
   732→      const attributes = await this.mispClient.searchIocs(value, type);
   733→
   734→      // Check warninglists
   735→      let onWarninglist = false;
   736→      try {
   737→        const wlCheck = await this.mispClient.checkWarninglist(value);
   738→        onWarninglist = wlCheck.hit;
   739→      } catch (e) {
   740→        // Warninglist check failure shouldn't block lookup
   741→        console.warn('[TI] Warninglist check failed:', e.message);
   742→      }
   743→
   744→      // Calculate risk scores for each match
   745→      const matches = attributes.map(attr => ({
   746→        attribute: attr,
   747→        event: {
   748→          id: attr._eventId || '',
   749→          info: attr._eventInfo || '',
   750→          date: attr._eventDate || '',
   751→          org: attr._eventOrg || '',
   752→        },
   753→        riskScore: this.calculateRiskScore(attr),
   754→      }));
   755→
   756→      // Sort by risk score descending
   757→      matches.sort((a, b) => b.riskScore - a.riskScore);
   758→
   759→      return {
   760→        found: matches.length > 0,
   761→        matches,
   762→        onWarninglist,
   763→      };
   764→    } catch (error) {
   765→      console.error('[TI] IOC lookup failed:', error);
   766→      throw error;
   767→    }
   768→  }

   770→  /**
   771→   * Sync feeds from configured sources
   772→   */
   773→  async syncFeeds(feedIds?: string[]): Promise<SyncResult[]> {
   774→    if (!this.isRunning) {
   775→      throw new ThreatIntelError('TI service is not running', 'SERVICE_NOT_RUNNING');
   776→    }
   777→
   778→    const results: SyncResult[] = [];
   779→
   780→    try {
   781→      const feeds = await this.mispClient.getFeeds(true);
   782→      
   783→      const targetFeeds = feedIds 
   784→        ? feeds.filter(f => feedIds.includes(f.id))
   785→        : feeds.filter(f => f.enabled);
   786→
   787→      for (const feed of targetFeeds) {
   788→        const syncResult: SyncResult = {
   789→          feedId: feed.id,
   790→          feedName: feed.name,
   791→          startedAt: new Date().toISOString(),
   792→          status: 'running',
   793→          itemsProcessed: 0,
   794→          itemsNew: 0,
   795→          itemsUpdated: 0,
   796→          itemsFailed: 0,
   797→        };
   798→
   799→        try {
   800→          console.log(`[TI] Syncing feed: ${feed.name}`);
   801→          
   802→          // Fetch feed content
   803→          const content = await this.mispClient.fetchFeedContent(feed.id);
   804→          
   805→          // Process each item from the feed
   806→          for (const item of content) {
   807→            try {
   808→              syncResult.itemsProcessed++;
   809→              
   810→              // Convert feed item to IOC format and ingest
   811→              // (Implementation depends on feed format)
   812→              
   813→              syncResult.itemsNew++; // Simplified - would actually check for existence
   814→            } catch (itemError) {
   815→              syncResult.itemsFailed++;
   816→              syncResult.errors?.push(String(itemError));
   817→            }
   818→          }
   819→
   820→          syncResult.completedAt = new Date().toISOString();
   821→          syncResult.status = syncResult.itemsFailed > 0 ? 'partial' : 'success';
   822→
   823→          this.emit('feed_synced', syncResult);
   824→        } catch (feedError) {
   825→          syncResult.completedAt = new Date().toISOString();
   826→          syncResult.status = 'error';
   827→          syncResult.errors = [String(feedError)];
   828→          
   829→          this.emit('sync_error', { feed, error: feedError });
   830→        }
   831→
   832→        results.push(syncResult);
   833→        
   834→        // Store in history
   835→        const history = this.syncHistory.get(feed.id) || [];
   836→        history.push(syncResult);
   837→        this.syncHistory.set(feed.id, history);
   838→      }
   839→
   840→      return results;
   841→    } catch (error) {
   842→      console.error('[TI] Feed sync failed:', error);
   843→      throw error;
   844→    }
   845→  }

   890→  private mapConfidenceToThreatLevel(confidence: number): 1 | 2 | 3 | 4 {
   891→    if (confidence >= 80) return 1; // High
   892→    if (confidence >= 60) return 2; // Medium
   893→    if (confidence >= 40) return 3; // Low
   894→    return 4; // Undefined
   895→  }

   941→  private calculateRiskScore(attribute: MispAttribute): number {
   942→    let score = 0;
   943→
   944→    // Base score from TLP
   945→    const tlpTags = (attribute.AttributeTag || [])
   946→      .filter(t => t.Tag.name.startsWith('tlp:'))
   947→      .map(t => t.Tag.name);
   948→
   949→    if (tlpTags.includes('tlp:red')) score += 40;
   950→    else if (tlpTags.includes('tlp:amber')) score += 25;
   951→    else if (tlpTags.includes('tlp:green')) score += 10;
   952→
   953→    // Score from IDS flag
   954→    if (attribute.to_ids) score += 20;
   955→
   956→    // Score from correlation count
   957→    if (attribute.Correlation && attribute.Correlation.length > 0) {
   958→      score += Math.min(attribute.Correlation.length * 5, 30);
   959→    }
   960→
   961→    // Score from tags indicating severity
   962→    const severeTags = ['malware', 'phishing', 'apt', 'c2', 'botnet'];
   963→    const hasSevereTag = (attribute.AttributeTag || [])
   964→      .some(t => severeTags.some(st => t.Tag.name.toLowerCase().includes(st)));
   965→    
   966→    if (hasSevereTag) score += 15;
   967→
   968→    return Math.min(score, 100); // Cap at 100
   969→  }
